def value(node,path,val):
    element=node.find(path);assert element is not None,path
    element.set('Value',str(val))


def remap(node):
    global next_id
    mapping={}
    for e in node.iter():
        if 'Id' in e.attrib and (e.tag.endswith('Target') or e.tag=='Pointee'):
            mapping[e.get('Id')]=str(next_id);e.set('Id',str(next_id));next_id+=1
    for e in node.iter('PointeeId'):
        if e.get('Value') in mapping:e.set('Value',mapping[e.get('Value')])
    return node

    auto=ET.SubElement(env,'Automation');events=ET.SubElement(auto,'Events')
    for i,(beat,val) in enumerate([(-63072000,points[0][1])]+points):
        ET.SubElement(events,'BoolEvent' if boolean else 'FloatEvent',Id=str(i),Time=str(beat),Value=str(val).lower())

    ET.SubElement(view,'IsTransformPending',Value='false');ET.SubElement(view,'TimeAndValueTransforms')
    param.find('Manual').set('Value',str(points[0][1]).lower())
    manifest['automation'].append({'lane':track.find('Name/EffectiveName').get('Value'),'control':label,'points':points})

def energy(beat):
    if beat<64:return .65
    if beat<128:return .8
    if beat<192:return .9
    if beat<256:return .82
    if beat<384:return 1
    if beat<448:return .45
    if beat<512:return .72
    if beat<640:return 1
    if beat<704:return .85
    return .6

import asyncio
import base64
import importlib.util
import json
from pathlib import Path
import shutil
import sys

spec = importlib.util.spec_from_file_location('bass', '/private/tmp/beatmind-original-bass.py')
bass = importlib.util.module_from_spec(spec)
spec.loader.exec_module(bass)
ROOT = Path('/Users/jeet/Music/BeatMind Projects/Original Groove - Bass 01 Project')


async def main():
    assert await bass.window_title() == 'Original Groove - Kick and Bass'
    live = bass.Live()
    try:
        names = live.get('/live/song/get/track_names')
        assert names[:2] == ['YOUR Kick', 'YOUR Bass'] and names[2:] in (
            ['YOUR Percussion', 'YOUR Synth'], ['YOUR Closed Hat', 'YOUR Synth'])
        assert live.get('/live/song/get/tempo') == [122.0]
        assert not live.get('/live/song/get/record_mode')[-1]
        assert not live.get('/live/song/get/session_record')[-1]
        live.send('/live/song/stop_playing')
        await asyncio.sleep(0.2)
        live.send('/live/song/set/current_song_time', [0.0])
        await asyncio.sleep(0.2)
        assert live.get('/live/song/get/current_song_time') == [0.0]
        mode = sys.argv[1]
        if mode == 'load':
            assert live.get('/live/track/get/devices/name', [2]) == [2]
            assert not live.get('/live/clip_slot/get/has_clip', [2, 0])[-1]
            backup = ROOT / 'Original Groove - Before Hats.als'
            assert not backup.exists(), 'Inspect the previous attempt, do not repeat it.'
            shutil.copy2(ROOT / 'Original Groove - Kick and Bass.als', backup)
            result = await live.tool('load_library_item', {'track': 2, 'kind': 'instrument',
                'category': 'instruments', 'folders': ['DS HH']})
            print(result['summary'], flush=True)
            result = await live.tool('get_device_control_map', {'track': 2, 'path': [0], 'limit': 32})
            (ROOT / 'hat-01-initial-map.json').write_text(json.dumps(result, indent=2))
            print(json.dumps({k: v for k, v in result.items() if k != 'steps'}, indent=2), flush=True)
        elif mode in ('make', 'audition'):
            assert live.get('/live/track/get/devices/name', [2]) == [2, 'DS HH']
            before_notes = [live.get('/live/clip/get/notes', [i, 0]) for i in (0, 1)]
            if mode == 'make':
                assert not live.get('/live/clip_slot/get/has_clip', [2, 0])[-1], 'Inspect existing clip; do not duplicate.'
                changes = []
                for name, value, unit in [('Noise', 'Pink', 'label'), ('Decay', 9, '%'),
                                          ('Pitch', 45, '%'), ('Tone', 75, 'native')]:
                    mapping = await live.tool('get_device_control_map', {'track': 2, 'path': [0], 'query': name})
                    result = await live.tool('set_device_control', {'track': 2, 'path': [0], 'map_id': mapping['map_id'],
                                          'control': name, 'value': value, 'unit': unit})
                    changes.append(result)
                    print(result['summary'], flush=True)
                live.send('/live/track/set/volume', [2, 0.72])
                await asyncio.sleep(0.2)
                assert abs(live.get('/live/track/get/volume', [2])[-1] - 0.72) < 0.001
                notes = []
                for bar in range(8):
                    for beat in range(4):
                        late = 0.016 if beat in (1, 3) else 0
                        notes.append([36, bar * 4 + beat + 0.5 + late, 0.07,
                                      float([80, 72, 84, 75][beat] - bar % 3), 0])
                    notes.append([36, bar * 4 + 1.77, 0.05, 42.0, 0])
                    if bar in (3, 7):
                        notes.append([36, bar * 4 + 3.77, 0.05, 50.0, 0])
                notes.sort(key=lambda n: n[1])
                live.send('/live/clip_slot/create_clip', [2, 0, 32.0])
                await asyncio.sleep(0.3)
                assert live.get('/live/clip_slot/get/has_clip', [2, 0])[-1]
                live.send('/live/clip/set/name', [2, 0, 'Hat 01 - Original - Awaiting Review'])
                live.send('/live/track/set/name', [2, 'YOUR Closed Hat'])
                live.send('/live/scene/set/name', [0, 'Groove 01 - Kick + Bass + Hat'])
                live.send('/live/clip/add/notes', [2, 0] + [v for note in notes for v in note])
                await asyncio.sleep(0.3)
                actual = live.get('/live/clip/get/notes', [2, 0])[2:]
                expected = [v for note in notes for v in note]
                assert len(actual) == len(expected) and all(abs(a-b) < 0.0001 for a,b in zip(actual, expected))
                assert [live.get('/live/clip/get/notes', [i, 0]) for i in (0, 1)] == before_notes
                (ROOT / 'hat-01-design.json').write_text(json.dumps({'source': 'Ableton DS HH synthesis',
                    'reference_audio_used': False, 'notes': notes, 'controls': changes,
                    'timing_offset_beats': 0.016, 'timing_offset_ms_at_122_bpm': 0.016 * 60000 / 122,
                    'approval': 'pending'}, indent=2))
            assert live.get('/live/clip/get/name', [2, 0])[-1] == 'Hat 01 - Original - Awaiting Review'
            preview = await bass.capture_part(live, 2, 0, 8)
            if preview.get('audio_base64'):
                (ROOT / 'Hat-01-Original.m4a').write_bytes(base64.b64decode(preview.pop('audio_base64')))
            (ROOT / 'hat-01-audition.json').write_text(json.dumps(preview, indent=2))
            assert preview['status'] == 'verified', preview.get('summary')
            print(json.dumps({k:v for k,v in preview['metrics'].items() if k != 'waveform'}), flush=True)
        else:
            raise ValueError('Unknown operation')
    finally:
        live.close()


if __name__ == '__main__':
    asyncio.run(main())

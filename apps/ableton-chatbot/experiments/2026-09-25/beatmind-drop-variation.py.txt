def clips(track):
    return track.findall('DeviceChain/MainSequencer/ClipTimeable/ArrangerAutomation/Events/MidiClip')


def at(track,beat):
    found=[c for c in clips(track) if float(c.get('Time'))==beat]
    assert len(found)==1
    return found[0]


def add_note(clip,pitch,time,duration,velocity):
    keys=clip.find('Notes/KeyTracks')
    found=[k for k in keys if int(k.find('MidiKey').get('Value'))==pitch]
    if found:
        key=found[0]
    else:
        key=ET.SubElement(keys,'KeyTrack',Id=str(max([int(k.get('Id')) for k in keys]+[-1])+1))
        ET.SubElement(key,'Notes')
        ET.SubElement(key,'MidiKey',Value=str(pitch))
    if any(abs(float(n.get('Time'))-time)<.04 for n in key.find('Notes')):
        return
    generator=clip.find('Notes/NoteIdGenerator/NextId')
    noteid=int(generator.get('Value'))
    ET.SubElement(key.find('Notes'),'MidiNoteEvent',Time=str(time),Duration=str(duration),
                  Velocity=str(velocity),OffVelocity='64',NoteId=str(noteid))
    generator.set('Value',str(noteid+1))

def envelope(track,parameter,points,label,boolean=False):
    target=parameter.find('AutomationTarget').get('Id')
    envs=track.find('AutomationEnvelopes/Envelopes')
    assert not any(e.find('EnvelopeTarget/PointeeId').get('Value')==target for e in envs)
    env=ET.SubElement(envs,'AutomationEnvelope',Id=str(max([int(e.get('Id')) for e in envs]+[-1])+1))
    ET.SubElement(ET.SubElement(env,'EnvelopeTarget'),'PointeeId',Value=target)
    auto=ET.SubElement(env,'Automation');events=ET.SubElement(auto,'Events')
    for i,(beat,value) in enumerate([(-63072000,points[0][1])]+points):
        ET.SubElement(events,'BoolEvent' if boolean else 'FloatEvent',Id=str(i),Time=str(beat),
                      Value=str(value).lower() if boolean else str(value))
    view=ET.SubElement(auto,'AutomationTransformViewState')
    ET.SubElement(view,'IsTransformPending',Value='false');ET.SubElement(view,'TimeAndValueTransforms')
    changes.append({'automation':label,'target':target,'points':points})

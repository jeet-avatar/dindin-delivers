import asyncio
import base64
import importlib.util
import json
from pathlib import Path
import shutil
import sys

spec = importlib.util.spec_from_file_location('bass', '/private/tmp/beatmind-original-bass.py')
bass = importlib.util.module_from_spec(spec)
spec.loader.exec_module(bass)
ROOT = Path('/Users/jeet/Music/BeatMind Projects/Original Groove - Bass 01 Project')
BASE = ['YOUR Kick', 'YOUR Bass', 'YOUR Closed Hat', 'YOUR Clap']


async def main():
    assert await bass.window_title() == 'Original Groove - Rhythm Section'
    live = bass.Live()
    try:
        mode = sys.argv[1]
        assert live.get('/live/song/get/tempo') == [122.0]
        assert not live.get('/live/song/get/record_mode')[-1]
        assert not live.get('/live/song/get/session_record')[-1]
        live.send('/live/song/stop_playing')
        await asyncio.sleep(0.2)
        live.send('/live/song/set/current_song_time', [0.0])
        await asyncio.sleep(0.2)
        assert live.get('/live/song/get/current_song_time') == [0.0]
        if mode == 'load':
            assert live.get('/live/song/get/track_names') == BASE
            backup = ROOT / 'Original Groove - Before Keys.als'
            assert not backup.exists(), 'Inspect existing attempt before any retry.'
            shutil.copy2(ROOT / 'Original Groove - Rhythm Section.als', backup)
            live.send('/live/song/create_midi_track', [-1])
            await asyncio.sleep(0.3)
            names = live.get('/live/song/get/track_names')
            assert names[:4] == BASE and len(names) == 5
            live.send('/live/track/set/name', [4, 'YOUR Plucked Keys'])
            await asyncio.sleep(0.2)
            assert live.get('/live/song/get/track_names') == BASE + ['YOUR Plucked Keys']
            assert live.get('/live/track/get/devices/name', [4]) == [4]
            for kind, category, device in [('instrument', 'instruments', 'Operator'),
                                            ('effect', 'audio_effects', 'Echo')]:
                result = await live.tool('load_library_item', {'track': 4, 'kind': kind,
                    'category': category, 'folders': [device]})
                print(result['summary'], flush=True)
            result = await live.tool('get_device_control_map', {'track': 4, 'path': [1], 'limit': 32})
            (ROOT / 'keys-01-echo-initial-map.json').write_text(json.dumps(result, indent=2))
            print(json.dumps({k: v for k, v in result.items() if k != 'steps'}, indent=2), flush=True)
        elif mode in ('make', 'audition'):
            assert live.get('/live/song/get/track_names') == BASE + ['YOUR Plucked Keys']
            assert live.get('/live/track/get/devices/name', [4]) == [4, 'Operator', 'Echo']
            if mode == 'make':
                assert not live.get('/live/clip_slot/get/has_clip', [4, 0])[-1], 'Inspect existing clip; do not duplicate.'
                previous = [live.get('/live/clip/get/notes', [i, 0]) for i in range(4)]
                changes = []
                controls = [(0, 'Osc-A Wave', 'Saw 4', 'label'),
                    (0, 'Filter Freq', 1800, 'Hz'), (0, 'Filter Res', 12, '%'),
                    (0, 'Ae Attack', 6, 'ms'), (0, 'Ae Decay', 420, 'ms'),
                    (0, 'Ae Sustain', 0.2, 'normalized'), (0, 'Ae Release', 260, 'ms'),
                    (0, 'Volume', 0.5, 'normalized'),
                    (1, 'Feedback', 24, '%'), (1, 'Dry Wet', 18, '%'),
                    (1, 'Channel Mode', 'Ping Pong', 'label'),
                    (1, 'HP Freq', 250, 'Hz'), (1, 'LP Freq', 3500, 'Hz')]
                for device, name, value, unit in controls:
                    mapping = await live.tool('get_device_control_map', {'track': 4, 'path': [device], 'query': name})
                    result = await live.tool('set_device_control', {'track': 4, 'path': [device], 'map_id': mapping['map_id'],
                                          'control': name, 'value': value, 'unit': unit})
                    changes.append(result)
                    print(result['summary'], flush=True)
                mapped = await live.tool('get_device_control_map', {'track': 4, 'path': [1], 'limit': 32})
                displays = {p['name']: p['display'] for p in mapped['parameters']}
                assert displays['L Sync'] == 'On' and displays['L Division'] == '1/8' and displays['L Sync Mode'] == 'Dotted'
                live.send('/live/track/set/volume', [4, 0.78])
                await asyncio.sleep(0.2)
                assert abs(live.get('/live/track/get/volume', [4])[-1] - 0.78) < 0.001
                voicings = [[63, 67, 70, 74], [60, 63, 67, 68], [60, 63, 65, 68], [62, 65, 67, 70]]
                notes = []
                for bar in range(8):
                    chord = voicings[bar // 2]
                    for onset, pair, length, velocity in [(0.75, (0, 2), 0.32, 72),
                                                         (1.5, (1, 3), 0.24, 64),
                                                         (2.75, (0, 1), 0.40, 76)]:
                        for voice, index in enumerate(pair):
                            notes.append([chord[index], bar * 4 + onset + voice * 0.008,
                                          length, float(velocity - voice * 5 - bar % 3), 0])
                    if bar in (3, 7):
                        notes.append([chord[3], bar * 4 + 3.5, 0.25, 58.0, 0])
                live.send('/live/clip_slot/create_clip', [4, 0, 32.0])
                await asyncio.sleep(0.3)
                assert live.get('/live/clip_slot/get/has_clip', [4, 0])[-1]
                live.send('/live/clip/set/name', [4, 0, 'Keys 01 - Original - Awaiting Review'])
                live.send('/live/scene/set/name', [0, 'Groove 01 - Rhythm + Keys'])
                live.send('/live/clip/add/notes', [4, 0] + [v for note in notes for v in note])
                await asyncio.sleep(0.3)
                actual = live.get('/live/clip/get/notes', [4, 0])[2:]
                actual = sorted([actual[i:i+5] for i in range(0, len(actual), 5)])
                assert len(actual) == len(notes) and all(all(abs(a-b)<0.0001 for a,b in zip(x,y))
                                                                       for x,y in zip(actual, sorted(notes)))
                assert [live.get('/live/clip/get/notes', [i, 0]) for i in range(4)] == previous
                (ROOT / 'keys-01-design.json').write_text(json.dumps({'source': 'Operator + Echo',
                    'reference_audio_used': False, 'midi_origin': 'Newly authored harmonic phrase; no reference transcription',
                    'voicings': voicings, 'notes': notes, 'controls': changes, 'echo_division': 'dotted eighth',
                    'approval': 'pending'}, indent=2))
            assert live.get('/live/clip/get/name', [4, 0])[-1] == 'Keys 01 - Original - Awaiting Review'
            preview = await bass.capture_part(live, 4, 0, 16)
            if preview.get('audio_base64'):
                (ROOT / 'Keys-01-Original.m4a').write_bytes(base64.b64decode(preview.pop('audio_base64')))
            (ROOT / 'keys-01-audition.json').write_text(json.dumps(preview, indent=2))
            assert preview['status'] == 'verified', preview.get('summary')
            print(json.dumps({k:v for k,v in preview['metrics'].items() if k != 'waveform'}), flush=True)
        else:
            raise ValueError('Unknown operation')
    finally:
        live.close()


if __name__ == '__main__':
    asyncio.run(main())

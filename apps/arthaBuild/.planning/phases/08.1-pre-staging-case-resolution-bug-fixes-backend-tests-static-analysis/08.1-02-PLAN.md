```python
def test_pip_audit_no_critical_vulnerabilities():
    """CASE-191: pip-audit must report zero CRITICAL or HIGH vulnerabilities."""
    import subprocess
    result = subprocess.run(
        ["pip-audit", "--format=json", "--progress-spinner=off"],
        capture_output=True, text=True
    )
    # pip-audit exits 1 if vulnerabilities found — check for CRITICAL/HIGH only
    if result.returncode != 0:
        import json
        try:
            audit_data = json.loads(result.stdout)
            critical_high = [
                v for dep in audit_data.get("dependencies", [])
                for v in dep.get("vulns", [])
                if v.get("severity", "").upper() in ("CRITICAL", "HIGH")
            ]
            assert len(critical_high) == 0, \
                f"pip-audit found CRITICAL/HIGH vulnerabilities: {critical_high}"
        except json.JSONDecodeError:
            pass  # pip-audit not installed or different format — skip
```

    Create scripts/smoke_test.sh — runs after every deployment to verify the instance is healthy:

    ```bash
    #!/bin/bash
    # ArthaBuild Smoke Test — run after deployment to verify all critical paths
    # Usage: ./scripts/smoke_test.sh [BASE_URL]
    # Exit code 0 = all pass, non-zero = failures
    BASE_URL=${1:-http://localhost}
    PASS=0
    FAIL=0
    ERRORS=()

    check() {
      local name="$1"
      local result="$2"
      local expected="$3"
      if echo "$result" | grep -q "$expected"; then
        echo "  ✓ $name"
        PASS=$((PASS+1))
      else
        echo "  ✗ $name (expected '$expected' in: $result)"
        FAIL=$((FAIL+1))
        ERRORS+=("$name")
      fi
    }

    echo "=== ArthaBuild Smoke Test === $(date)"
    echo "Target: $BASE_URL"
    echo ""

    # 1. Health endpoint
    R=$(curl -s "$BASE_URL/health")
    check "Health endpoint returns status:ok" "$R" '"status":"ok"'

    # 2. AI ready
    check "AI pipeline ready (ai_ready:true)" "$R" '"ai_ready":true'

    # 3. License valid
    check "License valid" "$R" '"license_valid":true'

    # 4. Register endpoint available
    R=$(curl -s -X POST "$BASE_URL/api/user/register" -H "Content-Type: application/json" \
      -d '{"first_name":"Smoke","last_name":"Test","email":"smoketest@artha.local","password":"SmokeTest123!"}')
    check "Register endpoint responds" "$R" '"user"'

    # 5. Login endpoint
    TOKEN=$(curl -s -X POST "$BASE_URL/api/auth/login" -H "Content-Type: application/json" \
      -d '{"email":"smoketest@artha.local","password":"SmokeTest123!"}' \
      | python3 -c "import json,sys; print(json.load(sys.stdin).get('access_token',''))" 2>/dev/null)
    check "Login returns access_token" "$TOKEN" "eyJ"

    # 6. JWT required on protected endpoint
    R=$(curl -s -X POST "$BASE_URL/api/chatbot/process" -H "Content-Type: application/json" \
      -d '{"message":"test"}')
    check "Protected endpoint requires JWT" "$R" '"detail"'

    # 7. Chat endpoint reachable with token
    R=$(curl -s -X POST "$BASE_URL/api/chatbot/process" \
      -H "Authorization: Bearer $TOKEN" \
      -H "Content-Type: application/json" \
      -d '{"message":"What is NetSuite?"}')
    check "Chat endpoint returns response" "$R" '"response"'

    # 8. NetSuite status endpoint
    R=$(curl -s "$BASE_URL/api/netsuite/status" -H "Authorization: Bearer $TOKEN")
    check "NetSuite status endpoint" "$R" '"authenticated"'

    # 9. License status endpoint
    R=$(curl -s "$BASE_URL/api/license/status")
    check "License status endpoint" "$R" '"valid"'

    # 10. Check-user endpoint
    R=$(curl -s -X POST "$BASE_URL/api/auth/check-user" -H "Content-Type: application/json" \
      -d '{"email":"smoketest@artha.local"}')
    check "Check-user endpoint" "$R" '"exists"'

    echo ""
    echo "Results: $PASS passed, $FAIL failed"
    if [ $FAIL -gt 0 ]; then
      echo "FAILED: ${ERRORS[*]}"
      exit 1
    else
      echo "All smoke tests PASSED"
      exit 0
    fi
    ```

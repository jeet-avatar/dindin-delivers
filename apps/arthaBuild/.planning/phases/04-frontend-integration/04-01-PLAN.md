       ```typescript
       const handleSend = async () => {
         if (!input.trim() || !activeChatId) return;
         setLoading(true);

         const userMsg: Message = {
           id: Date.now().toString(),
           role: "user",
           content: input.trim(),
           createdAt: new Date().toISOString(),
         };
         addMessage(activeChatId, userMsg);
         const question = input.trim();
         setInput("");

         try {
           const result = await sendChatMessage(question, sessionId);
           // Echo back the session_id so subsequent messages continue the conversation
           if (result.session_id) setSessionId(result.session_id);

           const botMsg: Message = {
             id: Date.now().toString(),
             role: "assistant",
             content: result.response,
             createdAt: new Date().toISOString(),
           };
           addMessage(activeChatId, botMsg);
         } catch (err: any) {
           const errMsg: Message = {
             id: Date.now().toString(),
             role: "assistant",
             content: err.message?.includes("loading")
               ? "⏳ AI model is still loading. Please wait a moment and try again."
               : err.message?.includes("expired")
               ? "🔒 Your session has expired. Please log in again."
               : `❌ Error: ${err.message || "Request failed"}`,
             createdAt: new Date().toISOString(),
           };
           addMessage(activeChatId, errMsg);
         } finally {
           setLoading(false);
         }
       };
       ```

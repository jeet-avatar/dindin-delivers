```typescript
// backend/src/onboarding/data-signals.ts — Phase 65.2 plan 02
// Pure read helper — 6 queries via Promise.all on the caller's PoolClient.
// MUST be invoked INSIDE a withTenantClient callback so RLS guarantees
// tenant isolation. Caller owns the txn; this helper never BEGIN/COMMITs.
//
// W5: signal cache is module-scope Map (survives Lambda warm reuse,
// recomputes on cold start). 5-min TTL. NO DB column.

import type { PoolClient } from 'pg';

export interface DataSignals {
  item_count: number;
  brand_breakdown: Record<string, number>;
  so_count: number;
  so_total_usd: number;
  bom_part_count: number;
  vendor_count: number;
  customer_count: number;
  has_corporate_facts: boolean;
}

// === W5: module-scope in-memory cache. Survives Lambda warm reuse; lost on cold start (acceptable per research Pitfall 3). ===
interface CacheEntry { signals: DataSignals; computedAt: number; }
const SIGNAL_CACHE: Map<string, CacheEntry> = new Map();
const TTL_MS = 5 * 60 * 1000; // 5 minutes

export function getCachedSignals(tenantId: string): DataSignals | null {
  const entry = SIGNAL_CACHE.get(tenantId);
  if (!entry) return null;
  if (Date.now() - entry.computedAt > TTL_MS) {
    SIGNAL_CACHE.delete(tenantId);
    return null;
  }
  return entry.signals;
}

export function setCachedSignals(tenantId: string, signals: DataSignals): void {
  SIGNAL_CACHE.set(tenantId, { signals, computedAt: Date.now() });
}

export function clearCachedSignals(tenantId: string): void {
  SIGNAL_CACHE.delete(tenantId);
}

export async function buildDataSignals(client: PoolClient): Promise<DataSignals> {
  // All 6 queries serialize on the same client (single PoolClient = single conn).
  // Promise.all is for clarity, not parallelism. Total cost: 6 × ~5-25ms.
  const [items, brands, sos, parts, vendors, customers, corp] = await Promise.all([
    client.query<{ c: string }>(`SELECT count(*)::text AS c FROM turion.items`),
    client.query<{ brand: string; c: string }>(
      `SELECT (source_data->>'brand') AS brand, count(*)::text AS c
         FROM turion.items
        WHERE source_data ? 'brand'
        GROUP BY (source_data->>'brand')`
    ),
    client.query<{ c: string; t: string }>(
      `SELECT count(*)::text AS c, COALESCE(sum(total),0)::text AS t FROM turion.sales_orders`
    ),
    client.query<{ c: string }>(`SELECT count(*)::text AS c FROM turion.parts`),
    client.query<{ c: string }>(`SELECT count(*)::text AS c FROM turion.vendors`),
    client.query<{ c: string }>(`SELECT count(*)::text AS c FROM turion.customers`),
    client.query<{ has: boolean }>(
      `SELECT (corporate_data IS NOT NULL) AS has
         FROM public.tenants
        WHERE id = current_setting('app.tenant_id')::uuid`
    ),
  ]);

  const brand_breakdown: Record<string, number> = {};
  for (const r of brands.rows) brand_breakdown[r.brand] = parseInt(r.c, 10);

  return {
    item_count: parseInt(items.rows[0]?.c ?? '0', 10),
    brand_breakdown,
    so_count: parseInt(sos.rows[0]?.c ?? '0', 10),
    so_total_usd: parseFloat(sos.rows[0]?.t ?? '0'),
    bom_part_count: parseInt(parts.rows[0]?.c ?? '0', 10),
    vendor_count: parseInt(vendors.rows[0]?.c ?? '0', 10),
    customer_count: parseInt(customers.rows[0]?.c ?? '0', 10),
    has_corporate_facts: !!corp.rows[0]?.has,
  };
}
```

```typescript
// === Phase 65.2: GET /api/onboarding/profile ===
// Any authenticated tenant member can load — no requireRole gate.
// Returns { question_set, prefilled_answers, data_signals, has_corporate_facts, corporate_summary }.

r.get('/profile', async (req: Request, res: Response) => {
  const refresh = req.query.refresh === 'true';
  const tenantId = req.tenant!.id;
  try {
    // W5: in-memory cache lookup (synchronous, no DB hit)
    let signals: DataSignals | null = refresh ? null : getCachedSignals(tenantId);
    const result = await withTenantClient(req, async (client) => {
      const p = await client.query<{
        question_set_code: string;
        answers: any;
        source: string;
      }>(
        `SELECT question_set_code, answers, source
           FROM public.tenant_onboarding_profile`
      );
      if (signals === null) {
        signals = await buildDataSignals(client);
        setCachedSignals(tenantId, signals);
      }
      const t = await client.query<{ corporate_data: any }>(
        `SELECT corporate_data FROM public.tenants WHERE id = $1`,
        [tenantId]
      );
      return {
        profile: p.rows[0] ?? null,
        corporate_data: t.rows[0]?.corporate_data ?? null,
      };
    });
    const code = result.profile?.question_set_code ?? 'd2c-multibrand';
    const questionSet = (QUESTION_SETS as any)[code] ?? (QUESTION_SETS as any)['d2c-multibrand'];

    res.json({
      question_set_code: code,
      question_set: questionSet,
      prefilled_answers: result.profile?.answers ?? {},
      source: result.profile?.source ?? 'seeded',
      data_signals: signals,
      has_corporate_facts: !!corp,
      corporate_summary: corp

            headcount: corp.organization?.employees_fy25 ?? null,

});
```

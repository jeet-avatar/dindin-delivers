    <script src="/turion-config.js"></script>

    let modified = 0;

    for (const f of files) {
      if (SKIP.has(f)) continue;
      const p = path.join(dir, f);
      const html = readFileSync(p, 'utf8');

      // Skip if already injected
      if (html.includes('ERP auth helpers (Phase 38)')) continue;

      // Find insertion point: after </title> if present, else after <head>
      let newHtml;
      if (html.includes('</title>')) {
        newHtml = html.replace('</title>', `</title>\n${HELPER_BLOCK}`);
      } else if (html.includes('<head>')) {
        newHtml = html.replace('<head>', `<head>\n${HELPER_BLOCK}`);
      } else {
        console.error(`SKIP ${f}: no <head> or </title> found`);
        continue;
      }

      writeFileSync(p, newHtml);
      modified++;
    }

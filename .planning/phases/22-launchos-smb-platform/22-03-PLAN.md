**Part A: generate-video endpoint in campaigns.ts**

Add `POST /api/campaigns/:id/generate-video` endpoint to the existing campaigns router. Place it after the existing campaign CRUD endpoints.

```typescript
// POST /api/campaigns/:id/generate-video
router.post('/:id/generate-video', authenticateToken, async (req, res) => {
  try {
    const { id } = req.params;
    const userId = (req as any).user.id; // from JWT middleware
    const launchosUserId = (req as any).user.launchos_user_id; // set at SSO time

    // 1. Fetch campaign from DB
    const campaign = await prisma.campaign.findFirst({
      where: { id, userId },
    });
    if (!campaign) return res.status(404).json({ error: 'Campaign not found' });
    if (!campaign.content) return res.status(400).json({ error: 'Campaign has no content to render' });

    // 2. Check entitlement — do NOT proceed if quota exceeded
    const entitlementUrl = process.env.ENTITLEMENT_SERVICE_URL;
    const entitlementResp = await axios.post(
      `${entitlementUrl}/entitlements/check`,
      { launchos_user_id: launchosUserId, action: 'render_video', quantity: 1 },
      { headers: { 'x-launchos-token': req.headers['x-launchos-token'] }, validateStatus: () => true }
    );
    if (entitlementResp.status === 402) {
      return res.status(402).json({
        error: 'Video render quota exceeded',
        upgrade_url: entitlementResp.data.upgrade_url,
        tier: entitlementResp.data.tier,
      });
    }

    // 3. Enqueue render job — POST to video server
    const videoServerUrl = process.env.VIDEO_SERVER_URL;
    const videoResp = await axios.post(
      `${videoServerUrl}/video/render`,
      {
        campaign_id: id,
        script: campaign.content, // campaign copy text
        voiceover: true,
        composition: 'DollorDemo',
        launchos_user_id: launchosUserId,
        tier: (req as any).user.tier || 'starter',
        webhook: `${process.env.BRANDMONKZ_API_URL}/api/webhooks/video-complete`,
      },
      { headers: { 'x-video-server-key': process.env.VIDEO_SERVER_SECRET! } }
    );

    // 4. Store job_id on campaign record for polling
    await prisma.campaign.update({
      where: { id },
      data: {
        videoJobId: videoResp.data.job_id,
        videoStatus: 'rendering',
      },
    });

    return res.json({
      job_id: videoResp.data.job_id,
      status: 'queued',
      poll_url: `${videoServerUrl}/video/jobs/${videoResp.data.job_id}`,
    });
  } catch (err: any) {
    console.error('[generate-video]', err.message);
    return res.status(500).json({ error: 'Failed to enqueue video render' });
  }
});
```

**Part B: video-complete webhook in webhooks.ts**

IMPORTANT: The video server fires its webhook to `POST /api/webhooks/video-complete`. This handler MUST be in a router mounted at `/api/webhooks` in `index.ts` — NOT inside the campaigns router (which is mounted at `/api/campaigns`). Adding it to the campaigns router would make the actual URL `/api/campaigns/webhooks/video-complete`, causing the video server to get a 404.

In `src/routes/webhooks.ts` (create if it doesn't exist):
```typescript
import { Router } from 'express';
import { prisma } from '../prisma';

const router = Router();

// POST /api/webhooks/video-complete
// Called by the video server when a render job completes or fails.
// Auth: x-video-server-key header (shared secret — same value as VIDEO_SERVER_SECRET)
router.post('/video-complete', async (req, res) => {
  const key = req.headers['x-video-server-key'];
  if (key !== process.env.VIDEO_SERVER_SECRET) {
    return res.status(401).json({ error: 'Invalid webhook key' });
  }
  const { job_id, status, output_url, error } = req.body;
  if (!job_id) return res.status(400).json({ error: 'job_id required' });

  // Find campaign by videoJobId
  const campaign = await prisma.campaign.findFirst({ where: { videoJobId: job_id } });
  if (!campaign) {
    // Return 200 to avoid webhook retries — campaign may have been deleted
    return res.json({ received: true, campaign_found: false });
  }

  await prisma.campaign.update({
    where: { id: campaign.id },
    data: {
      videoStatus: status === 'done' ? 'ready' : 'failed',
      videoUrl: output_url || null,
    },
  });

  return res.json({ updated: true });
});

export default router;
```

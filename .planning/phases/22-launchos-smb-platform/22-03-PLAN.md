**Endpoint implementation:**

```typescript
// POST /api/campaigns/:id/generate-video
router.post('/:id/generate-video', authenticateToken, async (req, res) => {
  try {
    const { id } = req.params;
    const userId = (req as any).user.id; // from JWT middleware
    const launchosUserId = (req as any).user.launchos_user_id; // set at SSO time

    // 1. Fetch campaign from DB
    const campaign = await prisma.campaign.findFirst({
      where: { id, userId },
    });
    if (!campaign) return res.status(404).json({ error: 'Campaign not found' });
    if (!campaign.content) return res.status(400).json({ error: 'Campaign has no content to render' });

    // 2. Check entitlement — do NOT proceed if quota exceeded
    const entitlementUrl = process.env.ENTITLEMENT_SERVICE_URL;
    const entitlementResp = await axios.post(
      `${entitlementUrl}/entitlements/check`,
      { launchos_user_id: launchosUserId, action: 'render_video', quantity: 1 },
      { headers: { 'x-launchos-token': req.headers['x-launchos-token'] }, validateStatus: () => true }
    );
    if (entitlementResp.status === 402) {
      return res.status(402).json({
        error: 'Video render quota exceeded',
        upgrade_url: entitlementResp.data.upgrade_url,
        tier: entitlementResp.data.tier,
      });
    }

    // 3. Enqueue render job — POST to video server
    const videoServerUrl = process.env.VIDEO_SERVER_URL;
    const videoResp = await axios.post(
      `${videoServerUrl}/video/render`,
      {
        campaign_id: id,
        script: campaign.content, // campaign copy text
        voiceover: true,
        composition: 'DollorDemo',
        launchos_user_id: launchosUserId,
        tier: (req as any).user.tier || 'starter',
        webhook: `${process.env.BRANDMONKZ_API_URL}/api/webhooks/video-complete`,
      },
      { headers: { 'x-video-server-key': process.env.VIDEO_SERVER_SECRET! } }
    );

    // 4. Store job_id on campaign record for polling
    await prisma.campaign.update({
      where: { id },
      data: {
        // Use Prisma JSON field or existing metadata column — check schema first
        // If no metadata column: add video_job_id to campaignMetadata JSON or use a new column
        // SAFE: store in existing `metadata` JSON field if present, else add `videoJobId String?`
        videoJobId: videoResp.data.job_id,
        videoStatus: 'rendering',
      },
    });

    return res.json({
      job_id: videoResp.data.job_id,
      status: 'queued',
      poll_url: `${videoServerUrl}/video/jobs/${videoResp.data.job_id}`,
    });
  } catch (err: any) {
    console.error('[generate-video]', err.message);
    return res.status(500).json({ error: 'Failed to enqueue video render' });
  }
});
```

**New file: `apps/zoom/server/services/meetingSummary.ts`**

```typescript
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY! });

interface TranscriptMessage {
  speaker: string;
  text: string;
  timestamp: string;
}

export async function summarizeMeeting(transcript: TranscriptMessage[]): Promise<string> {
  if (transcript.length === 0) return '';
  
  const transcriptText = transcript
    .map(m => `[${m.speaker}] ${m.text}`)
    .join('\n');
  
  const message = await client.messages.create({
    model: 'claude-sonnet-4-5',
    max_tokens: 1024,
    messages: [{
      role: 'user',
      content: `You are a CRM assistant. Summarize this business meeting transcript into a concise CRM note. Include: key topics discussed, action items, next steps, and any decisions made. Be specific and professional.\n\nTranscript:\n${transcriptText}`,
    }],
  });
  
  const content = message.content[0];
  return content.type === 'text' ? content.text : '';
}
```

Add `PATCH /api/deals/:id/meeting-notes` endpoint to the deals router.

**Auth pattern decision:** Use `jsonwebtoken.verify()` consistent with 22-01's `requireLaunchOSToken` pattern. The Zietra Meet server signs a short-lived JWT (Task 1 above) using the same `LAUNCHOS_JWT_SECRET`. This endpoint verifies that JWT — same approach as the entitlement service middleware.

```typescript
// PATCH /api/deals/:id/meeting-notes
// Called by Zietra Meet server after meeting ends
// Auth: x-launchos-token — a short-lived JWT signed with LAUNCHOS_JWT_SECRET
router.patch('/:id/meeting-notes', async (req, res) => {
  const { id } = req.params;
  const { notes, meeting_ended_at } = req.body;
  
  if (!notes) return res.status(400).json({ error: 'notes required' });
  
  // Authenticate via LaunchOS JWT (signed by calling service with LAUNCHOS_JWT_SECRET)
  const token = req.headers['x-launchos-token'] as string;
  if (!token) return res.status(401).json({ error: 'Missing x-launchos-token' });
  try {
    // Verify the JWT — rejects expired tokens and tokens with wrong secret
    jwt.verify(token, process.env.LAUNCHOS_JWT_SECRET!);
  } catch {
    return res.status(401).json({ error: 'Invalid or expired service token' });
  }
  
  const deal = await prisma.deal.findUnique({ where: { id } });
  if (!deal) return res.status(404).json({ error: 'Deal not found' });
  
  // Append to existing notes if any — don't overwrite
  const meetingSummaryBlock = `\n\n--- Meeting Summary (${meeting_ended_at || new Date().toISOString()}) ---\n${notes}`;
  const updatedNotes = (deal.notes || '') + meetingSummaryBlock;
  
  await prisma.deal.update({
    where: { id },
    data: { 
      notes: updatedNotes,
      lastMeetingAt: meeting_ended_at ? new Date(meeting_ended_at) : new Date(),
    },
  });
  
  return res.json({ updated: true, deal_id: id });
});
```

**Step 3: Create `.github/workflows/deploy-zietra-meet.yml`:**
```yaml
name: Deploy Zietra Meet

on:
  workflow_dispatch:
  push:
    paths:
      - 'apps/zoom/**'
    branches:
      - main

env:
  AWS_REGION: us-east-1
  ECR_REPOSITORY: zietra-meet
  ECS_SERVICE: zietra-meet-service
  ECS_CLUSTER: dollor-production
  CONTAINER_NAME: zietra-meet

jobs:
  deploy:
    name: Build and Deploy
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Login to Amazon ECR
        id: login-ecr
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build, tag, and push image to ECR
        id: build-image
        env:
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          IMAGE_TAG: ${{ github.sha }}
        run: |
          docker build \
            -f apps/zoom/deploy/Dockerfile \
            -t $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG \
            -t $ECR_REGISTRY/$ECR_REPOSITORY:latest \
            apps/zoom/
          docker push $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG
          docker push $ECR_REGISTRY/$ECR_REPOSITORY:latest
          echo "image=$ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG" >> $GITHUB_OUTPUT

      - name: Download task definition
        run: |
          aws ecs describe-task-definition \
            --task-definition zietra-meet \
            --query taskDefinition > task-definition.json

      - name: Update ECS task definition with new image
        id: task-def
        uses: aws-actions/amazon-ecs-render-task-definition@v1
        with:
          task-definition: task-definition.json
          container-name: ${{ env.CONTAINER_NAME }}
          image: ${{ steps.build-image.outputs.image }}

      - name: Deploy to ECS
        uses: aws-actions/amazon-ecs-deploy-task-definition@v1
        with:
          task-definition: ${{ steps.task-def.outputs.task-definition }}
          service: ${{ env.ECS_SERVICE }}
          cluster: ${{ env.ECS_CLUSTER }}
          wait-for-service-stability: true
```

import multer from 'multer';
import { parse } from 'csv-parse/sync'; // add "csv-parse": "^5.5.6" to package.json
import { prisma } from '../prisma'; // shared singleton — NEVER new PrismaClient()
import { authenticateToken } from '../middleware/auth';

const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: 10 * 1024 * 1024 } }); // 10MB max

// ActiveCampaign CSV field → BrandMonkz contact field mapping
const AC_FIELD_MAP: Record<string, string> = {
  'First Name': 'firstName',
  'Last Name': 'lastName',
  'Email': 'email',
  'Phone': 'phone',
  'Tags': 'tags',
  'Company': 'company',
  'Title': 'jobTitle',
  'City': 'city',
  'State': 'state',
  'Country': 'country',
};

router.post('/active-campaign', authenticateToken, upload.single('csv'), async (req, res) => {
  if (!req.file) return res.status(400).json({ error: 'CSV file required' });
  
  const userId = (req as any).user.id;
  
  // Parse CSV from buffer
  let rows: Record<string, string>[];
  try {
    rows = parse(req.file.buffer.toString('utf-8'), {
      columns: true,      // first row as headers
      skip_empty_lines: true,
      trim: true,
    });
  } catch (err: any) {
    return res.status(400).json({ error: 'Invalid CSV format: ' + err.message });
  }
  
  if (rows.length === 0) return res.status(400).json({ error: 'CSV is empty' });
  
  // Map AC fields to BrandMonkz fields
  const contacts = rows.map(row => {
    const contact: any = { userId };
    for (const [acField, bmField] of Object.entries(AC_FIELD_MAP)) {
      if (row[acField] !== undefined) {
        contact[bmField] = row[acField];
      }
    }
    // Derive full name if firstName + lastName present
    if (contact.firstName && contact.lastName) {
      contact.name = `${contact.firstName} ${contact.lastName}`;
    }
    return contact;
  }).filter(c => c.email); // must have email — skip rows without
  
  if (contacts.length === 0) {
    return res.status(400).json({ error: 'No valid contacts found (all rows missing email)' });
  }
  
  // Get existing emails for this user to detect duplicates
  const existingEmails = new Set(
    (await prisma.contact.findMany({
      where: { userId, email: { in: contacts.map(c => c.email) } },
      select: { email: true },
    })).map(c => c.email)
  );
  
  const newContacts = contacts.filter(c => !existingEmails.has(c.email));
  const duplicateCount = contacts.length - newContacts.length;
  
  // Bulk insert new contacts
  let importedCount = 0;
  let failedCount = 0;
  
  if (newContacts.length > 0) {
    try {
      const result = await prisma.contact.createMany({
        data: newContacts,
        skipDuplicates: true, // safety net
      });
      importedCount = result.count;
    } catch (err: any) {
      // If createMany fails (schema mismatch), fall back to individual creates
      for (const contact of newContacts) {
        try {
          await prisma.contact.create({ data: contact });
          importedCount++;
        } catch {
          failedCount++;
        }
      }
    }
  }
  
  return res.json({
    total: contacts.length,
    imported: importedCount,
    duplicates_skipped: duplicateCount,
    failed: failedCount,
  });
});

// GET /api/importer/preview — parse CSV and return first 5 rows without importing
router.post('/active-campaign/preview', authenticateToken, upload.single('csv'), async (req, res) => {
  if (!req.file) return res.status(400).json({ error: 'CSV file required' });
  
  let rows: Record<string, string>[];
  try {
    rows = parse(req.file.buffer.toString('utf-8'), { columns: true, skip_empty_lines: true, trim: true });
  } catch (err: any) {
    return res.status(400).json({ error: 'Invalid CSV: ' + err.message });
  }
  
  const preview = rows.slice(0, 5).map(row => {
    const mapped: any = {};
    for (const [acField, bmField] of Object.entries(AC_FIELD_MAP)) {
      if (row[acField]) mapped[bmField] = row[acField];
    }
    return mapped;
  });
  
  return res.json({ total_rows: rows.length, preview, detected_fields: Object.keys(rows[0] || {}) });
});

```tsx
import { useState, useRef } from 'react';

type Step = 'upload' | 'preview' | 'importing' | 'done';

interface PreviewData {
  total_rows: number;
  preview: Array<Record<string, string>>;
  detected_fields: string[];
}

interface ImportResult {
  total: number;
  imported: number;
  duplicates_skipped: number;
  failed: number;
}

export default function ActiveCampaignImporter() {
  const [step, setStep] = useState<Step>('upload');
  const [file, setFile] = useState<File | null>(null);
  const [preview, setPreview] = useState<PreviewData | null>(null);
  const [result, setResult] = useState<ImportResult | null>(null);
  const [error, setError] = useState('');
  const fileRef = useRef<HTMLInputElement>(null);

  const getToken = () => localStorage.getItem('crmToken');

  const handleFileChange = (f: File) => {
    setFile(f);
    setError('');
  };

  const handlePreview = async () => {
    if (!file) return;
    const form = new FormData();
    form.append('csv', file);
    const res = await fetch('/api/importer/active-campaign/preview', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${getToken()}` },
      body: form,
    });
    const data = await res.json();
    if (!res.ok) { setError(data.error); return; }
    setPreview(data);
    setStep('preview');
  };

  const handleImport = async () => {
    if (!file) return;
    setStep('importing');
    const form = new FormData();
    form.append('csv', file);
    const res = await fetch('/api/importer/active-campaign', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${getToken()}` },
      body: form,
    });
    const data = await res.json();
    if (!res.ok) { setError(data.error); setStep('preview'); return; }
    setResult(data);
    setStep('done');
  };

  return (
    <div className="max-w-3xl mx-auto p-8">
      <div className="mb-8">
        <h1 className="text-2xl font-bold">Import from ActiveCampaign</h1>
        <p className="text-gray-500 mt-1">
          Export your contacts from ActiveCampaign as CSV, then upload here. We'll map the fields automatically.
        </p>
        <div className="mt-3 bg-blue-50 border border-blue-200 rounded-lg p-3 text-sm text-blue-700">
          <strong>How to export from ActiveCampaign:</strong> Contacts → All Contacts → Export → CSV format → Download
        </div>
      </div>

      {step === 'upload' && (
        <div>
          <div
            className="border-2 border-dashed border-gray-300 rounded-xl p-12 text-center cursor-pointer hover:border-blue-400 transition-colors"
            onClick={() => fileRef.current?.click()}
            onDragOver={e => e.preventDefault()}
            onDrop={e => { e.preventDefault(); const f = e.dataTransfer.files[0]; if (f) handleFileChange(f); }}
          >
            <div className="text-4xl mb-4">📁</div>
            <p className="text-gray-600">{file ? file.name : 'Drop your ActiveCampaign CSV here, or click to browse'}</p>
            <p className="text-sm text-gray-400 mt-1">Max 10MB • CSV format only</p>
          </div>
          <input ref={fileRef} type="file" accept=".csv" className="hidden"
            onChange={e => { const f = e.target.files?.[0]; if (f) handleFileChange(f); }} />
          
          {error && <p className="text-red-500 mt-2 text-sm">{error}</p>}
          
          <button
            onClick={handlePreview}
            disabled={!file}
            className="mt-4 w-full bg-blue-600 text-white py-3 rounded-lg font-semibold disabled:opacity-50"
          >
            Preview Import →
          </button>
        </div>
      )}

      {step === 'preview' && preview && (
        <div>
          <div className="bg-green-50 border border-green-200 rounded-lg p-4 mb-6">
            <p className="font-semibold text-green-800">
              Found {preview.total_rows} contacts ready to import
            </p>
            <p className="text-sm text-green-600 mt-1">
              Detected fields: {preview.detected_fields.join(', ')}
            </p>
          </div>

          <h3 className="font-semibold mb-3">Preview (first 5 contacts)</h3>
          <div className="overflow-x-auto rounded-lg border">
            <table className="w-full text-sm">
              <thead className="bg-gray-50">
                <tr>
                  {Object.keys(preview.preview[0] || {}).map(k => (
                    <th key={k} className="px-3 py-2 text-left text-gray-500">{k}</th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {preview.preview.map((row, i) => (
                  <tr key={i} className="border-t">
                    {Object.values(row).map((v, j) => (
                      <td key={j} className="px-3 py-2 text-gray-700 max-w-xs truncate">{v}</td>
                    ))}
                  </tr>
                ))}
              </tbody>
            </table>
          </div>

          <div className="flex gap-3 mt-6">
            <button onClick={() => setStep('upload')} className="px-4 py-2 text-gray-600 border rounded-lg">
              Back
            </button>
            <button
              onClick={handleImport}
              className="flex-1 bg-blue-600 text-white py-3 rounded-lg font-semibold"
            >
              Import {preview.total_rows} Contacts
            </button>
          </div>
        </div>
      )}

      {step === 'importing' && (
        <div className="text-center py-16">
          <div className="text-4xl mb-4 animate-bounce">⬆️</div>
          <p className="text-gray-600 text-lg">Importing your contacts...</p>
        </div>
      )}

      {step === 'done' && result && (
        <div className="text-center">
          <div className="text-5xl mb-4">✅</div>
          <h2 className="text-2xl font-bold mb-4">Import Complete!</h2>
          <div className="grid grid-cols-3 gap-4 mb-8">
            <div className="bg-green-50 rounded-xl p-4">
              <div className="text-3xl font-bold text-green-600">{result.imported}</div>
              <div className="text-sm text-gray-500">Imported</div>
            </div>
            <div className="bg-yellow-50 rounded-xl p-4">
              <div className="text-3xl font-bold text-yellow-600">{result.duplicates_skipped}</div>
              <div className="text-sm text-gray-500">Duplicates skipped</div>
            </div>
            <div className="bg-red-50 rounded-xl p-4">
              <div className="text-3xl font-bold text-red-600">{result.failed}</div>
              <div className="text-sm text-gray-500">Failed</div>
            </div>
          </div>
          <a href="/contacts" className="bg-blue-600 text-white px-8 py-3 rounded-lg font-semibold">
            View Your Contacts →
          </a>
        </div>
      )}
    </div>
  );
}
```

**Endpoint: `POST /api/strategy-bot/generate`**

Auth: `authenticateToken` middleware (standard BrandMonkz JWT).

Body: `{ answers: Record<string, string> }` — keyed by question number `"q1"` through `"q10"`.

```typescript
import Anthropic from '@anthropic-ai/sdk';
import { prisma } from '../prisma'; // shared singleton — NEVER new PrismaClient()

const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY! });

router.post('/generate', authenticateToken, async (req, res) => {
  const { answers } = req.body;
  if (!answers || Object.keys(answers).length < 10) {
    return res.status(400).json({ error: 'All 10 answers required' });
  }

  const prompt = buildGTMPrompt(answers);
  
  const message = await anthropic.messages.create({
    model: 'claude-sonnet-4-5',
    max_tokens: 4096,
    messages: [{
      role: 'user',
      content: prompt,
    }],
  });
  
  const responseText = message.content[0].type === 'text' ? message.content[0].text : '';
  
  // Parse structured JSON from Claude response
  let gtmPlan: GTMPlan;
  try {
    const jsonMatch = responseText.match(/```json\n([\s\S]+?)\n```/);
    gtmPlan = jsonMatch ? JSON.parse(jsonMatch[1]) : JSON.parse(responseText);
  } catch {
    return res.status(500).json({ error: 'Failed to parse GTM plan from AI response' });
  }
  
  // Store plan on user record (use existing user metadata or a new UserGTMPlan table)
  // For v1: store as JSON in user's metadata field
  await prisma.user.update({
    where: { id: (req as any).user.id },
    data: { gtmPlan: JSON.stringify(gtmPlan) },
  });
  
  return res.json({ plan: gtmPlan, generated_at: new Date().toISOString() });
});
```

```tsx
import { useState } from 'react';
import { useMutation } from '@tanstack/react-query'; // already in BrandMonkz frontend

const QUESTIONS = [
  { id: 'q1', label: 'What industry or sector is your business in?', placeholder: 'e.g. SaaS, e-commerce, professional services...' },
  { id: 'q2', label: 'Who is your target customer? Be specific.', placeholder: 'e.g. Marketing managers at B2B SaaS companies with 50-200 employees' },
  { id: 'q3', label: "What's your biggest marketing challenge right now?", placeholder: 'e.g. Not enough leads, low email open rates, no time to create content...' },
  { id: 'q4', label: 'What tools do you currently use for marketing?', placeholder: 'e.g. ActiveCampaign, HeyGen, Buffer, Zoom...' },
  { id: 'q5', label: 'What is your monthly marketing budget?', placeholder: 'e.g. $500/mo, $2,000/mo...' },
  { id: 'q6', label: 'Primary goal: more leads, more sales, or more brand awareness?', placeholder: 'leads / sales / awareness' },
  { id: 'q7', label: 'How would you describe your content style?', placeholder: 'professional, casual, educational, entertaining...' },
  { id: 'q8', label: 'What geographic markets do you focus on?', placeholder: 'e.g. USA, UK, global B2B...' },
  { id: 'q9', label: 'Who are your top 1–3 competitors?', placeholder: 'e.g. HubSpot, ActiveCampaign, Mailchimp' },
  { id: 'q10', label: 'What is your timeline to first visible results?', placeholder: 'e.g. 30 days, 90 days...' },
];

export default function StrategyBotPage() {
  const [currentStep, setCurrentStep] = useState(0);
  const [answers, setAnswers] = useState<Record<string, string>>({});
  const [gtmPlan, setGtmPlan] = useState<any>(null);

  const generateMutation = useMutation({
    mutationFn: async (answers: Record<string, string>) => {
      const token = localStorage.getItem('crmToken');
      const res = await fetch('/api/strategy-bot/generate', {
        method: 'POST',
        headers: { 
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`,
        },
        body: JSON.stringify({ answers }),
      });
      if (!res.ok) throw new Error('Failed to generate plan');
      return res.json();
    },
    onSuccess: (data) => setGtmPlan(data.plan),
  });

  const handleNext = () => {
    if (currentStep < QUESTIONS.length - 1) {
      setCurrentStep(s => s + 1);
    } else {
      // All answered — submit
      generateMutation.mutate(answers);
    }
  };

  // Render wizard or GTM plan display
  if (gtmPlan) {
    return <GTMPlanDisplay plan={gtmPlan} />;
  }

  const q = QUESTIONS[currentStep];
  return (
    <div className="max-w-2xl mx-auto p-8">
      <div className="mb-6">
        <div className="flex justify-between text-sm text-gray-500 mb-2">
          <span>Question {currentStep + 1} of {QUESTIONS.length}</span>
          <span>{Math.round(((currentStep + 1) / QUESTIONS.length) * 100)}% complete</span>
        </div>
        <div className="w-full bg-gray-200 rounded-full h-2">
          <div 
            className="bg-blue-600 h-2 rounded-full transition-all"
            style={{ width: `${((currentStep + 1) / QUESTIONS.length) * 100}%` }}
          />
        </div>
      </div>
      
      <h2 className="text-2xl font-semibold mb-4">{q.label}</h2>
      <textarea
        className="w-full border rounded-lg p-3 h-32 text-gray-800"
        placeholder={q.placeholder}
        value={answers[q.id] || ''}
        onChange={e => setAnswers(prev => ({ ...prev, [q.id]: e.target.value }))}
      />
      
      <div className="flex justify-between mt-4">
        {currentStep > 0 && (
          <button onClick={() => setCurrentStep(s => s - 1)} className="px-4 py-2 text-gray-600">
            Back
          </button>
        )}
        <button
          onClick={handleNext}
          disabled={!answers[q.id] || generateMutation.isPending}
          className="ml-auto bg-blue-600 text-white px-6 py-2 rounded-lg disabled:opacity-50"
        >
          {currentStep === QUESTIONS.length - 1 
            ? (generateMutation.isPending ? 'Building your GTM plan...' : 'Generate My GTM Plan')
            : 'Next'}
        </button>
      </div>
    </div>
  );
}

function GTMPlanDisplay({ plan }: { plan: any }) {
  return (
    <div className="max-w-4xl mx-auto p-8 space-y-8">
      <div className="bg-blue-50 rounded-xl p-6">
        <h2 className="text-2xl font-bold text-blue-900 mb-2">Your 90-Day GTM Plan</h2>
        <p className="text-blue-700">{plan.summary}</p>
      </div>
      
      {/* First week actions */}
      <section>
        <h3 className="text-xl font-semibold mb-3">Start This Week</h3>
        <ul className="space-y-2">
          {plan.first_week_actions?.map((action: string, i: number) => (
            <li key={i} className="flex items-start gap-2">
              <span className="text-blue-600 font-bold">{i + 1}.</span>
              <span>{action}</span>
            </li>
          ))}
        </ul>
      </section>
      
      {/* Video topics */}
      <section>
        <h3 className="text-xl font-semibold mb-3">Video Topics for Social.Network</h3>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
          {plan.video_topics?.map((topic: string, i: number) => (
            <div key={i} className="border rounded-lg p-3 flex justify-between items-center">
              <span className="text-sm">{topic}</span>
              <button className="text-blue-600 text-sm ml-2 whitespace-nowrap">
                Generate Video →
              </button>
            </div>
          ))}
        </div>
      </section>
      
      {/* SEO keywords */}
      <section>
        <h3 className="text-xl font-semibold mb-3">Target SEO Keywords</h3>
        <div className="flex flex-wrap gap-2">
          {plan.seo_keywords?.map((kw: string, i: number) => (
            <span key={i} className="bg-gray-100 px-3 py-1 rounded-full text-sm">{kw}</span>
          ))}
        </div>
      </section>
    </div>
  );
}
```

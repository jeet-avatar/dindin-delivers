Add `POST /api/webhooks/video-watch` endpoint.

**Authentication:** Verify `X-Webhook-Secret` header against `process.env.VIDEO_WEBHOOK_SECRET`. Return 401 if missing or wrong. This is a machine-to-machine webhook (no user JWT needed).

**Payload schema:**
```typescript
interface VideoWatchEvent {
  contact_email: string;
  video_id: string;       // job_id from render server
  percentage_watched: number;  // 0-100
  campaign_id: string;
  watched_at: string;     // ISO timestamp
}
```

**Lead scoring logic:**
- `percentage_watched >= 75` → hot lead: increment `lead_score` by 15 points, set `lead_status` to `'hot'` if not already
- `percentage_watched >= 50` → warm: increment `lead_score` by 5 points
- `percentage_watched >= 25` → increment `lead_score` by 2 points
- `percentage_watched < 25` → no update (avoid noise from accidental opens)

**Implementation:**
```typescript
router.post('/video-watch', async (req, res) => {
  // Auth check
  if (req.headers['x-webhook-secret'] !== process.env.VIDEO_WEBHOOK_SECRET) {
    return res.status(401).json({ error: 'Invalid webhook secret' });
  }

  const { contact_email, percentage_watched, campaign_id, video_id } = req.body;
  if (!contact_email || percentage_watched === undefined) {
    return res.status(400).json({ error: 'contact_email and percentage_watched required' });
  }

  // Find contact — use shared prisma singleton
  const contact = await prisma.contact.findFirst({
    where: { email: contact_email },
  });
  if (!contact) {
    // Unknown contact — still return 200 to avoid webhook retries
    return res.json({ received: true, contact_found: false });
  }

  // Score increment
  let scoreIncrement = 0;
  let newStatus: string | undefined;
  if (percentage_watched >= 75) { scoreIncrement = 15; newStatus = 'hot'; }
  else if (percentage_watched >= 50) { scoreIncrement = 5; }
  else if (percentage_watched >= 25) { scoreIncrement = 2; }

  if (scoreIncrement > 0) {
    await prisma.contact.update({
      where: { id: contact.id },
      data: {
        lead_score: { increment: scoreIncrement },
        ...(newStatus ? { lead_status: newStatus } : {}),
      },
    });
  }

  // Log video engagement on campaign if campaign_id provided
  // Omit if Campaign model doesn't have a videoEngagements relation yet — add TODO
  
  return res.json({ received: true, contact_found: true, score_added: scoreIncrement });
});
```

**CRITICAL:** Use `import { prisma } from '../prisma'` — the shared singleton. Never `new PrismaClient()`.

Create `POST /api/video-embed/watch-event` endpoint. This receives events from browser-side video players (fired via fetch from the embed HTML) and forwards them to BrandMonkz:

```javascript
const express = require('express');
const axios = require('axios');
const router = express.Router();

// POST /api/video-embed/watch-event
// Called by browser-embedded video players via fetch (no auth — public endpoint)
// Rate limit: 10 events per IP per video per session (prevent spam)
const eventCounts = new Map(); // simple in-memory dedup

router.post('/watch-event', async (req, res) => {
  const { contact_email, video_id, percentage_watched, campaign_id } = req.body;
  
  if (!contact_email || !video_id || percentage_watched === undefined) {
    return res.status(400).json({ error: 'contact_email, video_id, percentage_watched required' });
  }

  // Dedup: only forward milestone percentages (25, 50, 75, 100) to avoid spam
  const milestones = [25, 50, 75, 100];
  const nearestMilestone = milestones.find(m => 
    percentage_watched >= m && percentage_watched < m + 5
  );
  if (!nearestMilestone) {
    return res.json({ received: true, forwarded: false, reason: 'not a milestone' });
  }

  // Forward to BrandMonkz webhook
  const brandmonkzWebhookUrl = process.env.BRANDMONKZ_WEBHOOK_URL;
  const webhookSecret = process.env.VIDEO_WEBHOOK_SECRET;
  
  try {
    await axios.post(
      `${brandmonkzWebhookUrl}/api/webhooks/video-watch`,
      {
        contact_email,
        video_id,
        percentage_watched: nearestMilestone,
        campaign_id,
        watched_at: new Date().toISOString(),
      },
      {
        headers: {
          'X-Webhook-Secret': webhookSecret,
          'Content-Type': 'application/json',
        },
        timeout: 5000,
      }
    );
    return res.json({ received: true, forwarded: true, milestone: nearestMilestone });
  } catch (err) {
    // Don't expose BrandMonkz errors to browser
    console.error('[video-embed/watch-event] Forward failed:', err.message);
    return res.json({ received: true, forwarded: false, reason: 'upstream error' });
  }
});

module.exports = router;
```

Create a new router. `GET /api/launchos/auth-token`:

```typescript
import jwt from 'jsonwebtoken';
import axios from 'axios';
import { authenticateToken } from '../middleware/auth'; // existing BrandMonkz auth middleware
import { prisma } from '../prisma'; // shared singleton

router.get('/auth-token', authenticateToken, async (req, res) => {
  const user = (req as any).user;
  
  // Sign a short-lived service JWT to call the entitlement service.
  // The entitlement service's requireLaunchOSToken middleware uses jwt.verify()
  // so we MUST send a properly signed JWT — the literal string 'internal' will be rejected with 401.
  const serviceToken = jwt.sign(
    { service: 'brandmonkz-dashboard', launchos_user_id: user.id },
    process.env.LAUNCHOS_JWT_SECRET!,
    { expiresIn: '30s' }  // short-lived — only for this single usage lookup
  );
  
  // Fetch current usage from entitlement service
  let usage: any = null;
  try {
    const usageResp = await axios.get(
      `${process.env.ENTITLEMENT_SERVICE_URL}/entitlements/usage/${user.id}`,
      {
        headers: { 'x-launchos-token': serviceToken }, // signed JWT, not literal 'internal'
        timeout: 3000,
      }
    );
    usage = usageResp.data;
  } catch {
    // Graceful degradation — usage meters won't show if entitlement service is down
    usage = null;
  }
  
  // Issue 5-minute LaunchOS JWT for the user (cross-app SSO token)
  const launchosToken = jwt.sign(
    {
      launchos_user_id: user.id,
      email: user.email,
      tier: user.tier || 'starter',
      exp: Math.floor(Date.now() / 1000) + 300, // 5 minutes
    },
    process.env.LAUNCHOS_JWT_SECRET!
  );
  
  return res.json({ launchos_token: launchosToken, usage, tier: user.tier || 'starter' });
});
```

**IMPORTANT:** The `serviceToken` (signed with `LAUNCHOS_JWT_SECRET`) is what gets sent to the entitlement service in `x-launchos-token`. The literal string `'internal'` must NOT be used here — the entitlement service's `requireLaunchOSToken` middleware runs `jwt.verify()` and will return 401 for any non-JWT value.

Mount at `/api/launchos` in main app.

**Frontend: `ToolCard.tsx`**

```tsx
interface Tool {
  name: string;
  description: string;
  icon: string;
  url: string;          // external URL (VibingTicket, Zietra Meet) or internal path
  isExternal: boolean;
  usageKey?: string;    // key into usage object (e.g. "render_video", "send_email")
  usageLabel?: string;  // human label (e.g. "Videos")
  color: string;        // Tailwind color class
}

interface ToolCardProps {
  tool: Tool;
  usage: any;
  tier: string;
  launchosToken: string;
}

export function ToolCard({ tool, usage, tier, launchosToken }: ToolCardProps) {
  const handleOpen = () => {
    if (tool.isExternal) {
      // Redirect-based SSO: append launchos_token to URL
      window.open(`${tool.url}?launchos_token=${launchosToken}`, '_blank');
    } else {
      window.location.href = tool.url;
    }
  };

  const used = usage?.[tool.usageKey || '']?.used ?? 0;
  const limit = usage?.[tool.usageKey || '']?.limit ?? null;
  const pct = limit && limit !== Infinity ? Math.min((used / limit) * 100, 100) : null;

  return (
    <div className={`rounded-2xl border-2 p-6 hover:shadow-lg transition-shadow cursor-pointer ${tool.color}`}
         onClick={handleOpen}>
      <div className="flex items-start justify-between mb-4">
        <span className="text-3xl">{tool.icon}</span>
        <span className="text-xs bg-white/60 px-2 py-1 rounded-full capitalize">{tier}</span>
      </div>
      <h3 className="font-bold text-lg mb-1">{tool.name}</h3>
      <p className="text-sm opacity-70 mb-4">{tool.description}</p>
      
      {pct !== null && (
        <div>
          <div className="flex justify-between text-xs mb-1">
            <span>{tool.usageLabel}</span>
            <span>{used} / {limit}</span>
          </div>
          <div className="w-full bg-black/10 rounded-full h-1.5">
            <div className="bg-white h-1.5 rounded-full transition-all"
                 style={{ width: `${pct}%` }} />
          </div>
        </div>
      )}
    </div>
  );
}
```

```tsx
    queryKey: ['launchos-auth'],

      const token = localStorage.getItem('crmToken');

        headers: { 'Authorization': `Bearer ${token}` },

      <div className="mb-8">

            tool={tool}

function UsageStat({ label, used, limit }: { label: string; used?: number; limit?: number }) {
  if (used === undefined) return null;

  );
```

export async function generateVoiceover(text: string, outputDir: string): Promise<string> {

  const response = await axios.post(
    `https://api.elevenlabs.io/v1/text-to-speech/${VOICE_ID}?output_format=mp3_44100_128`,

    }
  );

**src/queue/jobQueue.ts** — In-memory async job queue (sufficient for v1; upgrade to Redis-backed BullMQ if needed later):
```typescript
import { EventEmitter } from 'events';

export type JobStatus = 'queued' | 'rendering' | 'done' | 'failed';
export interface Job {
  id: string;
  status: JobStatus;
  output_url?: string;
  error?: string;
  created_at: string;
}

const jobs = new Map<string, Job>();
const queue = new EventEmitter();

export function enqueueJob(id: string, task: () => Promise<string>, webhookUrl?: string): void {
  jobs.set(id, { id, status: 'queued', created_at: new Date().toISOString() });
  // Run async without blocking HTTP response
  setImmediate(async () => {
    jobs.get(id)!.status = 'rendering';
    try {
      const output_url = await task();
      jobs.get(id)!.status = 'done';
      jobs.get(id)!.output_url = output_url;
      if (webhookUrl) await notifyWebhook(webhookUrl, { job_id: id, status: 'done', output_url });
    } catch (err: any) {
      jobs.get(id)!.status = 'failed';
      jobs.get(id)!.error = err.message;
      if (webhookUrl) await notifyWebhook(webhookUrl, { job_id: id, status: 'failed', error: err.message });
    }
  });
}

export function getJobStatus(id: string): Job | undefined {
  return jobs.get(id);
}

async function notifyWebhook(url: string, payload: object) {
  try {
    const { default: axios } = await import('axios');
    await axios.post(url, payload, { timeout: 5000 });
  } catch { /* best-effort webhook — do not throw */ }
}
```

**src/services/remotionRenderer.ts** — Wraps `npx remotion render` via child_process. The Remotion project is at `apps/dollor-video/` (confirmed: `apps/dollor-video/src/Root.tsx` with composition `DollorDemo`):
```typescript
import { execFile } from 'child_process';
import { promisify } from 'util';
import path from 'path';
import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';
import fs from 'fs';

const execFileAsync = promisify(execFile);
const s3 = new S3Client({ region: process.env.AWS_REGION! });

export async function renderVideo(opts: {
  composition: string;    // e.g. "DollorDemo" or "Marketing60"
  audioPath?: string;     // local MP3 path from ElevenLabs (passed via --props or env)
  jobId: string;
}): Promise<string> {
  const outDir = `/tmp/renders/${opts.jobId}`;
  fs.mkdirSync(outDir, { recursive: true });
  const outFile = path.join(outDir, 'output.mp4');
  const remotionProject = process.env.REMOTION_PROJECT_PATH!;

  // Pass audio path via props JSON if provided
  const props = opts.audioPath ? JSON.stringify({ audioPath: opts.audioPath }) : '{}';
  
  await execFileAsync('npx', [
    'remotion', 'render',
    `${remotionProject}/src/Root.tsx`,
    opts.composition,
    outFile,
    '--props', props,
    '--log', 'error',
  ], { 
    timeout: 120000, // 120s timeout — required per pitfall #5 (render is slow)
    cwd: remotionProject,
  });

  // Upload to S3 suiteflow-demo bucket (confirmed in RESEARCH.md)
  const s3Key = `launchos-videos/${opts.jobId}/output.mp4`;
  await s3.send(new PutObjectCommand({
    Bucket: process.env.S3_VIDEO_BUCKET!,
    Key: s3Key,
    Body: fs.createReadStream(outFile),
    ContentType: 'video/mp4',
    ACL: 'public-read',
  }));

  // Cleanup temp files
  fs.rmSync(outDir, { recursive: true, force: true });
  
  return `https://${process.env.S3_VIDEO_BUCKET}.s3.amazonaws.com/${s3Key}`;
}
```

**src/routes/render.ts** — Two endpoints:

`POST /video/render` — Validates `X-Video-Server-Key` header. Body:
```typescript
{
  campaign_id: string;
  script: string;         // campaign copy for ElevenLabs
  voiceover: boolean;     // false = skip ElevenLabs
  composition: string;    // default: "DollorDemo"
  launchos_user_id: string;
  tier: 'starter' | 'growth' | 'scale';
  webhook?: string;       // URL to POST when done
}
```

Logic:
1. Generate UUID job_id
2. Check render entitlement: `POST ${ENTITLEMENT_SERVICE_URL}/entitlements/check` body `{ launchos_user_id, action: 'render_video', quantity: 1 }`. If 402 → return 402 to caller.
3. Call `enqueueJob(job_id, async () => { ... }, webhook)` where the async task:
   a. If `voiceover=true`:
      - **Check tts_character entitlement FIRST**: `POST ${ENTITLEMENT_SERVICE_URL}/entitlements/check` body `{ launchos_user_id, action: 'tts_character', quantity: script.length }`. If response status is 402, skip ElevenLabs and set `audioPath = undefined` (music-only fallback — same path as any other ElevenLabs error).
      - If entitlement check allows: call `generateVoiceover(script, '/tmp')`. Catch any ElevenLabs API errors → set `audioPath = undefined` (music-only fallback, no error thrown).
   b. Call `renderVideo({ composition, audioPath, jobId: job_id })`
   c. Call entitlement consume for render_video: `POST /entitlements/consume { launchos_user_id, action: 'render_video', quantity: 1 }`
   d. If voiceover was used (audioPath is set): Call entitlement consume for tts_character: `POST /entitlements/consume { launchos_user_id, action: 'tts_character', quantity: script.length }`
   e. Return output_url
4. Respond immediately: `201 { job_id, status: 'queued', poll_url: '/video/jobs/${job_id}' }`

The tts_character entitlement check is gate logic only — a 402 response means "skip TTS and use music-only," NOT "fail the entire render job."

`GET /video/jobs/:id` — Return job from in-memory map: `{ job_id, status, output_url?, error? }`.

**Step 3: Create `.github/workflows/deploy-launchos-video-server.yml`:**
```yaml
name: Deploy LaunchOS Video Server

on:
  workflow_dispatch:
  push:
    paths:
      - 'apps/launchos-video-server/**'
    branches:
      - main

env:
  AWS_REGION: us-east-1
  ECR_REPOSITORY: launchos-video-server
  ECS_SERVICE: launchos-video-server-service
  ECS_CLUSTER: dollor-production
  CONTAINER_NAME: launchos-video-server

jobs:
  deploy:
    name: Build and Deploy
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Login to Amazon ECR
        id: login-ecr
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build, tag, and push image to ECR
        id: build-image
        env:
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          IMAGE_TAG: ${{ github.sha }}
        run: |
          docker build \
            -f apps/launchos-video-server/Dockerfile \
            -t $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG \
            -t $ECR_REGISTRY/$ECR_REPOSITORY:latest \
            apps/launchos-video-server/
          docker push $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG
          docker push $ECR_REGISTRY/$ECR_REPOSITORY:latest
          echo "image=$ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG" >> $GITHUB_OUTPUT

      - name: Download task definition
        run: |
          aws ecs describe-task-definition \
            --task-definition launchos-video-server \
            --query taskDefinition > task-definition.json

      - name: Update ECS task definition with new image
        id: task-def
        uses: aws-actions/amazon-ecs-render-task-definition@v1
        with:
          task-definition: task-definition.json
          container-name: ${{ env.CONTAINER_NAME }}
          image: ${{ steps.build-image.outputs.image }}

      - name: Deploy to ECS
        uses: aws-actions/amazon-ecs-deploy-task-definition@v1
        with:
          task-definition: ${{ steps.task-def.outputs.task-definition }}
          service: ${{ env.ECS_SERVICE }}
          cluster: ${{ env.ECS_CLUSTER }}
          wait-for-service-stability: true
```

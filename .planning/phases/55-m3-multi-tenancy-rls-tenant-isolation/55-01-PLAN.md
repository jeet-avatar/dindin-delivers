-- === Add column + backfill + index, schema-loop pattern (idempotent) ===
DO $$
DECLARE
  r record;
BEGIN
  FOR r IN
    SELECT t.table_schema, t.table_name
    FROM information_schema.tables t
    LEFT JOIN information_schema.columns c
      ON c.table_schema = t.table_schema
     AND c.table_name = t.table_name
     AND c.column_name = 'tenant_id'

-- Locks NOT NULL + adds FK ON DELETE RESTRICT on every multi-tenant table.

DO $$
DECLARE
  r record;
  has_fk int;
  has_notnull int;
BEGIN
  FOR r IN
    SELECT c.table_schema, c.table_name, c.is_nullable
    FROM information_schema.columns c
    WHERE c.column_name = 'tenant_id'

      EXECUTE format('ALTER TABLE %I.%I ALTER COLUMN tenant_id SET NOT NULL',
                     r.table_schema, r.table_name);
      RAISE NOTICE 'NOT NULL locked on %.%', r.table_schema, r.table_name;
    END IF;

    -- 2. Add FK (idempotent — check existence first via pg_constraint)
    SELECT COUNT(*) INTO has_fk
    FROM pg_constraint con
    JOIN pg_namespace n ON n.oid = con.connamespace
    JOIN pg_class cl ON cl.oid = con.conrelid
    WHERE n.nspname = r.table_schema
      AND cl.relname = r.table_name
      AND con.contype = 'f'
      AND con.conname = r.table_name || '_tenant_id_fkey';

    IF has_fk = 0 THEN
      EXECUTE format($q$
        ALTER TABLE %I.%I
          ADD CONSTRAINT %I FOREIGN KEY (tenant_id)
          REFERENCES public.tenants(id) ON DELETE RESTRICT
      $q$, r.table_schema, r.table_name, r.table_name || '_tenant_id_fkey');
      RAISE NOTICE 'FK added on %.% (ON DELETE RESTRICT)', r.table_schema, r.table_name;
    END IF;

    -- 3. Index — should already exist from migration 025 or 027, but belt-and-suspenders
    EXECUTE format('CREATE INDEX IF NOT EXISTS %I ON %I.%I (tenant_id)',
                   r.table_name || '_tenant_id_idx', r.table_schema, r.table_name);
  END LOOP;
END $$;

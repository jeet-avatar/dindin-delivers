-- migrations/029_provision_app_and_bypass_roles.sql · Phase 55 Plan 02 (M3)
-- Apply with master role zietra_admin (only role with CREATE ROLE privilege).
-- Creates zietra_app (NO BYPASSRLS) + zietra_admin_bypass (BYPASSRLS) + GRANTs + default privileges.
-- Passwords are RANDOM at create-time; Task 2 reads them from pg_authid for Secrets Manager.

-- === 1. zietra_app role (LAMBDA day-to-day; subject to RLS) ===
DO $$
DECLARE
  app_pw text := encode(gen_random_bytes(24), 'base64');
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'zietra_app') THEN
    EXECUTE format('CREATE ROLE zietra_app LOGIN PASSWORD %L NOINHERIT', app_pw);
    -- Persist the password in a one-row temp table for Task 2 to read
    CREATE TABLE IF NOT EXISTS _zietra_role_passwords (rolname text PRIMARY KEY, password text);
    INSERT INTO _zietra_role_passwords VALUES ('zietra_app', app_pw)
      ON CONFLICT (rolname) DO UPDATE SET password = EXCLUDED.password;
    RAISE NOTICE 'zietra_app created';
  ELSE
    RAISE NOTICE 'zietra_app already exists — SKIP create';
  END IF;
END $$;

-- Grants (idempotent — GRANT is non-destructive if already granted)
GRANT USAGE ON SCHEMA public, crm, turion, turion_satellite TO zietra_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO zietra_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA crm TO zietra_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA turion TO zietra_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA turion_satellite TO zietra_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO zietra_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA crm TO zietra_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA turion TO zietra_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA turion_satellite TO zietra_app;

ALTER DEFAULT PRIVILEGES IN SCHEMA crm GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO zietra_app;

END $$;

GRANT USAGE ON SCHEMA public, crm, turion, turion_satellite TO zietra_admin_bypass;
GRANT ALL ON ALL TABLES IN SCHEMA public TO zietra_admin_bypass;
GRANT ALL ON ALL TABLES IN SCHEMA crm TO zietra_admin_bypass;
GRANT ALL ON ALL TABLES IN SCHEMA turion TO zietra_admin_bypass;
GRANT ALL ON ALL TABLES IN SCHEMA turion_satellite TO zietra_admin_bypass;
GRANT ALL ON ALL SEQUENCES IN SCHEMA public TO zietra_admin_bypass;
GRANT ALL ON ALL SEQUENCES IN SCHEMA crm TO zietra_admin_bypass;

```bash
#!/usr/bin/env bash
# scripts/provision-rls-secrets-and-iam.sh — Phase 55 Plan 02
# Idempotently creates 2 Secrets Manager secrets for the new Postgres roles + drops temp pw table.
set -euo pipefail

source /Users/jeet/doordash-p2p/.planning/phases/54.6-enterprise-hardening-starter-pack-vpc-rds-proxy-waf-guardduty-close-sg/vpc-migration.env

REGION=us-east-1
PROXY_HOST=$(aws rds describe-db-proxies --db-proxy-name zietra-aurora-proxy \
  --query 'DBProxies[0].Endpoint' --output text --region $REGION)

# Resolve master pw, open temporary access to read pw table
NEW_MASTER_PW=$(aws secretsmanager get-secret-value --secret-id "$NEW_MASTER_SECRET_ARN" \
  --query SecretString --output text --region $REGION | jq -r .password)
OPERATOR_IP=$(curl -s https://checkip.amazonaws.com/)/32
aws ec2 authorize-security-group-ingress --group-id $AURORA_NEW_SG \
  --protocol tcp --port 5432 --cidr $OPERATOR_IP --region $REGION 2>/dev/null || true
aws rds modify-db-instance --db-instance-identifier ${NEW_CLUSTER}-writer \
  --publicly-accessible --apply-immediately --region $REGION
aws rds wait db-instance-available --db-instance-identifier ${NEW_CLUSTER}-writer --region $REGION
trap 'aws rds modify-db-instance --db-instance-identifier ${NEW_CLUSTER}-writer --no-publicly-accessible --apply-immediately --region $REGION; aws ec2 revoke-security-group-ingress --group-id $AURORA_NEW_SG --protocol tcp --port 5432 --cidr $OPERATOR_IP --region $REGION 2>/dev/null || true' EXIT

APP_PW=$(PGPASSWORD="$NEW_MASTER_PW" psql -h "$NEW_ENDPOINT" -U zietra_admin -d zietra -At \
  -c "SELECT password FROM _zietra_role_passwords WHERE rolname='zietra_app';")
BYPASS_PW=$(PGPASSWORD="$NEW_MASTER_PW" psql -h "$NEW_ENDPOINT" -U zietra_admin -d zietra -At \
  -c "SELECT password FROM _zietra_role_passwords WHERE rolname='zietra_admin_bypass';")

if [ -z "$APP_PW" ] || [ -z "$BYPASS_PW" ]; then
  echo "FAIL: could not retrieve role passwords from _zietra_role_passwords"; exit 1
fi

# Build JSON payloads (shape matches Lambda secrets.ts parser)
APP_JSON=$(jq -nc --arg pw "$APP_PW" --arg host "$PROXY_HOST" \
  '{username:"zietra_app", password:$pw, engine:"postgres", host:$host, port:5432, dbname:"zietra"}')
BYPASS_JSON=$(jq -nc --arg pw "$BYPASS_PW" --arg host "$PROXY_HOST" \
  '{username:"zietra_admin_bypass", password:$pw, engine:"postgres", host:$host, port:5432, dbname:"zietra"}')

create_or_update_secret() {
  local name=$1 json=$2 desc=$3
  if aws secretsmanager describe-secret --secret-id "$name" --region $REGION >/dev/null 2>&1; then
    echo "[secrets] $name exists — putting new value"
    aws secretsmanager put-secret-value --secret-id "$name" --secret-string "$json" --region $REGION >/dev/null
  else
    aws secretsmanager create-secret --name "$name" --description "$desc" --secret-string "$json" \
      --tags Key=Project,Value=Zietra Key=Phase,Value=55 --region $REGION >/dev/null
    echo "[secrets] $name created"
  fi
}

create_or_update_secret "zietra-aurora/app-role" "$APP_JSON" \
  "Postgres zietra_app role — used by 4 Lambdas day-to-day (subject to RLS). Phase 55."
create_or_update_secret "zietra-aurora/admin-bypass-role" "$BYPASS_JSON" \
  "Postgres zietra_admin_bypass role (BYPASSRLS) — used ONLY by migration scripts. Phase 55."

# DROP temp password table from Postgres (no plaintext lingering)
PGPASSWORD="$NEW_MASTER_PW" psql -h "$NEW_ENDPOINT" -U zietra_admin -d zietra -v ON_ERROR_STOP=1 \
  -c "DROP TABLE IF EXISTS _zietra_role_passwords;"

echo "[secrets] both secrets present; _zietra_role_passwords dropped"
```

  r record;
BEGIN

    SELECT c.table_schema, c.table_name
    FROM information_schema.columns c
    WHERE c.column_name = 'tenant_id'
      AND c.table_schema IN ('public','crm','turion','turion_satellite')

  LOOP
    -- 1. ENABLE
    EXECUTE format('ALTER TABLE %I.%I ENABLE ROW LEVEL SECURITY', r.table_schema, r.table_name);
    -- 2. FORCE (defense-in-depth — even table owner is RLS'd)
    EXECUTE format('ALTER TABLE %I.%I FORCE ROW LEVEL SECURITY', r.table_schema, r.table_name);
    -- 3. Drop+create policy (idempotent)
    EXECUTE format('DROP POLICY IF EXISTS tenant_isolation ON %I.%I', r.table_schema, r.table_name);
    EXECUTE format($q$
      CREATE POLICY tenant_isolation ON %I.%I
        AS PERMISSIVE FOR ALL TO public
        USING (tenant_id = current_setting('app.tenant_id')::uuid)
        WITH CHECK (tenant_id = current_setting('app.tenant_id')::uuid)
    $q$, r.table_schema, r.table_name);
    RAISE NOTICE 'RLS+policy applied to %.%', r.table_schema, r.table_name;
  END LOOP;
END $$;

if [ "$CURRENT_CERT" = "$CERT_ARN" ] && [ "$CURRENT_ALIASES_SORTED" = "$DESIRED_ALIASES_SORTED" ]; then

  | .ViewerCertificate.Certificate = $cert

  | del(.ViewerCertificate.IAMCertificateId)

' /tmp/53-02-cf-body.json > /tmp/53-02-cf-body-new.json

jq -e '.Aliases.Items | length == .Aliases.Quantity' /tmp/53-02-cf-body-new.json >/dev/null

  --id "$DIST_ID" \
  --if-match "$ETAG" \
  --distribution-config file:///tmp/53-02-cf-body-new.json \

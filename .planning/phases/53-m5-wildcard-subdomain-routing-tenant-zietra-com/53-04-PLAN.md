```bash
#!/usr/bin/env bash
# Phase 53-04 — End-to-end smoke for wildcard subdomain routing.
# Autonomous: signs up a fresh test tenant, verifies the new tenant subdomain works
# end-to-end across CloudFront + CF Function + tenant middleware on both Lambdas,
# verifies non-regression of legacy aliases + reserved-slug aliases,
# then cleans up everything it created.
#
# Anchor-guard: NEVER deletes the M1 admin (jm@techcloudpro.com) or the Turion tenant.
# Re-runnable: every run uses a fresh random slug.
#
# Requirements: WildcardACMCert, CloudFrontWildcardAlias, TenantSubdomainExtractor,
#               BackendTenantContextMiddleware, TenantConfigEndpoint
set -euo pipefail

AWS=${AWS:-/opt/homebrew/bin/aws}
[ -x "$AWS" ] || AWS=$(command -v aws)

# === Anchor-guard constants ===
ANCHOR_EMAIL="jm@techcloudpro.com"
ANCHOR_TENANT_SLUG="turion"
ANCHOR_TENANT_ID="00000000-0000-0000-0000-000000000001"

USER_POOL_ID=$($AWS secretsmanager get-secret-value \
  --secret-id zietra/cognito-config-yP3J9B \
  --query SecretString --output text | jq -r .user_pool_id)
[ -n "$USER_POOL_ID" ] && [ "$USER_POOL_ID" != "null" ] \
  || { echo "FATAL: failed to load user_pool_id from Secrets Manager"; exit 1; }

ERP_API="https://lo254mvukl.execute-api.us-east-1.amazonaws.com"
SAT_API="https://rjydekliee.execute-api.us-east-1.amazonaws.com"

# === Fresh test tenant identity ===
TS=$(date +%s)
RAND=$RANDOM
TEST_SLUG="smoke53-${RAND}"
TEST_EMAIL="phase53-smoke-${TS}-${RAND}@zietra.com"
TEST_NAME="Phase 53 Smoke User"
TEST_ORG="Phase 53 Smoke Org"
TEST_HOST="${TEST_SLUG}.zietra.com"

echo "============================================================"
echo " Phase 53 smoke — test tenant: $TEST_SLUG ($TEST_HOST)"
echo " Test email: $TEST_EMAIL"
echo " ANCHOR_EMAIL guard: $ANCHOR_EMAIL"
echo "============================================================"

# === Anchor-guard ===
if [ "$TEST_EMAIL" = "$ANCHOR_EMAIL" ] || [ "$TEST_SLUG" = "$ANCHOR_TENANT_SLUG" ]; then
  echo "FATAL: anchor guard tripped — would delete protected resource"
  exit 1
fi

# === Cleanup trap ===
cleanup() {
  local rc=$?
  echo
  echo "=== Cleanup (rc=$rc) ==="
  # Anchor-guard re-check before any DELETE
  if [ "$TEST_EMAIL" = "$ANCHOR_EMAIL" ]; then
    echo "[cleanup] SKIPPED Cognito delete — $TEST_EMAIL is the anchor admin"
  else
    $AWS cognito-idp admin-delete-user \
      --user-pool-id "$USER_POOL_ID" \
      --username "$TEST_EMAIL" 2>/dev/null \
      && echo "[cleanup] Cognito user $TEST_EMAIL deleted" \
      || echo "[cleanup] Cognito user $TEST_EMAIL not found (already gone or never created)"
  fi
  if [ "$TEST_SLUG" = "$ANCHOR_TENANT_SLUG" ]; then
    echo "[cleanup] SKIPPED tenants DELETE — $TEST_SLUG is the anchor Turion tenant"
  else
    # DELETE FROM public.tenants WHERE slug=$1 — FK ON DELETE CASCADE removes tenant_features
    # Use admin/dedicated endpoint if exposed, OR direct psql if DATABASE_URL available locally.
    # If neither, log instructive failure (smoke does NOT fail on cleanup-fail).
    if [ -n "${DATABASE_URL:-}" ]; then
      psql "$DATABASE_URL" -v ON_ERROR_STOP=1 -c \
        "DELETE FROM public.tenants WHERE slug = '$TEST_SLUG';" \
        && echo "[cleanup] public.tenants row '$TEST_SLUG' deleted (CASCADE removed tenant_features)" \
        || echo "[cleanup] DB DELETE failed — clean orphan row manually"
    else
      echo "[cleanup] DATABASE_URL not set locally — orphan row '$TEST_SLUG' must be cleaned manually:"
      echo "         DELETE FROM public.tenants WHERE slug = '$TEST_SLUG';"
    fi
  fi
  exit $rc
}
trap cleanup EXIT

# Pre-flight: snapshot starting state
echo
echo "=== Pre-flight ==="
echo "[pre] $TEST_HOST DNS resolves?"
dig +short "$TEST_HOST" @8.8.8.8 | head -3   # informational only

# === Assertion 1 — POST /api/tenants/signup creates a new tenant ===
echo
echo "=== A1 — POST /api/tenants/signup creates tenant '$TEST_SLUG' ==="
SIGNUP_BODY=$(cat <<EOF
{"email":"$TEST_EMAIL","name":"$TEST_NAME","organization_name":"$TEST_ORG","slug":"$TEST_SLUG"}
EOF
)
SIGNUP_RESP=$(curl -sS -X POST "$ERP_API/api/tenants/signup" \
  -H 'Content-Type: application/json' \
  -d "$SIGNUP_BODY")
echo "Signup response: $SIGNUP_RESP"
TENANT_ID=$(echo "$SIGNUP_RESP" | jq -r '.tenant.id // empty')
[ -n "$TENANT_ID" ] || { echo "A1 FAIL: signup did not return tenant.id"; exit 1; }
echo "A1 PASS: tenant created, id=$TENANT_ID"

# Wait for: Lambda warm-container cache (60s TTL, fresh tenant won't be cached) +
#           CloudFront wildcard propagation (~30s once cert+alias are deployed) +
#           DNS recursors (5-10s typically — wildcard ALIAS resolves at any time)
echo
echo "=== Waiting 90s for Lambda+CF+DNS to settle ==="
sleep 90

# === Assertion 2 — TLS handshake on new <slug>.zietra.com works ===
echo
echo "=== A2 — TLS handshake on $TEST_HOST ==="
TLS_RC=$(curl -sk -o /dev/null -w '%{http_code}|%{ssl_verify_result}|%{http_version}\n' "https://$TEST_HOST/")
echo "  -> $TLS_RC"
case "$TLS_RC" in
  200*|301*|302*) echo "A2 PASS";;
  *) echo "A2 FAIL: $TEST_HOST returned $TLS_RC"; exit 1;;
esac

# === Assertion 3 — <slug>.zietra.com / returns 200 + serves index HTML ===
echo
echo "=== A3 — <slug>.zietra.com / serves Turion index.html ==="
ROOT=$(curl -sS -o /tmp/53-04-root.html -w '%{http_code}' "https://$TEST_HOST/")
test "$ROOT" = "200" || { echo "A3 FAIL: $TEST_HOST / returned $ROOT"; exit 1; }
grep -q "<!doctype html\|<!DOCTYPE html" /tmp/53-04-root.html \
  && echo "A3 PASS: HTML body received" \
  || { echo "A3 FAIL: body is not HTML"; exit 1; }

# === Assertion 4 — ERP /api/tenants/current with new slug returns matching tenant ===
echo
echo "=== A4 — ERP GET /api/tenants/current returns the new tenant ==="
ERP_RESP=$(curl -sS -H "X-Tenant-Slug: $TEST_SLUG" "$ERP_API/api/tenants/current")
echo "ERP resp: $ERP_RESP"
RESP_SLUG=$(echo "$ERP_RESP" | jq -r '.slug // empty')
RESP_NAME=$(echo "$ERP_RESP" | jq -r '.name // empty')
RESP_PLAN=$(echo "$ERP_RESP" | jq -r '.plan // empty')
[ "$RESP_SLUG" = "$TEST_SLUG" ] || { echo "A4 FAIL: slug mismatch (got $RESP_SLUG)"; exit 1; }
[ "$RESP_NAME" = "$TEST_ORG" ] || { echo "A4 FAIL: name mismatch (got $RESP_NAME)"; exit 1; }
[ "$RESP_PLAN" = "trial" ] || { echo "A4 FAIL: plan mismatch (got $RESP_PLAN)"; exit 1; }
echo "A4 PASS: slug=$RESP_SLUG name=$RESP_NAME plan=$RESP_PLAN"

# === Assertion 5 — Satellite /api/tenants/current also returns the new tenant ===
echo
echo "=== A5 — Satellite GET /api/tenants/current returns the same tenant ==="
SAT_RESP=$(curl -sS -H "X-Tenant-Slug: $TEST_SLUG" "$SAT_API/api/tenants/current")
echo "Sat resp: $SAT_RESP"
SAT_SLUG=$(echo "$SAT_RESP" | jq -r '.slug // empty')
[ "$SAT_SLUG" = "$TEST_SLUG" ] || { echo "A5 FAIL: satellite slug mismatch (got $SAT_SLUG)"; exit 1; }
echo "A5 PASS (satellite mirror works)"

# === Assertion 6 — Unknown slug → 404 on BOTH Lambdas ===
echo
echo "=== A6 — Unknown slug 404 on both Lambdas ==="
BOGUS="nonexistent-slug-${TS}"
ERP_CODE=$(curl -sS -o /dev/null -w '%{http_code}' -H "X-Tenant-Slug: $BOGUS" "$ERP_API/api/tenants/current")
SAT_CODE=$(curl -sS -o /dev/null -w '%{http_code}' -H "X-Tenant-Slug: $BOGUS" "$SAT_API/api/tenants/current")
echo "ERP: $ERP_CODE  Sat: $SAT_CODE"
[ "$ERP_CODE" = "404" ] && [ "$SAT_CODE" = "404" ] \
  || { echo "A6 FAIL"; exit 1; }
echo "A6 PASS"

# === Assertion 7 — turionspace.zietra.com still 200 (legacy alias) ===
echo
echo "=== A7 — turionspace.zietra.com still serves (legacy alias) ==="
LEGACY_CODE=$(curl -sS -o /dev/null -w '%{http_code}' https://turionspace.zietra.com/)
test "$LEGACY_CODE" = "200" || { echo "A7 FAIL: legacy alias broke (code $LEGACY_CODE)"; exit 1; }
echo "A7 PASS"

# === Assertion 8 — marquee.zietra.com NOT shadowed by wildcard ===
echo
echo "=== A8 — marquee.zietra.com NOT shadowed (more-specific match wins) ==="
MARQ_CODE=$(curl -sS -o /dev/null -w '%{http_code}' https://marquee.zietra.com/)
test "$MARQ_CODE" = "200" || { echo "A8 FAIL: marquee broke (code $MARQ_CODE)"; exit 1; }
# Body check: must contain marquee-specific content, NOT the Turion 404 inline HTML
curl -sS https://marquee.zietra.com/ | grep -qi "marquee\|anni\|larc" \
  && echo "A8 PASS: marquee content intact" \
  || { echo "A8 FAIL: marquee body looks wrong (might be wildcard CF 404)"; exit 1; }

# === Assertion 9 — asc606.zietra.com NOT shadowed ===
echo
echo "=== A9 — asc606.zietra.com NOT shadowed ==="
ASC_CODE=$(curl -sS -o /dev/null -w '%{http_code}' https://asc606.zietra.com/)
test "$ASC_CODE" = "200" || { echo "A9 FAIL: asc606 broke (code $ASC_CODE)"; exit 1; }
echo "A9 PASS"

# === Regression — Phase 52 signup contract intact ===
echo
echo "=== R1 — Phase 52 signup contract: empty body → 400 'Valid email required' ==="
RESP=$(curl -sS -X POST "$ERP_API/api/tenants/signup" \
  -H 'Content-Type: application/json' -d '{}')
echo "$RESP" | grep -q "Valid email required" \
  && echo "R1 PASS" \
  || { echo "R1 FAIL: signup contract regressed"; exit 1; }

# === Regression — Phase 41 requireAuth still gates protected route ===
echo
echo "=== R2 — Phase 41 /api/data/all unauth → 401 ==="
CODE=$(curl -sS -o /dev/null -w '%{http_code}' "$ERP_API/api/data/all")
test "$CODE" = "401" \
  && echo "R2 PASS" \
  || { echo "R2 FAIL: Phase 41 auth regressed (got $CODE)"; exit 1; }

# === Regression — Phase 38 /api/health on both Lambdas ===
echo
echo "=== R3 — Phase 38 /api/health on both Lambdas → 200 ==="
ERPH=$(curl -sS -o /dev/null -w '%{http_code}' "$ERP_API/api/health")
SATH=$(curl -sS -o /dev/null -w '%{http_code}' "$SAT_API/api/health")
test "$ERPH" = "200" && test "$SATH" = "200" \
  && echo "R3 PASS" \
  || { echo "R3 FAIL: ERP=$ERPH Sat=$SATH"; exit 1; }

echo
echo "============================================================"
echo " Phase 53 smoke — ALL 9 ASSERTIONS + 3 REGRESSIONS PASS"
echo " Test tenant: $TEST_SLUG ($TENANT_ID)"
echo " Will be cleaned up on EXIT (trap)"
echo "============================================================"
```

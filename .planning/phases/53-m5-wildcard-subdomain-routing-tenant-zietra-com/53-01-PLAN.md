```bash
#!/usr/bin/env bash
# Phase 53-01 — Provision wildcard ACM cert *.zietra.com + zietra.com SANs
# in us-east-1, DNS-validate via Route 53, and add wildcard A + AAAA aliases
# pointing at CloudFront E37R9PT8IL44L2.
#
# Idempotent: re-runs reuse an existing ISSUED cert with matching SANs.
# Safe: only UPSERTs records (never DELETE), never touches apex or other-tenant aliases.
#
# Requirements: WildcardACMCert
set -euo pipefail

# --- Constants (per research §"Live AWS State") ---
REGION=us-east-1
ZONE_ID=Z090201115UMJZ8TIAX5G
DOMAIN='*.zietra.com'
APEX_SAN='zietra.com'
DISTRO_ID=E37R9PT8IL44L2
DISTRO_HOSTNAME=d2bl7vqyf3n9m5.cloudfront.net
# CloudFront's hosted zone ID for ALIAS records is fixed (AWS-published constant)
CLOUDFRONT_ZONE_ID=Z2FDTNDATAQYW2

# Pin AWS CLI v2 (Phase 52-04 Rule-3 auto-fix)
AWS=${AWS:-/opt/homebrew/bin/aws}
if ! [ -x "$AWS" ]; then AWS=$(command -v aws); fi
[ -x "$AWS" ] || { echo "FATAL: aws CLI not found"; exit 1; }

echo "[1/5] Sanity check region + zone exist"
$AWS sts get-caller-identity --query Account --output text >/dev/null
$AWS route53 get-hosted-zone --id "$ZONE_ID" --query 'HostedZone.Name' --output text \
  | grep -q '^zietra.com.$' || { echo "FATAL: zone $ZONE_ID is not zietra.com"; exit 1; }

# --- Idempotency: look up existing ISSUED wildcard cert ---
echo "[2/5] Looking for existing ISSUED *.zietra.com cert in $REGION"
EXISTING_ARN=$($AWS acm list-certificates --region "$REGION" \
  --query "CertificateSummaryList[?DomainName=='$DOMAIN' && Status=='ISSUED'].CertificateArn | [0]" \
  --output text)
if [ -n "$EXISTING_ARN" ] && [ "$EXISTING_ARN" != "None" ]; then
  # Belt-and-suspenders: confirm SANs include the apex too (skip otherwise)
  SAN_COUNT=$($AWS acm describe-certificate --region "$REGION" --certificate-arn "$EXISTING_ARN" \
    --query "Certificate.SubjectAlternativeNames[?@=='$APEX_SAN'] | length(@)" --output text)
  if [ "$SAN_COUNT" = "1" ]; then
    echo "[2/5] Reusing existing ISSUED cert: $EXISTING_ARN"
    CERT_ARN="$EXISTING_ARN"
  else
    echo "[2/5] Existing cert lacks apex SAN — requesting fresh"
    EXISTING_ARN=""
  fi
fi

# --- Provision fresh cert if none reusable ---
if [ -z "${CERT_ARN:-}" ]; then
  IDEM=phase-53-wildcard-$(date +%s)
  CERT_ARN=$($AWS acm request-certificate --region "$REGION" \
    --domain-name "$DOMAIN" \
    --subject-alternative-names "$APEX_SAN" \
    --validation-method DNS \
    --idempotency-token "$IDEM" \
    --query CertificateArn --output text)
  echo "[3/5] Requested cert $CERT_ARN — waiting 10s for ACM to generate validation records"
  sleep 10

  # Read validation CNAME (wildcard + apex SHARE the same validation record per AWS docs)
  VAL_NAME=$($AWS acm describe-certificate --region "$REGION" --certificate-arn "$CERT_ARN" \
    --query 'Certificate.DomainValidationOptions[0].ResourceRecord.Name' --output text)
  VAL_VALUE=$($AWS acm describe-certificate --region "$REGION" --certificate-arn "$CERT_ARN" \
    --query 'Certificate.DomainValidationOptions[0].ResourceRecord.Value' --output text)
  [ -n "$VAL_NAME" ] && [ "$VAL_NAME" != "None" ] || { echo "FATAL: no validation record yet"; exit 1; }
  echo "[3/5] Validation CNAME: $VAL_NAME → $VAL_VALUE"

  # UPSERT validation CNAME in Route 53
  cat > /tmp/53-01-acm-val.json <<EOF
{"Changes":[{"Action":"UPSERT","ResourceRecordSet":{
  "Name":"$VAL_NAME","Type":"CNAME","TTL":300,
  "ResourceRecords":[{"Value":"$VAL_VALUE"}]}}]}
EOF
  CHANGE_ID=$($AWS route53 change-resource-record-sets --hosted-zone-id "$ZONE_ID" \
    --change-batch file:///tmp/53-01-acm-val.json \
    --query 'ChangeInfo.Id' --output text)
  echo "[3/5] Submitted validation CNAME change: $CHANGE_ID"

  # Poll cert status until ISSUED (aws wait polls every 5s, max 40 attempts = ~200s)
  echo "[4/5] Waiting for cert ISSUED (up to ~200s, typically 1-5 min)"
  $AWS acm wait certificate-validated --region "$REGION" --certificate-arn "$CERT_ARN"
  echo "[4/5] Cert ISSUED: $CERT_ARN"
fi

# --- UPSERT wildcard A + AAAA aliases ---
echo "[5/5] Upserting wildcard A + AAAA aliases → $DISTRO_HOSTNAME"
cat > /tmp/53-01-wildcard-aliases.json <<EOF
{"Changes":[
  {"Action":"UPSERT","ResourceRecordSet":{
    "Name":"*.zietra.com","Type":"A",
    "AliasTarget":{
      "HostedZoneId":"$CLOUDFRONT_ZONE_ID",
      "DNSName":"$DISTRO_HOSTNAME",
      "EvaluateTargetHealth":false
    }
  }},
  {"Action":"UPSERT","ResourceRecordSet":{
    "Name":"*.zietra.com","Type":"AAAA",
    "AliasTarget":{
      "HostedZoneId":"$CLOUDFRONT_ZONE_ID",
      "DNSName":"$DISTRO_HOSTNAME",
      "EvaluateTargetHealth":false
    }
  }}
]}
EOF
$AWS route53 change-resource-record-sets --hosted-zone-id "$ZONE_ID" \
  --change-batch file:///tmp/53-01-wildcard-aliases.json \
  --query 'ChangeInfo.{Id:Id,Status:Status}' --output json

echo
echo "=== Phase 53-01 DONE ==="
echo "WILDCARD_CERT_ARN=$CERT_ARN"
echo "Export this ARN for plan 53-02:"
echo "  echo 'WILDCARD_CERT_ARN=$CERT_ARN' > /tmp/53-01-output.env"
echo "$CERT_ARN" > /tmp/53-01-cert-arn.txt
```

```typescript
// backend/src/middleware/tenant.ts — Phase 53-03 (M5)
// Resolves the tenant from the X-Tenant-Slug request header.
// 60s in-memory cache per warm Lambda container; 5s negative cache for unknown slugs.
// Mounted PER-ROUTE alongside requireAuth (NOT app-wide — Pitfall 8).
//
// Mirror file: /Users/jeet/turion-satellite/backend/src/middleware/tenant.ts
import { Request, Response, NextFunction } from 'express';
import { pool } from '../db';

export interface TenantContext {
  id: string;
  slug: string;
  name: string;
  plan: 'trial' | 'paid' | 'disabled';
  trial_ends_at: string | null;
}

declare global {
  // eslint-disable-next-line @typescript-eslint/no-namespace
  namespace Express {
    interface Request {
      tenant?: TenantContext;
    }
  }
}

// Positive cache 60s; negative cache 5s (typo-loop defense).
const CACHE_TTL_MS = 60_000;
const NEG_CACHE_TTL_MS = 5_000;
type CacheEntry = { tenant: TenantContext | null; expiresAt: number };
const cache = new Map<string, CacheEntry>();

async function loadTenant(slug: string): Promise<TenantContext | null> {
  const now = Date.now();
  const hit = cache.get(slug);
  if (hit && hit.expiresAt > now) return hit.tenant;

  const r = await pool.query<TenantContext>(
    `SELECT id, slug, name, plan, trial_ends_at
       FROM public.tenants WHERE slug = $1`,
    [slug],
  );
  const tenant = r.rowCount ? r.rows[0] : null;
  cache.set(slug, {
    tenant,
    expiresAt: now + (tenant ? CACHE_TTL_MS : NEG_CACHE_TTL_MS),
  });
  return tenant;
}

// Hard-required: 400 if header missing, 404 if slug unknown, 500 on DB error.
// Per-route mount: `router.get('/path', tenantContext, requireAuth, handler)` style.
export async function tenantContext(
  req: Request, res: Response, next: NextFunction,
): Promise<void> {
  // Express normalizes header names to lowercase (Pitfall 10).
  const raw = req.headers['x-tenant-slug'];
  const slug = typeof raw === 'string' ? raw.trim().toLowerCase() : '';
  if (!slug) {
    res.status(400).json({ error: 'Missing X-Tenant-Slug header' });
    return;
  }
  // Defensive: slugs are [a-z0-9-]{3,32} per Phase 52 contract.
  // Avoid SQL load on obviously-malformed values.
  if (!/^[a-z0-9-]{3,32}$/.test(slug)) {
    res.status(404).json({ error: 'Unknown tenant' });
    return;
  }
  try {
    const tenant = await loadTenant(slug);
    if (!tenant) {
      res.status(404).json({ error: 'Unknown tenant' });
      return;
    }
    req.tenant = tenant;
    next();
  } catch (e: any) {
    console.error('[tenantContext] DB lookup failed:', e?.message);
    res.status(500).json({ error: 'Tenant lookup failed' });
  }
}
```

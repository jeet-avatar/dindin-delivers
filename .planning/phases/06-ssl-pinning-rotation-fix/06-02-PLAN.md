**Step 2: Add two CloudWatch alarms to `infrastructure/terraform/modules/cloudwatch/main.tf`.**

Add a new section after the RDS alarms block (after the `rds_storage` alarm):

```hcl
# =============================================================================
# CLOUDWATCH ALARMS - ACM Certificate Expiry
# =============================================================================

resource "aws_cloudwatch_metric_alarm" "acm_expiry_warning" {
  count = var.acm_certificate_arn != "" ? 1 : 0

  alarm_name          = "dollor-${var.environment}-acm-cert-expiry-warning"
  comparison_operator = "LessThanThreshold"
  evaluation_periods  = 1
  metric_name         = "DaysToExpiry"
  namespace           = "AWS/CertificateManager"
  period              = 86400
  statistic           = "Minimum"
  threshold           = 30
  alarm_description   = "ACM certificate for dollor.ai expires in less than 30 days"
  alarm_actions       = [aws_sns_topic.alarms.arn]
  treat_missing_data  = "notBreaching"

  dimensions = {
    CertificateArn = var.acm_certificate_arn
  }

  tags = var.tags
}

resource "aws_cloudwatch_metric_alarm" "acm_expiry_critical" {
  count = var.acm_certificate_arn != "" ? 1 : 0

  alarm_name          = "dollor-${var.environment}-acm-cert-expiry-critical"
  comparison_operator = "LessThanThreshold"
  evaluation_periods  = 1
  metric_name         = "DaysToExpiry"
  namespace           = "AWS/CertificateManager"
  period              = 86400
  statistic           = "Minimum"
  threshold           = 7
  alarm_description   = "CRITICAL: ACM certificate for dollor.ai expires in less than 7 days"
  alarm_actions       = [aws_sns_topic.alarms.arn]
  ok_actions          = [aws_sns_topic.alarms.arn]
  treat_missing_data  = "notBreaching"

  dimensions = {
    CertificateArn = var.acm_certificate_arn
  }

  tags = var.tags
}
```

Key design decisions per research:
- `period = 86400` (1 day) because `DaysToExpiry` is only published twice daily
- `treat_missing_data = "notBreaching"` to avoid false alerts when metric is not published for far-out expiry dates
- `count` conditional so environments without `acm_certificate_arn` skip alarm creation
- Critical alarm has `ok_actions` to notify when the cert is renewed and the alarm clears
- Both alarms use the existing `aws_sns_topic.alarms` (sends to `production-alerts@dollor.ai` in production)

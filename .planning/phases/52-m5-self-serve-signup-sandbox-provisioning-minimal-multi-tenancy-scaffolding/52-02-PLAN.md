// backend/src/lib/ses-send.ts — Phase 52 plan 02
// Thin wrapper for non-Cognito SES emails. The Phase 52 welcome email reuses
// Phase 39's Create-Auth-Challenge Lambda (via AdminInitiateAuth CUSTOM_AUTH)
// instead of this helper, so this is kept narrow for Phase 53/54 callers.
import { SESClient, SendEmailCommand } from '@aws-sdk/client-ses';

const ses = new SESClient({ region: process.env.AWS_REGION ?? 'us-east-1' });

export async function sendEmail(opts: {
  to: string;
  from?: string;     // default 'noreply@zietra.com' (covered by zietra.com domain verification)
  subject: string;
  textBody: string;
  htmlBody?: string;
}): Promise<void> {
  await ses.send(new SendEmailCommand({
    Source: opts.from ?? 'noreply@zietra.com',
    Destination: { ToAddresses: [opts.to] },
    Message: {
      Subject: { Charset: 'UTF-8', Data: opts.subject },
      Body: {
        Text: { Charset: 'UTF-8', Data: opts.textBody },
        ...(opts.htmlBody ? { Html: { Charset: 'UTF-8', Data: opts.htmlBody } } : {}),
      },
    },
  }));
}

```typescript
// backend/src/routes/tenants.ts — Phase 52 plan 02 (M5)
// PUBLIC endpoint (no auth). Mounted in app.ts BEFORE any requireAuth-using router.
// TODO(M8): add IP-based rate limiting to prevent signup spam.
import { Router } from 'express';
import { randomBytes } from 'crypto';
import { pool } from '../db';
import {
  CognitoIdentityProviderClient,
  AdminCreateUserCommand,
  AdminSetUserPasswordCommand,
  AdminAddUserToGroupCommand,
  AdminDeleteUserCommand,
  AdminInitiateAuthCommand,
} from '@aws-sdk/client-cognito-identity-provider';
import { getCognitoUserPoolId, getCognitoAppClientId } from '../secrets';

const r = Router();

const RESERVED_SLUGS = new Set([
  'www','admin','app','api','static','mail','turion','zietra',
  'marquee','asc606','meet','docs','support','turionspace',
  'campaigns-api','login','signup',
]);
const SLUG_RE = /^[a-z0-9-]{3,32}$/;
const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const MODULE_CODES = [
  'crm','sales','purchase','items','plm','mes','quality',
  'lean-erp-pro','asc606','royalty','dropship','ai-agents','qb-migration',
];

const cognito = new CognitoIdentityProviderClient({
  region: process.env.AWS_REGION ?? 'us-east-1',
});

r.post('/signup', async (req, res) => {
  // === 1. Validate input ===
  const email = String(req.body?.email || '').trim().toLowerCase();
  const name = String(req.body?.name || '').trim();
  const organization_name = String(req.body?.organization_name || '').trim();
  const slug = String(req.body?.slug || '').trim().toLowerCase();

  if (!EMAIL_RE.test(email))
    return res.status(400).json({ error: 'Valid email required' });
  if (!name || name.length > 100)
    return res.status(400).json({ error: 'Name required (max 100 chars)' });
  if (!organization_name || organization_name.length > 100)
    return res.status(400).json({ error: 'Organization name required (max 100 chars)' });
  if (!SLUG_RE.test(slug))
    return res.status(400).json({ error: 'Slug must be 3-32 chars, lowercase letters/digits/hyphens' });
  if (slug.startsWith('-') || slug.endsWith('-') || slug.includes('--'))
    return res.status(400).json({ error: 'Slug cannot start/end with hyphen or contain "--"' });
  if (RESERVED_SLUGS.has(slug))
    return res.status(409).json({ error: 'Slug is reserved' });

  // === 2. Pre-flight slug uniqueness check ===
  const existing = await pool.query('SELECT 1 FROM public.tenants WHERE slug = $1', [slug]);
  if (existing.rowCount && existing.rowCount > 0)
    return res.status(409).json({ error: 'Slug already taken' });

  const USER_POOL_ID = getCognitoUserPoolId();
  const APP_CLIENT_ID = getCognitoAppClientId();
  if (!USER_POOL_ID || !APP_CLIENT_ID) {
    console.error('[signup] Cognito config not loaded');
    return res.status(500).json({ error: 'Signup failed' });
  }

  // === 3. Create Cognito user (idempotent if you catch UsernameExistsException as 409) ===
  let cognitoSub: string | null = null;
  try {
    const tmpPassword = randomBytes(24).toString('base64url') + 'A1!';
    const createResp = await cognito.send(new AdminCreateUserCommand({
      UserPoolId: USER_POOL_ID,
      Username: email,
      MessageAction: 'SUPPRESS',
      UserAttributes: [
        { Name: 'email', Value: email },
        { Name: 'email_verified', Value: 'true' },
        { Name: 'custom:role', Value: 'customer' },
        { Name: 'name', Value: name },
      ],
    }));
    cognitoSub = createResp.User?.Attributes?.find(a => a.Name === 'sub')?.Value ?? null;
    if (!cognitoSub) throw new Error('Cognito did not return sub');

    await cognito.send(new AdminSetUserPasswordCommand({
      UserPoolId: USER_POOL_ID, Username: email, Password: tmpPassword, Permanent: true,
    }));
    await cognito.send(new AdminAddUserToGroupCommand({
      UserPoolId: USER_POOL_ID, Username: email, GroupName: 'customer',
    }));
  } catch (e: any) {
    if (e?.name === 'UsernameExistsException')
      return res.status(409).json({ error: 'Email already registered' });
    console.error('[signup] Cognito create failed:', e?.name, e?.message);
    return res.status(500).json({ error: 'Signup failed' });
  }

  // === 4. DB transaction (tenant + features) — rollback Cognito user on failure ===
  const client = await pool.connect();
  let tenantId: string;
  try {
    await client.query('BEGIN');
    const ins = await client.query(
      `INSERT INTO public.tenants (slug, name, owner_cognito_sub, plan)
       VALUES ($1, $2, $3, 'trial') RETURNING id`,
      [slug, organization_name, cognitoSub]
    );
    tenantId = ins.rows[0].id;
    for (const m of MODULE_CODES) {
      await client.query(
        `INSERT INTO public.tenant_features (tenant_id, module_code, enabled) VALUES ($1, $2, true)`,
        [tenantId, m]
      );
    }
    await client.query('COMMIT');
  } catch (e: any) {
    await client.query('ROLLBACK').catch(() => {});
    await cognito.send(new AdminDeleteUserCommand({
      UserPoolId: USER_POOL_ID, Username: email,
    })).catch(err => console.error('[signup] Cognito rollback failed:', err?.name, err?.message));
    console.error('[signup] DB transaction failed:', e?.message);
    return res.status(500).json({ error: 'Signup failed' });
  } finally {
    client.release();
  }

  // === 5. Trigger welcome magic-link via Cognito CUSTOM_AUTH (best-effort) ===
  // Reuses Phase 39 Create-Auth-Challenge Lambda → SES SendEmail. No separate SES code path.
  // SES sandbox: delivery only to verified addresses; non-verified silently bounce.
  try {
    await cognito.send(new AdminInitiateAuthCommand({
      UserPoolId: USER_POOL_ID,
      ClientId: APP_CLIENT_ID,
      AuthFlow: 'CUSTOM_AUTH',
      AuthParameters: { USERNAME: email },
    }));
  } catch (e: any) {
    console.warn('[signup] welcome magic-link InitiateAuth failed (best-effort):', e?.name, e?.message);
  }

  return res.json({
    ok: true,
    tenant: { id: tenantId, slug, name: organization_name },
    message: `Check your inbox at ${email} to sign in.`,
  });
});

export default r;
```

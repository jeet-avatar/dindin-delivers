<script>
const SLUG_RE = /^[a-z0-9-]{3,32}$/;
document.getElementById('signupForm').addEventListener('submit', async (e) => {
  e.preventDefault();
  const email = document.getElementById('email').value.trim().toLowerCase();
  const name = document.getElementById('name').value.trim();
  const organization_name = document.getElementById('organization_name').value.trim();
  const slug = document.getElementById('slug').value.trim().toLowerCase();
  const btn = document.getElementById('submitBtn');
  const err = document.getElementById('error');
  err.textContent = '';

  if (!SLUG_RE.test(slug)) { err.textContent = 'Workspace URL must be 3-32 chars (a-z, 0-9, -)'; return; }
  if (slug.startsWith('-') || slug.endsWith('-') || slug.includes('--')) {
    err.textContent = 'Workspace URL cannot start/end with hyphen or contain "--"'; return;
  }

  btn.disabled = true; btn.textContent = 'Creating...';
  try {
    const r = await fetch(window.TURION_CONFIG.API_BASE + '/api/tenants/signup', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, name, organization_name, slug }),
    });
    const data = await r.json();
    if (!r.ok) {
      err.textContent = data.error || 'Signup failed';
      btn.disabled = false; btn.textContent = 'Create my workspace';
      return;
    }
    document.getElementById('signupForm').style.display = 'none';
    document.getElementById('successSlug').textContent = slug;
    document.getElementById('successEmail').textContent = email;
    document.getElementById('success').style.display = 'block';
  } catch (e) {
    err.textContent = e.message || 'Network error';
    btn.disabled = false; btn.textContent = 'Create my workspace';
  }
});
</script>
</body>
</html>

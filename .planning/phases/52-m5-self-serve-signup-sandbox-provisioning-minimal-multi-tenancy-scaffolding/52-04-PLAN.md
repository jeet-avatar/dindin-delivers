# === Cleanup trap (idempotent — runs even on failure) ===
cleanup() {
  set +e
  if [ "${TENANT_ID:-}" != "" ]; then
    PGPASSWORD="${PGPASSWORD:?PGPASSWORD env required}" $PSQL \
      -h "$DB_HOST" -p "$DB_PORT" -U "$DB_USER" -d "$DB_NAME" \
      -c "DELETE FROM public.tenants WHERE id='$TENANT_ID';" > /dev/null
    echo "[cleanup] Deleted tenant row $TENANT_ID (cascades to features)"
  fi
  if [ "${COGNITO_DELETE:-no}" = "yes" ] && [ "$EMAIL" != "$ANCHOR_EMAIL" ]; then
    aws cognito-idp admin-delete-user --user-pool-id "$USER_POOL_ID" --username "$EMAIL" 2>/dev/null \
      && echo "[cleanup] Deleted Cognito user $EMAIL" \
      || echo "[cleanup] Cognito user $EMAIL already absent"
  elif [ "$EMAIL" = "$ANCHOR_EMAIL" ]; then
    echo "[cleanup] SKIPPED Cognito delete — $EMAIL is the anchor admin"
  fi
}
trap cleanup EXIT

# === Pre-check: PGPASSWORD must be set ===
if [ -z "${PGPASSWORD:-}" ]; then
  echo "ERROR: PGPASSWORD must be set (export PGPASSWORD='Thirumala977!')"
  exit 2
fi

# === Pre-check: SES verification ===
SES_STATUS=$(aws ses get-identity-verification-attributes --identities "$EMAIL" \
  --query "VerificationAttributes.\"$EMAIL\".VerificationStatus" --output text 2>/dev/null || echo "NotFound")
if [ "$SES_STATUS" != "Success" ] && [ "$EMAIL" != "$ANCHOR_EMAIL" ]; then
  # zietra.com domain is verified, so non-anchor sub-addresses might still send

BODY=$(echo "$RESP" | head -n1)
CODE=$(echo "$RESP" | tail -n1)
echo "POST /signup → $CODE"

[ "$CODE" = "200" ] || { echo "FAIL: expected 200, got $CODE"; exit 1; }

echo "TENANT_ID=$TENANT_ID"

# === 2. Side effect 1: Cognito user exists with 'customer' group ===

echo "Side effect 1 OK — Cognito sub: $COG_SUB"

  --query 'Groups[].GroupName' --output text)
echo "$GROUPS" | grep -q '^customer$\| customer$\|^customer ' || \
  echo "$GROUPS" | tr '\t' '\n' | grep -q '^customer$' || \
  { echo "FAIL: customer group not assigned (got: $GROUPS)"; exit 1; }
echo "Side effect 1 OK — 'customer' group: yes"

ROLE=$(aws cognito-idp admin-get-user --user-pool-id "$USER_POOL_ID" --username "$EMAIL" \
  --query 'UserAttributes[?Name==`custom:role`].Value' --output text)

echo "Side effect 1 OK — custom:role=customer"

sleep 8

## What Phase 53 inherits from Phase 52

### 1. `public.tenants` table — LIVE on Supabase Postgres

```sql
column              type          notes
------------------  ------------  ------------------------------------------------
id                  uuid PK       gen_random_uuid(); fixed `00000000-...-001` for Turion
slug                text UNIQUE   3-32 chars, `^[a-z0-9-]+$`, no leading/trailing/double hyphen
name                text          organization display name (1-100 chars)
owner_cognito_sub   text          forward-link to Cognito user (NOT Supabase sub)
plan                text          'trial' | 'paid' | 'disabled' (default 'trial')
created_at          timestamptz   default now()
trial_ends_at       timestamptz   default now() + interval '30 days'
```

**Subdomain router input:** `SELECT id, plan, trial_ends_at FROM public.tenants WHERE slug = $1`.

### 2. `public.tenant_features` table — LIVE

```sql
column        type        notes
------------  ----------  -----------------------------------------------------
tenant_id     uuid FK     references public.tenants(id) ON DELETE CASCADE
module_code   text        13 values (see allowed list below)
enabled       boolean     default true (Phase 52 sets all 13 to true on signup)
enabled_at    timestamptz default now()
expires_at    timestamptz NULL (M4 fills this for paid-add-on expiry)
PRIMARY KEY (tenant_id, module_code)
```

**Allowed module_code values:** `crm`, `sales`, `purchase`, `items`, `plm`, `mes`, `quality`, `lean-erp-pro`, `asc606`, `royalty`, `dropship`, `ai-agents`, `qb-migration`.

**App shell query (Phase 54):** `SELECT module_code FROM public.tenant_features WHERE tenant_id = $1 AND enabled = true`.

### 3. `tenant_id UUID NULL` columns on existing tables

- **105 tables** total: 57 `turion.*` + 48 `turion_satellite.*` (full list in 52-RESEARCH.md A2/A3)

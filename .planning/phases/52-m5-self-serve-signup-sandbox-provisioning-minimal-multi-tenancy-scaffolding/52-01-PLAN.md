```sql
-- === Step 1: Seed Turion tenant row FIRST (so backfill has a target) ===
insert into public.tenants (id, slug, name, owner_cognito_sub, plan)
values (
  '00000000-0000-0000-0000-000000000001'::uuid,

  :'cognito_sub',

on conflict (id) do update set

  owner_cognito_sub = excluded.owner_cognito_sub,
  plan              = excluded.plan;

-- === Step 2: Seed Turion tenant_features (all 13 ON) ===
insert into public.tenant_features (tenant_id, module_code, enabled)
select '00000000-0000-0000-0000-000000000001'::uuid, code, true
from unnest(array[
  'crm','sales','purchase','items','plm','mes','quality',
  'lean-erp-pro','asc606','royalty','dropship','ai-agents','qb-migration'
]) as code
on conflict (tenant_id, module_code) do nothing;

-- === Step 3: Add tenant_id NULL column to every existing turion.*/turion_satellite.* table ===
DO $$
DECLARE
  r record;
BEGIN
  FOR r IN
    SELECT table_schema, table_name
    FROM information_schema.tables
    WHERE table_schema IN ('turion','turion_satellite')
      AND table_type = 'BASE TABLE'
    ORDER BY table_schema, table_name
  LOOP
    EXECUTE format(
      'ALTER TABLE %I.%I ADD COLUMN IF NOT EXISTS tenant_id uuid',
      r.table_schema, r.table_name
    );
  END LOOP;
END $$;

-- === Step 4: Backfill every row to Turion's UUID where tenant_id IS NULL ===
DO $$
DECLARE
  r record;
  turion_id uuid := '00000000-0000-0000-0000-000000000001';
BEGIN
  FOR r IN
    SELECT table_schema, table_name
    FROM information_schema.tables
    WHERE table_schema IN ('turion','turion_satellite')
      AND table_type = 'BASE TABLE'
    ORDER BY table_schema, table_name
  LOOP
    EXECUTE format(
      'UPDATE %I.%I SET tenant_id = $1 WHERE tenant_id IS NULL',
      r.table_schema, r.table_name
    ) USING turion_id;
  END LOOP;
END $$;

-- === Step 5: Index every tenant_id column (B-tree, simple; M3 replaces with RLS) ===
DO $$
DECLARE
  r record;
BEGIN
  FOR r IN
    SELECT table_schema, table_name
    FROM information_schema.tables
    WHERE table_schema IN ('turion','turion_satellite')
      AND table_type = 'BASE TABLE'
    ORDER BY table_schema, table_name
  LOOP
    EXECUTE format(
      'CREATE INDEX IF NOT EXISTS %I ON %I.%I (tenant_id)',
      r.table_name || '_tenant_id_idx', r.table_schema, r.table_name
    );
  END LOOP;
END $$;

-- === Step 6 (DEFERRED — M3 owns): SET NOT NULL + add FK to public.tenants(id). ===
-- M5 leaves tenant_id NULLABLE so M3 RLS rollout is a separate phase.
```

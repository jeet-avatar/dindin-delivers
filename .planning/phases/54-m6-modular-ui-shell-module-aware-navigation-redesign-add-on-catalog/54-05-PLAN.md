```typescript
import { test as setup, expect } from '@playwright/test';
import { writeFileSync, mkdirSync, existsSync } from 'node:fs';

setup('seed authenticated Turion session', async ({ page, baseURL }) => {
  // 54.3 hardens this to AdminInitiateAuth USER_PASSWORD_AUTH against a dedicated test user.
  // v1: read token from env (CI secret) or skip if already written.
  const tokenPath = 'playwright/.auth/turion.json';
  if (existsSync(tokenPath) && !process.env.FORCE_RESEED) {
    setup.skip(true, `Auth state already exists at ${tokenPath}. Set FORCE_RESEED=1 to re-seed.`);
    return;
  }
  const idToken = process.env.TURION_ID_TOKEN;
  if (!idToken) {
    setup.skip(true,
      'No TURION_ID_TOKEN env. To seed: get a fresh Cognito IdToken from a magic-link login on turionspace.zietra.com, ' +
      'then run: TURION_ID_TOKEN=<token> FORCE_RESEED=1 npm run test:e2e:turion -- --project=setup'
    );
    return;
  }
  await page.goto(baseURL!);
  await page.evaluate((t) => {
    localStorage.setItem('cognito_id_token', t);   // matches cognito-auth.js storage key
    localStorage.setItem('cognito_id_token_expiry', String(Date.now() + 3600_000));
  }, idToken);
  mkdirSync('playwright/.auth', { recursive: true });
  await page.context().storageState({ path: tokenPath });
});
```

test.describe('Phase 41 + 52 + 53 auth contracts', () => {
  test('anonymous request to /catalog redirects to /erp-login.html (Phase 41 requireSession)', async ({ page, browser }) => {
    // Use clean context (no storageState) for anon
    const ctx = await browser.newContext({ storageState: { cookies: [], origins: [] } });

    await ctx.close();

  test('signup with reserved slug "royalty" returns 409 (Phase 54-04 RESERVED expansion)', async ({ request }) => {

      'https://lo254mvukl.execute-api.us-east-1.amazonaws.com/api/tenants/signup',

    );
    expect(resp.status()).toBe(409);
    const body = await resp.json();
    expect(JSON.stringify(body).toLowerCase()).toContain('reserved');

  test('signup with reserved slug "salesforce" returns 409', async ({ request }) => {
    const resp = await request.post(
      'https://lo254mvukl.execute-api.us-east-1.amazonaws.com/api/tenants/signup',

3. Render block — wait for `window.__ZIETRA_TENANT` set by Wave-1 shell:
```javascript
(async () => {
  // Poll for shell-tenant up to 5s (shell is `defer`red, this script also defer → runs after)
  let attempts = 0;
  while (!window.__ZIETRA_TENANT && attempts < 100) {
    await new Promise(r => setTimeout(r, 50));
    attempts++;
  }
  const t = window.__ZIETRA_TENANT || { features: [], plan: 'trial' };
  const enabled = new Set(t.features || []);
  const grid = document.getElementById('catalog-grid');
  if (!grid) return;

  for (const m of MODULE_CATALOG) {
    const on = enabled.has(m.code);
    const card = document.createElement('article');
    card.className = 'z-catalog-card';
    card.id = m.code;
    const iconHtml = (window.__ZIETRA_ICONS && window.__ZIETRA_ICONS[m.icon]) || '';
    const statusClass = on ? 'enabled' : 'disabled';
    const statusLabel = on ? 'Enabled' : '+ Add to plan';
    const externalGlyph = m.external ? ' ↗' : '';

    let ctaHtml;
    if (on) {
      ctaHtml = m.external
        ? `<a class="z-cta-btn" target="_blank" rel="noopener" href="${m.open}">Open ↗</a>`
        : `<a class="z-cta-btn" href="${m.open}">Open</a>`;
    } else {
      const label = t.plan === 'trial' ? 'Try free' : 'Subscribe';
      ctaHtml = `<button class="z-cta-btn secondary" data-code="${m.code}">${label}</button>`;
    }
    card.innerHTML = `
      <div class="z-icon" style="width:28px;height:28px;color:var(--z-primary)">${iconHtml}</div>
      <h3>${m.name}${externalGlyph}</h3>
      <p>${m.desc}</p>
      <div class="z-source">${m.source}</div>
      <span class="z-status ${statusClass}">${statusLabel}</span>
      <br>${ctaHtml}
    `;
    grid.appendChild(card);
  }

  // Wire stub CTAs
  for (const btn of grid.querySelectorAll('button.z-cta-btn[data-code]')) {
    btn.addEventListener('click', () => {
      alert('Stripe checkout ships in M4. To explore this module today, sign up for a trial.');
    });
  }

  // Hash routing — /catalog#asc606 scrolls to that card
  if (location.hash) {
    const target = document.querySelector(location.hash);
    if (target) target.scrollIntoView({ behavior: 'smooth', block: 'center' });
  }
})();
```

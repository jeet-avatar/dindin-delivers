2. Idempotent main function:
```javascript
function processFile(absPath, relPath) {
  let html = readFileSync(absPath, 'utf8');
  const before = html;

  // Step 1: already injected? — no-op
  if (html.includes(MARKER)) return { status: 'already', file: relPath };

  // Step 2: strip OLD shell tags (53 pages have them)
  let stripped = false;
  for (const pat of OLD_PATTERNS) {
    if (pat.test(html)) {
      html = html.replace(pat, '');
      stripped = true;
    }
  }

  // Step 3: inject new tags into <head>. Require a literal '</head>' to anchor against.
  if (!/<\/head>/i.test(html)) {
    return { status: 'no-head', file: relPath };
  }
  html = html.replace(/<\/head>/i, NEW_INJECTION + '</head>');

  if (html === before) return { status: 'unchanged', file: relPath };

  writeFileSync(absPath, html, 'utf8');
  return { status: stripped ? 'stripped+injected' : 'injected', file: relPath };
}
```

3. Walk + invocation:
```javascript
function walk(rootAbs) {
  const out = [];
  for (const dir of WALK_DIRS) {
    const base = (dir === '.') ? rootAbs : join(rootAbs, dir);
    for (const entry of readdirSync(base)) {
      const abs = join(base, entry);
      const st = statSync(abs);
      if (st.isDirectory()) {
        // we don't recurse beyond WALK_DIRS for v1 — keeps the strip blast radius tight
        continue;
      }
      if (extname(entry).toLowerCase() !== '.html') continue;
      if (SKIP_FILES.has(entry)) continue;
      out.push({ abs, rel: (dir === '.') ? entry : `${dir}/${entry}` });
    }
  }
  return out;
}

function main() {
  const rootAbs = fileURLToPath(new URL('..', import.meta.url));
  const dry = process.argv.includes('--dry');
  const files = walk(rootAbs);

  const counts = { 'injected': 0, 'stripped+injected': 0, 'already': 0, 'no-head': 0, 'unchanged': 0 };
  const details = [];

  for (const f of files) {
    let result;
    if (dry) {
      // In dry mode, read but never write — emulate by NOT calling writeFileSync.
      const html = readFileSync(f.abs, 'utf8');
      if (html.includes(MARKER)) result = { status: 'already', file: f.rel };
      else if (!/<\/head>/i.test(html)) result = { status: 'no-head', file: f.rel };
      else result = { status: OLD_PATTERNS.some(p => p.test(html)) ? 'would-strip+inject' : 'would-inject', file: f.rel };
    } else {
      result = processFile(f.abs, f.rel);
    }
    counts[result.status] = (counts[result.status] || 0) + 1;
    details.push(result);
  }

  console.log(`[inject-shell] mode=${dry ? 'DRY' : 'LIVE'} total=${files.length}`);
  for (const [k, v] of Object.entries(counts)) console.log(`  ${k}: ${v}`);
  if (counts['no-head'] > 0) {
    console.log('\n[inject-shell] WARN — pages without </head>:');
    for (const d of details) if (d.status === 'no-head') console.log(`  - ${d.file}`);
  }
  // Exit non-zero on no-head OR if LIVE mode found zero changes AND zero already-done (clearly broken):
  if (counts['no-head'] > 0) process.exit(2);
}

main();
```

The middleware uses `pool.query` + an in-module Map cache. Tests mock `../../src/db` so `pool.query` is a `vi.fn()`. They also need to clear the role cache between tests (since it's module-scoped) — call `invalidateRoleCache` in `beforeEach` for the test sub/tenant ids used.

```ts
// backend/tests/unit/role-middleware.test.ts — Phase 54.1-04
import { describe, it, expect, vi, beforeEach } from 'vitest';

// Pool mock MUST be set up before the SUT import so the SUT's `import { pool } from '../db'` resolves to the mock.
vi.mock('../../src/db', () => ({
  pool: { query: vi.fn() },
}));

// Required by db.ts at import-time
process.env.DATABASE_URL = process.env.DATABASE_URL || 'postgresql://test:test@localhost/test';

import { requireRole, invalidateRoleCache } from '../../src/middleware/role';
import { pool } from '../../src/db';

const TENANT = 't-1';
const SUB = 'sub-1';

function mockReq(overrides: any = {}) {
  return { user: { id: SUB }, tenant: { id: TENANT }, headers: {}, ...overrides };
}
function mockRes() {
  const r: any = {};
  r.status = vi.fn(() => r);
  r.json = vi.fn(() => r);
  return r;
}

describe('requireRole — DB lookup + JWT fallback + cache', () => {
  beforeEach(() => {
    vi.clearAllMocks();
    invalidateRoleCache(TENANT, SUB);
  });

  it('1. allows admin when admin required (DB rowCount=1, role=admin)', async () => {
    (pool.query as any).mockResolvedValueOnce({ rowCount: 1, rows: [{ role: 'admin' }] });
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin')(mockReq() as any, res, next);
    expect(next).toHaveBeenCalled();
    expect(res.status).not.toHaveBeenCalled();
  });

  it('2. allows admin when manager-or-admin required (admin in allowed set)', async () => {
    (pool.query as any).mockResolvedValueOnce({ rowCount: 1, rows: [{ role: 'admin' }] });
    const next = vi.fn();
    await requireRole('admin', 'manager')(mockReq() as any, mockRes(), next);
    expect(next).toHaveBeenCalled();
  });

  it('3. rejects viewer when admin required → 403 with required + actual', async () => {
    (pool.query as any).mockResolvedValueOnce({ rowCount: 1, rows: [{ role: 'viewer' }] });
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin')(mockReq() as any, res, next);
    expect(next).not.toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(403);
    const body = res.json.mock.calls[0][0];
    expect(body.error).toMatch(/insufficient|Insufficient/i);
    expect(body.actual).toBe('viewer');
  });

  it('4. rejects member when manager required → 403', async () => {
    (pool.query as any).mockResolvedValueOnce({ rowCount: 1, rows: [{ role: 'member' }] });
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin', 'manager')(mockReq() as any, res, next);
    expect(next).not.toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(403);
  });

  it('5. returns 401 when req.user missing', async () => {
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin')(mockReq({ user: undefined }) as any, res, next);
    expect(res.status).toHaveBeenCalledWith(401);
    expect(next).not.toHaveBeenCalled();
  });

  it('6. returns 401 when req.tenant missing', async () => {
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin')(mockReq({ tenant: undefined }) as any, res, next);
    expect(res.status).toHaveBeenCalledWith(401);
  });

  it('7. on DB error, falls back to req.user.role (JWT) and proceeds when role matches', async () => {
    (pool.query as any).mockRejectedValueOnce(new Error('connection refused'));
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin')(mockReq({ user: { id: SUB, role: 'admin' } }) as any, res, next);
    expect(next).toHaveBeenCalled();
  });

  it('8. on DB error AND no JWT role → 403 no_tenant_membership', async () => {
    (pool.query as any).mockRejectedValueOnce(new Error('connection refused'));
    const next = vi.fn();
    const res = mockRes();
    await requireRole('admin')(mockReq({ user: { id: SUB } }) as any, res, next);
    expect(res.status).toHaveBeenCalledWith(403);
    const body = res.json.mock.calls[0][0];
    expect(body.actual).toBe('none');
  });

  it('9. invalidateRoleCache clears the cached entry (next call re-queries DB)', async () => {
    (pool.query as any).mockResolvedValueOnce({ rowCount: 1, rows: [{ role: 'admin' }] });
    await requireRole('admin')(mockReq() as any, mockRes(), vi.fn());
    expect(pool.query).toHaveBeenCalledTimes(1);
    // Second call within TTL — should hit cache, NOT re-query
    await requireRole('admin')(mockReq() as any, mockRes(), vi.fn());
    expect(pool.query).toHaveBeenCalledTimes(1);
    // Invalidate + third call → re-queries
    invalidateRoleCache(TENANT, SUB);
    (pool.query as any).mockResolvedValueOnce({ rowCount: 1, rows: [{ role: 'admin' }] });
    await requireRole('admin')(mockReq() as any, mockRes(), vi.fn());
    expect(pool.query).toHaveBeenCalledTimes(2);
  });

  it('10. negative cache prevents repeat DB hammering for unknown user (rowCount=0)', async () => {
    (pool.query as any).mockResolvedValueOnce({ rowCount: 0, rows: [] });
    await requireRole('admin')(mockReq() as any, mockRes(), vi.fn());
    expect(pool.query).toHaveBeenCalledTimes(1);
    // Second call — negative cache hit (5s TTL), no re-query
    await requireRole('admin')(mockReq() as any, mockRes(), vi.fn());
    expect(pool.query).toHaveBeenCalledTimes(1);
  });
});
```

```typescript
function drawBeatGrid(
  ctx: CanvasRenderingContext2D, W: number, H: number,
  beats: TrackAnlzData['beat_grid'],
  startMs: number, endMs: number,
) {
  let barCount = 0;
  for (const beat of beats) {
    const isDown = beat.beat === 1;
    if (isDown) barCount++;
    if (beat.time_ms < startMs || beat.time_ms > endMs) continue;

    const x = ((beat.time_ms - startMs) / (endMs - startMs)) * W;

    if (isDown && barCount % 16 === 1) {
      // Phrase marker
      ctx.strokeStyle = 'rgba(255,255,255,0.4)';
      ctx.lineWidth = 2;
    } else if (isDown) {
      // Downbeat
      ctx.strokeStyle = 'rgba(255,255,255,0.25)';
      ctx.lineWidth = 1.5;
    } else {
      // Regular beat
      ctx.strokeStyle = 'rgba(255,255,255,0.06)';
      ctx.lineWidth = 1;
    }

    ctx.beginPath();
    ctx.moveTo(x, 0);
    ctx.lineTo(x, H);
    ctx.stroke();
  }
}
```

**IMPORTANT**: Count bars even for beats outside the visible range (before `startMs`) so phrase markers align correctly. The counting must happen before the `continue` skip.

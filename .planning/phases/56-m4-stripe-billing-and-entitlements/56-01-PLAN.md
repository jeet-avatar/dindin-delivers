```bash
#!/usr/bin/env bash
set -euo pipefail
REGION=us-east-1

prompt_and_write() {
  local secret_name="$1"
  local prefix_check="$2"
  local description="$3"
  echo
  echo "=== $description ==="
  echo "Secret name: $secret_name"
  echo "Expected prefix: $prefix_check"
  read -srp "Paste value (input hidden): " VALUE
  echo
  if [[ ! "$VALUE" =~ ^${prefix_check} ]]; then
    echo "ERROR: value does not start with '$prefix_check'. Aborting."
    exit 1
  fi
  aws secretsmanager put-secret-value --secret-id "$secret_name" --secret-string "$VALUE" --region $REGION > /dev/null
  echo "OK — wrote $secret_name"
}

echo "This script populates 3 Stripe TEST-mode secrets in AWS Secrets Manager."
```

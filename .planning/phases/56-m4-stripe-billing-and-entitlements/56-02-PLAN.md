```typescript
import { getStripe } from './stripe-client';
import { pool } from '../db';

export async function createCheckoutSession(
  tenantId: string,
  addonModuleIds: string[],   // e.g. ['plm', 'asc606']  — base is always included
): Promise<{ url: string; sessionId: string }> {
  const stripe = await getStripe();

  // 1. Resolve tenant + ensure stripe_customer_id exists (lazy create).
  const t = await pool.query(
    `SELECT id, slug, name, stripe_customer_id, trial_ends_at, plan
       FROM public.tenants WHERE id=$1`, [tenantId]);
  if (t.rowCount === 0) throw new Error(`Tenant ${tenantId} not found`);
  const tenant = t.rows[0];

  let customerId = tenant.stripe_customer_id;
  if (!customerId) {
    const c = await stripe.customers.create({
      metadata: { tenant_id: tenant.id, tenant_slug: tenant.slug },
      name: tenant.name,
    });
    customerId = c.id;
    await pool.query(
      `UPDATE public.tenants SET stripe_customer_id=$1 WHERE id=$2`,
      [customerId, tenant.id]);
  }

  // 2. Build line_items: base + add-ons (looked up from public.pricing — Pitfall 3).
  const moduleCodes = ['__base__', ...addonModuleIds];
  const prices = await pool.query(
    `SELECT stripe_price_id, module_code FROM public.pricing
      WHERE module_code = ANY($1) AND active = true`,
    [moduleCodes]);
  if (prices.rowCount === 0) throw new Error('No active prices found for selection');
  const lineItems = prices.rows.map(p => ({ price: p.stripe_price_id, quantity: 1 }));

  // 3. Trial carry-over (UTC epoch math — Pitfall 13).
  let trialDays: number | undefined;
  if (tenant.trial_ends_at) {
    const msLeft = new Date(tenant.trial_ends_at).getTime() - Date.now();
    const days = Math.ceil(msLeft / 86400000);
    if (days >= 1) trialDays = days;
  }

  // 4. Create session — metadata.tenant_id on BOTH customer and subscription (Pattern 4 defense in depth).
  const session = await stripe.checkout.sessions.create({
    mode: 'subscription',
    customer: customerId,
    line_items: lineItems,
    subscription_data: {
      metadata: { tenant_id: tenant.id, tenant_slug: tenant.slug },
      trial_period_days: trialDays,
    },
    metadata: { tenant_id: tenant.id, tenant_slug: tenant.slug },
    payment_method_collection: 'always',   // Pitfall 9 — capture card AT trial start
    success_url: `https://${tenant.slug}.zietra.com/billing/success?session_id={CHECKOUT_SESSION_ID}`,
    cancel_url:  `https://${tenant.slug}.zietra.com/billing/upgrade?canceled=true`,
    allow_promotion_codes: false,
  });
  return { url: session.url as string, sessionId: session.id };
}
```

```typescript
import Stripe from 'stripe';
import { pool } from '../db';

const BASE_MODULES = ['crm', 'sales', 'purchase', 'items'];

export async function syncSubscription(sub: Stripe.Subscription): Promise<void> {
  const tenantId = sub.metadata?.tenant_id;
  if (!tenantId) {
    console.warn(`[entitlement-sync] subscription ${sub.id} has no tenant_id metadata; skipping`);
    return;
  }

  // 1. Map subscription items → module_codes via public.pricing.
  const subscribedPriceIds = sub.items.data.map(i => i.price.id);
  if (subscribedPriceIds.length === 0) return;
  const priceMap = await pool.query(
    `SELECT stripe_price_id, module_code FROM public.pricing WHERE stripe_price_id = ANY($1)`,
    [subscribedPriceIds]);
  const subscribedModules = new Set<string>(priceMap.rows.map(r => r.module_code));
  const hasBase = priceMap.rows.some(r => r.module_code === '__base__');
  if (hasBase) BASE_MODULES.forEach(m => subscribedModules.add(m));

  await pool.query('BEGIN');
  try {
    // 2. UPSERT public.subscriptions cache.
    await pool.query(
      `INSERT INTO public.subscriptions
         (tenant_id, stripe_subscription_id, stripe_customer_id, status, cancel_at_period_end,
          trial_end, current_period_start, current_period_end, updated_at)
       VALUES ($1,$2,$3,$4,$5,to_timestamp($6),to_timestamp($7),to_timestamp($8), NOW())
       ON CONFLICT (stripe_subscription_id) DO UPDATE SET
         status=EXCLUDED.status,
         cancel_at_period_end=EXCLUDED.cancel_at_period_end,
         trial_end=EXCLUDED.trial_end,
         current_period_start=EXCLUDED.current_period_start,
         current_period_end=EXCLUDED.current_period_end,
         updated_at=NOW()`,
      [tenantId, sub.id, sub.customer, sub.status, sub.cancel_at_period_end,
       sub.trial_end, sub.current_period_start, sub.current_period_end]);

    // 3. Atomic disable-then-enable on tenant_features.
    await pool.query(`UPDATE public.tenant_features SET enabled=false WHERE tenant_id=$1`, [tenantId]);
    if (subscribedModules.size > 0) {
      await pool.query(
        `UPDATE public.tenant_features SET enabled=true
           WHERE tenant_id=$1 AND module_code = ANY($2)`,
        [tenantId, [...subscribedModules]]);
    }
    if (hasBase) {
      await pool.query(`UPDATE public.tenants SET plan='paid' WHERE id=$1 AND plan != 'paid'`, [tenantId]);
    }
    await pool.query('COMMIT');
  } catch (e) {
    await pool.query('ROLLBACK');
    throw e;
  }
}

export async function downgradeSubscription(sub: Stripe.Subscription): Promise<void> {
  // Only fully downgrade on customer.subscription.deleted (period end, NOT cancel_at_period_end).
  const tenantId = sub.metadata?.tenant_id;
  if (!tenantId) return;
  await pool.query('BEGIN');
  try {
    await pool.query(`UPDATE public.subscriptions SET status='canceled', updated_at=NOW()
                        WHERE stripe_subscription_id=$1`, [sub.id]);
    await pool.query(`UPDATE public.tenant_features SET enabled=false WHERE tenant_id=$1`, [tenantId]);
    await pool.query(`UPDATE public.tenants SET plan='disabled' WHERE id=$1`, [tenantId]);
    await pool.query('COMMIT');
  } catch (e) {
    await pool.query('ROLLBACK');
    throw e;
  }
}
```

Step 1 — Modify stripe-client.ts to honor `STRIPE_SECRET_NAME` env var (defaults to test):

```typescript
import Stripe from 'stripe';
import { getSecret } from '../secrets';

let _stripe: Stripe | null = null;
let _webhookSecret: string | null = null;

export async function getStripe(): Promise<Stripe> {
  if (_stripe) return _stripe;
  const secretName = process.env.STRIPE_SECRET_NAME ?? 'zietra/stripe/test-secret-key';
  const key = await getSecret(secretName);
  if (!key || key.startsWith('PLACEHOLDER_')) {
    throw new Error(`Stripe key at ${secretName} is still placeholder — populate via the helper script`);
  }
  _stripe = new Stripe(key, { apiVersion: '2026-04-22.dahlia', typescript: true });
  return _stripe;
}

export async function getWebhookSecret(): Promise<string> {
  if (_webhookSecret) return _webhookSecret;
  const secretName = process.env.STRIPE_WEBHOOK_SECRET_NAME ?? 'zietra/stripe/test-webhook-secret';
  _webhookSecret = await getSecret(secretName);
  return _webhookSecret;
}
```

```typescript
import { pool } from '../db';
import { sendEmail } from '../notify';   // existing helper

export async function runTrialReminders(): Promise<{ sent: number; skipped: number }> {
  let sent = 0, skipped = 0;
  // Day -3: trial ends in 3 days
  // Day 0:  trial ends today
  // Day +7: 7 days after trial ended (read-only grace expiring)
  const buckets = [
    { days_until: 3,  template: 'trial-ending-3d',  subject: 'Your Zietra trial ends in 3 days' },
    { days_until: 0,  template: 'trial-ending-today', subject: 'Your Zietra trial ends today' },
    { days_until: -7, template: 'trial-grace-expiring', subject: 'Action required — reactivate your Zietra subscription' },
  ];

  for (const b of buckets) {
    const r = await pool.query(
      `SELECT t.id, t.slug, t.email, t.trial_ends_at
         FROM public.tenants t
         LEFT JOIN public.subscriptions s ON s.tenant_id = t.id AND s.status IN ('active','trialing')
         WHERE s.id IS NULL
           AND DATE(t.trial_ends_at) = DATE(NOW() + INTERVAL '${b.days_until} days')
           AND NOT EXISTS (
             SELECT 1 FROM public.email_log el
              WHERE el.tenant_id = t.id
                AND el.template = '${b.template}'
                AND el.sent_at > NOW() - INTERVAL '20 hours'
           )`,
    );
    for (const tenant of r.rows) {
      try {
        await sendEmail({
          to: tenant.email,
          subject: b.subject,
          template: b.template,
          vars: { tenant_slug: tenant.slug, trial_ends_at: tenant.trial_ends_at,
                  upgrade_url: `https://${tenant.slug}.zietra.com/billing/upgrade` },
        });
        await pool.query(
          `INSERT INTO public.email_log (tenant_id, template, sent_at) VALUES ($1, $2, NOW())`,
          [tenant.id, b.template]);
        sent++;
      } catch (e) {
        console.error('[trial-reminders] send failed', tenant.id, e);
        skipped++;
      }
    }
  }
  return { sent, skipped };
}
```

```bash
#!/usr/bin/env bash
set -euo pipefail
source /Users/jeet/doordash-p2p/.planning/phases/54.6-enterprise-hardening-starter-pack-vpc-rds-proxy-waf-guardduty-close-sg/vpc-migration.env
REGION=us-east-1

# 1. IAM role for proxy (get-or-create)
ROLE_NAME=zietra-rds-proxy-role
if ! aws iam get-role --role-name $ROLE_NAME --query 'Role.Arn' --output text 2>/dev/null; then
  cat > /tmp/proxy-trust.json <<'EOF'
{"Version":"2012-10-17","Statement":[{"Effect":"Allow","Principal":{"Service":"rds.amazonaws.com"},"Action":"sts:AssumeRole"}]}
EOF
  aws iam create-role --role-name $ROLE_NAME \
    --assume-role-policy-document file:///tmp/proxy-trust.json \
    --tags Key=Phase,Value=54.6 --region $REGION
fi
PROXY_ROLE_ARN=$(aws iam get-role --role-name $ROLE_NAME --query 'Role.Arn' --output text)

# 2. Inline policy: get-secret-value on rds!cluster-* + kms:Decrypt via secretsmanager
cat > /tmp/proxy-policy.json <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": ["secretsmanager:GetSecretValue"],
      "Resource": "arn:aws:secretsmanager:us-east-1:134607809447:secret:rds!cluster-*"
    },
    {
      "Effect": "Allow",
      "Action": ["kms:Decrypt"],
      "Resource": "arn:aws:kms:us-east-1:134607809447:key/*",
      "Condition": {"StringEquals": {"kms:ViaService": "secretsmanager.us-east-1.amazonaws.com"}}
    }
  ]
}
EOF
aws iam put-role-policy --role-name $ROLE_NAME \
  --policy-name secrets-and-kms --policy-document file:///tmp/proxy-policy.json

# 3. Create proxy (get-or-create by name)
PROXY_NAME=zietra-aurora-proxy
if ! aws rds describe-db-proxies --db-proxy-name $PROXY_NAME --region $REGION 2>/dev/null; then
  aws rds create-db-proxy \
    --db-proxy-name $PROXY_NAME \
    --engine-family POSTGRESQL \
    --auth "AuthScheme=SECRETS,SecretArn=$NEW_MASTER_SECRET_ARN,IAMAuth=DISABLED" \
    --role-arn $PROXY_ROLE_ARN \
    --vpc-subnet-ids $PRIV_1A $PRIV_1B \
    --vpc-security-group-ids $PROXY_SG \
    --require-tls \
    --idle-client-timeout 1800 \
    --tags Key=Phase,Value=54.6 \
    --region $REGION
  aws rds wait db-proxy-available --db-proxy-name $PROXY_NAME --region $REGION
fi

# NOTE: IAMAuth=DISABLED per critical_constraints #2 — clients use password auth toward Proxy.
# Proxy still uses Secrets Manager toward Aurora. IAM token client conversion deferred to M3.

# 4. Register Aurora cluster as proxy target (get-or-register)
if ! aws rds describe-db-proxy-targets --db-proxy-name $PROXY_NAME --region $REGION \
  --query 'Targets[?RdsResourceId==`'$NEW_CLUSTER'`]' --output text | grep -q .; then
  aws rds register-db-proxy-targets \
    --db-proxy-name $PROXY_NAME \
    --db-cluster-identifiers $NEW_CLUSTER \
    --region $REGION
fi

# 5. Wait for target health
for i in $(seq 1 30); do
  STATE=$(aws rds describe-db-proxy-targets --db-proxy-name $PROXY_NAME --region $REGION \
    --query 'Targets[0].TargetHealth.State' --output text 2>/dev/null || echo INIT)
  [ "$STATE" = "AVAILABLE" ] && break
  sleep 5
done

# 6. Capture proxy endpoint
PROXY_ENDPOINT=$(aws rds describe-db-proxies --db-proxy-name $PROXY_NAME \
  --query 'DBProxies[0].Endpoint' --output text --region $REGION)
echo "PROXY_ENDPOINT=$PROXY_ENDPOINT"   # expect zietra-aurora-proxy.proxy-XXXXX.us-east-1.rds.amazonaws.com
```

Build `scripts/smoke-phase-54-6.sh` — a cross-cutting verification script that probes EVERY Phase 54.6 surface. Mirror the structure of `scripts/aurora-cutover-smoke.sh` (used in 54.5-03).

```bash
#!/usr/bin/env bash
set -uo pipefail
# Do NOT use -e — we want to collect all failures, exit non-zero at the end.

FAILS=0
log() { printf "[%s] %s\n" "$1" "$2"; }
pass() { log "PASS" "$1"; }
fail() { log "FAIL" "$1"; FAILS=$((FAILS+1)); }

# 1. Lambda health: 4 Lambdas should return 200 via /api/health
for L in turion-demo-api turion-satellite-api zietra-crm-api zietra-api; do
  RC=$(aws lambda invoke --function-name $L \
    --payload '{"rawPath":"/api/health","requestContext":{"http":{"method":"GET"}}}' \
    --region us-east-1 --cli-binary-format raw-in-base64-out \
    /tmp/sm-$L.json --query StatusCode --output text 2>/dev/null || echo 500)
  BODY=$(cat /tmp/sm-$L.json | jq -r '.statusCode // .body // .errorMessage' 2>/dev/null || echo error)
  if [ "$RC" = "200" ] && echo "$BODY" | grep -qiE "200|ok|healthy"; then
    pass "Lambda $L /api/health"
  else
    fail "Lambda $L /api/health (RC=$RC BODY=$BODY)"
  fi
done

# 2. CloudFront distros: 3 should return 200 + have x-amz-cf-id header
for HOST in turionspace.zietra.com marquee.zietra.com asc606.zietra.com; do
  HEAD=$(curl -sI "https://$HOST/" || echo "")
  if echo "$HEAD" | head -1 | grep -qE "HTTP/[12]\\.?[01]? 200"; then
    if echo "$HEAD" | grep -qi "x-amz-cf-id:"; then
      pass "CloudFront $HOST"
    else
      fail "CloudFront $HOST (200 but missing x-amz-cf-id)"
    fi
  else
    fail "CloudFront $HOST"
  fi
done

# 3. Trust page reachable
TRUST_RC=$(curl -s -o /dev/null -w "%{http_code}" https://zietra.com/security.html)
if [ "$TRUST_RC" = "200" ]; then
  pass "zietra.com/security.html"
else
  fail "zietra.com/security.html (RC=$TRUST_RC)"
fi

# 4. Old Aurora endpoint should be UNREACHABLE from public internet (proves SG hardening worked)
OLD_ENDPOINT=zietra-aurora-prod.cluster-c23qcukqe810.us-east-1.rds.amazonaws.com
if timeout 8 bash -c "exec 3<>/dev/tcp/$OLD_ENDPOINT/5432" 2>/dev/null; then
  fail "Old Aurora endpoint $OLD_ENDPOINT:5432 is REACHABLE from public — SG hardening failed"
else
  pass "Old Aurora endpoint blocked from public internet (timeout/refused as expected)"
fi

# 5. New Aurora cluster endpoint should also be unreachable from public (no public IP)
source /Users/jeet/doordash-p2p/.planning/phases/54.6-enterprise-hardening-starter-pack-vpc-rds-proxy-waf-guardduty-close-sg/vpc-migration.env
if timeout 8 bash -c "exec 3<>/dev/tcp/$NEW_ENDPOINT/5432" 2>/dev/null; then
  fail "New Aurora endpoint $NEW_ENDPOINT:5432 is REACHABLE from public — should be private-only"
else
  pass "New Aurora endpoint blocked from public internet"
fi

# 6. RDS Proxy endpoint should also be unreachable from public
if timeout 8 bash -c "exec 3<>/dev/tcp/$PROXY_ENDPOINT/5432" 2>/dev/null; then
  fail "RDS Proxy endpoint $PROXY_ENDPOINT:5432 is REACHABLE from public — should be private-only"
else
  pass "RDS Proxy endpoint blocked from public internet"
fi

# 7. GuardDuty detector enabled
DET_STATUS=$(aws guardduty get-detector \
  --detector-id $(aws guardduty list-detectors --region us-east-1 --query 'DetectorIds[0]' --output text) \
  --region us-east-1 --query Status --output text 2>/dev/null)
[ "$DET_STATUS" = "ENABLED" ] && pass "GuardDuty ENABLED" || fail "GuardDuty status=$DET_STATUS"

# 8. Security Hub has ≥2 standards subscribed
STD_CT=$(aws securityhub get-enabled-standards --region us-east-1 \
  --query 'length(StandardsSubscriptions)' --output text 2>/dev/null || echo 0)
[ "$STD_CT" -ge 2 ] && pass "Security Hub $STD_CT standards" || fail "Security Hub $STD_CT standards (need ≥2)"

# 9. AWS Config recorder recording=True
REC=$(aws configservice describe-configuration-recorder-status --region us-east-1 \
  --query 'ConfigurationRecordersStatus[0].recording' --output text 2>/dev/null)
[ "$REC" = "True" ] && pass "AWS Config recording" || fail "AWS Config recording=$REC"

# 10. WAFv2 Web ACL exists + 3 associations
ACL_ARN=$(aws wafv2 list-web-acls --scope CLOUDFRONT --region us-east-1 \
  --query "WebACLs[?Name=='zietra-prod-waf'].ARN | [0]" --output text 2>/dev/null)
if [ -n "$ACL_ARN" ] && [ "$ACL_ARN" != "None" ]; then
  ASSOC_CT=$(aws wafv2 list-resources-for-web-acl --web-acl-arn $ACL_ARN \
    --resource-type CLOUDFRONT --region us-east-1 --query 'length(ResourceArns)' --output text)
  [ "$ASSOC_CT" = "3" ] && pass "WAF Web ACL + 3 associations" || fail "WAF associations=$ASSOC_CT (need 3)"
else
  fail "WAF Web ACL zietra-prod-waf not found"
fi

# Final verdict
echo
if [ $FAILS -eq 0 ]; then
  echo "ALL CHECKS PASS — Phase 54.6 surface verified end-to-end"
  exit 0
else
  echo "FAILED $FAILS check(s) — investigate before declaring phase complete"
  exit 1
fi
```

```bash
#!/usr/bin/env bash
set -euo pipefail
REGION=us-east-1

# Helper: get-or-create-vpc by Name tag
get_or_create_vpc() {
  local existing
  existing=$(aws ec2 describe-vpcs --filters Name=tag:Name,Values=zietra-prod-vpc \
    --query 'Vpcs[0].VpcId' --output text --region $REGION 2>/dev/null || echo None)
  if [ "$existing" != "None" ] && [ "$existing" != "null" ] && [ -n "$existing" ]; then
    echo "$existing"
    return
  fi
  aws ec2 create-vpc --cidr-block 10.0.0.0/16 \
    --tag-specifications "ResourceType=vpc,Tags=[{Key=Name,Value=zietra-prod-vpc},{Key=Phase,Value=54.6},{Key=Environment,Value=production}]" \
    --region $REGION --query 'Vpc.VpcId' --output text
}

VPC_ID=$(get_or_create_vpc)
aws ec2 modify-vpc-attribute --vpc-id $VPC_ID --enable-dns-hostnames --region $REGION
aws ec2 modify-vpc-attribute --vpc-id $VPC_ID --enable-dns-support --region $REGION

# IGW (get-or-create-by-name)
# Subnets x4 (get-or-create-by-name) — see RESEARCH §A.3 for full commands
# NAT GW + EIP (get-or-create-by-name, single in public-1a)
# Route tables (public → IGW, private → NAT) + associations
# 3 Security Groups (zietra-prod-lambda-sg, zietra-prod-rds-proxy-sg, zietra-prod-aurora-sg)
#   - lambda-sg: NO ingress, default egress allow-all
#   - proxy-sg: 5432 ingress from lambda-sg
#   - aurora-sg: 5432 ingress from proxy-sg ONLY (NO 0.0.0.0/0, NO operator IP)

# At end, echo all values for runbook capture
echo "VPC_ID=$VPC_ID"
echo "PUB_1A=$PUB_1A"
echo "PUB_1B=$PUB_1B"
echo "PRIV_1A=$PRIV_1A"
echo "PRIV_1B=$PRIV_1B"
echo "IGW_ID=$IGW_ID"
echo "NAT_ID=$NAT_ID"
echo "LAMBDA_SG=$LAMBDA_SG"
echo "PROXY_SG=$PROXY_SG"
echo "AURORA_NEW_SG=$AURORA_NEW_SG"
```

# Source VPC env from Task 1 runbook captures
source /tmp/vpc-provision.log   # exports VPC_ID, PRIV_1A, PRIV_1B, AURORA_NEW_SG, etc.

SNAP_ID=zietra-aurora-pre-vpc-migration-$(date -u +%Y-%m-%d)
NEW_CLUSTER=zietra-aurora-prod-v2

# 1. Create DB subnet group in new VPC
aws rds create-db-subnet-group \
  --db-subnet-group-name zietra-aurora-private-subnets \
  --db-subnet-group-description "Zietra Aurora in zietra-prod-vpc private subnets (Phase 54.6)" \
  --subnet-ids $PRIV_1A $PRIV_1B \
  --tags Key=Project,Value=Zietra Key=Phase,Value=54.6 \
  --region us-east-1

# 2. Restore cluster INTO the private VPC subnet group
aws rds restore-db-cluster-from-snapshot \
  --db-cluster-identifier $NEW_CLUSTER \
  --snapshot-identifier $SNAP_ID \
  --engine aurora-postgresql \
  --engine-version 16.4 \
  --db-subnet-group-name zietra-aurora-private-subnets \
  --vpc-security-group-ids $AURORA_NEW_SG \
  --serverless-v2-scaling-configuration MinCapacity=0.5,MaxCapacity=4 \
  --enable-iam-database-authentication \
  --enable-cloudwatch-logs-exports '["postgresql"]' \
  --storage-encrypted \
  --tags Key=Project,Value=Zietra Key=Phase,Value=54.6 Key=Environment,Value=production \
  --region us-east-1

# Wait for cluster available
aws rds wait db-cluster-available --db-cluster-identifier $NEW_CLUSTER --region us-east-1

# 3. Rotate the new cluster's master password to a fresh secret (snapshot-restore PRESERVES old password,

aws rds modify-db-cluster \
  --db-cluster-identifier $NEW_CLUSTER \
  --manage-master-user-password \
  --apply-immediately \
  --region us-east-1
aws rds wait db-cluster-available --db-cluster-identifier $NEW_CLUSTER --region us-east-1

# 4. Create writer instance (db.serverless, NOT publicly-accessible)
aws rds create-db-instance \
  --db-cluster-identifier $NEW_CLUSTER \
  --db-instance-identifier ${NEW_CLUSTER}-writer \
  --db-instance-class db.serverless \
  --engine aurora-postgresql \
  --no-publicly-accessible \
  --tags Key=Project,Value=Zietra Key=Phase,Value=54.6 \
  --region us-east-1
aws rds wait db-instance-available --db-instance-identifier ${NEW_CLUSTER}-writer --region us-east-1

# 5. Capture new endpoint + new master secret ARN
NEW_ENDPOINT=$(aws rds describe-db-clusters --db-cluster-identifier $NEW_CLUSTER \
  --query 'DBClusters[0].Endpoint' --output text --region us-east-1)
NEW_MASTER_SECRET_ARN=$(aws rds describe-db-clusters --db-cluster-identifier $NEW_CLUSTER \
  --query 'DBClusters[0].MasterUserSecret.SecretArn' --output text --region us-east-1)
echo "NEW_ENDPOINT=$NEW_ENDPOINT"
echo "NEW_MASTER_SECRET_ARN=$NEW_MASTER_SECRET_ARN"

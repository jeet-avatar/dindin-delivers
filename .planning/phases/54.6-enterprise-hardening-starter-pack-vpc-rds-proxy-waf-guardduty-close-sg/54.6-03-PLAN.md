```bash
#!/usr/bin/env bash
set -euo pipefail
REGION=us-east-1

# 1. Get-or-create detector
DET=$(aws guardduty list-detectors --region $REGION --query 'DetectorIds[0]' --output text 2>/dev/null || echo None)
if [ "$DET" = "None" ] || [ -z "$DET" ]; then
  DET=$(aws guardduty create-detector \
    --enable \
    --finding-publishing-frequency FIFTEEN_MINUTES \
    --data-sources '{"S3Logs": {"Enable": true}, "Kubernetes": {"AuditLogs": {"Enable": false}}, "MalwareProtection": {"ScanEc2InstanceWithFindings": {"EbsVolumes": false}}}' \
    --tags Phase=54.6 \
    --region $REGION --query 'DetectorId' --output text)
fi
echo "DETECTOR_ID=$DET"

# 2. Get-or-create SNS topic
SNS_ARN=$(aws sns list-topics --region $REGION \
  --query "Topics[?contains(TopicArn, 'zietra-security-findings')].TopicArn | [0]" --output text)
if [ -z "$SNS_ARN" ] || [ "$SNS_ARN" = "None" ]; then
  SNS_ARN=$(aws sns create-topic --name zietra-security-findings --region $REGION \
    --query 'TopicArn' --output text)
fi
echo "SNS_ARN=$SNS_ARN"

# 3. Subscribe security@zietra.com (idempotent: check first)
EXISTING_SUB=$(aws sns list-subscriptions-by-topic --topic-arn $SNS_ARN --region $REGION \
  --query "Subscriptions[?Endpoint=='security@zietra.com'].SubscriptionArn | [0]" --output text)
if [ -z "$EXISTING_SUB" ] || [ "$EXISTING_SUB" = "None" ] || [ "$EXISTING_SUB" = "PendingConfirmation" ]; then
  aws sns subscribe --topic-arn $SNS_ARN --protocol email \
    --notification-endpoint security@zietra.com --region $REGION
  echo "Subscription created for security@zietra.com — operator must confirm via email link"
fi

# 4. EventBridge rule: severity >=7.0 → SNS
cat > /tmp/eb-rule.json <<'EOF'
{
  "source": ["aws.guardduty"],
  "detail-type": ["GuardDuty Finding"],
  "detail": {"severity": [{"numeric": [">=", 7.0]}]}
}
EOF
aws events put-rule --name zietra-gd-high-severity --region $REGION \
  --event-pattern file:///tmp/eb-rule.json

aws events put-targets --rule zietra-gd-high-severity --region $REGION \
  --targets "Id=1,Arn=$SNS_ARN"

# 5. Grant EventBridge permission to publish to SNS (idempotent: get-topic-attributes for policy, append if missing)
SID=AllowEventBridgePublish
aws sns set-topic-attributes --topic-arn $SNS_ARN \
  --attribute-name Policy \
  --attribute-value "{
    \"Version\": \"2012-10-17\",
    \"Statement\": [{
      \"Sid\": \"$SID\",
      \"Effect\": \"Allow\",
      \"Principal\": {\"Service\": \"events.amazonaws.com\"},
      \"Action\": \"sns:Publish\",
      \"Resource\": \"$SNS_ARN\"
    }]
  }" --region $REGION
```

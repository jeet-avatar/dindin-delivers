```bash
#!/usr/bin/env bash
# scripts/smoke-erp-pages.sh — Phase 65.3 ERP-page tenant matrix smoke
set -uo pipefail
WEB_TURION="https://turion.zietra.com"
WEB_SB="https://solobrands.zietra.com"
WEB_EMPTY="https://test65empty.zietra.com"

# Counters
PASS=0; FAIL=0; WARN=0
ok()   { echo "PASS $*"; PASS=$((PASS+1)); }
fail() { echo "FAIL $*"; FAIL=$((FAIL+1)); }
warn() { echo "WARN $*"; WARN=$((WARN+1)); }

  [arena-bom]="SAT-001\|ADCS"

  [executive-cockpit]="Solo Brands"

SB_FORBIDDEN="SAT-001\|ADCS-RW-MEDIUM\|Blue Canyon\|PROJ-ANDROMEDA\|CLIN-004\|BCT-RW-\|Cygnus\|Honeywell · Aerospace"

    "$WEB_TURION/${page}.html" -o "/tmp/${page}-t.html"
  if [ -n "${TURION_TOKENS[$page]:-}" ]; then
    grep -Eq "${TURION_TOKENS[$page]}" "/tmp/${page}-t.html" \
      && ok "Turion ${page} contains expected Turion token" \
      || warn "Turion ${page} missing expected Turion token (data-bound; visible in browser only)"

    [ -s "/tmp/${page}-t.html" ] && ok "Turion ${page} returns non-empty HTML"

    "$WEB_SB/${page}.html" -o "/tmp/${page}-s.html"

  else
    ok "Solo Brands ${page} no Turion tokens in static shell"
  fi

  curl -sf -H "Authorization: Bearer $EMPTY_TOKEN" -H "X-Tenant-Slug: test65empty" \
    "$WEB_EMPTY/${page}.html" -o "/tmp/${page}-e.html"
  [ -s "/tmp/${page}-e.html" ] && ok "Empty tenant ${page} returns non-empty HTML"

# Hybrid Puppeteer smoke for the 5 heaviest pages (where dynamic content matters per RESEARCH §Hybrid smoke)

echo
echo "=== Puppeteer dynamic-content checks (5 heavy pages × 3 tenants) ==="
for tenant in turion solobrands test65empty; do
  case $tenant in
    turion) export COGNITO_ID_TOKEN=$TURION_TOKEN;;
    solobrands) export COGNITO_ID_TOKEN=$SB_TOKEN;;
    test65empty) export COGNITO_ID_TOKEN=$EMPTY_TOKEN;;
  esac
  node scripts/snapshot-turion-erp-views.mjs "$tenant" > /tmp/snap-${tenant}-heavies.json 2>/dev/null || warn "snapshot failed for $tenant"
  # Assert per-page tokens
  for p in "${HEAVIES[@]}"; do
    if [ "$tenant" = "solobrands" ]; then
      tokens=$(jq -r ".pages[\"$p\"].hardcodedTokens | length" /tmp/snap-${tenant}-heavies.json)
      [ "$tokens" = "0" ] && ok "Puppeteer: Solo Brands $p — 0 hardcoded Turion tokens" || fail "Puppeteer: Solo Brands $p LEAKS $tokens Turion tokens"
    fi
    if [ "$tenant" = "test65empty" ]; then
      empties=$(jq -r ".pages[\"$p\"].rowCounts.emptyStates" /tmp/snap-${tenant}-heavies.json)
      [ "$empties" -gt 0 ] && ok "Puppeteer: Empty tenant $p shows empty-state" || warn "Puppeteer: Empty tenant $p has no empty-state element (may be acceptable for platform pages)"
    fi
    if [ "$tenant" = "turion" ]; then
      bodyLen=$(jq -r ".pages[\"$p\"].bodyText | length" /tmp/snap-${tenant}-heavies.json)
      [ "$bodyLen" -gt 500 ] && ok "Puppeteer: Turion $p has $bodyLen bytes of content" || fail "Puppeteer: Turion $p too sparse ($bodyLen bytes)"
    fi
  done
done

elif [ $WARN -gt 0 ]; then
  echo "VERDICT: YELLOW — $WARN documented warnings"
  exit 0
else
  echo "VERDICT: GREEN — all clean"
```

**Step 2.2 — Diff PRE vs POST for Turion (TurionDataPreservedCheck).** Per page, compute line-overlap %:
```bash
node -e "
const before = require('/Users/jeet/doordash-p2p/.planning/phases/65.3-tenant-data-driven-erp-detail-pages/65.3-01-TURION-SNAPSHOT-PRE.json');
const after  = require('/Users/jeet/doordash-p2p/.planning/phases/65.3-tenant-data-driven-erp-detail-pages/65.3-05-TURION-SNAPSHOT-POST.json');
const results = [];
const fails = [];
for (const page of Object.keys(before.pages)) {
  const b = before.pages[page]; const a = after.pages[page];
  if (!a) { fails.push({page, reason: 'missing in POST'}); continue; }
  const bL = new Set(b.bodyText.split('\n').map(s => s.trim()).filter(s => s.length > 4));
  const aL = new Set(a.bodyText.split('\n').map(s => s.trim()).filter(s => s.length > 4));
  const overlap = [...bL].filter(l => aL.has(l)).length;
  const pct = bL.size ? (overlap / bL.size) * 100 : 100;
  const threshold = ['netsuite-financials','netsuite-bs','netsuite-pl','netsuite-tb'].includes(page) ? 60 : 75;
  results.push({page, pct: pct.toFixed(1), threshold, status: pct >= threshold ? 'PASS' : 'FAIL'});
  if (pct < threshold) fails.push({page, pct: pct.toFixed(1), threshold});
}
require('fs').writeFileSync('/tmp/turion-diff.json', JSON.stringify({results, fails}, null, 2));
console.log('TURION DIFF:', results.filter(r => r.status === 'PASS').length, '/', results.length, 'PASS');
if (fails.length > 0) { console.error('FAILS:', JSON.stringify(fails, null, 2)); process.exit(1); }
"
```

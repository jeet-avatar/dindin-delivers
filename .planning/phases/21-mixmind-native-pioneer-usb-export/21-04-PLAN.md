Step 2 — DeviceSQL string encoder. Three variants per Deep Symmetry spec:

```python
def encode_devicesql(s: str) -> bytes:
    """DeviceSQL string format used in PDB rows.
    
    Three variants based on length and character set:
    - Short ASCII (<=126 chars, all ASCII): 1-byte length flag + payload + NUL
    - Long ASCII: marker 0x40 + 4-byte length + payload + NUL  
    - UTF-16: marker 0x90 + 4-byte byte-length + UTF-16-LE payload + NUL NUL
    """
    if not s:
        return b"\x03"  # empty short string per spec
    try:
        ascii_bytes = s.encode("ascii")
        if len(ascii_bytes) <= 126:
            # Short ASCII: high bit clear, length << 1 | 1
            return bytes([(len(ascii_bytes) << 1) | 1]) + ascii_bytes + b"\x00"
        else:
            # Long ASCII: 0x40 marker + Int32ul total length
            length = 4 + len(ascii_bytes) + 1  # 4-byte len + payload + NUL
            return b"\x40" + length.to_bytes(4, "little") + ascii_bytes + b"\x00"
    except UnicodeEncodeError:
        # UTF-16 LE
        utf16 = s.encode("utf-16-le")
        length = 4 + len(utf16) + 2
        return b"\x90" + length.to_bytes(4, "little") + utf16 + b"\x00\x00"

def _self_test_strings():
    # Round-trip: encode + decode by re-reading the length flag
    for s in ["", "Track Title", "A" * 200, "ñoño", "日本語"]:
        encoded = encode_devicesql(s)
        # First byte tells us the variant
        first = encoded[0]
        if first == 0x90:
            assert s.encode("utf-16-le") in encoded
        elif first == 0x40:
            assert s.encode("ascii") in encoded
        elif (first & 1) == 1:
            assert s.encode("ascii") in encoded
        print(f"OK: {s!r} -> {len(encoded)} bytes (variant={hex(first)})")

if __name__ == "__main__":
    _self_test_strings()
```

Step 3 — Create `usb_layout.py`. Pure functions only — no I/O:

```python
"""USB filesystem layout: ID minting + analyze path generation.

CDJ-3000 expects: PIONEER/USBANLZ/Pxxx/yyyyyyyy/ANLZ0000.{DAT,EXT}
where Pxxx = upper 12 bits of track ID as 3-char hex,
      yyyyyyyy = full 32-bit track ID as 8-char hex (lowercase).

Track IDs are sequential 32-bit unsigned integers minted starting at 0x10000000
to avoid collisions with any existing IDs in the user's local Rekordbox library
(which mints from 0). See RESEARCH.md Pitfall 8.
"""
from pathlib import Path
from threading import Lock

_TRACK_ID_BASE = 0x10000000
_id_counter = _TRACK_ID_BASE
_id_lock = Lock()


def reset_id_counter(start: int = _TRACK_ID_BASE) -> None:
    """Reset the in-process counter — used by tests and at start of each export."""
    global _id_counter
    with _id_lock:
        _id_counter = start


def mint_track_id() -> int:
    """Mint the next sequential track ID. Thread-safe."""
    global _id_counter
    with _id_lock:
        tid = _id_counter
        _id_counter += 1
    return tid


def analyze_path(track_id: int) -> str:
    """Convert a track ID to its CDJ analyze path: 'Pxxx/yyyyyyyy'.
    
    Pxxx = 'P' + (track_id >> 20) as 3-char hex (covers upper 12 bits)
    yyyyyyyy = track_id as 8-char lowercase hex
    """
    bucket = f"P{(track_id >> 20):03X}"
    full = f"{track_id:08x}"
    return f"{bucket}/{full}"


def usbanlz_dir(track_id: int, usb_root: Path) -> Path:
    """Full directory containing ANLZ0000.{DAT,EXT} for one track."""
    return usb_root / "PIONEER" / "USBANLZ" / analyze_path(track_id)


_UNKNOWN_ARTIST = "UnknownArtist"
_UNKNOWN_ALBUM = "UnknownAlbum"


def _sanitize_path_segment(s: str) -> str:
    """Pioneer sanitisation (observed on /Volumes/Untitled/Contents/):
    - Only `/` → `_` (filesystem requirement)
    - Apostrophes, spaces, unicode, special chars PRESERVED
    - Empty string → caller decides UnknownArtist/UnknownAlbum sentinel
    """
    return s.replace("/", "_") if s else ""


def audio_path_on_usb(
    artist: str | None,
    album: str | None,
    original_filename: str,
    usb_root: Path,
) -> Path:
    """Where to copy the audio file on the USB.

    Pioneer convention (verified against real CDJ-ready USB at /Volumes/Untitled/):
        Contents/<Artist>/<Album>/<original_filename>

    - Missing artist → `UnknownArtist` literal
    - Missing album  → `UnknownAlbum` literal
    - Original filename preserved (no renaming) — CDJ relies on the PDB
      Track row's file_path field being exact.
    - Only `/` → `_` substitution observed; all other chars preserved.

    See 21-REFERENCE-DATASET.md for the audio path convention derivation.
    """
    artist_seg = _sanitize_path_segment(artist) or _UNKNOWN_ARTIST
    album_seg = _sanitize_path_segment(album) or _UNKNOWN_ALBUM
    return usb_root / "Contents" / artist_seg / album_seg / original_filename
```

```python
Output: PIONEER/rekordbox/export.pdb that pyrekordbox.Rekordbox6Database can
parse and the CDJ-3000 firmware can read.

def write_pdb(
    out_path: Path,
    tracks: list[TrackRecord],
    playlists: Optional[list[PlaylistRecord]] = None,
) -> dict:
    """Write a Rekordbox-compatible export.pdb file.
    
    Returns a dict {tables_written: [...], total_pages: N, file_size: bytes}.
    """
    playlists = playlists or [_default_all_tracks_playlist(tracks)]
    
    # Step 1: Resolve cross-table foreign keys (artists, albums, genres, keys, colors)
    artists = _collect_artists(tracks)        # dict[name -> id]
    albums = _collect_albums(tracks)
    genres = _collect_genres(tracks)
    keys = _collect_keys(tracks)
    colors = _collect_colors(tracks)
    
    # Step 2: Build pages per table
    pages: list[bytes] = []
    table_infos: list[dict] = []
    
    table_infos.append(_build_table(pages, PAGE_TYPE_TRACKS, _serialize_track_rows(tracks, artists, albums, genres, keys)))
    table_infos.append(_build_table(pages, PAGE_TYPE_GENRES, _serialize_simple_rows(genres, GenreRow)))
    table_infos.append(_build_table(pages, PAGE_TYPE_ARTISTS, _serialize_artist_rows(artists)))
    table_infos.append(_build_table(pages, PAGE_TYPE_ALBUMS, _serialize_album_rows(albums, artists)))
    table_infos.append(_build_table(pages, PAGE_TYPE_KEYS, _serialize_simple_rows(keys, KeyRow)))
    table_infos.append(_build_table(pages, PAGE_TYPE_COLORS, _serialize_color_rows(colors)))
    table_infos.append(_build_table(pages, PAGE_TYPE_PLAYLISTS, _serialize_playlist_rows(playlists)))
    table_infos.append(_build_table(pages, PAGE_TYPE_PLAYLIST_ENTRIES, _serialize_playlist_entry_rows(playlists)))
    
    # Step 3: Build header
    header = PdbHeader.build({
        "unknown1": 0,
        "len_page": PAGE_SIZE,
        "num_tables": 20,        # tables 0..19
        "next_unused_page": len(pages),
        "unknown2": 0,
        "sequence": 1,
        "gap": b"\x00" * 4,
    })
    
    # Step 4: Pad table info section to 20 entries (empty for unused page types)
    info_bytes = b""
    for ti in table_infos:
        info_bytes += TableInfo.build(ti)
    # Pad remaining slots with zero TableInfo (empty/unused tables)
    empty_ti = TableInfo.build({"type": 0, "empty_candidate": 0, "first_page": 0, "last_page": 0})
    while len(info_bytes) < 20 * TableInfo.sizeof():
        info_bytes += empty_ti
    
    # Step 5: Concatenate header + table_infos + pages → file
    out_path.parent.mkdir(parents=True, exist_ok=True)
    with open(out_path, "wb") as f:
        f.write(header)
        f.write(info_bytes)
        # Pad first page to PAGE_SIZE
        first_page_pad = PAGE_SIZE - (len(header) + len(info_bytes))
        if first_page_pad > 0:
            f.write(b"\x00" * first_page_pad)
        # All other pages
        for page_bytes in pages:
            assert len(page_bytes) == PAGE_SIZE, f"page must be {PAGE_SIZE} bytes, got {len(page_bytes)}"
            f.write(page_bytes)
    
    return {
        "tables_written": [PAGE_TYPE_TRACKS, PAGE_TYPE_GENRES, PAGE_TYPE_ARTISTS,
                           PAGE_TYPE_ALBUMS, PAGE_TYPE_KEYS, PAGE_TYPE_COLORS,
                           PAGE_TYPE_PLAYLISTS, PAGE_TYPE_PLAYLIST_ENTRIES],
        "total_pages": len(pages) + 1,    # +1 for the header page
        "file_size": out_path.stat().st_size,
    }

# Internal helpers

def _build_table(pages: list[bytes], page_type: int, row_blobs: list[bytes]) -> dict:
    """Pack rows into pages, append to `pages` list, return TableInfo dict.
    
    Each page is exactly PAGE_SIZE bytes:
      [PageHeader (40 bytes)] [row data ...] [padding to PAGE_SIZE] [row offset table at end]
    """
    if not row_blobs:
        # Empty table: write one empty page
        empty = _build_empty_page(page_type, len(pages))
        first_page = len(pages) + 1   # +1 for the header page that's at index 0 logically
        pages.append(empty)
        return {"type": page_type, "empty_candidate": first_page, "first_page": first_page, "last_page": first_page}
    
    header_size = PageHeader.sizeof()  # 40 bytes
    first_page_idx = len(pages) + 1
    current_page_rows = []
    current_used = 0
    
    for blob in row_blobs:
        # Each row also needs a 2-byte offset entry at end of page
        if header_size + current_used + len(blob) + (len(current_page_rows) + 1) * 2 > PAGE_SIZE:
            # Flush current page
            pages.append(_pack_page(page_type, len(pages), current_page_rows))
            current_page_rows = []
            current_used = 0
        current_page_rows.append(blob)
        current_used += len(blob)
    
    # Flush last page
    if current_page_rows:
        pages.append(_pack_page(page_type, len(pages), current_page_rows))
    
    last_page_idx = len(pages)
    return {"type": page_type, "empty_candidate": first_page_idx,
            "first_page": first_page_idx, "last_page": last_page_idx}


def _pack_page(page_type: int, page_index: int, rows: list[bytes]) -> bytes:
    """Pack rows into a single PAGE_SIZE-byte page.
    
    Layout: [PageHeader (40B)] [row1] [row2] ... [padding] [offset_table at end]
    Offset table: one Int16ul per row, pointing from page start to row start.
    """
    header_size = PageHeader.sizeof()
    rows_data = b"".join(rows)
    offsets = []
    cursor = header_size
    for r in rows:
        offsets.append(cursor)
        cursor += len(r)
    
    # Offset table: stored at end of page, last row's offset first (LIFO)
    offset_table = b"".join(o.to_bytes(2, "little") for o in reversed(offsets))
    
    used_size = len(rows_data) + len(offset_table)
    free_size = PAGE_SIZE - header_size - used_size
    
    if free_size < 0:
        raise ValueError(f"Rows don't fit in page: used={used_size}, available={PAGE_SIZE - header_size}")
    
    header_bytes = PageHeader.build({
        "page_index": page_index,
        "page_type": page_type,
        "next_page": 0,                # 0 = last page in chain
        "unknown1": 0, "unknown2": 0,
        "num_rows_small": min(len(rows), 0xFF),
        "unknown3": 0, "unknown4": 1,
        "page_flags": 0x24,            # data page
        "free_size": free_size,
        "used_size": used_size,
        "unknown5": 0,
        "num_rows_large": len(rows) if len(rows) > 0xFF else 0x100,  # see RESEARCH.md Pitfall 6
        "unknown6": 0, "unknown7": 0,
    })
    
    return header_bytes + rows_data + (b"\x00" * free_size) + offset_table

    return _pack_page(page_type, page_index, [])

def _serialize_track_rows(tracks, artists, albums, genres, keys) -> list[bytes]:
    """Build TrackRow byte blobs. Each row = fixed-size header + 21 string offsets + DeviceSQL strings."""
    blobs = []
    for i, t in enumerate(tracks):
        # Compute string offsets — the 21 strings live at the END of the row,
        # offsets are Int16ul from row start to each string.
        strings = [
            t.isrc, "", "", "", "", "", "",     # isrc, texter, kuvo_public, mix_name, reserved, autoload_hot_cues, kuvo_artist_id
            "", "", "", str(t.sample_rate),     # date_added, release_date, mix2, sample_rate
            t.file_path_on_usb, t.file_name, t.title,    # file_path, filename, title
            "", t.comment, "", "", t.key_name, str(t.year), "",  # sample_rate2, comment, hot_cue_autoload, kuvo_pub2, key_text, year_text, ofa
        ]
        encoded_strings = [encode_devicesql(s) for s in strings]
        
        # Fixed-size header: ~84 bytes per Kaitai spec, then 21 × 2-byte offsets = 42 bytes
        # Total fixed = 126 bytes; strings start at offset 126 from row start.
        offsets = []
        cursor = 126
        for s in encoded_strings:
            offsets.append(cursor)
            cursor += len(s)
        
        header_dict = {
            "row_index": i,
            "bitmask": 0,                              # all fields present per spec
            "sample_rate": t.sample_rate,
            "composer_id": 0, "file_size": t.file_size,
            "unknown1": 3, "unknown2": 0, "unknown3": 0,
            "unknown4": 0, "artwork_id": 0,
            "key_id": t.key_id,
            "original_artist_id": 0,
            "label_id": 0, "remixer_id": 0,
            "bitrate": t.bitrate, "track_number": t.track_number,
            "tempo": int(round(t.bpm * 100)),
            "genre_id": t.genre_id,
            "album_id": t.album_id,
            "artist_id": t.artist_id,
            "id": t.id,
            "disc_number": 0, "play_count": 0,
            "year": t.year, "sample_depth": t.sample_depth,
            "duration": t.duration_sec, "unknown5": 41,
            "color_id": t.color_id, "rating": t.rating,
            "unknown6": 0, "unknown7": 0,
        }
        header_bytes = TrackRow.build(header_dict)
        offset_bytes = b"".join(o.to_bytes(2, "little") for o in offsets)
        string_bytes = b"".join(encoded_strings)
        blobs.append(header_bytes + offset_bytes + string_bytes)
    return blobs
```

```python
"""USB export orchestrator: stage → validate → atomic move.

Pattern: two-phase write per RESEARCH.md Architecture Pattern 1.
1. Build the entire PIONEER/ tree in ~/.mixmind/staging/{uuid}/
2. Validate by parsing back with pyrekordbox + anlz_parser
3. Pre-export USB clean: wipe target PIONEER/ to avoid Pitfall #8 ID collisions
4. Atomically move staging → /Volumes/{USB}/PIONEER/

License: MIT. No rbox import (Phase 21 Option C decision).
"""
from __future__ import annotations
import hashlib
import os
import shutil
import subprocess
import uuid as _uuid
from dataclasses import dataclass
from pathlib import Path
from typing import Optional

from pdb_writer import write_pdb, TrackRecord, PlaylistRecord
from anlz_writer import write_dat, write_ext, BeatEntry, CueEntry, SectionEntry
from usb_layout import mint_track_id, reset_id_counter, analyze_path, usbanlz_dir, audio_path_on_usb


# Filesystem support per CDJ-3000 spec (RESEARCH.md Pitfall 4)
SUPPORTED_FILESYSTEMS = {"msdos", "exfat", "hfs", "hfs+"}   # macOS diskutil names
UNSUPPORTED_FILESYSTEMS = {"ntfs"}
FAT32_NAMES = {"msdos", "fat32"}
FAT32_MAX_BYTES = 32 * 1024**3


@dataclass
class ExportResult:
    staging_dir: Path
    usb_target: Path
    exported_count: int
    validation_status: str       # "ok" | "warning" | "failed"
    validation_messages: list[str]
    bytes_written: int


def export_to_usb(
    track_records: list[dict],   # each dict carries title/artist/bpm/file_path/analysis_data
    usb_path: Path,
    playlists: Optional[list[dict]] = None,
    progress_cb=None,           # callable(step: str, current: int, total: int)
) -> ExportResult:
    """Main entry. track_records are loaded from state.py (StateDB) by the caller."""
    usb_path = Path(usb_path).resolve()
    
    # Step 1: Filesystem check
    fs_status = validate_filesystem(usb_path)
    if fs_status["status"] == "rejected":
        raise ValueError(f"Filesystem unsupported: {fs_status['message']}")
    
    # Step 2: Reset ID counter — fresh sequential IDs for this export
    reset_id_counter()
    
    # Step 3: Stage in ~/.mixmind/staging/{uuid}/
    staging = Path.home() / ".mixmind" / "staging" / str(_uuid.uuid4())
    staging.mkdir(parents=True, exist_ok=True)
    
    try:
        # Step 4: Build PDB Track records + write ANLZ files for each track
        pdb_tracks = []
        for i, rec in enumerate(track_records):
            if progress_cb:
                progress_cb("write_anlz", i + 1, len(track_records))
            
            tid = mint_track_id()
            
            # Copy audio file into staging
            src_audio = Path(rec["file_path"]).expanduser()
            ext = src_audio.suffix
            dst_audio = audio_path_on_usb(tid, ext, staging)
            dst_audio.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src_audio, dst_audio)
            
            # Verify byte-for-byte copy (Pitfall: audio corruption silent until playback)
            assert _sha256(src_audio) == _sha256(dst_audio), f"Audio copy mismatch: {src_audio.name}"
            
            # Write ANLZ files for this track at PIONEER/USBANLZ/Pxxx/yyyyyyyy/
            anlz_dir = usbanlz_dir(tid, staging)
            anlz_dir.mkdir(parents=True, exist_ok=True)
            
            beats = _build_beats_from_record(rec)
            hot_cues = _build_hot_cues_from_record(rec)
            memory_cues = _build_memory_cues_from_record(rec)
            sections = _build_sections_from_record(rec)
            preview = _build_waveform_preview(rec)
            waveform_3band = _build_waveform_3band(rec)
            
            write_dat(anlz_dir / "ANLZ0000.DAT", beats, memory_cues, hot_cues, preview, sections)
            write_ext(anlz_dir / "ANLZ0000.EXT", waveform_3band)
            
            pdb_tracks.append(TrackRecord(
                id=tid,
                title=rec.get("title", src_audio.stem),
                artist_id=0,           # resolved by pdb_writer._collect_artists
                artist_name=rec.get("artist", ""),
                bpm=float(rec.get("bpm") or 0.0),
                duration_sec=int(rec.get("duration_sec") or 0),
                key_id=0, key_name=rec.get("camelot", ""),
                color_id=0, bitrate=int(rec.get("bitrate") or 0),
                sample_rate=int(rec.get("sample_rate") or 44100),
                sample_depth=int(rec.get("sample_depth") or 16),
                file_path_on_usb=f"/Contents/{tid:08x}{ext.lower()}",
                file_name=f"{tid:08x}{ext.lower()}",
                file_size=src_audio.stat().st_size,
            ))
        
        # Step 5: Write PDB
        if progress_cb:
            progress_cb("write_pdb", len(track_records), len(track_records))
        pdb_path = staging / "PIONEER" / "rekordbox" / "export.pdb"
        write_pdb(pdb_path, pdb_tracks, playlists=_to_playlist_records(playlists, pdb_tracks))
        
        # Step 6: Validate by parsing back
        if progress_cb:
            progress_cb("validate", 1, 1)
        validation = _validate_export(staging, expected_count=len(pdb_tracks))
        if validation["status"] == "failed":
            raise RuntimeError(f"Export validation failed: {validation['messages']}")
        
        # Step 7: Pre-export USB clean — wipe existing PIONEER/ on target (Pitfall #8)
        if progress_cb:
            progress_cb("clean_target", 1, 1)
        clean_target(usb_path)
        
        # Step 8: Atomic move staging → USB
        if progress_cb:
            progress_cb("move_to_usb", 1, 1)
        bytes_written = _move_to_usb(staging, usb_path)
        
        return ExportResult(
            staging_dir=staging, usb_target=usb_path,
            exported_count=len(pdb_tracks),
            validation_status=validation["status"],
            validation_messages=validation["messages"],
            bytes_written=bytes_written,
        )
    
    finally:
        if staging.exists():
            shutil.rmtree(staging, ignore_errors=True)


def validate_filesystem(usb_path: Path) -> dict:
    """Check that usb_path is mounted with a CDJ-supported filesystem."""
    if not usb_path.is_mount() and not usb_path.exists():
        return {"status": "rejected", "message": f"{usb_path} not mounted"}
    
    # macOS: diskutil info -plist gives us filesystem
    try:
        out = subprocess.run(
            ["diskutil", "info", str(usb_path)],
            capture_output=True, text=True, check=True
        )
        lines = out.stdout.lower()
        if "ntfs" in lines:
            return {"status": "rejected", "message": "NTFS not supported by CDJ-3000 — reformat to exFAT"}
        if any(fs in lines for fs in FAT32_NAMES):
            # Check size — FAT32 caps at 32GB
            if any(t in lines for t in ["msdos"]):
                # diskutil reports total size; warn if > 32GB
                return {"status": "warning",
                        "message": "FAT32 detected; recommend exFAT for libraries > 32GB"}
        if "exfat" in lines or "hfs" in lines:
            return {"status": "ok", "message": "exFAT or HFS+ — recommended"}
    except (subprocess.CalledProcessError, FileNotFoundError):
        # diskutil not available (e.g., test environment) — pass with warning
        return {"status": "warning", "message": "Could not verify filesystem (test/non-macOS env)"}
    
    return {"status": "ok", "message": "filesystem OK"}


def clean_target(usb_path: Path) -> None:
    """Wipe PIONEER/rekordbox/ + PIONEER/USBANLZ/ on target USB.
    
    Per RESEARCH.md Pitfall #8: never mix MixMind + prior Rekordbox export
    on same USB — track ID collisions silently corrupt the library.
    """
    pioneer_dir = usb_path / "PIONEER"
    if not pioneer_dir.exists():
        return
    for sub in ("rekordbox", "USBANLZ"):
        target = pioneer_dir / sub
        if target.exists():
            shutil.rmtree(target)


def _move_to_usb(staging: Path, usb_path: Path) -> int:
    """Atomic move via os.rename (same FS) or rsync (cross-FS).
    
    For cross-FS we rsync then sync-fsync each file to mitigate FAT32
    torn-write risk per RESEARCH.md Pitfall #7.
    """
    src_pioneer = staging / "PIONEER"
    dst_pioneer = usb_path / "PIONEER"
    dst_pioneer.parent.mkdir(parents=True, exist_ok=True)
    
    # Try os.rename first (same filesystem)
    try:
        os.rename(src_pioneer, dst_pioneer)
        return _dir_size(dst_pioneer)
    except OSError:
        # Cross-filesystem: rsync with --fsync (or shutil.copytree)
        if dst_pioneer.exists():
            shutil.rmtree(dst_pioneer)
        shutil.copytree(src_pioneer, dst_pioneer)
        # Force sync to disk before returning success
        try:
            subprocess.run(["sync"], check=True, timeout=30)
        except Exception:
            pass
        return _dir_size(dst_pioneer)


def _validate_export(staging: Path, expected_count: int) -> dict:
    """Parse-back round-trip validation."""
    messages = []
    pdb_path = staging / "PIONEER" / "rekordbox" / "export.pdb"
    if not pdb_path.exists():
        return {"status": "failed", "messages": ["export.pdb not created"]}
    try:
        from pyrekordbox.db6 import Rekordbox6Database
        db = Rekordbox6Database(str(pdb_path))
        contents = db.get_content()
        if len(contents) != expected_count:
            return {"status": "failed",
                    "messages": [f"PDB has {len(contents)} tracks, expected {expected_count}"]}
        messages.append(f"PDB parsed OK: {len(contents)} tracks")
        # Spot-check ANLZ files exist for first 3 tracks
        for t in contents[:3]:
            anlz_dir = usbanlz_dir(t.ID, staging)
            if not (anlz_dir / "ANLZ0000.DAT").exists():
                messages.append(f"Missing ANLZ0000.DAT for track {t.ID}")
            if not (anlz_dir / "ANLZ0000.EXT").exists():
                messages.append(f"Missing ANLZ0000.EXT for track {t.ID}")
        return {"status": "ok", "messages": messages}
    except Exception as e:
        return {"status": "failed", "messages": [f"pyrekordbox parse error: {e}"]}


def _sha256(p: Path) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def _dir_size(p: Path) -> int:
    total = 0
    for f in p.rglob("*"):
        if f.is_file(): total += f.stat().st_size
    return total

def _build_beats_from_record(rec: dict) -> list[BeatEntry]:
    grid = rec.get("beat_grid_mm") or rec.get("beat_grid") or []
    return [BeatEntry(time_ms=int(b["time_ms"]),
                      beat_number=int(b.get("beat_number", 1)),
                      bpm=float(b.get("bpm") or rec.get("bpm") or 120.0))
            for b in grid]


def _build_hot_cues_from_record(rec: dict) -> list[CueEntry]:
    cues = rec.get("auto_cues_mm") or rec.get("hot_cues") or []
    out = []
    for i, c_ in enumerate(cues[:8]):
        slot = chr(ord('A') + i)
        out.append(CueEntry(
            time_ms=int(c_["time_ms"]), slot=slot,
            label=c_.get("label", f"Cue{slot}"),
            color_id=int(c_.get("color_id", i % 8)),
            is_loop=bool(c_.get("is_loop", False)),
            loop_end_ms=c_.get("loop_end_ms"),
        ))
    return out


def _build_memory_cues_from_record(rec: dict) -> list[CueEntry]:
    cues = rec.get("memory_cues") or []
    return [CueEntry(time_ms=int(c_["time_ms"]), slot=None,
                     label=c_.get("label", f"Mem{i}"),
                     color_id=int(c_.get("color_id", 0)))
            for i, c_ in enumerate(cues)]


def _build_sections_from_record(rec: dict) -> list[SectionEntry]:
    sections = rec.get("sections_mm") or []
    out = []
    for i, s in enumerate(sections):
        out.append(SectionEntry(
            start_beat=int(s.get("start_beat", 0)),
            end_beat=int(s.get("end_beat", 0)),
            kind=int(s.get("kind", 1)),
            name=s.get("name", "section"),
        ))
    return out


def _build_waveform_preview(rec: dict) -> bytes:
    """800-byte mono preview from waveform_4stem (downsample if needed)."""
    wf = rec.get("waveform_4stem") or []
    if not wf:
        return b"\x00" * 800
    out = bytearray()
    step = max(1, len(wf) // 800)
    for i in range(800):
        idx = min(i * step, len(wf) - 1)
        col = wf[idx]
        # Mono = max of all 4 stems, clamped to 0..255
        amp = max(int(col.get("drums", 0)), int(col.get("bass", 0)),
                  int(col.get("vocals", 0)), int(col.get("other", 0)))
        out.append(min(255, amp))
    while len(out) < 800: out.append(0)
    return bytes(out[:800])


def _build_waveform_3band(rec: dict) -> list[dict]:
    wf = rec.get("waveform_4stem") or []
    if not wf:
        return [{"drums": 0, "bass": 0, "vocals": 0, "other": 0}] * 1200
    return wf


def _to_playlist_records(playlists, pdb_tracks):
    if playlists:
        return [PlaylistRecord(id=p["id"], name=p["name"],
                               track_ids=p.get("track_ids", []))
                for p in playlists]
    return None
```

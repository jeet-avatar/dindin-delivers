Step 1 — Create import_routes.py. Full module:

```python
"""Library import — folder scan OR existing Rekordbox DB."""
from __future__ import annotations
import logging
import uuid
from pathlib import Path
from typing import Annotated

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel

from folder_scanner import scan_folder, ScannedTrack
from state import StateDB, ImportedTrack
from rekordbox import try_load_library_db, load_library_xml
from library import XML_PATH

logger = logging.getLogger(__name__)
router = APIRouter(prefix="/api/library")

class ImportRequest(BaseModel):
    folder_path: str | None = None    # required when from_rekordbox=False
    from_rekordbox: bool = False      # if True, use existing pyrekordbox path

class ImportResponse(BaseModel):
    batch_id: str
    imported: int
    skipped_duplicates: int
    warnings: list[str]
    source: str   # "folder" or "rekordbox"

def _get_state_db() -> StateDB:
    db = StateDB()
    try:
        yield db
    finally:
        db.close()

@router.post("/import", response_model=ImportResponse)
async def import_library(req: ImportRequest, db: Annotated[StateDB, Depends(_get_state_db)]):
    batch_id = str(uuid.uuid4())
    warnings: list[str] = []

    if req.from_rekordbox:
        # PRESERVATION: use existing pyrekordbox path unchanged
        tracks = try_load_library_db()
        if tracks is None and XML_PATH.exists():
            tracks = load_library_xml(XML_PATH)
        if not tracks:
            raise HTTPException(status_code=404, detail="No Rekordbox library found (master.db or rekordbox.xml)")
        # Convert Track (from rekordbox.py) -> ImportedTrack rows
        imported = 0
        skipped = 0
        for t in tracks:
            if not t.file_path:
                continue
            cid = f"rbximport_{t.content_id}"
            existing = db.get_imported_track(cid)
            if existing:
                skipped += 1
                continue
            db.add_imported_track(ImportedTrack(
                content_id=cid, file_path=t.file_path, title=t.title,
                artist=t.artist, album="", genre=t.genre or "",
                duration_sec=t.duration_sec, bitrate_kbps=0,
                sample_rate_hz=0, file_size_bytes=0,
                import_batch_id=batch_id, wav_extensible=False,
            ))
            imported += 1
        return ImportResponse(batch_id=batch_id, imported=imported,
                              skipped_duplicates=skipped, warnings=warnings,
                              source="rekordbox")

    # Folder scan path
    if not req.folder_path:
        raise HTTPException(status_code=400, detail="folder_path required when from_rekordbox=false")
    folder = Path(req.folder_path).expanduser().resolve()
    if not folder.is_dir():
        raise HTTPException(status_code=400, detail=f"Not a directory: {folder}")

    scanned: list[ScannedTrack] = scan_folder(folder)
    imported = 0
    skipped = 0
    for s in scanned:
        existing = db.get_imported_track(s.content_id)
        if existing:
            skipped += 1
            continue
        db.add_imported_track(ImportedTrack(
            content_id=s.content_id, file_path=s.file_path, title=s.title,
            artist=s.artist, album=s.album, genre=s.genre,
            duration_sec=s.duration_sec, bitrate_kbps=s.bitrate_kbps,
            sample_rate_hz=s.sample_rate_hz, file_size_bytes=s.file_size_bytes,
            import_batch_id=batch_id, wav_extensible=s.wav_extensible,
        ))
        imported += 1
        if s.wav_extensible:
            warnings.append(f"WAV_EXTENSIBLE detected (CDJ-3000 may refuse): {s.file_path}")
    return ImportResponse(batch_id=batch_id, imported=imported,
                          skipped_duplicates=skipped, warnings=warnings,
                          source="folder")
```

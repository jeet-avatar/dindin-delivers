```python
"""ANLZ (.DAT + .EXT) file writer — hand-rolled per Phase 21 Option C decision.

**License posture:** Uses ONLY construct (MIT) + Deep Symmetry public spec.
rbox 0.1.7 was EVALUATED and REJECTED due to GPL-3.0 license contamination.
DO NOT import rbox here or anywhere else in MixMind.

**Round-trip oracle:** apps/mixmind/sidecar/anlz_parser.py is the reference
parser. Every byte this writer produces must parse back through anlz_parser
with field-equality.

**Written against spec:** https://djl-analysis.deepsymmetry.org/rekordbox-export-analysis/anlz.html (fetched 2026-04-19)
"""
from __future__ import annotations
import struct as _struct
from dataclasses import dataclass
from pathlib import Path
from typing import Literal, Optional

import construct as c

from anlz_structs import (
    AnlzFileHeader, AnlzTagHeader,
    PQTZBody, PCO2Body, PCOBBody, PWAVBody, PWV3Body, PSSIBody,
)

# ---------------------------------------------------------------------------
# Input data classes — what callers pass in
# ---------------------------------------------------------------------------

@dataclass
class BeatEntry:
    time_ms: int
    beat_number: int           # 1..4 — 1 = downbeat
    bpm: float                 # BPM at this beat (essentia + madmom output)

@dataclass
class CueEntry:
    time_ms: int
    slot: Optional[str]        # 'A'..'H' for hot cues; None for memory cues
    label: str
    color_id: int              # 0..7 per Pioneer palette (see anlz_parser._CDJ_CUE_COLORS)
    is_loop: bool = False
    loop_end_ms: Optional[int] = None

@dataclass
class SectionEntry:
    start_beat: int
    end_beat: int
    kind: int                  # 1..8 per anlz_parser._SECTION_KINDS
    name: str

# ---------------------------------------------------------------------------
# .DAT writer — beatgrid + cues + mono preview waveform + sections
# ---------------------------------------------------------------------------

def write_dat(
    out_path: Path,
    beats: list[BeatEntry],
    memory_cues: list[CueEntry],
    hot_cues: list[CueEntry],
    waveform_preview: bytes,    # 800 bytes, 0..255 amplitude
    sections: list[SectionEntry],
) -> None:
    """Write a Rekordbox-compatible ANLZ0000.DAT file."""
    tags: list[bytes] = []

    # 1. PQTZ — beat grid
    tags.append(_build_tag(b"PQTZ", PQTZBody, {
        "len_header_extra": 0x18,
        "unknown1": 0,
        "len_entry_bytes": 8,
        "len_entries": len(beats),
        "beats": [{"beat_number": b.beat_number, "tempo": int(round(b.bpm * 100)),
                   "time_ms": b.time_ms} for b in beats],
    }))

    # 2. PCOB — legacy uncolored memory cues (CDJ falls back to this if PCO2 missing)
    if memory_cues:
        tags.append(_build_tag(b"PCOB", PCOBBody, _pcob_dict(memory_cues, cue_type=0)))
    # 3. PCOB — legacy uncolored hot cues
    if hot_cues:
        tags.append(_build_tag(b"PCOB", PCOBBody, _pcob_dict(hot_cues, cue_type=1)))

    # 4. PCO2 — colored memory cues (modern CDJ preferred)
    if memory_cues:
        tags.append(_build_tag(b"PCO2", PCO2Body, _pco2_dict(memory_cues, cue_type=0)))
    # 5. PCO2 — colored hot cues
    if hot_cues:
        tags.append(_build_tag(b"PCO2", PCO2Body, _pco2_dict(hot_cues, cue_type=1)))

    # 6. PWAV — mono waveform preview (800 bytes)
    assert len(waveform_preview) == 800, f"preview must be 800 bytes, got {len(waveform_preview)}"
    tags.append(_build_tag(b"PWAV", PWAVBody, {
        "len_preview": 800, "unknown": 0x00100000, "waveform": waveform_preview,
    }))

    # 7. PSSI — song structure
    if sections:
        tags.append(_build_tag(b"PSSI", PSSIBody, _pssi_dict(sections)))

    _write_anlz_file(out_path, tags)


# ---------------------------------------------------------------------------
# .EXT writer — 3-band waveform (PRO) + optional colored preview
# ---------------------------------------------------------------------------

def write_ext(
    out_path: Path,
    waveform_3band: list[dict],   # from analyzer.stems_to_waveform: [{"drums":0-255,"bass":..,"vocals":..,"other":..}, ...]
    n_columns: int = 1200,
) -> None:
    """Write a Rekordbox-compatible ANLZ0000.EXT file.

    Bands mapping: drums+bass → low, other → mid, vocals → high. This matches
    the existing analyzer.stems_to_waveform() convention documented in
    analyzer.py:33-70.
    """
    # Resample input (which may be 800 columns) to n_columns
    columns_3band = _resample_bands(waveform_3band, n_columns)
    # Pack: [low_hi, low_lo, mid_hi, mid_lo, high_hi, high_lo] per column
    packed = bytearray()
    for col in columns_3band:
        low  = min(255, col["drums"] // 2 + col["bass"] // 2)   # drums + bass averaged
        mid  = col["other"]
        high = col["vocals"]
        packed.extend(_struct.pack(">HHH", low, mid, high))

    tags: list[bytes] = []
    tags.append(_build_tag(b"PWV3", PWV3Body, {
        "len_entry_bytes": 6,
        "len_entries": n_columns,
        "unknown": 0x00960000,
        "columns": bytes(packed),
    }))

    _write_anlz_file(out_path, tags)


# ---------------------------------------------------------------------------
# Internal helpers
# ---------------------------------------------------------------------------

def _build_tag(magic: bytes, body_struct: c.Struct, body_dict: dict) -> bytes:
    body = body_struct.build(body_dict)
    header = AnlzTagHeader.build({
        "magic": magic,
        "len_header": 0x10,
        "len_tag": 0x10 + len(body),
    })
    return header + body

def _write_anlz_file(out_path: Path, tags: list[bytes]) -> None:
    body = b"".join(tags)
    header = AnlzFileHeader.build({
        "magic": b"PMAI", "len_header": 0x1C,
        "len_file": 0x1C + len(body),
        "unknown1": 0, "unknown2": 0, "unknown3": 0, "unknown4": 0,
    })
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_bytes(header + body)

def _pco2_dict(cues: list[CueEntry], cue_type: int) -> dict:
    entries = []
    for i, c_entry in enumerate(cues):
        slot_idx = 0 if c_entry.slot is None else (ord(c_entry.slot) - ord('A') + 1)
        loop_end = c_entry.loop_end_ms if c_entry.is_loop and c_entry.loop_end_ms else 0xFFFFFFFF
        label_utf16 = c_entry.label.encode("utf-16-be")
        # Pad to 4-byte multiple if needed; label_len is char count (not byte count)
        entries.append({
            "magic": b"PCPT", "len_header": 0x1C,
            "len_entry": 0x48 + len(label_utf16),
            "hot_cue": slot_idx,
            "type": 4 if cue_type == 1 else (2 if c_entry.is_loop else 1),
            "unknown1": b"\x00\x00\x00",
            "time_ms": c_entry.time_ms,
            "loop_time_ms": loop_end,
            "unknown2": b"\x00" * 8,
            "color_id": c_entry.color_id,
            "unknown3": b"\x00\x00\x00",
            "unknown4": b"\x00\x00\x00\x00",
            "loop_numerator": 0,
            "loop_denominator": 0,
            "label_len": len(c_entry.label) + 1,   # include null terminator
            "label": c_entry.label,
        })
    return {"type": cue_type, "unknown1": 0, "len_entries": len(entries), "entries": entries}

def _pcob_dict(cues: list[CueEntry], cue_type: int) -> dict:
    # Simpler legacy format — implement per Deep Symmetry PCOB section
    # ... (mirror PCO2 minus the color/label block)
    raise NotImplementedError("Fill in per Deep Symmetry spec during implementation")

def _pssi_dict(sections: list[SectionEntry]) -> dict:
    entries = [{
        "index": i, "start_beat": s.start_beat, "kind": s.kind, "unknown": 0,
    } for i, s in enumerate(sections)]
    return {
        "len_entry_bytes": 8,
        "len_entries": len(entries),
        "mood": 1, "unknown1": b"\x00" * 6,
        "end_beat": sections[-1].end_beat if sections else 0,
        "unknown2": 0, "bank": 0, "unknown3": b"\x00\x00\x00",
        "entries": entries,
    }

def _resample_bands(columns: list[dict], n: int) -> list[dict]:
    if len(columns) == n: return columns
    out = []
    for i in range(n):
        src = int(i * len(columns) / n)
        out.append(columns[min(src, len(columns) - 1)])
    return out
```

```python
def test_header_magic_and_length(tmp_path):
    out = tmp_path / "ANLZ0000.DAT"
    write_dat(out, beats=[BeatEntry(0, 1, 120.0)], memory_cues=[], hot_cues=[],
              waveform_preview=b'\x00'*800, sections=[])
    data = out.read_bytes()
    assert data[:4] == b"PMAI"
    # len_file in header must equal actual file size
    header = AnlzFileHeader.parse(data[:28])
    assert header.len_file == len(data)


def test_pqtz_beat_roundtrip():
    beats_in = [BeatEntry(time_ms=i*500, beat_number=(i % 4) + 1, bpm=128.0) for i in range(16)]
    body_bytes = PQTZBody.build({
        "len_header_extra": 0x18, "unknown1": 0,
        "len_entry_bytes": 8, "len_entries": len(beats_in),
        "beats": [{"beat_number": b.beat_number, "tempo": int(b.bpm*100), "time_ms": b.time_ms} for b in beats_in],
    })
    parsed = PQTZBody.parse(body_bytes)
    assert parsed.len_entries == 16
    assert parsed.beats[0].beat_number == 1
    assert parsed.beats[0].tempo == 12800
    # Downbeats at every 4th beat (0, 4, 8, 12)
    for i in range(0, 16, 4):
```

```python
    from anlz_writer import write_dat, BeatEntry, CueEntry, SectionEntry
    beats = [BeatEntry(i*500, (i % 4) + 1, 128.0) for i in range(n_beats)]
    hot_cues = [CueEntry(i*10000, chr(ord('A')+i), f'HotCue{i}', i) for i in range(n_hot)]
    memory_cues = [CueEntry(i*15000, None, f'Mem{i}', 0) for i in range(n_mem)]
    sections = [SectionEntry(i*32, (i+1)*32, ((i % 8)+1), 'sec') for i in range(n_sec)]
```

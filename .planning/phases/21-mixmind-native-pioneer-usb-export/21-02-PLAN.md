```python
@router.post("/library/analyze")
async def analyze_library(force: bool = Query(False)):
    """Run analysis on all imported tracks (source='import').

    Pulls tracks from state.imported_tracks, filters out ones already analyzed
    (status='complete' in analysis_cache with source='import'), queues the rest
    into AnalysisBatchRunner.

    Returns {status, total, queued} — progress via GET /api/analyze/status (existing endpoint).
    """
    global _batch_runner
    db = _get_db()
    try:
        if _batch_runner and _batch_runner.running:
            return {"status": "already_running", "progress": _batch_runner.status.to_dict()}

        imported = db.get_imported_tracks()
        if not imported:
            return {"status": "no_imported_tracks", "total": 0}

        pending = []
        already_done = 0
        for t in imported:
            if not force:
                existing = db.get_analysis(t.content_id, "import")
                if existing and existing["status"] == "complete":
                    already_done += 1
                    continue
            pending.append({
                "content_id": t.content_id,
                "source": "import",
                "file_path": t.file_path,
                "title": t.title or Path(t.file_path).stem,
                "artist": t.artist or "",
            })

        if not pending:
            return {"status": "all_analyzed", "total": len(imported), "already_done": already_done}

        _batch_runner = AnalysisBatchRunner(db)
        _batch_runner.start(pending)
        return {"status": "started", "total": len(pending), "already_done": already_done}
    finally:
        # AnalysisBatchRunner holds its own db reference; don't close here
        pass
```

```python
"""Integration test: imported track → analysis → state.db has all required fields."""
import numpy as np
import soundfile as sf
import pytest
from pathlib import Path
from fastapi.testclient import TestClient

from main import app
from state import StateDB, ImportedTrack

SAMPLE_RATE = 44100
BPM = 128
BEAT_INTERVAL_SEC = 60.0 / BPM   # 0.46875s
DURATION_SEC = 10                 # 10s = ~21 beats


def _synth_track(path: Path):
    """Generate a 10s track: 440Hz sine (A note) + kick click every 0.46875s (128 BPM)."""
    t = np.linspace(0, DURATION_SEC, SAMPLE_RATE * DURATION_SEC, False)
    # Musical tone at A4 (440Hz)
    tone = 0.2 * np.sin(2 * np.pi * 440 * t)
    # Kick: short 80Hz sine at every beat
    kick = np.zeros_like(t)
    kick_len = int(0.05 * SAMPLE_RATE)
    for beat_t in np.arange(0, DURATION_SEC, BEAT_INTERVAL_SEC):
        beat_samples = int(beat_t * SAMPLE_RATE)
        kick[beat_samples:beat_samples + kick_len] += 0.8 * np.sin(2 * np.pi * 80 * t[:kick_len])
    audio = np.clip(tone + kick, -1.0, 1.0)
    sf.write(str(path), audio, SAMPLE_RATE)


@pytest.fixture
def tmp_audio_folder(tmp_path):
    for i in range(3):
        _synth_track(tmp_path / f"track_{i:02d}.wav")
    return tmp_path


def test_analyze_imported_populates_all_required_fields(tmp_audio_folder, monkeypatch, tmp_path):
    # Point StateDB at a temp sqlite
    monkeypatch.setattr("state._DEFAULT_PATH", tmp_path / "state.db")
    client = TestClient(app)

    # 1. Import
    r = client.post("/api/library/import", json={"folder_path": str(tmp_audio_folder)})
    assert r.status_code == 200, r.text
    assert r.json()["imported"] == 3

    # 2. Analyze
    r = client.post("/api/library/analyze")
    assert r.status_code == 200
    # Poll status until done (max 5 min for 3 short tracks)
    import time
    for _ in range(300):
        time.sleep(1)
        s = client.get("/api/analyze/status").json()
        if not s.get("in_progress") and s.get("analyzed", 0) + s.get("failed", 0) >= 3:
            break
    else:
        pytest.fail(f"Analysis timeout. Status: {s}")

    # 3. Verify every required field exists for each track
    db = StateDB()
    imported = db.get_imported_tracks()
    assert len(imported) == 3
    for t in imported:
        row = db.get_analysis(t.content_id, "import")
        assert row is not None, f"No analysis row for {t.content_id}"
        assert row["status"] == "complete", f"{t.content_id}: status={row['status']}, err={row.get('error_message')}"
        # BPM: essentia should find ~128 (tolerate ±5 for synthetic)
        assert row["bpm"], "bpm missing"
        assert 120 <= row["bpm"] <= 135, f"BPM {row['bpm']} outside expected 128±7"
        # Camelot key
        assert row["camelot"], "camelot missing"
        # 3-band waveform
        assert row["waveform_4stem"], "waveform_4stem missing"
        # Beat grid (from madmom)
        assert row["beat_grid_mm"], "beat_grid_mm missing"
        # Auto cues
        assert row["auto_cues_mm"], "auto_cues_mm missing"
        # Sections
        assert row["sections_mm"], "sections_mm missing"
    db.close()


def test_analyze_imported_idempotent(tmp_audio_folder, monkeypatch, tmp_path):
    """Second /api/library/analyze after completion should return already_done count."""
    monkeypatch.setattr("state._DEFAULT_PATH", tmp_path / "state.db")
    client = TestClient(app)
    client.post("/api/library/import", json={"folder_path": str(tmp_audio_folder)})
    client.post("/api/library/analyze")
    # Wait for completion
    import time
    for _ in range(300):
        time.sleep(1)
        s = client.get("/api/analyze/status").json()
        if not s.get("in_progress") and s.get("analyzed", 0) + s.get("failed", 0) >= 3:
            break
    # Re-trigger analyze
    r = client.post("/api/library/analyze")
    # Since all 3 are complete, expect status=all_analyzed OR status=started with already_done=3
    j = r.json()
    assert j.get("status") == "all_analyzed" or j.get("already_done", 0) == 3, j
```

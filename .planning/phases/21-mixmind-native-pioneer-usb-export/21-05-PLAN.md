```tsx
import { useState } from 'react';

declare global {
  interface Window {
    mixmind: {
      chooseFolder: () => Promise<string | null>;
      openExternal: (url: string) => void;
      openRekordbox: () => void;
    };
  }
}

interface Props {
  port: number;
  onImported: (queued: number) => void;
}

export function ImportFolderButton({ port, onImported }: Props) {
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [fromRekordbox, setFromRekordbox] = useState(false);

  async function handleClick() {
    setBusy(true);
    setError(null);
    try {
      const folder = await window.mixmind.chooseFolder();
      if (!folder) { setBusy(false); return; }
      const res = await fetch(`http://127.0.0.1:${port}/api/library/import`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ folder, from_rekordbox: fromRekordbox }),
      });
      if (!res.ok) throw new Error(await res.text());
      const data = await res.json();
      onImported(data.queued || 0);
    } catch (e: any) {
      setError(e.message || String(e));
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="import-folder">
      <button onClick={handleClick} disabled={busy}>
        {busy ? 'Importing…' : '📁 Import Folder'}
      </button>
      <label style={{ marginLeft: 12, fontSize: 12 }}>
        <input
          type="checkbox"
          checked={fromRekordbox}
          onChange={(e) => setFromRekordbox(e.target.checked)}
        />
        {' '}Use existing Rekordbox library
      </label>
      {error && <div style={{ color: 'red', marginTop: 8 }}>{error}</div>}
    </div>
  );
}
```

Step 4 — Create `apps/mixmind/src/components/AnalyzeProgress.tsx`. Polls existing `/api/analyze/status` (defined in `analyze_routes.py`):

```tsx
import { useEffect, useState } from 'react';

interface AnalyzeStatus {
  running: boolean;
  total: number;
  completed: number;
  failed: number;
  current_track?: { title: string; bpm?: number; key?: string };
  eta_seconds?: number;
}

export function AnalyzeProgress({ port }: { port: number }) {
  const [status, setStatus] = useState<AnalyzeStatus | null>(null);

  useEffect(() => {
    let mounted = true;
    const tick = async () => {
      try {
        const res = await fetch(`http://127.0.0.1:${port}/api/analyze/status`);
        if (!res.ok) return;
        const data = await res.json();
        if (mounted) setStatus(data);
      } catch (e) {
        // sidecar may be restarting — ignore transient errors
      }
    };
    tick();
    const interval = setInterval(tick, 2000);
    return () => { mounted = false; clearInterval(interval); };
  }, [port]);

  if (!status || (!status.running && status.completed === 0)) return null;

  const pct = status.total > 0 ? Math.floor((status.completed / status.total) * 100) : 0;
  const etaText = status.eta_seconds
    ? `~${Math.ceil(status.eta_seconds / 60)} min remaining`
    : '';

  return (
    <div style={{
      padding: 12, background: '#1a1a1a', border: '1px solid #333',
      borderRadius: 8, marginTop: 12, color: '#eee',
    }}>
      <div style={{ fontWeight: 'bold', marginBottom: 6 }}>
        {status.running ? '⚙️ Analyzing' : '✅ Analysis complete'}
      </div>
      <div style={{ background: '#333', height: 8, borderRadius: 4, overflow: 'hidden' }}>
        <div style={{
          width: `${pct}%`, height: '100%', background: '#4af',
          transition: 'width 0.3s ease',
        }} />
      </div>
      <div style={{ marginTop: 6, fontSize: 12, color: '#aaa' }}>
        {status.completed} of {status.total} ({pct}%)
        {status.failed > 0 && <span style={{ color: '#f66' }}> · {status.failed} failed</span>}
        {etaText && <span> · {etaText}</span>}
      </div>
      {status.current_track && (
        <div style={{ marginTop: 6, fontSize: 11, color: '#888' }}>
          Now: {status.current_track.title}
          {status.current_track.bpm && ` · ${status.current_track.bpm.toFixed(1)} BPM`}
          {status.current_track.key && ` · ${status.current_track.key}`}
        </div>
      )}
    </div>
  );
}
```

Step 1 — Create `apps/mixmind/src/components/UsbExportWizard.tsx`. 3-step wizard:
1. Detect USB (calls existing GET /api/usb/status)
2. Confirm wipe of existing PIONEER/ on USB (Pitfall #8 user-facing safety)
3. Run export with progress + completion summary

```tsx
import { useState, useEffect } from 'react';

interface Props {
  port: number;
  selectedTrackIds: string[];
  onClose: () => void;
}

interface UsbStatus {
  detected: boolean;
  mount_path?: string;
  filesystem?: string;
  has_existing_pioneer?: boolean;
}

interface ExportResult {
  status: string;
  exported_count: number;
  bytes_written: number;
  validation_status: string;
  validation_messages: string[];
  filesystem: { status: string; message: string };
}

export function UsbExportWizard({ port, selectedTrackIds, onClose }: Props) {
  const [step, setStep] = useState<'detect'|'confirm'|'exporting'|'done'|'error'>('detect');
  const [usbStatus, setUsbStatus] = useState<UsbStatus | null>(null);
  const [result, setResult] = useState<ExportResult | null>(null);
  const [errorMsg, setErrorMsg] = useState<string>('');

  useEffect(() => {
    if (step !== 'detect') return;
    fetch(`http://127.0.0.1:${port}/api/usb/status`)
      .then(r => r.json())
      .then(d => {
        setUsbStatus(d);
        if (d.detected) setStep('confirm');
      })
      .catch(e => { setErrorMsg(String(e)); setStep('error'); });
  }, [step, port]);

  async function runExport() {
    if (!usbStatus?.mount_path) return;
    setStep('exporting');
    try {
      const res = await fetch(`http://127.0.0.1:${port}/api/usb/export`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          usb_mount_path: usbStatus.mount_path,
          track_ids: selectedTrackIds,
        }),
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data.detail || JSON.stringify(data));
      setResult(data);
      setStep('done');
    } catch (e: any) {
      setErrorMsg(e.message || String(e));
      setStep('error');
    }
  }

  return (
    <div className="usb-export-wizard" style={{
      position: 'fixed', top: 0, left: 0, right: 0, bottom: 0,
      background: 'rgba(0,0,0,0.85)', display: 'flex',
      alignItems: 'center', justifyContent: 'center', zIndex: 100,
    }}>
      <div style={{
        background: '#1a1a1a', color: '#eee', padding: 32,
        borderRadius: 12, maxWidth: 600, width: '90%',
      }}>
        <h2 style={{ marginTop: 0 }}>Export to Pioneer USB</h2>

        {step === 'detect' && (
          <div>🔍 Looking for a mounted Pioneer USB drive…</div>
        )}

        {step === 'confirm' && usbStatus && (
          <div>
            <div>USB detected: <strong>{usbStatus.mount_path}</strong></div>
            <div>Filesystem: {usbStatus.filesystem || 'unknown'}</div>
            <div>Tracks to export: <strong>{selectedTrackIds.length}</strong></div>
            {usbStatus.has_existing_pioneer && (
              <div style={{
                background: '#3a2a00', padding: 12, borderRadius: 6, marginTop: 12,
              }}>
                ⚠️ This USB has existing Pioneer files (likely from Rekordbox).<br/>
                Export will <strong>wipe</strong> PIONEER/rekordbox/ and PIONEER/USBANLZ/
                before writing. Continue?
              </div>
            )}
            <div style={{ marginTop: 24, display: 'flex', gap: 12 }}>
              <button onClick={onClose}>Cancel</button>
              <button onClick={runExport} style={{ background: '#4af' }}>
                {usbStatus.has_existing_pioneer ? 'Wipe & Export' : 'Export'}
              </button>
            </div>
          </div>
        )}

        {step === 'exporting' && (
          <div>
            <div>⚙️ Exporting {selectedTrackIds.length} tracks to USB…</div>
            <div style={{ marginTop: 12, fontSize: 12, color: '#aaa' }}>
              Building staging area · Writing PDB · Writing ANLZ files · Copying audio · Atomic move to USB
            </div>
            <div style={{
              background: '#333', height: 4, borderRadius: 2, marginTop: 16,
              overflow: 'hidden',
            }}>
              <div style={{
                width: '100%', height: '100%', background: '#4af',
                animation: 'pulse 1.5s ease infinite',
              }} />
            </div>
          </div>
        )}

        {step === 'done' && result && (
          <div>
            <div style={{ color: '#4f4', fontSize: 18 }}>✅ Export complete</div>
            <div style={{ marginTop: 8 }}>
              {result.exported_count} tracks · {(result.bytes_written / 1024 / 1024 / 1024).toFixed(2)} GB written
            </div>
            <div style={{ fontSize: 12, color: '#888', marginTop: 8 }}>
              Filesystem: {result.filesystem.status} ({result.filesystem.message})<br/>
              Validation: {result.validation_status}
              {result.validation_messages.map((m, i) => (
                <div key={i}>· {m}</div>
              ))}
            </div>
            <div style={{ marginTop: 24, fontSize: 14 }}>
              Safely eject USB: macOS Finder → click eject icon next to drive name.<br/>
              Plug into CDJ-3000 and select rekordbox source.
            </div>
            <button onClick={onClose} style={{ marginTop: 16 }}>Close</button>
          </div>
        )}

        {step === 'error' && (
          <div>
            <div style={{ color: '#f66' }}>❌ Export failed</div>
            <pre style={{
              background: '#000', padding: 12, marginTop: 12,
              fontSize: 11, overflow: 'auto', maxHeight: 200,
            }}>{errorMsg}</pre>
            <button onClick={onClose} style={{ marginTop: 16 }}>Close</button>
          </div>
        )}
      </div>
    </div>
  );
}
```

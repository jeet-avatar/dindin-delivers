import type { SubsystemPalette } from './palettes.js';

export interface CabinetBoxOpts {
  cx: number; cy: number;        // center of front face
  w: number; h: number;          // front face width, height (60-unit space)
  depth: number;                 // pre-foreshortening depth
  prefix: string;                // gradient ID prefix (e.g. 'strhinge-')
  palette: SubsystemPalette;
}

export function cabinetBox(opts: CabinetBoxOpts): string {
  const { cx, cy, w, h, depth, prefix, palette } = opts;
  const dx = depth * 0.866;      // cos(30°) × depth
  const dy = -depth * 0.5;       // -sin(30°) × depth (up is negative y)
  const fl = cx - w/2, fr = cx + w/2;
  const ft = cy - h/2, fb = cy + h/2;
  const topPolygon = `${fl},${ft} ${fr},${ft} ${fr + dx},${ft + dy} ${fl + dx},${ft + dy}`;
  const rightPolygon = `${fr},${ft} ${fr + dx},${ft + dy} ${fr + dx},${fb + dy} ${fr},${fb}`;
  return `<defs>
  <linearGradient id="${prefix}top" x1="0" y1="0" x2="0" y2="1">
    <stop offset="0" stop-color="${palette.topLight}"/>
    <stop offset="1" stop-color="${palette.topDark}"/>
  </linearGradient>
  <linearGradient id="${prefix}front" x1="0" y1="0" x2="0" y2="1">
    <stop offset="0" stop-color="${palette.frontLight}"/>
    <stop offset="1" stop-color="${palette.frontDark}"/>
  </linearGradient>
  <linearGradient id="${prefix}right" x1="0" y1="0" x2="1" y2="0">
    <stop offset="0" stop-color="${palette.rightLight}"/>
    <stop offset="1" stop-color="${palette.rightDark}"/>
  </linearGradient>
</defs>
<polygon points="${topPolygon}"   fill="url(#${prefix}top)"   stroke="${palette.edge}" stroke-width="0.25"/>
<polygon points="${rightPolygon}" fill="url(#${prefix}right)" stroke="${palette.edge}" stroke-width="0.25"/>
<rect x="${fl}" y="${ft}" width="${w}" height="${h}" fill="url(#${prefix}front)" stroke="${palette.edge}" stroke-width="0.3"/>`;
}

export function dropShadowFilter(prefix: string): string {
  // Use x="-20%" y="-20%" width="140%" height="140%" to avoid clipping (research §Pitfall 2).
  return `<filter id="${prefix}shadow" x="-20%" y="-20%" width="140%" height="140%">
  <feGaussianBlur in="SourceAlpha" stdDeviation="0.8"/>
  <feOffset dx="0" dy="1.2"/>
  <feComponentTransfer><feFuncA type="linear" slope="0.45"/></feComponentTransfer>
  <feMerge><feMergeNode/><feMergeNode in="SourceGraphic"/></feMerge>
</filter>`;
}

export function partLabel(text: string): string {
  // Bottom-right corner, monospace, 2.4px font (renders ~10pt at 4.7x page scale).
  return `<text x="58" y="58.5" text-anchor="end" font-family="Fira Code, monospace" font-size="2.4" fill="#cde" opacity="0.7">${escapeSvgText(text)}</text>`;
}

function escapeSvgText(s: string): string {
  return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

export interface Dims { L: number; W: number; H: number; }
export function normalizeDims(spec: any): Dims {
  const d = spec?.dimensions_mm;
  if (!d) return { L: 40, W: 40, H: 40 };          // sensible default per research §Pitfall 5
  if (Array.isArray(d) && d.length >= 3) return { L: Number(d[0]) || 40, W: Number(d[1]) || 40, H: Number(d[2]) || 40 };
  if (typeof d === 'object') {
    return {
      L: Number(d.length) || 40,
      W: Number(d.width)  || 40,
      H: Number(d.height) || 40,
    };
  }
  return { L: 40, W: 40, H: 40 };
}

export function makePrefix(partNumber: string): string {
  // Per-part-prefixed IDs (research §Pitfall 1: avoid SVG ID collisions in gallery).
  // CONTEXT.md line 80 specifies 4-char prefix; this implementation uses 8 chars to
  // reduce (not eliminate) collision risk among sibling parts. See <context-deviation>
  // block in Task 2 action — body-hash uniqueness gate in Plan 27-04 is the authoritative
  // collision guard regardless of prefix length.
  return partNumber.toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 8) + '-';
}

/**
 * Deterministic per-part-number perturbation for guaranteed visual uniqueness.
 *
 * Returns `baseValue + ((hashCode(partNumber) % 7) - 3)` — a stable integer in
 * the range [-3, +3] derived purely from the part_number string. Same input
 * always returns the same output (no Date.now / Math.random), so templates
 * remain deterministic and migration 017 idempotency holds.
 *
 * Templates MUST call this for width, height, AND at least one other
 * geometric axis so sibling parts with identical (or missing) dimensions_mm
 * still produce visually distinct silhouettes — different aspect ratios,
 * different LED/connector positions, different seam-ring spacings, etc.
 *
 * The ±3 range is small enough that the part still looks like its template
 * family but large enough to be perceptible against a 30-60 unit viewBox.
 *
 * @example
 *   perturbForPartNumber("STR-HINGE-PIN-A", 20)     // → 22 (always)
 *   perturbForPartNumber("STR-HINGE-SPRING", 20)    // → 18 (always, ≠ PIN-A)
 *   perturbForPartNumber("STR-HINGE-PIN-A", 20)     // → 22 (deterministic re-call)
 */
export function perturbForPartNumber(partNumber: string, baseValue: number): number {
  // djb2 hash (simple, deterministic, no deps)
  let hash = 5381;
  for (let i = 0; i < partNumber.length; i++) {
    hash = ((hash * 33) ^ partNumber.charCodeAt(i)) >>> 0;
  }
  // Map to [-3, +3] range
  const offset = (hash % 7) - 3;
  return baseValue + offset;
}

export function groundShadowEllipse(cx: number, cy: number, rx: number, ry: number): string {
  return `<ellipse cx="${cx}" cy="${cy}" rx="${rx}" ry="${ry}" fill="#000" opacity="0.22"/>`;
}

import { cabinetBox, dropShadowFilter, partLabel, normalizeDims, makePrefix, perturbForPartNumber, groundShadowEllipse } from './primitives.js';
import { paletteFor } from './palettes.js';

export function fastenerTemplate(part: { part_number: string; subsystem_code?: string; specifications?: any }): string {
  const palette = paletteFor(part.subsystem_code);
  const prefix = makePrefix(part.part_number);
  const dims = normalizeDims(part.specifications);
  // Hex head: scale relative to dims.L (M3 → L~6mm, M5 → L~10mm). Map mm to viewBox units (1 unit ≈ 1mm at 60-unit ViewBox).
  // B2 — perturbForPartNumber on width, height, AND shaft length so sibling fasteners with identical
  //      dimensions_mm produce visually distinct silhouettes (different head size + shaft length).
  const headSize = perturbForPartNumber(part.part_number, Math.max(8, Math.min(16, dims.L * 0.4)));
  const headHeight = perturbForPartNumber(part.part_number + ':h', Math.max(6, Math.min(12, dims.L * 0.28)));
  const shaftLen = perturbForPartNumber(part.part_number + ':s', Math.max(20, Math.min(40, dims.L * 1.2)));
  const shaftR = Math.max(2, Math.min(5, dims.W * 0.3));
  const cx = 30, cy = 22;
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 60 60">
${dropShadowFilter(prefix)}
<g filter="url(#${prefix}shadow)">
  ${groundShadowEllipse(30, 54, 16, 2)}
  ${cabinetBox({ cx, cy, w: headSize, h: headHeight, depth: headSize * 0.5, prefix: prefix + 'head', palette })}
  <!-- threaded shaft as a vertical cylinder below the head -->
  <rect x="${30 - shaftR}" y="${cy + headHeight * 0.5}" width="${shaftR * 2}" height="${shaftLen}" fill="url(#${prefix}headfront)" stroke="${palette.edge}" stroke-width="0.2"/>
  <!-- thread ridges: 6 horizontal lines -->
  ${Array.from({ length: 6 }, (_, i) => `<line x1="${30 - shaftR}" y1="${cy + headHeight * 0.5 + (i + 1) * shaftLen / 7}" x2="${30 + shaftR}" y2="${cy + headHeight * 0.5 + (i + 1) * shaftLen / 7}" stroke="${palette.edge}" stroke-width="0.15" opacity="0.6"/>`).join('')}
</g>
${partLabel(part.part_number)}
</svg>`;
}

export function plateTemplate(part: { part_number: string; subsystem_code?: string; specifications?: any }): string {
  const palette = paletteFor(part.subsystem_code);
  const prefix = makePrefix(part.part_number);
  const dims = normalizeDims(part.specifications);
  // Plate: wide and shallow (thin extrusion). Aspect from dims.L × dims.W × small dims.H.
  // Map: w = clamp(dims.L * 0.6, 24, 44), h = clamp(dims.W * 0.4, 8, 20), depth = clamp(dims.H * 0.3, 2, 6).
  // B2 — perturbForPartNumber on width, height, AND mounting-hole spacing offset so sibling plates
  //      with identical dimensions_mm produce visually distinct silhouettes.
  const w = perturbForPartNumber(part.part_number, Math.max(24, Math.min(44, dims.L * 0.6)));
  const h = perturbForPartNumber(part.part_number + ':h', Math.max(8,  Math.min(20, dims.W * 0.4)));
  const depth = Math.max(2, Math.min(6, dims.H * 0.3));
  const holeOffsetFrac = 0.35 + (perturbForPartNumber(part.part_number + ':o', 0) / 30); // ±0.1 perturbation
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 60 60">
${dropShadowFilter(prefix)}

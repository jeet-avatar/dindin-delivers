```bash
# CRITICAL W6: psql -e flag echoes per-statement row counts. Without it, grep
# "^UPDATE 1" against default psql output returns 0 regardless of behavior
# (because default psql doesn't print "UPDATE N" per statement — only at txn end).
psql -e "$DATABASE_URL" -f /Users/jeet/turion-satellite/migrations/017_redraw_cad_phase27.sql > /tmp/phase27-reapply-017.txt 2>&1
REAPPLY_EXIT=$?
test "$REAPPLY_EXIT" -eq 0 || { echo "FATAL: re-apply failed"; tail -20 /tmp/phase27-reapply-017.txt; exit 1; }
grep -q "^COMMIT" /tmp/phase27-reapply-017.txt || { echo "FATAL: re-apply did not COMMIT"; tail -20 /tmp/phase27-reapply-017.txt; exit 1; }

# Now count per-statement results:
#   "^UPDATE 1" = a row was actually changed (BAD on re-apply — idempotency broken)
#   "^UPDATE 0" = the WHERE clause filtered it out (GOOD on re-apply — no-op as expected)
REAPPLY_UPDATE1=$(grep -c "^UPDATE 1$" /tmp/phase27-reapply-017.txt || echo 0)
REAPPLY_UPDATE0=$(grep -c "^UPDATE 0$" /tmp/phase27-reapply-017.txt || echo 0)
echo "Re-apply: UPDATE 1 (touched) = $REAPPLY_UPDATE1"
echo "Re-apply: UPDATE 0 (no-op)   = $REAPPLY_UPDATE0"

# Pass criteria: REAPPLY_UPDATE1 == 0 (idempotency holds)
# AND REAPPLY_UPDATE0 should be close to APPLY_UPDATE1 from step 2 (every row that was
# updated on first pass should be a no-op on second pass).
if [ "$REAPPLY_UPDATE1" -gt 0 ]; then
  echo "FAIL: re-apply touched $REAPPLY_UPDATE1 rows — idempotency broken"
  echo "Investigate: SVG string round-trip drift (Postgres normalizing whitespace) OR generator non-determinism"
  exit 1
fi
if [ "$REAPPLY_UPDATE0" -lt 50 ]; then
  echo "WARN: re-apply shows only $REAPPLY_UPDATE0 UPDATE 0 lines (expected ~79)"
  echo "psql -e may not be echoing per-statement results as expected. Falling back to verification query."
  # Fallback W6 (c): verify via timestamp query that no rows were touched in the last minute.
  # (Requires turion_satellite.part_definitions to have updated_at; verify schema first.)
  HAS_UPDATED_AT=$(psql -t -A "$DATABASE_URL" -c "SELECT EXISTS(SELECT 1 FROM information_schema.columns WHERE table_schema='turion_satellite' AND table_name='part_definitions' AND column_name='updated_at');")
  if [ "$HAS_UPDATED_AT" = "t" ]; then
    RECENTLY_TOUCHED=$(psql -t -A "$DATABASE_URL" -c "SELECT COUNT(*) FROM turion_satellite.part_definitions WHERE updated_at > NOW() - INTERVAL '60 seconds';")
    echo "Recently-touched rows (last 60s): $RECENTLY_TOUCHED"
    if [ "$RECENTLY_TOUCHED" -gt 0 ]; then
      echo "FAIL: $RECENTLY_TOUCHED rows updated_at-bumped in the last minute — idempotency broken"
      exit 1
    fi
    echo "PASS: idempotency proven via updated_at timestamp fallback"
  else
    echo "WARN: part_definitions has no updated_at column; cannot use timestamp fallback"
    echo "Idempotency proof depends on the -e UPDATE 0 count above. Manual inspection of re-apply log advised."
  fi
fi
echo "PASS: idempotency proven on live DB"
```

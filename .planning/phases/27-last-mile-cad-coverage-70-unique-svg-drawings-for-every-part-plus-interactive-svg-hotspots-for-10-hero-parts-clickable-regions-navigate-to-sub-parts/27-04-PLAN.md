#!/usr/bin/env -S node --import tsx
// scripts/generate-cad-svgs.ts · Phase 27 CAD drawing generator
//
// Connects to production Postgres, fetches all SAT-003 part_definitions, skips rows
// already populated by migration 016 (sentinel '<!-- v=016 -->'), dispatches each
// remaining row to one of 8 part-family templates, and writes migrations/017_redraw_cad_phase27.sql
// plus a preview JSON for the local QA gallery.
//
// Hard gates: uniqueness (no two parts share body geometry) + determinism (two runs byte-equal).
//
// Usage: node --import tsx scripts/generate-cad-svgs.ts
// Env:   DATABASE_URL (Postgres connection string)

import { Client } from 'pg';
import * as crypto from 'node:crypto';
import * as fs from 'node:fs';
import * as path from 'node:path';
import { fileURLToPath } from 'node:url';

import { assemblyTemplate } from './cad-templates/assembly.js';
import { subassemblyTemplate } from './cad-templates/subassembly.js';
import { cylindricalTemplate } from './cad-templates/cylindrical.js';
import { lensOpticalTemplate } from './cad-templates/lens-optical.js';
import { antennaDishTemplate } from './cad-templates/antenna-dish.js';
import { solarCellTemplate } from './cad-templates/solar-cell.js';
import { fastenerTemplate } from './cad-templates/fastener.js';
import { plateTemplate } from './cad-templates/plate.js';
import { makePrefix } from './cad-templates/primitives.js';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const REPO_ROOT = path.resolve(__dirname, '..');

// SAT-003 satellite UUID is stable per Phase 26 plan 26-05 line 80
// (`SELECT id FROM satellites WHERE name='SAT-003'` returns 24587565-b15b-42ce-b590-87ecf9b6bb99).
const SAT_003_ID = '24587565-b15b-42ce-b590-87ecf9b6bb99';
const SENTINEL_V016 = '<!-- v=016 -->';

interface PartDef {
  id: string;
  part_number: string;
  subsystem_code: string | null;
  specifications: any;
  drawing_svg: string | null;
}

type TemplateFn = (p: PartDef) => string;

// Dispatch table (research §Pattern 3). Order matters — longer/more-specific regex first.
// W5 FIX: SOLAR-* MUST come before plate regex (`EPS-SOLAR-PANEL-A` would otherwise
// match `/-PANEL-/` in the plate rule and land on plate template instead of solar-cell).
function chooseTemplate(part: PartDef): { name: string; fn: TemplateFn } {
  const pn = part.part_number;
  if (/^FASTENER-|HINGE-PIVOT-PIN|^.*-PIN-A$|-SCREW-/.test(pn))
    return { name: 'fastener',    fn: fastenerTemplate };
  if (/^[A-Z]+-ASSY$/.test(pn))
    return { name: 'assembly',    fn: assemblyTemplate };
  if (/SPRING|DAMPER|TANK|THRUSTER|VALVE|FILTER|PUMP|XDUCER|HARNESS|CABLE|WAVEGUIDE|REGULATOR/.test(pn))
    return { name: 'cylindrical', fn: cylindricalTemplate };
  // SOLAR-* before plate to avoid -PANEL- regex stealing solar parts (W5)
  if (/SOLAR-CELL|SOLAR-PANEL|SOLAR-WING|SOLAR-ARRAY/.test(pn))
    return { name: 'solar',       fn: solarCellTemplate };
  if (/TELESCOPE|FOCAL|BAFFLE|LENS|MIRROR|SENSOR-OPT/.test(pn))
    return { name: 'lens',        fn: lensOpticalTemplate };
  if (/ANT-|ANTENNA|DISH-/.test(pn))
    return { name: 'antenna',     fn: antennaDishTemplate };
  if (/^.*-(BUSBAR|BRACKET|RING|PANEL)-/.test(pn) || /PANEL-RAD|MOUNT-PLATE/.test(pn))
    return { name: 'plate',       fn: plateTemplate };
  return { name: 'subassembly',   fn: subassemblyTemplate };  // default catch-all
}

function escapeSqlDollarQuote(s: string): string {
  // Use $svg$ … $svg$ dollar-quoting to avoid escaping single quotes inside SVG strings.
  // If the SVG string ever contains literal '$svg$', use $svg27$ instead. Check for collision.
  if (s.includes('$svg$')) return `$svg27$${s}$svg27$`;
  return `$svg$${s}$svg$`;
}

/**
 * Strip per-part artefacts so we can compare pure body geometry across parts.
 * Removes:
 *   1. The per-part-prefix from every id="..." and url(#...) reference (uses makePrefix).
 *   2. The bottom-right partLabel <text ... font-family="Fira Code">{part_number}</text>.
 *   3. Trailing/leading whitespace per line (cosmetic noise).
 *
 * What's left is the pure geometric shape: polygons, rects, ellipses, paths, circles,
 * lines, gradient stops, filter primitives — without ID names or part-number text.
 * Two parts producing the SAME stripped body = visually identical drawings.
 */
function stripPartArtefacts(svg: string, partNumber: string): string {
  const prefix = makePrefix(partNumber);                 // e.g. "strhinge-"
  // Replace the prefix anywhere it appears (id=, url(#, href= references).
  let stripped = svg.split(prefix).join('PREFIX-');
  // Strip the bottom-right partLabel <text ...>part_number</text> entirely.
  // The label uses font-family="Fira Code, monospace" (from primitives.partLabel).
  stripped = stripped.replace(/<text[^>]*font-family="Fira Code[^>]*>[^<]*<\/text>/g, '<text>LABEL</text>');
  // Also belt-and-suspenders: replace any literal occurrence of the part_number string.
  if (partNumber) stripped = stripped.split(partNumber).join('PARTNUM');
  return stripped;
}

function sha256(s: string): string {
  return crypto.createHash('sha256').update(s, 'utf8').digest('hex');
}

async function main() {
  const dbUrl = process.env.DATABASE_URL;
  if (!dbUrl) {
    console.error('FATAL: DATABASE_URL not set. Export it from .env or pass inline.');
    process.exit(1);
  }
  const client = new Client({ connectionString: dbUrl });
  await client.connect();
  console.log('[gen] connected to Postgres');

  // Sanity: confirm SAT-003 exists with expected UUID
  const sat = await client.query(`SELECT id, name FROM turion_satellite.satellites WHERE name = 'SAT-003'`);
  if (sat.rowCount === 0) {
    console.error('FATAL: SAT-003 not found. Phase 26 must be applied first.');
    process.exit(1);
  }
  const satId = sat.rows[0].id;
  if (satId !== SAT_003_ID) {
    console.warn(`[gen] SAT-003 UUID is ${satId}, not the expected ${SAT_003_ID}. Continuing with live UUID.`);
  }

  // Fetch all part_definitions that have ≥1 instance on SAT-003.
  // B1 FIX: part_definitions has NO subsystem_code column. JOIN to subsystems table
  // and SELECT s.code AS subsystem_code (canonical pattern from backend/src/routes/parts.ts).
  const result = await client.query<PartDef>(`
    SELECT DISTINCT pd.id,
           pd.part_number,
           s.code AS subsystem_code,
           pd.specifications,
           pd.drawing_svg
      FROM turion_satellite.part_definitions pd
      INNER JOIN turion_satellite.part_instances pi ON pi.part_definition_id = pd.id
      LEFT JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
     WHERE pi.satellite_id = $1
     ORDER BY s.code NULLS LAST, pd.part_number
  `, [satId]);

  console.log(`[gen] found ${result.rowCount} part_definitions on SAT-003`);
  await client.end();

  // Stats
  const stats = { total: result.rowCount, skipped_v016: 0, generated: 0, by_template: {} as Record<string, number> };
  const updates: { part_number: string; svg: string }[] = [];
  const previewRows: { part_number: string; subsystem_code: string | null; template: string; svg: string; bodyHash: string }[] = [];

  // Body-hash map for uniqueness gate (B2)
  const bodyHashToParts = new Map<string, string[]>();
  // First-run SVG map for determinism gate (W7)
  const firstRunSvg = new Map<string, string>();

  // ============================================================================
  // First pass: generate, build hash map
  // ============================================================================
  for (const part of result.rows) {
    if (part.drawing_svg && part.drawing_svg.includes(SENTINEL_V016)) {
      stats.skipped_v016++;
      console.log(`[gen] SKIP ${part.part_number} (v=016 sentinel)`);
      continue;
    }
    if (!part.subsystem_code) {
      console.warn(`[gen] WARN ${part.part_number} has NULL subsystem_code (subsystem_id may be orphan). Falling back to STR palette.`);
    }
    const { name, fn } = chooseTemplate(part);
    let svg: string;
    try {
      svg = fn(part);
    } catch (e: any) {
      console.error(`[gen] FAILED to generate ${part.part_number} via ${name}:`, e.message);
      throw e;  // hard fail — every part must generate cleanly
    }
    if (!svg.startsWith('<svg') || !svg.includes('</svg>')) {
      throw new Error(`[gen] ${part.part_number} produced invalid SVG (missing tags)`);
    }
    stats.generated++;
    stats.by_template[name] = (stats.by_template[name] ?? 0) + 1;
    updates.push({ part_number: part.part_number, svg });

    // Hash the prefix+label-stripped body for uniqueness gate
    const bodyHash = sha256(stripPartArtefacts(svg, part.part_number));
    const existing = bodyHashToParts.get(bodyHash) ?? [];
    existing.push(part.part_number);
    bodyHashToParts.set(bodyHash, existing);

    firstRunSvg.set(part.part_number, svg);
    previewRows.push({
      part_number: part.part_number,
      subsystem_code: part.subsystem_code,
      template: name,
      svg,
      bodyHash: bodyHash.slice(0, 12),  // short hash for display in preview
    });
  }

  // ============================================================================
  // GATE 1: Uniqueness — no two parts may share body geometry (B2)
  // ============================================================================
  const duplicates: { hash: string; parts: string[] }[] = [];
  for (const [hash, parts] of bodyHashToParts.entries()) {
    if (parts.length > 1) duplicates.push({ hash, parts });
  }
  if (duplicates.length > 0) {
    console.error('\n========== UNIQUENESS GATE FAILED ==========');
    console.error(`Found ${duplicates.length} body-hash collision(s). Every part must have a UNIQUE drawing.`);
    console.error('User directive: "every part must have a UNIQUE drawing, no shared/duplicate art across parts of the same subsystem"');
    for (const dup of duplicates) {
      console.error(`  hash ${dup.hash.slice(0, 12)}…  →  ${dup.parts.join(', ')}`);
    }
    console.error('\nFix: add perturbForPartNumber() variance to templates (see scripts/cad-templates/primitives.ts)');
    console.error('or tighten dispatch regex so visually-similar parts route to different templates.');
    console.error('=============================================\n');
    process.exit(2);
  }
  console.log(`[gen] uniqueness gate PASS: ${bodyHashToParts.size} distinct body hashes for ${stats.generated} parts`);

  // ============================================================================
  // GATE 2: Determinism — second run must produce byte-equal output (W7)
  // ============================================================================
  console.log('[gen] running determinism check (second pass)…');
  for (const [partNumber, firstSvg] of firstRunSvg.entries()) {
    // Find the original part row to re-feed identical inputs
    const original = result.rows.find(p => p.part_number === partNumber);
    if (!original) continue;
    const { fn } = chooseTemplate(original);
    const secondSvg = fn(original);
    if (secondSvg !== firstSvg) {
      console.error('\n========== DETERMINISM GATE FAILED ==========');
      console.error(`Part ${partNumber} produced different SVG across two runs.`);
      console.error('Templates MUST be pure functions of (part_number, subsystem_code, specifications).');
      console.error('Common culprits: Date.now(), Math.random(), Set/Map iteration order, undefined input handling.');
      console.error('=============================================\n');
      process.exit(3);
    }
  }
  console.log(`[gen] determinism gate PASS: ${firstRunSvg.size} parts produced byte-equal output across two runs`);

  // ============================================================================
  // Emit migration 017
  // ============================================================================
  const migrationPath = path.join(REPO_ROOT, 'migrations', '017_redraw_cad_phase27.sql');
  const header = `-- 017_redraw_cad_phase27.sql · Phase 27 CAD redraw
-- Auto-generated by scripts/generate-cad-svgs.ts on ${new Date().toISOString()}
-- Updates drawing_svg for ${stats.generated} part_definitions on SAT-003.
-- Skips ${stats.skipped_v016} rows protected by the v=016 sentinel from migration 016.
--
-- Idempotency: each UPDATE includes 'AND drawing_svg IS DISTINCT FROM <new>' so
-- re-applying the migration is a no-op (changes 0 rows).
-- Apply: psql "$DATABASE_URL" -e -f migrations/017_redraw_cad_phase27.sql
-- ============================================================================

SET search_path TO turion_satellite, public;

BEGIN;
`;
  const body = updates.map(u => {
    const quoted = escapeSqlDollarQuote(u.svg);
    return `UPDATE turion_satellite.part_definitions
   SET drawing_svg = ${quoted}
 WHERE part_number = ${escapeSqlDollarQuote(u.part_number)}
   AND drawing_svg IS DISTINCT FROM ${quoted};`;
  }).join('\n\n');
  const footer = `\n\nCOMMIT;\n`;
  fs.writeFileSync(migrationPath, header + body + footer, 'utf8');
  console.log(`[gen] wrote ${migrationPath} (${updates.length} UPDATEs)`);

  // ============================================================================
  // Emit preview JSON (includes bodyHash for human spot-checks in the gallery)
  // ============================================================================
  const previewDir = path.join(REPO_ROOT, 'scripts', 'cad-preview');
  fs.mkdirSync(previewDir, { recursive: true });
  const previewJson = path.join(previewDir, 'generated.json');
  fs.writeFileSync(previewJson, JSON.stringify(previewRows, null, 2), 'utf8');
  console.log(`[gen] wrote ${previewJson} (${previewRows.length} parts for preview)`);

  // ============================================================================
  // Stats
  // ============================================================================
  console.log('\n========== Phase 27 Generator Stats ==========');
  console.log(`Total SAT-003 part_definitions:  ${stats.total}`);
  console.log(`Skipped (v=016 sentinel):        ${stats.skipped_v016}`);
  console.log(`Generated:                       ${stats.generated}`);
  console.log(`Distinct body hashes:            ${bodyHashToParts.size} (must equal Generated)`);
  console.log('By template:');
  for (const [name, count] of Object.entries(stats.by_template).sort((a, b) => b[1] - a[1])) {
    console.log(`  ${name.padEnd(14)} ${count}`);
  }
  console.log('Gates passed: uniqueness ✓  determinism ✓');
  console.log('==============================================\n');
}

main().catch(e => { console.error(e); process.exit(1); });

- **W7 — Determinism gate**: Second pass over the same input rows; per-part byte equality assertion. Catches any `Date.now()`, `Math.random()`, or iteration-order non-determinism that would break migration 017's idempotency guard.

    <p>Hand review every generated SVG before applying migration 017. Use the cluster filter to pin a (template, subsystem) pair and visually confirm sibling parts look distinct. Each tile shows its body-hash (prefix+label stripped) — sibling tiles must have DIFFERENT hashes for the uniqueness guarantee to hold visually.</p>
    <div class="filter">
      <label>Filter by template: <select id="templateFilter"><option value="">All</option></select></label>
      <label>Filter by subsystem: <select id="subsystemFilter"><option value="">All</option></select></label>

  <script>
    async function load() {
      const data = await fetch('./generated.json').then(r => r.json());
      const grid = document.getElementById('grid');
      const templateFilter = document.getElementById('templateFilter');
      const subsystemFilter = document.getElementById('subsystemFilter');
      const count = document.getElementById('count');

      // Populate filter dropdowns
      const templates = [...new Set(data.map(d => d.template))].sort();
      const subsystems = [...new Set(data.map(d => d.subsystem_code).filter(Boolean))].sort();
      templates.forEach(t => templateFilter.add(new Option(t, t)));
      subsystems.forEach(s => subsystemFilter.add(new Option(s, s)));

      function render() {
        const t = templateFilter.value;
        const s = subsystemFilter.value;
        const filtered = data.filter(d => (!t || d.template === t) && (!s || d.subsystem_code === s));
        // Track uniqueness within this filtered view
        const hashSeen = new Map();
        filtered.forEach(d => {
          const arr = hashSeen.get(d.bodyHash) ?? [];
          arr.push(d.part_number);
          hashSeen.set(d.bodyHash, arr);
        });
        const collisions = [...hashSeen.values()].filter(a => a.length > 1).length;
        count.textContent = `${filtered.length} / ${data.length} parts · ${hashSeen.size} distinct hashes · ${collisions} collision(s)`;
        count.className = collisions > 0 ? 'count count-bad' : 'count';

        grid.innerHTML = filtered.map(d => {
          const dupClass = (hashSeen.get(d.bodyHash) || []).length > 1 ? ' tile-duplicate' : '';
          return `
          <figure class="tile${dupClass}" data-template="${d.template}" data-hash="${d.bodyHash}">
            <div class="svg-host">${d.svg}</div>
            <figcaption>
              <span class="pn">${d.part_number}</span>
              <span class="meta">${d.subsystem_code ?? '—'} · ${d.template}</span>
              <span class="hash">hash: ${d.bodyHash}</span>
            </figcaption>
          </figure>`;
        }).join('');
      }

      templateFilter.addEventListener('change', render);
      subsystemFilter.addEventListener('change', render);
      render();
    }
    load().catch(e => { document.body.innerHTML = `<pre>Failed: ${e}</pre>`; });
  </script>
</body>
</html>

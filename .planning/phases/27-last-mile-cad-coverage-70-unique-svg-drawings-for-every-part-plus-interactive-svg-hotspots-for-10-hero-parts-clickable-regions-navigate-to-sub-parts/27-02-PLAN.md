```javascript
/**
 * Overlay clickable BOM-child callouts onto an injected parent SVG.
 *
 * @param {SVGElement} svgEl - the parent <svg> element already in the DOM
 *                              (the one inside #cadCenter that was injected by part.html)
 * @param {Array<{child_part_definition_id, part_number, qty}>} children
 * @param {string} satId - current satellite UUID, used to build callout hrefs
 *
 * No-op if children is empty/null. Appends a <g class="callouts">...</g> sibling group
 * to the injected silhouette wrapper <g>. The page-side CSS controls visibility via the
 * .callouts-hidden class on a parent element (toggled by the Show/Hide labels button).
 */
function renderCalloutsOnSvg(svgEl, children, satId) {
  if (!svgEl || !children || children.length === 0) return;

  // The CAD silhouette is injected as <g transform="translate(-140,-140) scale(4.7)">...</g>
  // inside #cadCenter, which lives inside the OUTER frame svg. We place callouts at the
  // SAME outer-svg coordinate space — but to keep math simple, we append callouts to the
  // injected silhouette's <g> so they live in the 0-60 viewBox coordinate space.
  // Find the inner silhouette group: it's the <g> inside #cadCenter.
  const inner = svgEl.querySelector('#cadCenter > g') || svgEl.querySelector('#cadCenter');
  if (!inner) {
    console.warn('[satellite-cad] renderCalloutsOnSvg: no #cadCenter group found');
    return;
  }

  // 60×60 viewBox: parent silhouette is centered around (30, 30). Use radial layout.
  const cx = 30, cy = 30;
  const ringRadius = 22;          // distance from centroid to callout text
  const anchorRadius = 12;        // distance from centroid to leader-line terminus on body
  const n = children.length;

  // Greedy non-overlap: if n > 8, use a 2-tier ring (inner half + outer half).
  const positions = [];
  for (let i = 0; i < n; i++) {
    const tier = n > 8 && i % 2 === 1 ? 1 : 0;
    const r = ringRadius + tier * 4;
    const angle = (2 * Math.PI * i) / n - Math.PI / 2;  // start at top, clockwise
    const tx = cx + r * Math.cos(angle);
    const ty = cy + r * Math.sin(angle);
    const ax = cx + anchorRadius * Math.cos(angle);
    const ay = cy + anchorRadius * Math.sin(angle);
    positions.push({ tx, ty, ax, ay, angle });
  }

  const escapeText = (s) => String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
                                     .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
  const truncate = (s, max) => s.length > max ? s.slice(0, max - 1) + '…' : s;

  const calloutMarkup = children.map((c, i) => {
    const { tx, ty, ax, ay } = positions[i];
    const label = truncate(`${c.part_number || c.child_part_definition_id} × ${c.qty || 1}`, 18);
    const href = `part.html?id=${encodeURIComponent(c.child_part_definition_id)}&sat=${encodeURIComponent(satId)}`;
    // Background rect: width scales with text length (rough: ~1.4 px per char).
    const bgW = Math.max(14, Math.min(22, label.length * 1.4));
    const bgX = tx - bgW / 2;
    const bgY = ty - 2;
    return `<a xlink:href="${escapeText(href)}" href="${escapeText(href)}" class="callout-link" tabindex="0" role="link" aria-label="Navigate to child part ${escapeText(c.part_number || '')}">
  <line x1="${ax.toFixed(2)}" y1="${ay.toFixed(2)}" x2="${tx.toFixed(2)}" y2="${ty.toFixed(2)}" class="leader-line"/>
  <circle cx="${ax.toFixed(2)}" cy="${ay.toFixed(2)}" r="0.6" class="leader-dot"/>
  <rect x="${bgX.toFixed(2)}" y="${bgY.toFixed(2)}" width="${bgW.toFixed(2)}" height="3.5" rx="0.6" class="callout-bg"/>
  <text x="${tx.toFixed(2)}" y="${(ty + 0.6).toFixed(2)}" text-anchor="middle" class="callout-text">${escapeText(label)}</text>
</a>`;
  }).join('');

  // Use SVG namespace when creating the group via insertAdjacentHTML — the <a> elements
  // inherit the parent <svg>'s namespace correctly.
  inner.insertAdjacentHTML('beforeend', `<g class="callouts">${calloutMarkup}</g>`);
}
```

```javascript
  if (children && children.length > 0) {
    // Find the inline SVG containing #cadCenter (the outer frame SVG)
    const frameSvg = document.querySelector('#cadFrame svg') || document.querySelector('.cad-frame svg');
    if (frameSvg && window.satelliteCad && typeof window.satelliteCad.renderCalloutsOnSvg === 'function') {
      try {
        window.satelliteCad.renderCalloutsOnSvg(frameSvg, children, satId);
        document.getElementById('toggleCallouts').style.display = 'inline-block';
      } catch (e) {
        console.warn('[part.html] renderCalloutsOnSvg failed:', e);
      }
    }
  } else {
    const tog = document.getElementById('toggleCallouts');
    if (tog) tog.style.display = 'none';
  }

  // Wire the Show/Hide toggle (idempotent — safe to attach even if no children)
  const toggleBtn = document.getElementById('toggleCallouts');
  if (toggleBtn && !toggleBtn.dataset.wired) {
    toggleBtn.dataset.wired = '1';
    toggleBtn.addEventListener('click', () => {
      const frame = document.getElementById('cadFrame') || document.querySelector('.cad-frame');
      if (!frame) return;
      const nowHidden = frame.classList.toggle('callouts-hidden');
      toggleBtn.setAttribute('aria-pressed', String(!nowHidden));
      toggleBtn.textContent = nowHidden ? 'labels: off' : 'labels: on';
    });
  }
```

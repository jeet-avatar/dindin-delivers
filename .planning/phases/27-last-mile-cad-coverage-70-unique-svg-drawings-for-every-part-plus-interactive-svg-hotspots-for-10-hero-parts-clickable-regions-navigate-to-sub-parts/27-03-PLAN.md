```typescript
import { cabinetBox, dropShadowFilter, partLabel, normalizeDims, makePrefix, perturbForPartNumber, groundShadowEllipse } from './primitives.js';
import { paletteFor } from './palettes.js';

export function assemblyTemplate(part: { part_number: string; subsystem_code?: string; specifications?: any }): string {
  const palette = paletteFor(part.subsystem_code);
  const prefix = makePrefix(part.part_number);
  const dims = normalizeDims(part.specifications);
  // B2 — perturbForPartNumber on w, h, depth so sibling ASSY parts with identical dims still differ.
  const w = perturbForPartNumber(part.part_number, Math.max(30, Math.min(44, dims.L * 0.6)));
  const h = perturbForPartNumber(part.part_number + ':h', Math.max(22, Math.min(32, dims.W * 0.5)));
  const depth = perturbForPartNumber(part.part_number + ':d', Math.max(10, Math.min(18, dims.H * 0.4)));
  const cx = 28, cy = 30;
  // Compute corner positions for screw markers
  const fl = cx - w/2, fr = cx + w/2;
  const ft = cy - h/2, fb = cy + h/2;
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 60 60">
${dropShadowFilter(prefix)}
<g filter="url(#${prefix}shadow)">
  ${groundShadowEllipse(30, 54, w * 0.55, 3)}
  ${cabinetBox({ cx, cy, w, h, depth, prefix: prefix + 'asm', palette })}
  <!-- 4 corner screw markers on the front face -->
  <circle cx="${fl + 2}" cy="${ft + 2}" r="0.7" fill="#0a0e16" stroke="${palette.edge}" stroke-width="0.12"/>
  <circle cx="${fr - 2}" cy="${ft + 2}" r="0.7" fill="#0a0e16" stroke="${palette.edge}" stroke-width="0.12"/>
  <circle cx="${fl + 2}" cy="${fb - 2}" r="0.7" fill="#0a0e16" stroke="${palette.edge}" stroke-width="0.12"/>
  <circle cx="${fr - 2}" cy="${fb - 2}" r="0.7" fill="#0a0e16" stroke="${palette.edge}" stroke-width="0.12"/>
  <!-- Center seam line: subtle horizontal cut across front face -->
  <line x1="${fl + 4}" y1="${cy}" x2="${fr - 4}" y2="${cy}" stroke="${palette.edge}" stroke-width="0.15" opacity="0.5"/>
</g>
${partLabel(part.part_number)}
</svg>`;
}
```

**`cylindrical.ts`** — for springs, dampers, tanks, valves, filters, pumps, thrusters, transducers, harnesses, waveguides (regex: `SPRING|DAMPER|TANK|THRUSTER|VALVE|FILTER|PUMP|XDUCER|HARNESS|CABLE|WAVEGUIDE`). Render a vertical cylinder with elliptical top/bottom caps and side gradient. Apply `perturbForPartNumber` to width, height, AND seam-ring spacing so sibling cylinders look distinct.

```typescript
import { dropShadowFilter, partLabel, normalizeDims, makePrefix, perturbForPartNumber, groundShadowEllipse } from './primitives.js';
import { paletteFor } from './palettes.js';

export function cylindricalTemplate(part: { part_number: string; subsystem_code?: string; specifications?: any }): string {
  const palette = paletteFor(part.subsystem_code);
  const prefix = makePrefix(part.part_number);
  const dims = normalizeDims(part.specifications);
  // Cylinder: prefer tall+narrow. H > W typical for thrusters/tanks; reverse for valves.
  const isTall = dims.H >= dims.W;
  // B2 — perturbForPartNumber on width, height, AND seam-ring fractions
  const w = perturbForPartNumber(part.part_number,
    isTall ? Math.max(12, Math.min(20, dims.W * 0.4)) : Math.max(18, Math.min(28, dims.W * 0.5)));
  const h = perturbForPartNumber(part.part_number + ':h',
    isTall ? Math.max(26, Math.min(38, dims.H * 0.5)) : Math.max(14, Math.min(22, dims.H * 0.4)));
  // Seam-ring vertical positions vary per part
  const ring1Frac = 0.33 + (perturbForPartNumber(part.part_number + ':r1', 0) / 60); // ±0.05
  const ring2Frac = 0.66 + (perturbForPartNumber(part.part_number + ':r2', 0) / 60); // ±0.05
  const cx = 30, cy = 32;
  const rx = w / 2;
  const topY = cy - h/2;
  const botY = cy + h/2;
  const capRy = rx * 0.32;
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 60 60">
${dropShadowFilter(prefix)}
<defs>
  <linearGradient id="${prefix}cylside" x1="0" y1="0" x2="1" y2="0">
    <stop offset="0" stop-color="${palette.rightDark}"/>
    <stop offset="0.5" stop-color="${palette.frontLight}"/>
    <stop offset="1" stop-color="${palette.rightDark}"/>
  </linearGradient>
  <radialGradient id="${prefix}cyltop" cx="0.4" cy="0.4" r="0.7">
    <stop offset="0" stop-color="${palette.topLight}"/>
    <stop offset="1" stop-color="${palette.topDark}"/>
  </radialGradient>
</defs>
<g filter="url(#${prefix}shadow)">
  ${groundShadowEllipse(cx, botY + 3, rx * 1.1, capRy * 0.9)}
  <ellipse cx="${cx}" cy="${botY}" rx="${rx}" ry="${capRy}" fill="${palette.rightDark}" stroke="${palette.edge}" stroke-width="0.2"/>
  <rect x="${cx - rx}" y="${topY}" width="${w}" height="${h}" fill="url(#${prefix}cylside)" stroke="${palette.edge}" stroke-width="0.25"/>
  <ellipse cx="${cx}" cy="${topY}" rx="${rx}" ry="${capRy}" fill="url(#${prefix}cyltop)" stroke="${palette.edge}" stroke-width="0.25"/>
  <!-- 2 horizontal seam rings for visual interest, positions vary per part -->

</g>
${partLabel(part.part_number)}
</svg>`;
}
```

```typescript
import { dropShadowFilter, partLabel, normalizeDims, makePrefix, perturbForPartNumber, groundShadowEllipse } from './primitives.js';
import { paletteFor } from './palettes.js';

export function lensOpticalTemplate(part: { part_number: string; subsystem_code?: string; specifications?: any }): string {
  const palette = paletteFor(part.subsystem_code);
  const prefix = makePrefix(part.part_number);
  const dims = normalizeDims(part.specifications);
  // B2 — perturbForPartNumber on objective r1, focal r2 fraction, AND glint position
  const r1 = perturbForPartNumber(part.part_number, Math.max(14, Math.min(22, dims.L * 0.4)));   // objective (largest)
  const r2Frac = 0.75 + (perturbForPartNumber(part.part_number + ':r2', 0) / 60); // ±0.05
  const r2 = r1 * r2Frac;                                                          // focal
  const r3 = r1 * 0.55;                                                            // back
  const glintDxFrac = -0.3 + (perturbForPartNumber(part.part_number + ':g', 0) / 30); // ±0.1
  const cx = 30, cy = 32;
  const accent = palette.accent || '#8ed0ff';
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 60 60">
${dropShadowFilter(prefix)}
<defs>
  <radialGradient id="${prefix}objective" cx="0.5" cy="0.5" r="0.55">
    <stop offset="0" stop-color="${accent}" stop-opacity="0.6"/>
    <stop offset="0.5" stop-color="${palette.frontLight}"/>
    <stop offset="1" stop-color="${palette.frontDark}"/>
  </radialGradient>
  <radialGradient id="${prefix}aperture" cx="0.5" cy="0.5" r="0.5">
    <stop offset="0" stop-color="${accent}"/>
    <stop offset="0.7" stop-color="#0a0a14"/>
    <stop offset="1" stop-color="#000"/>
  </radialGradient>
</defs>
<g filter="url(#${prefix}shadow)">
  ${groundShadowEllipse(cx, cy + r1 + 4, r1 * 1.1, 2)}
  <!-- Back ring (smallest, deepest) -->
  <circle cx="${cx}" cy="${cy}" r="${r3}" fill="${palette.rightDark}" stroke="${palette.edge}" stroke-width="0.25"/>
  <!-- Focal ring -->
  <circle cx="${cx}" cy="${cy}" r="${r2}" fill="${palette.frontDark}" stroke="${palette.edge}" stroke-width="0.25" opacity="0.92"/>
  <!-- Objective (front lens) -->
  <circle cx="${cx}" cy="${cy}" r="${r1}" fill="url(#${prefix}objective)" stroke="${palette.edge}" stroke-width="0.3"/>
  <!-- Aperture center -->
  <circle cx="${cx}" cy="${cy}" r="${r1 * 0.45}" fill="url(#${prefix}aperture)"/>
  <!-- Glint highlight on objective, position varies per part -->
  <ellipse cx="${cx + r1 * glintDxFrac}" cy="${cy - r1 * 0.4}" rx="${r1 * 0.18}" ry="${r1 * 0.08}" fill="#fff" opacity="0.4"/>
</g>
${partLabel(part.part_number)}
</svg>`;
}
```

**`solar-cell.ts`** — for solar arrays/cells/wings (regex: `SOLAR-CELL|SOLAR-PANEL|SOLAR-WING|SOLAR-ARRAY`). Render a thin cabinet box substrate + a hexagonal-cell grid on the top face. Apply `perturbForPartNumber` to substrate width, height, AND cell-grid column count.

```typescript
import { cabinetBox, dropShadowFilter, partLabel, normalizeDims, makePrefix, perturbForPartNumber, groundShadowEllipse } from './primitives.js';
import { paletteFor } from './palettes.js';

export function solarCellTemplate(part: { part_number: string; subsystem_code?: string; specifications?: any }): string {
  const palette = paletteFor(part.subsystem_code);
  const prefix = makePrefix(part.part_number);
  const dims = normalizeDims(part.specifications);
  // B2 — perturbForPartNumber on w, h, AND cell-grid cols so sibling solar panels look distinct
  const w = perturbForPartNumber(part.part_number, Math.max(28, Math.min(44, dims.L * 0.55)));
  const h = perturbForPartNumber(part.part_number + ':h', Math.max(18, Math.min(28, dims.W * 0.45)));
  const depth = 4;
  // Cell grid cols varies between 3-5 per part
  const colsBase = 4;
  const cols = Math.max(3, Math.min(5, colsBase + perturbForPartNumber(part.part_number + ':c', 0) % 3 - 1));
  const rows = 3;
  const cx = 28, cy = 30;
  const fl = cx - w/2, ft = cy - h/2;
  const cellW = w / (cols + 0.5);
  const cellH = h / (rows + 0.5);
  let cellsMarkup = '';
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      const offsetX = (r % 2 === 1) ? cellW / 2 : 0;
      const hx = fl + cellW * 0.7 + c * cellW + offsetX;
      const hy = ft + cellH * 0.7 + r * cellH;
      const hs = Math.min(cellW, cellH) * 0.42;
      // Hexagon points (pointy-top)
      const points = [
        [hx, hy - hs],
        [hx + hs * 0.866, hy - hs * 0.5],
        [hx + hs * 0.866, hy + hs * 0.5],
        [hx, hy + hs],
        [hx - hs * 0.866, hy + hs * 0.5],
        [hx - hs * 0.866, hy - hs * 0.5],
      ].map(p => p.map(v => v.toFixed(2)).join(',')).join(' ');
      cellsMarkup += `<polygon points="${points}" fill="${palette.accent || '#3a5d8c'}" stroke="${palette.edge}" stroke-width="0.12" opacity="0.85"/>`;
    }
  }
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 60 60">
${dropShadowFilter(prefix)}
<g filter="url(#${prefix}shadow)">
  ${groundShadowEllipse(30, 54, w * 0.55, 2)}
  ${cabinetBox({ cx, cy, w, h, depth, prefix: prefix + 'sub', palette })}
  <!-- Hex cell grid on the front face, col count varies per part -->
  ${cellsMarkup}
</g>
${partLabel(part.part_number)}
</svg>`;
}
```

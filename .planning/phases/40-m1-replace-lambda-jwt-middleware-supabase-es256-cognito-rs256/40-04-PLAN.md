Structure (each case is a function; main runs all + prints a summary table):

```bash
#!/usr/bin/env bash
set -euo pipefail

REGION=us-east-1
ERP_API=https://lo254mvukl.execute-api.us-east-1.amazonaws.com
SAT_API=https://rjydekliee.execute-api.us-east-1.amazonaws.com
ERP_PROTECTED=/api/data/all
SAT_PROTECTED=/api/satellites
TEST_USER=jm@techcloudpro.com

# Read Cognito config from secret (Rule 1 — no hardcoded IDs in script)
COGNITO_CFG=$(aws secretsmanager get-secret-value --region "$REGION" \
  --secret-id zietra/cognito-config --query SecretString --output text)
POOL_ID=$(echo "$COGNITO_CFG" | jq -r .user_pool_id)
CLIENT_ID=$(echo "$COGNITO_CFG" | jq -r .app_client_id)

mint_cognito_idtoken() {
  # Step 1: admin-initiate-auth
  local RESP
  RESP=$(aws cognito-idp admin-initiate-auth \
    --user-pool-id "$POOL_ID" --client-id "$CLIENT_ID" \
    --auth-flow CUSTOM_AUTH \
    --auth-parameters USERNAME="$TEST_USER" --region "$REGION")
  local SESSION
  SESSION=$(echo "$RESP" | jq -r '.Session')
  [ -z "$SESSION" ] || [ "$SESSION" = "null" ] && { echo "FAIL: no Session"; return 1; }

  # Step 2: scrape nonce from CloudWatch — preferred over inbox check (autonomous)
  sleep 4
  local NONCE
  NONCE=$(aws logs filter-log-events \
    --log-group-name /aws/lambda/zietra-cognito-create-auth-challenge \
    --start-time $(($(date +%s) * 1000 - 60000)) \
    --filter-pattern '"nonce="' --region "$REGION" \
    --query 'events[-1].message' --output text \
    | grep -oE 'nonce=[A-Za-z0-9_-]+' | head -1 | cut -d= -f2)
  [ -z "$NONCE" ] && { echo "FAIL: no nonce in CloudWatch"; return 1; }

  # Step 3: respond-to-auth-challenge
  local TOKENS
  TOKENS=$(aws cognito-idp admin-respond-to-auth-challenge \
    --user-pool-id "$POOL_ID" --client-id "$CLIENT_ID" \
    --challenge-name CUSTOM_CHALLENGE --session "$SESSION" \
    --challenge-responses USERNAME="$TEST_USER",ANSWER="$NONCE" --region "$REGION")
  echo "$TOKENS" | jq -r '.AuthenticationResult.IdToken'
}

forge_cognito_jwt() {
  # Take a real IdToken; mutate last 8 chars of signature segment.
  local REAL="$1"
  local PARTS
  IFS='.' read -r H P S <<< "$REAL"
  # Replace last 8 chars of signature with junk that's still base64url-shaped
  local LEN=${#S}
  local TRUNK="${S:0:$((LEN-8))}"
  echo "${H}.${P}.${TRUNK}AAAAAAAA"
}

case_a_valid_cognito() {
  local IDTOKEN="$1"
  local API_BASE="$2"
  local PROTECTED="$3"
  curl -s -o /dev/null -w '%{http_code}' -H "Authorization: Bearer $IDTOKEN" "$API_BASE$PROTECTED"
}

case_c_forged_cognito() {
  local FORGED="$1"
  local API_BASE="$2"
  local PROTECTED="$3"
  curl -s -o /dev/null -w '%{http_code}' -H "Authorization: Bearer $FORGED" "$API_BASE$PROTECTED"
}

case_d_valid_supabase() {
  # If $SUPABASE_TEST_TOKEN env var set, use it. Otherwise skip with rationale.
  local TOKEN="${SUPABASE_TEST_TOKEN:-}"
  local API_BASE="$1"
  local PROTECTED="$2"
  if [ -z "$TOKEN" ]; then echo SKIP; return; fi
  curl -s -o /dev/null -w '%{http_code}' -H "Authorization: Bearer $TOKEN" "$API_BASE$PROTECTED"
}

case_e_forged_supabase() {
  # Junk bearer that fails ES256 signature check
  local API_BASE="$1"
  local PROTECTED="$2"
  curl -s -o /dev/null -w '%{http_code}' -H "Authorization: Bearer aaaa.bbbb.cccc" "$API_BASE$PROTECTED"
}

case_phase38_regression() {
  # 5 curls from Phase 38 transcript
  echo "ERP /api/health         = $(curl -s -o /dev/null -w '%{http_code}' $ERP_API/api/health)"
  echo "ERP /api/data/all unauth= $(curl -s -o /dev/null -w '%{http_code}' $ERP_API/api/data/all)"
  echo "Sat /api/health         = $(curl -s -o /dev/null -w '%{http_code}' $SAT_API/api/health)"
  echo "Sat /api/satellites unauth= $(curl -s -o /dev/null -w '%{http_code}' $SAT_API/api/satellites)"
  echo "ERP forged ES256        = $(curl -s -o /dev/null -w '%{http_code}' -H 'Authorization: Bearer aaaa.bbbb.cccc' $ERP_API/api/data/all)"
}

main() {
  echo "=== Phase 40 smoke transcript ==="
  echo "Date: $(date -u +%FT%TZ)"
  echo "Pool: $POOL_ID  ClientId: $CLIENT_ID"
  echo
  echo "Minting real Cognito IdToken for $TEST_USER..."
  local IDTOKEN
  IDTOKEN=$(mint_cognito_idtoken)
  echo "IdToken length: ${#IDTOKEN}"
  local FORGED
  FORGED=$(forge_cognito_jwt "$IDTOKEN")
  echo

  echo "--- ERP (turion-demo-api) ---"
  echo "(a) Valid Cognito IdToken                = $(case_a_valid_cognito "$IDTOKEN" "$ERP_API" "$ERP_PROTECTED")  [expect 200]"
  echo "(c) Forged Cognito (mutated signature)   = $(case_c_forged_cognito "$FORGED" "$ERP_API" "$ERP_PROTECTED")  [expect 401]"
  echo "(d) Valid Supabase ES256                 = $(case_d_valid_supabase "$ERP_API" "$ERP_PROTECTED")  [expect 200 or SKIP]"
  echo "(e) Forged Supabase ES256                = $(case_e_forged_supabase "$ERP_API" "$ERP_PROTECTED")  [expect 401]"
  echo

  echo "--- Satellite (turion-satellite-api) ---"
  echo "(a) Valid Cognito IdToken                = $(case_a_valid_cognito "$IDTOKEN" "$SAT_API" "$SAT_PROTECTED")  [expect 200]"
  echo "(c) Forged Cognito (mutated signature)   = $(case_c_forged_cognito "$FORGED" "$SAT_API" "$SAT_PROTECTED")  [expect 401]"
  echo "(d) Valid Supabase ES256                 = $(case_d_valid_supabase "$SAT_API" "$SAT_PROTECTED")  [expect 200 or SKIP]"
  echo "(e) Forged Supabase ES256                = $(case_e_forged_supabase "$SAT_API" "$SAT_PROTECTED")  [expect 401]"
  echo

  echo "--- Phase 38 regression ---"
  case_phase38_regression
  echo

  echo "=== END ==="
}

main "$@"
```

- **Case (b) DEFERRED with rationale** — expired-token requires waiting >3600s. Decision: skip (b); the signature-mutation case (c) directly tests the signature check, which is the more security-critical assertion.
- **Case (d) Valid Supabase ES256**: opportunistic. If `$SUPABASE_TEST_TOKEN` is set in the shell env, the script uses it; otherwise reports SKIP. Document this trade-off in CHECKPOINT.md — Phase 38 already proved this works; reproducing requires either (i) re-running Phase 38's smoke or (ii) capturing a real Supabase session token. Skip is acceptable per CONTEXT.md note 5 (planner discretion).
- **Forgery (c)** uses signature-mutation, which exercises `jwt.verify`'s sig check directly (Pitfall 8 — without this, you'd be testing malformed-token detection, not the verifier itself).

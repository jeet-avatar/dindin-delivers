**Endpoint 1 — POST /sync-sales-order/:salesOrderId  (path param, IDs are clean SO-* strings):**

```typescript
// POST /api/integration/sync-sales-order/:salesOrderId
// Body: { satelliteId: string }  (UUID of target satellite)
// Reads turion.sales_orders.source_data->'lineItems' (defensive Array.isArray guard:
//   legacy seeded SOs have NO lineItems array — clean exit with reason='no_line_items').
// For each line with a part_number that matches turion_satellite.part_definitions.part_number,
// either creates a new part_instance on (satelliteId, partDef) with sales_order_id set, OR
// links the existing instance to this SO. Idempotent: re-running with same (SO, sat) is a no-op.
// Returns { created, linked, skipped, matches, reason? }.
router.post('/sync-sales-order/:salesOrderId', requireAuth, async (req, res) => {
  const soId = String(req.params.salesOrderId || '');
  const satelliteId = String((req.body || {}).satelliteId || '');
  if (!soId) { res.status(400).json({ error: 'salesOrderId path param required' }); return; }
  if (!satelliteId) { res.status(400).json({ error: 'satelliteId required in body' }); return; }

  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    // 1. Fetch the SO. NOT 404 if not found — that's a "valid request, no data" case → 200 with reason.
    //    But we DO 404 here since the path-param specifies an exact ID and a missing one is malformed.
    //    (Distinct from "found SO but no lineItems" which IS a clean 200 no-op.)
    const soRows = await client.query(
      'SELECT id, source_data FROM turion.sales_orders WHERE id = $1',
      [soId]
    );
    if (soRows.rowCount === 0) {
      await client.query('ROLLBACK');
      res.status(404).json({ error: 'Sales order not found' });
      return;
    }
    const so = soRows.rows[0];

    // 2. Pull line items defensively. Legacy SOs lack source_data.lineItems entirely.
    const lineItems: any[] = Array.isArray(so.source_data?.lineItems)
      ? so.source_data.lineItems
      : [];
    if (lineItems.length === 0) {
      await client.query('COMMIT');
      res.json({ created: 0, linked: 0, skipped: 0, matches: 0, reason: 'no_line_items' });
      return;
    }

    // 3. Extract candidate part_numbers (try several field names — see RESEARCH §Open Q1).
    const candidates: string[] = lineItems
      .map((li: any) => li?.partNumber || li?.part_number || li?.itemId || li?.item)
      .filter((s: any): s is string => typeof s === 'string' && s.length > 0);
    if (candidates.length === 0) {
      await client.query('COMMIT');
      res.json({ created: 0, linked: 0, skipped: 0, matches: 0, reason: 'no_part_numbers' });
      return;
    }

    // 4. Match against part_definitions by exact part_number (case-sensitive per CONTEXT.md).
    const defRows = await client.query(
      `SELECT id, part_number FROM turion_satellite.part_definitions
       WHERE part_number = ANY($1::text[])`,
      [candidates]
    );
    if (defRows.rowCount === 0) {
      await client.query('COMMIT');
      res.json({
        created: 0, linked: 0, skipped: 0, matches: 0,
        reason: 'no_definition_matches',
        candidates_tried: candidates,  // for debugging
      });
      return;
    }

    // 5. For each matching def, link existing OR create new instance.
    let created = 0, linked = 0, skipped = 0;
    for (const def of defRows.rows) {
      const existing = await client.query(
        `SELECT id, sales_order_id FROM turion_satellite.part_instances
         WHERE satellite_id = $1 AND part_definition_id = $2
         ORDER BY instance_index LIMIT 1`,
        [satelliteId, def.id]
      );
      if (existing.rowCount === 0) {
        // Create new instance with default instance_index=1, sales_order_id set.
        // (instance_index defaults exist in schema; if first INSERT collides on UNIQUE, the
        // SELECT-LIMIT-1 above would have returned the row instead — defensive enough.)
        await client.query(
          `INSERT INTO turion_satellite.part_instances
             (satellite_id, part_definition_id, sales_order_id, cross_links_updated_at)
           VALUES ($1, $2, $3, NOW())`,
          [satelliteId, def.id, soId]
        );
        created++;
      } else if (existing.rows[0].sales_order_id === soId) {
        skipped++;  // Idempotent re-run: already linked to this SO.
      } else {
        await client.query(
          `UPDATE turion_satellite.part_instances
           SET sales_order_id = $1, cross_links_updated_at = NOW()
           WHERE id = $2`,
          [soId, existing.rows[0].id]
        );
        linked++;
      }
    }

    // 6. Audit log (Phase 25 migration 010 enables this).
    const result = { created, linked, skipped, matches: defRows.rowCount };
    await client.query(
      `INSERT INTO turion_satellite.audit_log (entity_type, entity_id, action, payload, actor_user_id, actor_email)
       VALUES ('sales_order', $1, 'sync_sales_order', $2::jsonb, $3, $4)`,
      [soId, JSON.stringify({ ...result, satelliteId }), req.user?.id ?? null, req.user?.id ?? null]
    );

    await client.query('COMMIT');
    res.json(result);
  } catch (err: any) {
    try { await client.query('ROLLBACK'); } catch (_) { /* ignore */ }
    console.error('[integration] sync-sales-order failed:', err);
    res.status(500).json({ error: 'Failed to sync sales order' });
  } finally {
    client.release();
  }
});
```

**Endpoint 2 — POST /sync-ns-invoice/:invoiceId (links vendor_orders.ns_invoice_id):**

```typescript
// POST /api/integration/sync-ns-invoice/:invoiceId
// Reads turion.invoices.source_data (expecting {vendor, lineItems:[{partNumber}, ...]} or
// {vendor, item}). Links matching turion_satellite.vendor_orders rows by part_number+vendor pair.
// Returns { linked, skipped, matches, reason? }.
router.post('/sync-ns-invoice/:invoiceId', requireAuth, async (req, res) => {
  const invoiceId = String(req.params.invoiceId || '');
  if (!invoiceId) { res.status(400).json({ error: 'invoiceId path param required' }); return; }

  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    const invRows = await client.query(
      'SELECT id, source_data FROM turion.invoices WHERE id = $1',
      [invoiceId]
    );
    if (invRows.rowCount === 0) {
      await client.query('ROLLBACK');
      res.status(404).json({ error: 'Invoice not found' });
      return;
    }
    const inv = invRows.rows[0];

    // Extract part_number candidates from invoice source_data (try lineItems[*].partNumber, then top-level item/itemId).
    const lineItems: any[] = Array.isArray(inv.source_data?.lineItems) ? inv.source_data.lineItems : [];
    const candidates: string[] = lineItems
      .map((li: any) => li?.partNumber || li?.part_number || li?.itemId || li?.item)
      .filter((s: any): s is string => typeof s === 'string' && s.length > 0);
    if (candidates.length === 0) {
      const topItem = inv.source_data?.item || inv.source_data?.itemId;
      if (typeof topItem === 'string' && topItem.length > 0) candidates.push(topItem);
    }

    if (candidates.length === 0) {
      await client.query('COMMIT');
      res.json({ linked: 0, skipped: 0, matches: 0, reason: 'no_part_numbers' });
      return;
    }

    // Find matching vendor_orders by part_number (via part_definitions JOIN). Optional vendor filter
    // skipped here because invoice.source_data.vendor is free text and vendor_orders.vendor_id is a UUID;
    // exact vendor match would need a separate name lookup. v1: match by part_number alone — Phase 26
    // can refine.
    const linkRows = await client.query(
      `SELECT vo.id, vo.ns_invoice_id
       FROM turion_satellite.vendor_orders vo
       JOIN turion_satellite.part_instances pi ON pi.id = vo.part_instance_id
       JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
       WHERE pd.part_number = ANY($1::text[])`,
      [candidates]
    );

    if (linkRows.rowCount === 0) {
      await client.query('COMMIT');
      res.json({ linked: 0, skipped: 0, matches: 0, reason: 'no_vendor_order_matches', candidates_tried: candidates });
      return;
    }

    let linked = 0, skipped = 0;
    for (const row of linkRows.rows) {
      if (row.ns_invoice_id === invoiceId) { skipped++; continue; }
      await client.query(
```

**Endpoint 3 — POST /sync-arena-doc (body-param style, IDs may contain `·` and spaces):**

```typescript
// POST /api/integration/sync-arena-doc  Body: { docId: string }
// Body-param style (not path) because legacy arena_docs.id can contain Unicode middle dot
// and spaces (e.g. '5318-A · Bus integ GA'), which is fragile in URL paths even with encoding.
// Match strategy: extract a part_number via regex /^([A-Z]+(?:-[A-Z0-9]+)+)/ from
// source_data->>'linked'; fallback to source_data->>'whereUsed'. Returns { linked, skipped, matches, reason? }.
router.post('/sync-arena-doc', requireAuth, async (req, res) => {
  const docId = String((req.body || {}).docId || '');
  if (!docId) { res.status(400).json({ error: 'docId required in body' }); return; }

  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    const docRows = await client.query(
      'SELECT id, source_data FROM turion.arena_docs WHERE id = $1',
      [docId]
    );
    if (docRows.rowCount === 0) {
      await client.query('ROLLBACK');
      res.status(404).json({ error: 'Arena doc not found' });
      return;
    }
    const doc = docRows.rows[0];

    // Extract part_number candidate. Regex matches things like 'STR-ASSY', 'EPS-SOLAR-PANEL', 'ADCS-RW-MEDIUM-A'.
    const PN_REGEX = /^([A-Z]+(?:-[A-Z0-9]+)+)/;
    let candidate: string | null = null;
    let matchVia: string | null = null;
    const linked = String(doc.source_data?.linked || '');
    const whereUsed = String(doc.source_data?.whereUsed || '');
    let m = linked.match(PN_REGEX);
    if (m) { candidate = m[1]; matchVia = 'linked'; }
    if (!candidate) {
      m = whereUsed.match(PN_REGEX);
      if (m) { candidate = m[1]; matchVia = 'whereUsed'; }
    }
    if (!candidate) {
      await client.query('COMMIT');
      res.json({ linked: 0, skipped: 0, matches: 0, reason: 'no_part_number_extracted' });
      return;
    }

    // Find part_instances by part_number across all satellites (sync-arena-doc is system-wide,
    // not satellite-scoped — one arena doc applies to every satellite that uses the part).
    const piRows = await client.query(
      `SELECT pi.id, pi.arena_doc_id
       FROM turion_satellite.part_instances pi
       JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
       WHERE pd.part_number = $1`,
      [candidate]
    );
    if (piRows.rowCount === 0) {
      await client.query('COMMIT');
      res.json({ linked: 0, skipped: 0, matches: 0, reason: 'no_part_instance_matches', candidate_tried: candidate, match_via: matchVia });
      return;
    }

    let linkedCount = 0, skipped = 0;
    for (const row of piRows.rows) {
      if (row.arena_doc_id === docId) { skipped++; continue; }
      await client.query(
        `UPDATE turion_satellite.part_instances
         SET arena_doc_id = $1, cross_links_updated_at = NOW()
         WHERE id = $2`,
        [docId, row.id]
      );
      linkedCount++;
    }

    const result = { linked: linkedCount, skipped, matches: piRows.rowCount, match_via: matchVia };
    await client.query(
      `INSERT INTO turion_satellite.audit_log (entity_type, entity_id, action, payload, actor_user_id, actor_email)
       VALUES ('arena_doc', $1, 'sync_arena_doc', $2::jsonb, $3, $4)`,
      [docId, JSON.stringify(result), req.user?.id ?? null, req.user?.id ?? null]
    );

    await client.query('COMMIT');
    res.json(result);
  } catch (err: any) {
    try { await client.query('ROLLBACK'); } catch (_) {}
    console.error('[integration] sync-arena-doc failed:', err);
    res.status(500).json({ error: 'Failed to sync arena doc' });
  } finally {
    client.release();
  }
});
```

- All four endpoints follow the same skeleton: BEGIN → fetch source → extract candidates → match part_definitions → mutate FK → audit_log → COMMIT.

Restore the dump into temporary schemas on Aurora (NOT into the real public/crm/turion/turion_satellite — those stay empty until 54.5-03). Verify parity. Then DROP the dry-run schemas so Aurora returns to empty baseline.

**Strategy:** pg_restore doesn't support `--schema-mapping`. We restore into the real schema names but FIRST move them out of the way:
1. Rename real (currently empty) schemas to `<name>_baseline`
2. Restore (creates real names from dump)
3. Rename real to `_dryrun_<name>`
4. Rename `_baseline` back to original name (restoring empty Aurora baseline)
5. Verify parity from `_dryrun_<name>` against Supabase baseline
6. DROP the `_dryrun_*` schemas CASCADE

```bash
source /Users/jeet/doordash-p2p/.planning/phases/54.5-aurora-postgres-migration-leave-supabase/aurora-cutover.env
MASTER_PW=$(aws secretsmanager get-secret-value --secret-id "$MASTER_SECRET_ARN" \
  --query SecretString --output text --region us-east-1 | jq -r .password)
ENC_PW=$(printf '%s' "$MASTER_PW" | jq -sRr @uri)
AURORA_URL="postgres://zietra_admin:${ENC_PW}@${WRITER}:5432/zietra"

# Step 1: Rename empty schemas out of the way
psql "$AURORA_URL" -v ON_ERROR_STOP=1 <<'SQL'
ALTER SCHEMA public RENAME TO public_baseline;
ALTER SCHEMA crm RENAME TO crm_baseline;
ALTER SCHEMA turion RENAME TO turion_baseline;
ALTER SCHEMA turion_satellite RENAME TO turion_satellite_baseline;
CREATE SCHEMA public;  -- pg_restore needs target to exist or it auto-creates; safer to pre-create
SQL

# Step 2: Restore dump (creates schemas + tables + indexes + FKs + COPY data)
pg_restore \
  --dbname="$AURORA_URL" \
  --no-owner --no-privileges \
  --jobs=4 --verbose --exit-on-error \
  /tmp/zietra-supabase-dryrun.dump 2>&1 | tee /tmp/restore.log

# Hard-fail on any ERROR
grep -iE "^pg_restore: error|^ERROR:" /tmp/restore.log && \
  { echo "FAIL: restore had errors"; exit 1; } || \
  echo "PASS: restore clean"

# Step 3+4: Rename real → _dryrun_*; rename _baseline → real (empty)
psql "$AURORA_URL" -v ON_ERROR_STOP=1 <<'SQL'
ALTER SCHEMA public RENAME TO _dryrun_public;
ALTER SCHEMA crm RENAME TO _dryrun_crm;
ALTER SCHEMA turion RENAME TO _dryrun_turion;
ALTER SCHEMA turion_satellite RENAME TO _dryrun_turion_satellite;

ALTER SCHEMA public_baseline RENAME TO public;
ALTER SCHEMA crm_baseline RENAME TO crm;
ALTER SCHEMA turion_baseline RENAME TO turion;
ALTER SCHEMA turion_satellite_baseline RENAME TO turion_satellite;
SQL

# Step 5: Sequence resync against _dryrun_* (RESEARCH §C.4 adapted for dryrun naming)
psql "$AURORA_URL" -At -c "
  SELECT 'SELECT setval(' ||
         quote_literal(quote_ident(n.nspname) || '.' || quote_ident(s.relname)) ||
         ', COALESCE(MAX(' || quote_ident(a.attname) || '), 1)) FROM ' ||
         quote_ident(tn.nspname) || '.' || quote_ident(t.relname) || ';'
  FROM pg_class s
  JOIN pg_namespace n ON n.oid = s.relnamespace
  JOIN pg_depend d ON d.objid = s.oid AND d.deptype = 'a'
  JOIN pg_class t ON t.oid = d.refobjid
  JOIN pg_namespace tn ON tn.oid = t.relnamespace
  JOIN pg_attribute a ON a.attrelid = t.oid AND a.attnum = d.refobjsubid
  WHERE s.relkind = 'S'
    AND n.nspname IN ('_dryrun_public', '_dryrun_crm', '_dryrun_turion', '_dryrun_turion_satellite');
" > /tmp/sequence-resync-dryrun.sql
psql "$AURORA_URL" -f /tmp/sequence-resync-dryrun.sql 2>&1 | tee /tmp/sequence-resync.log

# Step 6: Row-count parity check (REWRITE schema names to match _dryrun_*)
psql "$AURORA_URL" -At -c "
  SELECT replace(schemaname,'_dryrun_','') || '.' || tablename || ',' ||
         (xpath('/row/c/text()', query_to_xml(
            format('SELECT COUNT(*) AS c FROM %I.%I', schemaname, tablename),
            false, true, '')))[1]::text::int
  FROM pg_tables
  WHERE schemaname IN ('_dryrun_public', '_dryrun_crm', '_dryrun_turion', '_dryrun_turion_satellite')
  ORDER BY 1;
" > /tmp/aurora-dryrun-counts.csv

# Diff — MUST be empty
diff /tmp/supabase-baseline-counts.csv /tmp/aurora-dryrun-counts.csv > /tmp/parity-diff.txt
if [ -s /tmp/parity-diff.txt ]; then
  echo "FAIL: row-count parity drift"; cat /tmp/parity-diff.txt; exit 1
else
  echo "PASS: row-count parity (153 tables match)"
fi
```

```bash
    INSERT INTO ${TURION_SCHEMA}.audit_log (id, action, target_type, target_id, actor_id, payload, created_at)
    VALUES (gen_random_uuid(), 'SMOKE_TEST', 'cutover', '$SENTINEL', NULL, '{}'::jsonb, now());

    "SELECT count(*) FROM ${TURION_SCHEMA}.audit_log WHERE target_id='$SENTINEL';")

  PGPASSWORD="$MASTER_PW" psql -h "$WRITER" -U zietra_admin -d zietra -c \
```

Create the Aurora Serverless v2 cluster + writer instance. Exact commands from RESEARCH §A.6 + §A.7. Writer endpoint and auto-generated master secret ARN are captured into the operator env file.

**No CI/CD here.** Aurora provisioning is direct `aws rds` — NOT a Lambda deploy. Do NOT use `gh workflow run`.

```bash
# Cluster (uses SG_ID from Task 1)
aws rds create-db-cluster \
  --db-cluster-identifier zietra-aurora-prod \
  --engine aurora-postgresql \
  --engine-version 16.4 \
  --master-username zietra_admin \
  --manage-master-user-password \
  --database-name zietra \
  --db-subnet-group-name zietra-aurora-subnets \
  --vpc-security-group-ids $SG_ID \
  --serverless-v2-scaling-configuration MinCapacity=0.5,MaxCapacity=4 \
  --backup-retention-period 7 \
  --preferred-backup-window "07:00-09:00" \
  --preferred-maintenance-window "sun:09:00-sun:11:00" \
  --storage-encrypted \
  --enable-cloudwatch-logs-exports '["postgresql"]' \
  --enable-iam-database-authentication \
  --tags Key=Project,Value=Zietra Key=Environment,Value=production Key=Phase,Value=54.5 \
  --region us-east-1

# Writer instance
aws rds create-db-instance \
  --db-cluster-identifier zietra-aurora-prod \
  --db-instance-identifier zietra-aurora-prod-writer \
  --db-instance-class db.serverless \
  --engine aurora-postgresql \
  --publicly-accessible \
  --tags Key=Project,Value=Zietra Key=Environment,Value=production Key=Phase,Value=54.5 \
  --region us-east-1

# Wait — cluster + instance both available (each waiter polls every 30s, exits when ready)
aws rds wait db-cluster-available --db-cluster-identifier zietra-aurora-prod --region us-east-1
aws rds wait db-instance-available --db-instance-identifier zietra-aurora-prod-writer --region us-east-1

# Capture writer endpoint
WRITER=$(aws rds describe-db-clusters \
  --db-cluster-identifier zietra-aurora-prod \
  --query 'DBClusters[0].Endpoint' --output text --region us-east-1)
echo "WRITER=$WRITER"
# Expected pattern: zietra-aurora-prod.cluster-XXXXXXXXXXXX.us-east-1.rds.amazonaws.com

# Capture auto-managed master secret ARN
MASTER_SECRET_ARN=$(aws rds describe-db-clusters \
  --db-cluster-identifier zietra-aurora-prod \
  --query 'DBClusters[0].MasterUserSecret.SecretArn' --output text --region us-east-1)
echo "MASTER_SECRET_ARN=$MASTER_SECRET_ARN"

# Resolve the master password (operator-only — never log or commit)
MASTER_PW=$(aws secretsmanager get-secret-value --secret-id $MASTER_SECRET_ARN \
  --query SecretString --output text --region us-east-1 | jq -r .password)

# URL-encode for connection string
ENC_PW=$(printf '%s' "$MASTER_PW" | jq -sRr @uri)
AURORA_URL="postgres://zietra_admin:${ENC_PW}@${WRITER}:5432/zietra"

# Enable Performance Insights on writer (free-tier 7-day retention)
aws rds modify-db-instance \
  --db-instance-identifier zietra-aurora-prod-writer \
  --enable-performance-insights \
  --performance-insights-retention-period 7 \
  --apply-immediately \
  --region us-east-1
```

Build the daily soak check script. It runs the read-only smoke (no SMOKE_WRITE), greps Lambda CloudWatch logs for any Supabase-host references in the last 24 hours, captures CloudWatch alarm states, and writes a per-day report to the runbook.

**`scripts/aurora-soak-day-check.sh`:**
```bash
#!/usr/bin/env bash
# Phase 54.5-04 daily soak check — argv1 = day number (1, 3, 5, 7)
set -euo pipefail
DAY="${1:-?}"
REPORT="/tmp/aurora-soak-day-${DAY}.txt"
echo "=== Aurora soak — Day $DAY — $(date -u) ===" | tee "$REPORT"

# 1. Read-only smoke (no writes — production traffic continues normally)
SMOKE_WRITE=0 SMOKE_SCHEMA_PREFIX="" \
  bash /Users/jeet/doordash-p2p/scripts/aurora-cutover-smoke.sh 2>&1 \
  | tee -a "$REPORT" \
  || { echo "SMOKE FAIL on Day $DAY" | tee -a "$REPORT"; SMOKE_PASS=0; }
SMOKE_PASS=${SMOKE_PASS:-1}

# 2. Grep last 24h Lambda logs for Supabase host references (must be 0)
END=$(date -u +%s)
START=$((END - 86400))   # 24h ago
echo "" >> "$REPORT"
echo "=== Supabase-host references in last 24h ===" >> "$REPORT"
TOTAL_REFS=0
for L in turion-demo-api turion-satellite-api zietra-crm-api zietra-api; do
  COUNT=$(aws logs filter-log-events \
    --log-group-name "/aws/lambda/$L" \
    --start-time $((START * 1000)) --end-time $((END * 1000)) \
    --filter-pattern "pooler.supabase.com" \
    --region us-east-1 \
    --query 'events[].message' --output text 2>/dev/null | wc -l)
  echo "$L: $COUNT" | tee -a "$REPORT"
  TOTAL_REFS=$((TOTAL_REFS + COUNT))
done

# 3. CloudWatch alarm states (any in ALARM = investigate)
echo "" >> "$REPORT"
echo "=== CloudWatch alarm states ===" >> "$REPORT"
aws cloudwatch describe-alarms --alarm-name-prefix zietra-aurora --region us-east-1 \
  --query 'MetricAlarms[].{Name:AlarmName,State:StateValue}' --output table | tee -a "$REPORT"

ALARMS_FIRED=$(aws cloudwatch describe-alarms --alarm-name-prefix zietra-aurora --region us-east-1 \
  --state-value ALARM --query 'MetricAlarms[].AlarmName' --output text | wc -w)

# 4. Verdict
echo "" >> "$REPORT"
if [ "$SMOKE_PASS" = "1" ] && [ "$TOTAL_REFS" = "0" ] && [ "$ALARMS_FIRED" = "0" ]; then
  echo "VERDICT: Day $DAY CLEAN" | tee -a "$REPORT"
  exit 0
else
  echo "VERDICT: Day $DAY DIRTY — smoke=$SMOKE_PASS supabase_refs=$TOTAL_REFS alarms=$ALARMS_FIRED" | tee -a "$REPORT"
  exit 1
fi
```

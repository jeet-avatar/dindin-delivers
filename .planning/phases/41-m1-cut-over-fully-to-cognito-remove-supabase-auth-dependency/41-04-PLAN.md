**Step B: Remove `SUPABASE_JWT_SECRET_ARN` env var from both Lambdas**
(research §"Pattern 7" — use `file://` JSON form to avoid the
comma-mangling bug from Phase 40 deviation #1):

```bash
for fn in turion-demo-api turion-satellite-api; do
  # Read current env vars
  aws lambda get-function-configuration --function-name "$fn" \
    --query 'Environment.Variables' --output json > /tmp/env-$fn.json

  # Strip SUPABASE_JWT_SECRET_ARN (and any related Supabase keys)
  jq 'del(.SUPABASE_JWT_SECRET_ARN) | del(.SUPABASE_JWT_PUBLIC_KEY) | del(.SUPABASE_JWT_SECRET)' \
    /tmp/env-$fn.json > /tmp/env-$fn-new.json

  # Build the update payload
  jq -n --slurpfile vars /tmp/env-$fn-new.json '{Variables: $vars[0]}' > /tmp/lambda-env-$fn.json

  # Confirm what's about to be applied — verify COGNITO_CONFIG_SECRET_ARN + DATABASE_URL still present
  jq -e '.Variables.COGNITO_CONFIG_SECRET_ARN' /tmp/lambda-env-$fn.json > /dev/null
  jq -e '.Variables.DATABASE_URL // .Variables.DATABASE_URL_ARN' /tmp/lambda-env-$fn.json > /dev/null

  aws lambda update-function-configuration \
    --function-name "$fn" \
    --environment "file:///tmp/lambda-env-$fn.json"

  # Wait for LastUpdateStatus: Successful
  for i in 1 2 3 4 5 6; do
    status=$(aws lambda get-function-configuration --function-name "$fn" --query 'LastUpdateStatus' --output text)
    [ "$status" = "Successful" ] && break
    sleep 5
  done
  echo "$fn LastUpdateStatus: $status"

  # Verify the env var is GONE
  aws lambda get-function-configuration --function-name "$fn" \
    --query 'Environment.Variables.SUPABASE_JWT_SECRET_ARN' --output text
  # Expect: "None" (jq-style null) — meaning the key isn't present.
done
```

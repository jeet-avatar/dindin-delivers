// Phase 58-03 — In-memory rate limiter. Lambda warm containers retain state.
// Cold start resets (acceptable — attacker gets ~limit more requests per cold start).
// Migrate to ElastiCache if RPS ever exceeds 100 (M8).

const buckets = new Map<string, number[]>()
const MAX_KEYS = 10_000 // bound memory (Phase 55 X-Forwarded-For spoof guard pattern)

export function rateLimitOk(key: string, opts: { limit: number; windowMs: number }): boolean {
  const now = Date.now()
  const cutoff = now - opts.windowMs

  // Bound the map
  if (buckets.size > MAX_KEYS) {
    // Drop oldest entries (best-effort — actual LRU is overkill here)
    const oldest = buckets.keys().next().value
    if (oldest) buckets.delete(oldest)
  }

  const arr = (buckets.get(key) ?? []).filter(ts => ts > cutoff)
  if (arr.length >= opts.limit) {
    buckets.set(key, arr)
    return false
  }
  arr.push(now)
  buckets.set(key, arr)
  return true
}
```

  }

  let body: any
  try { body = await c.req.json() } catch { return c.json({ error: 'Invalid JSON' }, 400, { 'Access-Control-Allow-Origin': origin }) }

  // Manual validation (zod alternative)
  const name = String(body.name ?? '').trim()
  const email = String(body.email ?? '').trim().toLowerCase()
  const company = body.company ? String(body.company).trim() : null
  const intent = String(body.intent ?? '').trim()
  const message = String(body.message ?? '').trim()
  const website = String(body.website ?? '')

  if (name.length < 1 || name.length > 200) return c.json({ error: 'Invalid name.' }, 400, { 'Access-Control-Allow-Origin': origin })
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || email.length > 320) return c.json({ error: 'Invalid email.' }, 400, { 'Access-Control-Allow-Origin': origin })
  if (company && company.length > 200) return c.json({ error: 'Invalid company.' }, 400, { 'Access-Control-Allow-Origin': origin })
  if (!VALID_INTENTS.has(intent)) return c.json({ error: 'Invalid intent.' }, 400, { 'Access-Control-Allow-Origin': origin })
  if (message.length < 10 || message.length > 5000) return c.json({ error: 'Invalid message.' }, 400, { 'Access-Control-Allow-Origin': origin })

  // Honeypot — silently succeed without persist/send
  if (website.length > 0) {
    return c.json({ ok: true }, 200, { 'Access-Control-Allow-Origin': origin })
  }

  // Persist
  const { rows } = await pool.query(
    `INSERT INTO public.contact_submissions (name, email, company, intent, message, ip, user_agent)
     VALUES ($1,$2,$3,$4,$5,$6,$7) RETURNING id`,
    [name, email, company, intent, message, ip, c.req.header('user-agent') ?? null],
  )

  // SES send (best-effort; failure does NOT roll back the row)
  try {
    await ses.send(new SendEmailCommand({
      FromEmailAddress: 'noreply@zietra.com',
      Destination: { ToAddresses: ['support@zietra.com'] },
      ReplyToAddresses: [email],
      Content: { Simple: {
        Subject: { Data: `[${intent.toUpperCase()}] ${name}${company ? ` (${company})` : ''} — Zietra contact form` },
        Body: { Text: { Data:
`From: ${name} <${email}>
Company: ${company ?? '—'}
Intent: ${intent}
IP: ${ip}
Submission ID: ${rows[0].id}

Message:
${message}
` } },
      }},
    }))
  } catch (err) {
    console.error('[contact] SES send failed (row persisted):', (err as any)?.message ?? err)
  }

  return c.json({ ok: true, id: rows[0].id }, 200, { 'Access-Control-Allow-Origin': origin })
})
```

**Step C — Build `src/components/PageHelmet.tsx`** (~30 LOC, provided for M8 retrofit):

```tsx
import { Helmet } from 'react-helmet-async'

interface Props {
  title: string             // Page-specific (e.g., "Salesforce CRM")
  description: string
  path: string              // e.g., "/modules/crm" (canonical URL path)
  ogImage?: string          // defaults to /og/default.png
}

export function PageHelmet({ title, description, path, ogImage }: Props) {
  const fullTitle = title.endsWith('Zietra') ? title : `${title} · Zietra`
  const canonical = `https://zietra.com${path}`
  const og = ogImage ?? '/og/default.png'
  return (
    <Helmet>
      <title>{fullTitle}</title>
      <meta name="description" content={description} />
      <meta property="og:title" content={fullTitle} />
      <meta property="og:description" content={description} />
      <meta property="og:image" content={og} />
      <meta property="og:url" content={canonical} />
      <meta property="og:type" content="website" />
      <link rel="canonical" href={canonical} />
    </Helmet>
  )
}
```

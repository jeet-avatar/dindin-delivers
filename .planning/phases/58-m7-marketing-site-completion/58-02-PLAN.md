```javascript
import { readFile, writeFile } from 'node:fs/promises'
import { fileURLToPath } from 'node:url'
import { dirname, resolve } from 'node:path'

const __dirname = dirname(fileURLToPath(import.meta.url))
const CATALOG_PATH = process.env.MODULE_CATALOG_PATH
  ?? '/Users/jeet/turion-space-demo/lib/module-catalog.js'

const src = await readFile(CATALOG_PATH, 'utf8')

// Upstream is browser-side: `window.MODULE_CATALOG = [...]`
const mod = { MODULE_CATALOG: [] }
new Function('window', src)(mod)
if (!Array.isArray(mod.MODULE_CATALOG) || mod.MODULE_CATALOG.length === 0) {
  throw new Error(`sync-modules: upstream MODULE_CATALOG empty (path: ${CATALOG_PATH})`)
}
const upstreamCodes = new Set(mod.MODULE_CATALOG.map(m => m.code))

const { MARKETING_MODULES } = await import('./marketing-copy.mjs')
const marketingCodes = new Set(MARKETING_MODULES.map(m => m.slug))

const missing = [...upstreamCodes].filter(c => !marketingCodes.has(c))
const extra   = [...marketingCodes].filter(c => !upstreamCodes.has(c))
if (missing.length) throw new Error(`sync-modules: marketing copy MISSING for upstream codes: ${missing.join(', ')}. Add to scripts/marketing-copy.mjs.`)
if (extra.length)   throw new Error(`sync-modules: marketing copy is STALE for codes not in upstream: ${extra.join(', ')}. Remove from scripts/marketing-copy.mjs OR add to upstream.`)

// Merge upstream metadata with marketing copy
const merged = mod.MODULE_CATALOG.map(m => {
  const mc = MARKETING_MODULES.find(x => x.slug === m.code)
  return {
    slug: m.code,
    code: m.code,
    title: mc.title,
    shortName: mc.shortName,
    source: m.source ?? '',
    icon: m.icon ?? 'box',
    tagline: mc.tagline,
    description: mc.description,
    useCases: mc.useCases,
    whyWeBuilt: mc.whyWeBuilt,
    bottomCta: mc.bottomCta,
    inAppUrl: mc.inAppUrl,
    screenshot: null, // M8 fills in real screenshots
  }
})

const outPath = resolve(__dirname, '../src/data/modules.ts')
const ts = `// AUTOGENERATED by scripts/sync-modules.mjs — DO NOT EDIT MANUALLY.
// Upstream catalog: ${CATALOG_PATH}
// Marketing copy: scripts/marketing-copy.mjs
// Regenerate: \`npm run build\` (runs sync-modules first).

export interface ModuleUseCase {
  industry: 'D2C' | 'SaaS' | 'Manufacturing' | 'Aerospace' | 'Services' | 'Distribution'
  bullets: string[]
}

export interface ModuleEntry {
  slug: string
  code: string
  title: string
  shortName: string
  source: string
  icon: string
  tagline: string
  description: string
  useCases: ModuleUseCase[]
  whyWeBuilt: string
  bottomCta: string
  inAppUrl: string
  screenshot: string | null
}

export const MODULES: ModuleEntry[] = ${JSON.stringify(merged, null, 2)} as const satisfies readonly ModuleEntry[]
`

await writeFile(outPath, ts)
console.log(`✅ Synced ${merged.length} modules → ${outPath}`)
```

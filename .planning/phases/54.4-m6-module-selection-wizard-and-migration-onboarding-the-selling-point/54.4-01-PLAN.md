```typescript
// backend/src/onboarding/rule-engine.ts — Phase 54.4-01
// PURE — no imports from db/cognito/secrets. Testable in isolation. Runs on server OR client identically.
import RULES from './recommend-rules.json';

export type Industry = 'aerospace'|'d2c-ecommerce'|'saas'|'manufacturing'|'distribution'|'services'|'other';
export type TeamSize = 'solo'|'2-10'|'11-50'|'51-200'|'201-1000'|'1000+';
export type PainPoint = 'no-erp'|'spreadsheet-hell'|'qb-too-small'|'compliance-asc606'|'no-crm'|'inventory-chaos';
export type Tool = 'quickbooks'|'salesforce'|'netsuite'|'excel'|'sap'|'none';
export type Asc606Needs = 'yes-now'|'maybe-2026'|'no';

export type Answers = {
  industry: Industry;
  team_size: TeamSize;
  pain_point?: PainPoint;
  tools_today?: Tool[];
  asc606_needs?: Asc606Needs;
};

export type Recommendation = {
  module_code: string;
  score: number;
  reasons: string[];
};

export const ALL_MODULES = [
  'crm','sales','purchase','items','plm','mes','quality',
  'lean-erp-pro','asc606','royalty','dropship','ai-agents','qb-migration',
] as const;

export function recommend(answers: Answers): Recommendation[] {
  const scores = new Map<string, { score: number; reasons: string[] }>();
  for (const m of ALL_MODULES) scores.set(m, { score: 0, reasons: [] });

  for (const [questionKey, answer] of Object.entries(answers)) {
    if (answer === undefined || answer === null) continue;
    const answerList = Array.isArray(answer) ? answer : [answer];
    for (const ans of answerList) {
      const contribution = (RULES as any)[questionKey]?.[ans] ?? {};
      for (const [moduleCode, sr] of Object.entries<any>(contribution)) {
        const entry = scores.get(moduleCode);
        if (!entry) continue;
        entry.score += sr.score;
        if (sr.reason) entry.reasons.push(sr.reason);
      }
    }
  }

  // Default threshold: include modules scoring >= 3, cap at top 7.
  let included = [...scores.entries()]
    .map(([code, v]) => ({ module_code: code, score: v.score, reasons: v.reasons }))
    .filter(r => r.score >= 3)
    .sort((a, b) => b.score - a.score)
    .slice(0, 7);

  // Fallback: if < 3 recommendations, lower threshold to >= 2.
  if (included.length < 3) {
    included = [...scores.entries()]
      .map(([code, v]) => ({ module_code: code, score: v.score, reasons: v.reasons }))
      .filter(r => r.score >= 2)
      .sort((a, b) => b.score - a.score)
      .slice(0, 7);
  }

  return included;
}
```

```typescript
// backend/src/routes/onboarding.ts — Phase 54.4 (this plan: 54.4-01 routes only)
import { Router, Request, Response } from 'express';
import { withTenantClient } from '../db';
import { requireAuth } from '../middleware/auth';
import { requireRole } from '../middleware/role';
import { tenantContext } from '../middleware/tenant';
import { recommend, type Answers, ALL_MODULES } from '../onboarding/rule-engine';

const r = Router();

// All routes require tenantContext + requireAuth. Mutating routes also require admin role.
r.use(tenantContext, requireAuth);

// GET /api/onboarding/rules — return the rules JSON for client-side preview (read-only).
r.get('/rules', (_req, res) => {
  // require() works at runtime because tsconfig has resolveJsonModule + esModuleInterop.
  const RULES = require('../onboarding/recommend-rules.json');
  res.json(RULES);
});

// POST /api/onboarding/recommend — pure compute. Any authenticated tenant user can call.
r.post('/recommend', (req: Request, res: Response) => {
  const answers = req.body as Answers;
  if (!answers || !answers.industry || !answers.team_size) {
    return res.status(400).json({ error: 'industry and team_size are required' });
  }
  const recommendations = recommend(answers);
  return res.json({ recommendations, total_recommended: recommendations.length });
});

// POST /api/onboarding/finalize — admin-only — writes tenant_features + onboarding_state.
r.post('/finalize', requireRole('admin'), async (req: Request, res: Response) => {
  const selected: string[] = Array.isArray(req.body?.selected_modules)
    ? req.body.selected_modules
    : [];
  if (selected.length < 1 || selected.length > 13) {
    return res.status(400).json({ error: 'Select between 1 and 13 modules' });
  }
  const VALID = new Set<string>(ALL_MODULES);
  if (!selected.every(c => VALID.has(c))) {
    return res.status(400).json({ error: 'Unknown module code in selection' });
  }
  try {
    await withTenantClient(req, async (client) => {
      // Single transaction: disable all, enable the chosen ones, update onboarding_state.
      await client.query(
        `UPDATE public.tenant_features SET enabled = false WHERE tenant_id = $1`,
        [(req as any).tenant.id],
      );
      for (const code of selected) {
        await client.query(
          `UPDATE public.tenant_features SET enabled = true
             WHERE tenant_id = $1 AND module_code = $2`,
          [(req as any).tenant.id, code],
        );
      }
      await client.query(
        `UPDATE public.tenants
           SET onboarding_state = jsonb_set(
             jsonb_set(onboarding_state, '{checklist,modules}', 'true'::jsonb),
             '{wizard_completed_at}', to_jsonb(now()::text)
           )
           WHERE id = $1`,
        [(req as any).tenant.id],
      );
    });
    return res.json({ ok: true, redirect: '/' });
  } catch (e: any) {
    console.error('[POST /api/onboarding/finalize] DB error:', e?.message);
    return res.status(500).json({ error: 'Finalize failed' });
  }
});

export default r;
```

// Generic CSV importer. Pure data-shape — caller provides rows already parsed by papaparse client-side.
import type { Request } from 'express';
import { withTenantClient } from '../db';

const MAX_ROWS_PER_REQUEST = 5000;

type ItemRow    = { sku: string; description?: string; list_price?: number; cost?: number };
type VendorRow  = { name: string; email?: string; payment_terms?: string };
type CustomerRow = { name: string; email?: string; billing_address?: string };

export async function importItems(req: Request, rows: ItemRow[]): Promise<{ inserted: number }> {
  if (rows.length > MAX_ROWS_PER_REQUEST) throw new Error(`Max ${MAX_ROWS_PER_REQUEST} rows per request`);
  const valid = rows.filter(r => r && typeof r.sku === 'string' && r.sku.trim().length > 0);
  let inserted = 0;
  await withTenantClient(req, async (client) => {
    for (const row of valid) {
      await client.query(
        `INSERT INTO turion.ns_items (id, tenant_id, sku, description, list_price, cost)
           VALUES (gen_random_uuid(), $1, $2, $3, $4, $5)
           ON CONFLICT DO NOTHING`,
        [(req as any).tenant.id, row.sku.trim(), row.description ?? null, row.list_price ?? null, row.cost ?? null],
      );
      inserted += 1;
    }
    // Mark checklist data step complete.
    await client.query(
      `UPDATE public.tenants SET onboarding_state = jsonb_set(onboarding_state, '{checklist,data}', 'true'::jsonb) WHERE id = $1`,
      [(req as any).tenant.id],
    );
  });
  return { inserted };
}

// backend/src/onboarding/sf-csv-import.ts — Phase 54.4-02
// Salesforce account import — FIELD_MAPS pattern mirroring Phase 37 quickbooks.ts:41.
import type { Request } from 'express';
import { withTenantClient } from '../db';

const MAX_ROWS_PER_REQUEST = 5000;

// FIELD_MAPS: SF column-label-candidates → our DB column. Used by the frontend column-mapper UI
// for the auto-map step; the backend just receives already-mapped rows.
export const FIELD_MAPS: Record<string, string[]> = {
  name:           ['Account Name', 'Name', 'AccountName'],
  email:          ['Email', 'Email Address'],
  phone:          ['Phone', 'Phone Number'],
  industry:       ['Industry', 'Industry__c'],
  annual_revenue: ['Annual Revenue', 'AnnualRevenue'],
  description:   ['Description', 'Notes'],
};

type SfAccountRow = {
  name: string;
  email?: string;
  phone?: string;
  industry?: string;
  annual_revenue?: number | string | null;
  description?: string;
};

export async function importSalesforceAccounts(
  req: Request,
  rows: SfAccountRow[],
): Promise<{ inserted: number; skipped: number }> {
  if (rows.length > MAX_ROWS_PER_REQUEST) throw new Error(`Max ${MAX_ROWS_PER_REQUEST} rows per request`);
  const valid = rows.filter(r => r && typeof r.name === 'string' && r.name.trim().length > 0);
  const skipped = rows.length - valid.length;
  let inserted = 0;

  await withTenantClient(req, async (client) => {
    for (const row of valid) {
      // Coerce annual_revenue: allow numeric strings, ignore unparseable.
      let revenue: number | null = null;
      if (typeof row.annual_revenue === 'number') revenue = row.annual_revenue;
      else if (typeof row.annual_revenue === 'string' && row.annual_revenue.trim()) {
        const n = parseFloat(row.annual_revenue.replace(/[^\d.-]/g, ''));
        revenue = Number.isFinite(n) ? n : null;
      }
      await client.query(
        `INSERT INTO turion.sf_accounts (id, tenant_id, name, email, phone, industry, annual_revenue, description)
           VALUES (gen_random_uuid(), $1, $2, $3, $4, $5, $6, $7)
           ON CONFLICT DO NOTHING`,
        [(req as any).tenant.id, row.name.trim(), row.email ?? null, row.phone ?? null,
         row.industry ?? null, revenue, row.description ?? null],
      );
      inserted += 1;
    }
    await client.query(
      `UPDATE public.tenants SET onboarding_state = jsonb_set(onboarding_state, '{checklist,data}', 'true'::jsonb) WHERE id = $1`,
      [(req as any).tenant.id],
    );
  });

  return { inserted, skipped };
}

```typescript
// backend/src/onboarding/sample-data-clone.ts — Phase 54.4-02
// SAFETY: bypassPool for SELECT (cross-tenant read), withTenantClient for INSERT (RLS-enforced).
// PK remap dictionary handles FK rewrites between parent/child tables.
import type { Request } from 'express';
import { randomUUID } from 'crypto';
import { withTenantClient } from '../db';
import { bypassPool } from '../db-bypass';

const TURION_TENANT_ID = '00000000-0000-0000-0000-000000000001';

// Clone in parent-first order. Children that have FK to a cloned parent must come AFTER the parent.
// Adjust to match the actual `turion.*` schema dependency graph in Aurora.
export const CLONE_TABLES_IN_ORDER: Array<{ table: string; fkRewrites?: Record<string, string> }> = [
  { table: 'turion.sf_accounts' },
  { table: 'turion.sf_contacts',       fkRewrites: { account_id: 'turion.sf_accounts' } },
  { table: 'turion.sf_opportunities',  fkRewrites: { account_id: 'turion.sf_accounts' } },
  { table: 'turion.ns_customers' },
  { table: 'turion.ns_vendors' },
  { table: 'turion.ns_items' },
  { table: 'turion.ns_journal_entries' },
  { table: 'turion.arena_parts' },
  { table: 'turion.arena_ecos',        fkRewrites: { part_id: 'turion.arena_parts' } },
  { table: 'turion.mes_work_orders',   fkRewrites: { part_id: 'turion.arena_parts' } },
  { table: 'turion.mes_build_steps',   fkRewrites: { work_order_id: 'turion.mes_work_orders' } },
];

export const BLOCKLIST = new Set(['audit_log', 'tenant_users', 'tenant_features', 'api_keys', 'secrets']);

export async function cloneSampleData(req: Request): Promise<{
  rowsByTable: Record<string, number>;
  totalRows: number;
}> {
  const result: Record<string, number> = {};
  let total = 0;
  // Remap: tableName → Map<oldId, newId>.
  const remap: Record<string, Map<string, string>> = {};

  for (const { table, fkRewrites } of CLONE_TABLES_IN_ORDER) {
    if (BLOCKLIST.has(table.split('.')[1])) continue;

    // 1. Cross-tenant SELECT via bypass pool (admin_bypass role has BYPASSRLS).
    const sourceRows = await bypassPool.query(
      `SELECT * FROM ${table} WHERE tenant_id = $1`,
      [TURION_TENANT_ID],
    );

    remap[table] = new Map();

    // 2. INSERT via withTenantClient — RLS sets app.tenant_id; every INSERT row must match.
    await withTenantClient(req, async (client) => {
      for (const row of sourceRows.rows) {
        const cleaned: Record<string, any> = { ...row };
        const oldId = cleaned.id;
        const newId = randomUUID();
        if (oldId) remap[table].set(String(oldId), newId);
        cleaned.id = newId;
        cleaned.tenant_id = (req as any).tenant.id;

        // Rewrite FK columns by looking up parent-table remap.
        if (fkRewrites) {
          for (const [col, parentTable] of Object.entries(fkRewrites)) {
            const parentRemap = remap[parentTable];
            if (cleaned[col] && parentRemap && parentRemap.has(String(cleaned[col]))) {
              cleaned[col] = parentRemap.get(String(cleaned[col]));
            }
          }
        }

        const cols = Object.keys(cleaned);
        const vals = cols.map(c => cleaned[c]);
        const placeholders = cols.map((_, i) => `$${i + 1}`).join(', ');
        await client.query(
          `INSERT INTO ${table} (${cols.join(', ')}) VALUES (${placeholders}) ON CONFLICT DO NOTHING`,
          vals,
        );
      }

      // Mark checklist data step complete (idempotent — same write per table is harmless).
      await client.query(
        `UPDATE public.tenants SET onboarding_state = jsonb_set(onboarding_state, '{checklist,data}', 'true'::jsonb) WHERE id = $1`,
        [(req as any).tenant.id],
      );
    });

    result[table] = sourceRows.rows.length;
    total += sourceRows.rows.length;
  }

  return { rowsByTable: result, totalRows: total };
}
```

const MAX_ROWS = 5000;

function chunkGuard(rows: any[], res: Response): boolean {
  if (!Array.isArray(rows)) {
    res.status(400).json({ error: 'rows array is required' });

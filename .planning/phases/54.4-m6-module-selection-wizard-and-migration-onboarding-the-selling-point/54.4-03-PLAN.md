// === 54.4-03: GET /api/onboarding/state — read-only snapshot of onboarding_state for the current tenant. ===
r.get('/state', async (req: Request, res: Response) => {
  try {
    const state = await withTenantClient(req, async (client) => {
      const result = await client.query(
        `SELECT onboarding_state FROM public.tenants WHERE id = $1`,
        [(req as any).tenant.id],
      );
      return result.rows[0]?.onboarding_state ?? null;
    });
    if (!state) return res.status(404).json({ error: 'Tenant not found' });
    const cl = state.checklist || {};
    const complete = !!cl.modules && !!cl.team && !!cl.data && !!cl.agents;
    return res.json({
      checklist: cl,
      wizard_completed_at: state.wizard_completed_at,
      last_wizard_answers: state.last_wizard_answers,
      complete,
    });
  } catch (e: any) {
    console.error('[GET /api/onboarding/state] error:', e?.message);
    return res.status(500).json({ error: 'Failed to read onboarding state' });
  }
});

// === 54.4-03: PATCH /api/onboarding/checklist — update one checklist item. ===
r.patch('/checklist', requireRole('admin', 'manager'), async (req: Request, res: Response) => {
  const item = String(req.body?.item || '');
  const completed = req.body?.completed === true;
  const VALID = new Set(['modules', 'team', 'data', 'agents']);
  if (!VALID.has(item)) return res.status(400).json({ error: 'Unknown checklist item' });
  try {
    await withTenantClient(req, async (client) => {
      await client.query(
        `UPDATE public.tenants
           SET onboarding_state = jsonb_set(onboarding_state, ARRAY['checklist', $1], $2::jsonb)
           WHERE id = $3`,
        [item, JSON.stringify(completed), (req as any).tenant.id],
      );
    });
    return res.json({ ok: true });
  } catch (e: any) {
    console.error('[PATCH /api/onboarding/checklist] error:', e?.message);
    return res.status(500).json({ error: 'Checklist update failed' });
  }
});
```

// /lib/onboarding-checklist.js — Phase 54.4-03
// Self-contained checklist widget. Reads tenant.onboarding_state (preloaded by app-shell.js into
// window.__ZIETRA_TENANT) and renders 3-4 steps. PATCHes /api/onboarding/checklist on Mark-done.
(function () {
  const ITEMS_BASE = [
    { key: 'modules', title: 'Pick your modules', cta: '/onboarding/recommend' },
    { key: 'team',    title: 'Invite your team',  cta: '/team' },
    { key: 'data',    title: 'Migrate your data', cta: '/onboarding/migrate' },
  ];
  // 4th step renders ONLY when tenant has ai-agents enabled.
  const AGENTS_ITEM = { key: 'agents', title: 'Connect to AI agents', cta: '/agents' };

  function escapeHtml(s){return String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));}

  async function init() {
    const mountEl = document.getElementById('onboarding-checklist');
    if (!mountEl) return;

    // Wait for app-shell to load the tenant; fall back to direct fetch if shell hasn't run.
    let tenant = window.__ZIETRA_TENANT;
    if (!tenant) {
      try { const r = await window.erpApi.get('/api/tenants/current'); tenant = r.tenant; }
      catch { return; }   // not signed in or API down — silently hide.
    }
    const state = tenant?.onboarding_state || null;
    if (!state || !state.checklist) return;

    const featureCodes = new Set((tenant.features || []).filter(f => f.enabled).map(f => f.module_code));
    const items = featureCodes.has('ai-agents') ? [...ITEMS_BASE, AGENTS_ITEM] : ITEMS_BASE;

    const cl = state.checklist;
    // Treat agents as complete if module not enabled (so "complete" computation still works).
    const effectiveAgents = featureCodes.has('ai-agents') ? !!cl.agents : true;
    const allDone = !!cl.modules && !!cl.team && !!cl.data && effectiveAgents;

    // Hide entirely if complete AND wizard finished > 24h ago.
    if (allDone) {
      const completedAt = state.wizard_completed_at ? new Date(state.wizard_completed_at).getTime() : null;
      const ageHours = completedAt ? (Date.now() - completedAt) / 3.6e6 : 0;
      if (!completedAt || ageHours > 24) { mountEl.style.display = 'none'; return; }
      mountEl.innerHTML = '<div class="z-ocl-done">Setup complete</div>';
      return;
    }

    // Render checklist.
    const myRole = (tenant.tenant_users || []).find(u => u.is_me)?.role || 'viewer';
    const canMark = myRole === 'admin' || myRole === 'manager';

    mountEl.innerHTML = `
      <h2 class="z-ocl-title">Get set up</h2>
      <ol class="z-ocl-list">
        ${items.map(it => {
          const done = it.key === 'agents' ? effectiveAgents : !!cl[it.key];
          return `<li class="z-ocl-item ${done ? 'done' : ''}">
            <span class="z-ocl-check">${done ? '✓' : ''}</span>
            <a class="z-ocl-link" href="${escapeHtml(it.cta)}">${escapeHtml(it.title)}</a>
            ${!done && canMark ? `<button class="z-ocl-mark" data-item="${escapeHtml(it.key)}">Mark done</button>` : ''}
          </li>`;
        }).join('')}
      </ol>
    `;

    mountEl.addEventListener('click', async (e) => {
      const btn = e.target.closest('.z-ocl-mark');
      if (!btn) return;
      const item = btn.dataset.item;
      btn.disabled = true;
      try {
        await window.erpApi.patch('/api/onboarding/checklist', { item, completed: true });
        // Re-fetch to re-render (simplest correct path).
        const r = await window.erpApi.get('/api/tenants/current');
        window.__ZIETRA_TENANT = r.tenant;
        init();
      } catch (err) {
        btn.disabled = false;
        alert(err?.message || 'Failed to update checklist');
      }
    });
  }

  // Run on DOMContentLoaded or immediately if already loaded.
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
```

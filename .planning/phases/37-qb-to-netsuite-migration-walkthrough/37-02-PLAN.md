    HANDLER SKELETON (~80 lines — research has this drafted; reproduce):
      r.post('/:type/migrate', async (req, res) => {
        const t = req.params.type;
        if (!assertType(t)) return res.status(400).json({ error: 'unknown qb_type' });
        const qbIds: string[] = Array.isArray(req.body?.qbIds) ? req.body.qbIds : [];
        if (qbIds.length === 0) return res.status(400).json({ error: 'qbIds required (non-empty array)' });
        const nsTable = NS_TABLE[t];
        const runIdRow = await query(`SELECT gen_random_uuid()::text AS u`);
        const runId = (runIdRow as any[])[0].u;
        const startedAt = new Date().toISOString();
        const client = await pool.connect();
        const result = { run_id: runId, migrated: [] as string[], skipped: [] as string[], errors: [] as {qbId:string;reason:string}[] };
        try {
          await client.query('BEGIN');
          for (const qbId of qbIds) {
            const src = await client.query(
              `SELECT source_data, status FROM turion.qb_records WHERE qb_type=$1 AND qb_id=$2`, [t, qbId]);
            if (src.rowCount === 0) { result.errors.push({qbId, reason:'not found'}); continue; }
            if (src.rows[0].status === 'migrated') { result.skipped.push(qbId); continue; }
            const qbRow = src.rows[0].source_data;
            let nsRow: any;
            try { nsRow = await applyMapping(client, t, qbId, qbRow); }
            catch (e: any) { result.errors.push({qbId, reason: e?.message || 'mapping failed'}); continue; }
            const dup = await client.query(`SELECT id FROM turion.${nsTable} WHERE id=$1`, [nsRow.id]);
            if ((dup.rowCount ?? 0) > 0) { result.errors.push({qbId, reason: `${nsTable} id ${nsRow.id} already exists`}); continue; }
            if (nsTable === 'customers') {
              await client.query(`INSERT INTO turion.customers (id, name, source_data) VALUES ($1,$2,$3)`,
                [nsRow.id, nsRow.name || nsRow.id, nsRow]);
            } else {
              await client.query(`INSERT INTO turion.${nsTable} (id, source_data) VALUES ($1,$2)`,
                [nsRow.id, nsRow]);
            }
            await client.query(`INSERT INTO turion.audit_log (entity, entity_id, action, before_data, after_data)
              VALUES ($1,$2,'CREATE',null,$3)`, [nsTable, nsRow.id, nsRow]);
            await client.query(`UPDATE turion.qb_records SET status='migrated', migrated_at=now(),
              migration_run_id=$1::uuid, updated_at=now() WHERE qb_type=$2 AND qb_id=$3`, [runId, t, qbId]);
            result.migrated.push(qbId);
          }
          const summary = {
            actor: 'demo-user', run_type: `qb-${t}`, run_id: runId,
            started_at: startedAt, finished_at: new Date().toISOString(),
            source_count: qbIds.length,
            migrated_count: result.migrated.length,
            skipped_count: result.skipped.length,
            error_count: result.errors.length,
            error_details: result.errors,
            summary: `Migrated ${result.migrated.length}/${qbIds.length} QB ${t} records to turion.${nsTable} (${result.skipped.length} already done, ${result.errors.length} failed).`
          };
          await client.query(
            `INSERT INTO turion.migration_runs (id, run_type, source_data) VALUES ($1::uuid, $2, $3)`,
            [runId, `qb-${t}`, summary]);
          await client.query('COMMIT');
          res.json({ ok: true, ...result, summary });
        } catch (e: any) {
          try { await client.query('ROLLBACK'); } catch { /* ignore */ }
          // hardened catch — do NOT leak err.message (Phase 36-08 audit rule)
          console.error('[quickbooks/migrate] ', e?.message);
          res.status(500).json({ error: 'migration failed' });
        } finally { client.release(); }
      });

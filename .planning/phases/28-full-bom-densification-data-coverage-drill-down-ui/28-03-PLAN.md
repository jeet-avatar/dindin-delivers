1. `GET /api/satellites/:satId/bom/tree` — returns a hierarchical JSON tree of every part_instance on the satellite, nested by bom_lines parent→child relationships. Includes drawing_svg inline (no per-node refetch) so the frontend can render thumbnails without 100+ extra API hits.

```typescript
// === Phase 28-03: recursive BOM tree ===

interface TreeNode {
  instance_id: string;
  part_definition_id: string;
  instance_index: number;
  serial_number: string | null;
  part_number: string;
  description: string | null;
  drawing_svg: string | null;
  default_make_buy: 'make' | 'buy' | null;
  itar_flag: boolean;
  subsystem_code: string | null;
  subsystem_label: string | null;
  parent_instance_id: string | null;
  qty: string | null;
  ref_designator: string | null;
  depth: number;
  children: TreeNode[];
}

router.get('/:satId/tree', requireAuth, async (req, res) => {
  const { satId } = req.params;
  try {
    const rows = await query<Omit<TreeNode, 'children'>>(`
      WITH RECURSIVE tree AS (
        -- Roots: instances on this satellite that are NOT a child in any bom_line
        SELECT
          pi.id AS instance_id,
          pi.part_definition_id,
          pi.instance_index,
          pi.serial_number,
          pd.part_number,
          pd.description,
          pd.drawing_svg,
          pd.default_make_buy,
          pd.itar_flag,
          s.code AS subsystem_code,
          s.label AS subsystem_label,
          NULL::uuid AS parent_instance_id,
          NULL::numeric AS qty,
          NULL::text AS ref_designator,
          0 AS depth,
          ARRAY[pi.id] AS path
        FROM turion_satellite.part_instances pi
        JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
        LEFT JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
        WHERE pi.satellite_id = $1
          AND NOT EXISTS (
            SELECT 1 FROM turion_satellite.bom_lines bl
            WHERE bl.child_part_instance_id = pi.id
              AND bl.parent_part_instance_id IS NOT NULL
              AND bl.satellite_id = $1
          )

        UNION ALL

        SELECT
          c_pi.id, c_pi.part_definition_id, c_pi.instance_index, c_pi.serial_number,
          c_pd.part_number, c_pd.description, c_pd.drawing_svg, c_pd.default_make_buy, c_pd.itar_flag,
          s.code, s.label,
          t.instance_id, bl.qty, bl.ref_designator,
          t.depth + 1,
          t.path || c_pi.id
        FROM tree t
        JOIN turion_satellite.bom_lines bl ON bl.parent_part_instance_id = t.instance_id
        JOIN turion_satellite.part_instances c_pi ON c_pi.id = bl.child_part_instance_id
        JOIN turion_satellite.part_definitions c_pd ON c_pd.id = c_pi.part_definition_id
        LEFT JOIN turion_satellite.subsystems s ON s.id = c_pd.subsystem_id
        WHERE bl.satellite_id = $1
          AND bl.status = 'released'
          AND c_pi.id <> ALL(t.path)  -- cycle guard
      )
      SELECT instance_id, part_definition_id, instance_index, serial_number,
             part_number, description, drawing_svg, default_make_buy, itar_flag,
             subsystem_code, subsystem_label, parent_instance_id, qty, ref_designator, depth
      FROM tree
      ORDER BY depth, subsystem_code NULLS LAST, part_number
    `, [satId]);

    // Assemble tree from flat rows (one pass)
    const byId = new Map<string, TreeNode>();
    const roots: TreeNode[] = [];
    for (const row of rows) {
      byId.set(row.instance_id, { ...row, children: [] });
    }
    for (const row of rows) {
      const node = byId.get(row.instance_id)!;
      if (row.parent_instance_id === null) {
        roots.push(node);
      } else {
        const parent = byId.get(row.parent_instance_id);
        if (parent) parent.children.push(node);
        else roots.push(node);  // orphan defensive: parent not in same satellite
      }
    }

    res.json({
      satellite_id: satId,
      node_count: rows.length,
      root_count: roots.length,
      max_depth: rows.reduce((m, r) => Math.max(m, r.depth), 0),
      roots
    });
  } catch (err: any) {
    console.error('[bom/tree] failed:', err);
    res.status(500).json({ error: 'Failed to compute BOM tree' });
  }
});
```

```typescript
// === Phase 28-03: recursive cost rollup at an instance ===

interface InstanceRollupResponse {
  instance_id: string;
  satellite_id: string;
  self_cost_usd: string;        // Decimal string
  descendants_cost_usd: string; // Decimal string
  subtree_cost_usd: string;     // Decimal string
  descendants_count: number;
  by_descendant: Array<{
    instance_id: string;
    part_number: string;
    depth: number;
    decision: 'make' | 'buy' | null;
    cost_usd: string;
  }>;
}

router.get('/instance/:instId', requireAuth, async (req, res) => {
  const { instId } = req.params;
  const satId = (req.query.sat as string) ?? '';
  if (!satId) {
    return res.status(400).json({ error: 'Missing required query param: sat' });
  }
  try {
    const rows = await query<{
      instance_id: string;
      part_number: string;
      depth: number;
      decision: 'make' | 'buy' | null;
      make_cost: string | null;
      buy_cost: string | null;
    }>(`
      WITH RECURSIVE subtree AS (
        SELECT $1::uuid AS instance_id, 0 AS depth, ARRAY[$1::uuid] AS path
        UNION ALL
        SELECT c_pi.id, st.depth + 1, st.path || c_pi.id
        FROM subtree st
        JOIN turion_satellite.bom_lines bl ON bl.parent_part_instance_id = st.instance_id
        JOIN turion_satellite.part_instances c_pi ON c_pi.id = bl.child_part_instance_id
        WHERE bl.satellite_id = $2
          AND bl.status = 'released'
          AND c_pi.id <> ALL(st.path)
      )
      SELECT
        st.instance_id,
        pd.part_number,
        st.depth,
        d.decision,
        mc.total_cost_usd AS make_cost,
        COALESCE(
          bc.invoiced_value_usd,
          bc.po_value_usd,
          bc.quoted_unit_cost_usd * COALESCE(bc.ordered_qty, 1)
        ) AS buy_cost
      FROM subtree st
      JOIN turion_satellite.part_instances pi ON pi.id = st.instance_id
      JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
      LEFT JOIN turion_satellite.make_buy_decisions d
        ON d.satellite_id = $2
       AND d.part_definition_id = pi.part_definition_id
       AND d.superseded_by IS NULL
      -- W3 fix: use *_current views (mig 005) — they encapsulate the
      -- `superseded_by IS NULL` filter, so the route never has to know about supersession.
      -- Pre-flight P3 (Plan 28-02 Task 1) confirmed both views exist in production.
      LEFT JOIN turion_satellite.make_costs_current mc
        ON mc.part_instance_id = pi.id
       AND mc.satellite_id = pi.satellite_id
      LEFT JOIN turion_satellite.buy_costs_current bc
        ON bc.part_instance_id = pi.id
       AND bc.satellite_id = pi.satellite_id
      ORDER BY st.depth, pd.part_number
    `, [instId, satId]);

    if (rows.length === 0) {
      return res.status(404).json({ error: 'Instance not found in subtree' });
    }

    // Decision-aware cost selection per node
    // RESEARCH Pitfall 9: a part_definition can have BOTH make_costs and buy_costs rows
    // (templates may exist for both even when the decision selects one side). Naïve
    // COALESCE(make, buy) would double-count or lose data. The decision row IS the source
    // of truth for which side to sum. Null decision = SKIP ($0 contribution), NOT a buy
    // fallback — post-mig-019, no part on SAT-003 should have a null decision, but if one
    // appears (e.g., a new part added between mig 019 apply and rollup query), treating
    // it as $0 fails-safe (loud absence in the rollup makes the gap discoverable) rather
    // than fails-quiet (hidden cost via implicit buy fallback).
    const byDescendant = rows.map(r => {
      // Use decision to switch — only "make" parts contribute make_cost; only "buy" parts contribute buy_cost
      const cost = r.decision === 'make'
        ? (r.make_cost ?? '0')
        : r.decision === 'buy'
        ? (r.buy_cost ?? '0')
        : '0';  // null decision = skip — see RESEARCH Pitfall 9 + comment above
      return {
        instance_id: r.instance_id,
        part_number: r.part_number,
        depth: r.depth,
        decision: r.decision,
        cost_usd: String(cost)
      };
    });

    const { sum, toMoney } = await import('../lib/money');
    const selfRow = byDescendant.find(d => d.depth === 0)!;
    const selfCost = toMoney(selfRow.cost_usd);
    const descendantsCost = sum(...byDescendant.filter(d => d.depth > 0).map(d => d.cost_usd));
    const subtreeCost = sum(selfCost, descendantsCost);

    res.json({
      instance_id: instId,
      satellite_id: satId,
      self_cost_usd: String(selfCost),
      descendants_cost_usd: String(descendantsCost),
      subtree_cost_usd: String(subtreeCost),
      descendants_count: byDescendant.length - 1,
      by_descendant: byDescendant
    } satisfies InstanceRollupResponse);
  } catch (err: any) {
    console.error('[cost-rollup/instance] failed:', err);
    res.status(500).json({ error: 'Failed to compute instance cost rollup' });
  }
});
```

```javascript
// Phase 28-05: fetch subtree cost rollup + walk parent trail
(async function loadSubtreeRollup() {
  try {
    const rollup = await window.satelliteApi.get(
      `/api/analytics/cost-rollup/instance/${encodeURIComponent(inst.id)}?sat=${encodeURIComponent(satId)}`
    );

    // Compute parent-trail running total client-side by fetching the full tree
    // and walking from this node up to the root. Locked decision #5.
    let parentTrailHTML = '';
    try {
      const tree = await window.satelliteApi.get(
        `/api/satellites/${encodeURIComponent(satId)}/bom/tree`
      );
      // Find this node and its ancestor chain
      const parents = findParentChain(tree.roots, inst.id);
      if (parents.length > 0) {
        // For each ancestor, the running total is its own subtree_cost_usd.
        // Render a compact chain: root → ... → parent → THIS
        parentTrailHTML = `
          <div class="cost-row" style="margin-top:12px; padding-top:12px; border-top:1px dotted #d2d2d7;">
            <span class="cost-row-label">Parent trail</span>
            <span class="cost-row-value" style="font-size:11px; color:#6e6e73;">
              ${parents.map(p => esc(p.part_number)).join(' → ')} → <strong>${esc(inst.part_number || 'this')}</strong>
            </span>
          </div>
        `;
      }
    } catch (e) {
      console.warn('[28-05] parent-trail fetch failed (non-fatal):', e);
    }

    const fmtUSD = (s) => {
      const n = Number(s);
      if (Number.isNaN(n)) return '—';
      return n.toLocaleString('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0 });
    };

    document.getElementById('subtreeMeta').textContent =
      `${rollup.descendants_count} descendant${rollup.descendants_count === 1 ? '' : 's'}`;

    document.getElementById('subtreeRollupPanel').innerHTML = `
      <div class="cost-row">
        <span class="cost-row-label">This instance (self)</span>
        <span class="cost-row-value">${fmtUSD(rollup.self_cost_usd)}</span>
      </div>
      <div class="cost-row">
        <span class="cost-row-label">Descendants</span>
        <span class="cost-row-value">${fmtUSD(rollup.descendants_cost_usd)} <span class="muted">across ${rollup.descendants_count}</span></span>
      </div>
      <div class="cost-row" style="border-top:1px solid #e5e5e7; margin-top:8px; padding-top:8px;">
        <span class="cost-row-label"><strong>Subtree total</strong></span>
        <span class="cost-row-value"><strong>${fmtUSD(rollup.subtree_cost_usd)}</strong></span>
      </div>
      ${parentTrailHTML}
    `;
  } catch (e) {
    console.error('[28-05] subtree rollup failed:', e);
    document.getElementById('subtreeRollupPanel').innerHTML =
      `<div class="muted">Subtree cost data unavailable: ${esc(e.message || String(e))}</div>`;
  }
})();

// Helper: walk the tree to find the ancestor chain of a target instance_id
function findParentChain(roots, targetId, chain = []) {
  for (const root of roots) {
    if (root.instance_id === targetId) return chain;
    if (root.children && root.children.length > 0) {
      const found = findParentChain(root.children, targetId, [...chain, root]);
      if (found) return found;
    }
  }
  return null;
}
```

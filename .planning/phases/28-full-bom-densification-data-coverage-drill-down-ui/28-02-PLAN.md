-- 019_backfill_data_coverage_for_phase28_parts.sql
-- Phase 28-02: backfill Phase 26-03 data layers (decisions, work_orders, build_steps,
-- procurement_requests, vendor_orders, make_costs, buy_costs) for ALL parts on SAT-003

--   1. The 7 mig-016 PCDU children that mig 013 ran BEFORE

-- =============================================================================
-- Block 1: make_buy_decisions (one per part_definition on SAT-003)
-- =============================================================================

INSERT INTO turion_satellite.make_buy_decisions
  (satellite_id, part_definition_id, decision, status, rationale, decided_at)
SELECT
  '24587565-b15b-42ce-b590-87ecf9b6bb99'::uuid,
  pd.id,
  pd.default_make_buy,
  'approved',
  CASE pd.default_make_buy
    WHEN 'make' THEN CASE s.code
      WHEN 'EPS' THEN 'In-house fabrication leverages our 6-layer PCB line; vendor lead times for COTS modules exceed 18 weeks for the qualification cycle this part requires.'
      WHEN 'ADCS' THEN 'Make-decision driven by ITAR-sensitive component count and the proprietary control-law firmware that must remain in-house per export-control assessment.'
      WHEN 'PROP' THEN 'Propulsion housing geometry is mission-specific; tooling already amortized across prior 4 builds keeps unit cost competitive vs single-vendor quote.'
      WHEN 'PAY' THEN 'Payload optical-bench tolerances (±0.005 mm) exceed vendor process capability; in-house turning + EDM line delivers reliably.'
      WHEN 'COMM' THEN 'Internal RF-shielded housing assembly avoids the 14-week vendor backlog and maintains hermetic-seal QC under our existing AS9100 envelope.'
      WHEN 'CDH' THEN 'Internal-make harnesses + brackets keep our shop loaded between major assembly milestones; quoted vendor markup exceeds 3.2× material cost.'
      WHEN 'TCS' THEN 'Radiator panel manufacturing uses heritage tooling from prior 3 builds; vendor sole-source on this part type triggers single-point-failure risk.'
      WHEN 'STR' THEN 'Structural bracket families are produced on our 5-axis CNC under flight-heritage process spec; vendor quotes 9-week lead time vs our 2-week turn.'
      ELSE 'In-house fabrication competitive on lead time and capability for this part type.'
    END
    WHEN 'buy' THEN CASE s.code
      WHEN 'EPS' THEN 'Commercial off-the-shelf component with mature flight heritage and qualified vendor; in-house production not economically viable for low-volume specialized parts.'
      WHEN 'ADCS' THEN 'Vendor-qualified flight component with proven performance envelope; in-house engineering and qualification cost exceeds 8× procurement cost.'
      WHEN 'PROP' THEN 'Specialized propulsion component requiring vendor-specific certification and test infrastructure not available in-house.'
      WHEN 'PAY' THEN 'High-precision optical component requires vendor-specific manufacturing process (e.g., diamond-turning) not available in-house.'
      WHEN 'COMM' THEN 'RF component with FCC/ITU-certified design requires vendor-qualified production line and EMC chamber not available in-house.'
      WHEN 'CDH' THEN 'Mil-spec or rad-tolerant semiconductor with vendor-specific test program; sole-source qualified part with established flight heritage.'
      WHEN 'TCS' THEN 'Thermal control component (heater patch, thermistor, MLI material) procured from qualified vendor with established space-flight history.'
      WHEN 'STR' THEN 'Structural fastener / standard hardware procured from qualified aerospace distributor; in-house production not cost-competitive at this volume.'
      ELSE 'Vendor-qualified flight component competitive on lead time and qualification cost.'
    END
  END,
  NOW() - (INTERVAL '30 days' + (RANDOM() * INTERVAL '60 days'))
FROM turion_satellite.part_definitions pd
JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
WHERE EXISTS (
  SELECT 1 FROM turion_satellite.part_instances pi
  WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
    AND pi.part_definition_id = pd.id
)
AND NOT EXISTS (
  SELECT 1 FROM turion_satellite.make_buy_decisions d
  WHERE d.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
    AND d.part_definition_id = pd.id
    AND d.superseded_by IS NULL
);

-- Block 2: work_orders + build_steps (every make-part instance gets WO with 6 steps)
-- =============================================================================

  (part_instance_id, wo_number, status, lifecycle_stage_id, created_at)
SELECT

  'WO-' || pd.part_number || '-' || pi.instance_index,
  'in_progress',
  (SELECT id FROM turion_satellite.lifecycle_stages WHERE code = 'manufacturing' LIMIT 1),
  NOW() - (INTERVAL '14 days' + (RANDOM() * INTERVAL '30 days'))
FROM turion_satellite.part_instances pi
JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
  AND pd.default_make_buy = 'make'
  AND NOT EXISTS (
    SELECT 1 FROM turion_satellite.work_orders wo WHERE wo.part_instance_id = pi.id
  );

-- Add 6 build_steps per new work_order (only for WOs without any steps)
INSERT INTO turion_satellite.build_steps (work_order_id, step_number, description, status)
SELECT wo.id, v.step_num, v.desc, v.status
FROM turion_satellite.work_orders wo
JOIN turion_satellite.part_instances pi ON pi.id = wo.part_instance_id
CROSS JOIN (VALUES
  (1, 'Material kit pull + verification against BOM', 'completed'),
  (2, 'Setup + first-article inspection', 'completed'),
  (3, 'Primary fabrication (machining / soldering / bonding)', 'completed'),
  (4, 'In-process QA (dimensional / functional check)', 'in_progress'),
  (5, 'Surface treatment + final cleaning', 'pending'),
  (6, 'Final inspection + sign-off', 'pending')
) AS v(step_num, desc, status)
WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
  AND NOT EXISTS (
    SELECT 1 FROM turion_satellite.build_steps bs WHERE bs.work_order_id = wo.id
  );

  150,  -- 150% overhead

    WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'

               WHEN 'PROP' THEN 4500 WHEN 'PAY' THEN 8000 ELSE 1500 END) * (1.0 + RANDOM() * 0.15),

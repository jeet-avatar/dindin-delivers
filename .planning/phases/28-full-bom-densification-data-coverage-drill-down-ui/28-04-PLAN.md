```javascript
// === Phase 28-04: shared integrations panel for cost-detail.html + instance.html ===
// Map cross-system FK kind → deep-link path on turionspace.zietra.com legacy pages
function _linkPath(kind, id) {
  if (!id) return null;
  const enc = encodeURIComponent(id);
  switch (kind) {
    case 'sales':    return `/sales/account/${enc}`;
    case 'netsuite': return `/finance/invoice/${enc}`;
    case 'arena':    return `/records/arena-doc/${enc}`;
    case 'mes':      return `/manufacturing/work-order/${enc}`;
    default:         return null;
  }
}

window.satelliteRender = window.satelliteRender || {};
window.satelliteRender.renderIntegrationsPanel = function(inst, opts) {
  opts = opts || {};
  const baseHost = opts.legacyHost || 'https://turionspace.zietra.com';
  const esc = window.satelliteRender.escapeHtml || (s => String(s ?? ''));
  const fmtDate = window.satelliteRender.fmtDate || (d => String(d ?? ''));

  const link = (kind, id) => {
    if (!id) return '<span class="muted">—</span>';
    const path = _linkPath(kind, id);
    const truncated = String(id).length > 18 ? String(id).slice(0, 18) + '…' : String(id);
    return path
      ? `<a href="${baseHost}${path}" target="_blank" rel="noopener" class="mono">${esc(truncated)} <span class="muted">↗</span></a>`
      : `<span class="mono">${esc(truncated)}</span>`;
  };

  const updatedLine = inst.cross_links_updated_at
    ? `updated ${fmtDate(inst.cross_links_updated_at)}`
    : 'never synced';

  return `
    <div class="panel" style="margin-bottom:18px;">
      <div class="panel-header">
        <strong>Cross-system links</strong>
        <span class="subtitle">${esc(updatedLine)}</span>
      </div>
      <div style="padding:12px;">
        <div class="cost-row">
          <span class="cost-row-label">Salesforce SO</span>
          <span class="cost-row-value">${link('sales', inst.sales_order_id)}</span>
        </div>
        <div class="cost-row">
          <span class="cost-row-label">NetSuite invoice</span>
          <span class="cost-row-value">${link('netsuite', inst.ns_invoice_id)}</span>
        </div>
        <div class="cost-row">
          <span class="cost-row-label">Arena doc</span>
          <span class="cost-row-value">${link('arena', inst.arena_doc_id)}</span>
        </div>
        <div class="cost-row">
          <span class="cost-row-label">MES work order</span>
          <span class="cost-row-value">${link('mes', inst.mes_work_order_id)}</span>
        </div>
      </div>
    </div>
  `;
};
```

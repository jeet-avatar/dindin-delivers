Run 7 acceptance queries against the live production DB. These verify Phase 28 truths regardless of UI/API status.

```bash
psql "$DATABASE_URL" <<'SQL' | tee /tmp/phase28-06-db-verify.txt
\echo === DB-DIRECT VERIFICATION (Phase 28 acceptance gate) ===

-- Q1: every part_definition with a SAT-003 instance has a make_buy_decision
\echo --- Q1: parts missing decision ---
SELECT
  COUNT(*) AS parts_missing_decision,
  CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.part_definitions pd
JOIN turion_satellite.part_instances pi
  ON pi.part_definition_id = pd.id
 AND pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
WHERE NOT EXISTS (
  SELECT 1 FROM turion_satellite.make_buy_decisions d
  WHERE d.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99'
    AND d.part_definition_id=pd.id AND d.superseded_by IS NULL
);

-- Q2: every make-part instance has ≥1 work_order
\echo --- Q2: make-parts missing work_order ---
SELECT
  COUNT(*) AS make_parts_missing_wo,
  CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.part_instances pi
JOIN turion_satellite.part_definitions pd ON pd.id=pi.part_definition_id
WHERE pi.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99'
  AND pd.default_make_buy='make'
  AND NOT EXISTS (SELECT 1 FROM turion_satellite.work_orders wo WHERE wo.part_instance_id=pi.id);

-- Q3: every buy-part instance has ≥1 procurement_request
\echo --- Q3: buy-parts missing procurement_request ---
SELECT
  COUNT(*) AS buy_parts_missing_pr,
  CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.part_instances pi
JOIN turion_satellite.part_definitions pd ON pd.id=pi.part_definition_id
WHERE pi.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99'
  AND pd.default_make_buy='buy'
  AND NOT EXISTS (SELECT 1 FROM turion_satellite.procurement_requests pr WHERE pr.part_instance_id=pi.id);

-- Q4: BOM tree max depth ≥ 3 (drill-down meaningful)
\echo --- Q4: max BOM depth ---
WITH RECURSIVE t AS (
  SELECT pi.id, 0 AS depth FROM turion_satellite.part_instances pi
   WHERE pi.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99'
     AND NOT EXISTS (
       SELECT 1 FROM turion_satellite.bom_lines bl
        WHERE bl.child_part_instance_id=pi.id AND bl.parent_part_instance_id IS NOT NULL AND bl.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99'
     )
  UNION ALL
  SELECT c.id, t.depth+1
    FROM t JOIN turion_satellite.bom_lines bl ON bl.parent_part_instance_id=t.id
            JOIN turion_satellite.part_instances c ON c.id=bl.child_part_instance_id
   WHERE bl.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99' AND bl.status='released'
)
SELECT MAX(depth) AS max_depth,
       CASE WHEN MAX(depth) >= 3 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM t;

-- Q5: new mig-018 parts each have specifications JSONB with ≥9 common keys
\echo --- Q5: new parts spec coverage ---
SELECT
  COUNT(*) AS new_parts_missing_specs,
  CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.part_definitions pd
WHERE pd.drawing_svg LIKE '%v=018%'
  AND (
    pd.specifications = '{}'::jsonb
    OR jsonb_array_length(jsonb_path_query_array(pd.specifications, '$.keyvalue()')) < 9
  );

-- Q6: total cost rollup still in plausible range ($5M-$30M; could be higher than Phase 26 due to new sub-component costs)
\echo --- Q6: cost rollup total ---
WITH costs AS (
  SELECT COALESCE(mc.total_cost_usd, 0) AS c
  FROM turion_satellite.make_costs mc
  JOIN turion_satellite.part_instances pi ON pi.id = mc.part_instance_id
  WHERE pi.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99' AND mc.superseded_by IS NULL
  UNION ALL
  SELECT COALESCE(bc.invoiced_value_usd, bc.po_value_usd, bc.quoted_unit_cost_usd * COALESCE(bc.ordered_qty, 1), 0)
  FROM turion_satellite.buy_costs bc
  JOIN turion_satellite.part_instances pi ON pi.id = bc.part_instance_id
  WHERE pi.satellite_id='24587565-b15b-42ce-b590-87ecf9b6bb99' AND bc.superseded_by IS NULL
)
SELECT
  ROUND(SUM(c)::numeric, 2) AS total_usd,
  CASE WHEN SUM(c) BETWEEN 5000000 AND 30000000 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM costs;

-- Q7: cross-system FK distribution unchanged (mig-018 parts correctly have NULL FKs per RESEARCH Pitfall 4)
\echo --- Q7: cross-system FK distribution ---
SELECT
```

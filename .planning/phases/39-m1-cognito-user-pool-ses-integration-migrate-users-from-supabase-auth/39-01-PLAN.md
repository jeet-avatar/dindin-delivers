Write `provision-cognito.sh`. Mirror the style of `/Users/jeet/turion-satellite/scripts/provision-aws.sh:1-46` (set -euo pipefail, `REGION=us-east-1`, `ACCOUNT=$(aws sts get-caller-identity ...)`). Idempotency: every step checks for existence first and skips if found.

Script structure (executed top-to-bottom):

```bash
#!/usr/bin/env bash
# Phase 39 Plan 01 — Provision Cognito user pool + KMS CMK + Secrets Manager config.
# Idempotent: safe to re-run.
set -euo pipefail
cd "$(dirname "$0")"

REGION=us-east-1
ACCOUNT=$(aws sts get-caller-identity --query Account --output text)
POOL_NAME=zietra-platform-users
APP_CLIENT_NAME=zietra-platform-web
KMS_ALIAS=alias/zietra-cognito-email-sender
SECRET_NAME=zietra/cognito-config
GROUPS=(admin customer driver vendor)

echo "=== Account: $ACCOUNT · Region: $REGION ==="
[ "$ACCOUNT" = "134607809447" ] || { echo "ABORT: not on the Zietra AWS account"; exit 1; }

echo "=== 1. KMS CMK — alias $KMS_ALIAS ==="
KEY_ID=$(aws kms describe-key --key-id "$KMS_ALIAS" --region "$REGION" --query 'KeyMetadata.KeyId' --output text 2>/dev/null || echo "")
if [ -z "$KEY_ID" ]; then
  KEY_ID=$(aws kms create-key \
    --description "Cognito Custom Email Sender — Zietra platform" \
    --tags TagKey=Service,TagValue=cognito-email TagKey=Owner,TagValue=zietra-platform \
    --policy file://cognito-kms-policy-bootstrap.json \
    --region "$REGION" \
    --query 'KeyMetadata.KeyId' --output text)
  aws kms create-alias --alias-name "$KMS_ALIAS" --target-key-id "$KEY_ID" --region "$REGION"
  echo "Created KMS key: $KEY_ID"
else
  echo "KMS key exists: $KEY_ID"
fi
KEY_ARN=$(aws kms describe-key --key-id "$KEY_ID" --region "$REGION" --query 'KeyMetadata.Arn' --output text)

echo "=== 2. Cognito user pool — $POOL_NAME ==="
POOL_ID=$(aws cognito-idp list-user-pools --max-results 60 --region "$REGION" \
  --query "UserPools[?Name=='${POOL_NAME}'].Id" --output text)
if [ -z "$POOL_ID" ] || [ "$POOL_ID" = "None" ]; then
  POOL_ID=$(aws cognito-idp create-user-pool \
    --cli-input-json file://user-pool-config.json \
    --region "$REGION" \
    --query 'UserPool.Id' --output text)
  echo "Created user pool: $POOL_ID"
else
  echo "User pool exists: $POOL_ID"
fi

echo "=== 3. Tighten KMS policy with pool-ID EncryptionContext ==="
sed "s|POOL_ID_PLACEHOLDER|$POOL_ID|g" cognito-kms-policy-final.json > /tmp/cognito-kms-policy-final-$POOL_ID.json
aws kms put-key-policy --key-id "$KEY_ID" --policy-name default \
  --policy "file:///tmp/cognito-kms-policy-final-$POOL_ID.json" --region "$REGION"
rm /tmp/cognito-kms-policy-final-$POOL_ID.json

echo "=== 4. App client — $APP_CLIENT_NAME ==="
CLIENT_ID=$(aws cognito-idp list-user-pool-clients --user-pool-id "$POOL_ID" --max-results 20 --region "$REGION" \
  --query "UserPoolClients[?ClientName=='${APP_CLIENT_NAME}'].ClientId" --output text)
if [ -z "$CLIENT_ID" ] || [ "$CLIENT_ID" = "None" ]; then
  CLIENT_ID=$(aws cognito-idp create-user-pool-client \
    --user-pool-id "$POOL_ID" \
    --client-name "$APP_CLIENT_NAME" \
    --no-generate-secret \
    --explicit-auth-flows ALLOW_ADMIN_USER_PASSWORD_AUTH ALLOW_CUSTOM_AUTH ALLOW_REFRESH_TOKEN_AUTH \
    --refresh-token-validity 30 \
    --access-token-validity 60 \
    --id-token-validity 60 \
    --token-validity-units AccessToken=minutes,IdToken=minutes,RefreshToken=days \
    --prevent-user-existence-errors ENABLED \
    --enable-token-revocation \
    --region "$REGION" \
    --query 'UserPoolClient.ClientId' --output text)
  echo "Created app client: $CLIENT_ID"
else
  echo "App client exists: $CLIENT_ID"
fi

echo "=== 5. Cognito Groups — ${GROUPS[*]} ==="
for g in "${GROUPS[@]}"; do
  aws cognito-idp get-group --user-pool-id "$POOL_ID" --group-name "$g" --region "$REGION" >/dev/null 2>&1 \
    || aws cognito-idp create-group --user-pool-id "$POOL_ID" --group-name "$g" --description "Zietra platform role: $g" --region "$REGION"
  echo "Group: $g"
done

echo "=== 6. Secrets Manager — $SECRET_NAME ==="
SECRET_JSON=$(printf '{"user_pool_id":"%s","app_client_id":"%s","kms_key_arn":"%s","region":"%s"}' \
  "$POOL_ID" "$CLIENT_ID" "$KEY_ARN" "$REGION")
aws secretsmanager describe-secret --secret-id "$SECRET_NAME" --region "$REGION" >/dev/null 2>&1 \
  && aws secretsmanager put-secret-value --secret-id "$SECRET_NAME" --secret-string "$SECRET_JSON" --region "$REGION" >/dev/null \
  || aws secretsmanager create-secret --name "$SECRET_NAME" --secret-string "$SECRET_JSON" --region "$REGION" >/dev/null
echo "Secret $SECRET_NAME populated"

echo "=== 7. SES verify demo@zietra.com (if not already verified) ==="
if ! aws ses list-identities --region "$REGION" --query 'Identities' --output text | tr '\t' '\n' | grep -qx "demo@zietra.com"; then
  aws ses verify-email-identity --email-address demo@zietra.com --region "$REGION"
  echo "Verification email sent to demo@zietra.com — recipient must click the link"
else
  echo "demo@zietra.com already verified in SES"
fi

echo ""
echo "=== DONE ==="
echo "POOL_ID=$POOL_ID"
echo "CLIENT_ID=$CLIENT_ID"
echo "KEY_ARN=$KEY_ARN"
echo "Secret: $SECRET_NAME"
echo ""
echo "Next: run lambdas/cognito-custom-email-sender/deploy.sh (Plan 39-02)"
```

```typescript
// TEMPORARY: delete after Phase 41 cutover — Cognito will be source-of-truth for users from M2 onward.
// Phase 39 Plan 03 — One-shot migration of confirmed Supabase auth.users → AWS Cognito user pool.
// Run: tsx scripts/migrate-supabase-users-to-cognito.ts
//   DRY_RUN=1   — pre-flight: prints what would happen, no Cognito writes
//   AWS_REGION  — defaults to us-east-1
//   DATABASE_URL — Supabase Postgres pooler URL (read from Lambda env or paste from Secrets Manager)
//
// Rule 1: pool ID is read from Secrets Manager 'zietra/cognito-config', NEVER hardcoded.
// Rule 3: each user is verified via AdminGetUser after creation (no trusting AdminCreateUser stdout).
// Rule 5: this script self-marks TEMPORARY and is dropped in Phase 41 cleanup.
// Rule 6: no MFA setup, no group hierarchy, no federated identity — magic-link via CUSTOM_AUTH only.

import { Client as PgClient } from 'pg';
import {
  CognitoIdentityProviderClient,
  AdminCreateUserCommand,
  AdminSetUserPasswordCommand,
  AdminAddUserToGroupCommand,
  AdminGetUserCommand,
  UserNotFoundException,
  UsernameExistsException,
} from '@aws-sdk/client-cognito-identity-provider';
import {
  SecretsManagerClient,
  GetSecretValueCommand,
} from '@aws-sdk/client-secrets-manager';
import crypto from 'node:crypto';

const REGION = process.env.AWS_REGION ?? 'us-east-1';
const DRY_RUN = process.env.DRY_RUN === '1';
const DEPRECATED_ALIAS_DROPS = new Set([
  'jeetnair.in+zietra-c8@gmail.com',  // RESEARCH §"Confirmed but deprecated test alias"
]);

// Role mapping — see RESEARCH §"Migration Script Skeleton". All 4 real users are platform builders/demo accounts → admin.
const ROLE_MAP: Record<string, 'admin' | 'customer' | 'driver' | 'vendor'> = {
  'jm@techcloudpro.com':    'admin',
  'jeetnair.in@gmail.com':  'admin',
  'demo@zietra.com':        'admin',
  'gteshnair@gmail.com':    'admin',
};

async function loadPoolId(sm: SecretsManagerClient): Promise<string> {
  const res = await sm.send(new GetSecretValueCommand({ SecretId: 'zietra/cognito-config' }));
  if (!res.SecretString) throw new Error('zietra/cognito-config has no SecretString');
  const parsed = JSON.parse(res.SecretString);
  if (!parsed.user_pool_id) throw new Error('zietra/cognito-config missing user_pool_id');
  return parsed.user_pool_id as string;
}

async function main() {
  if (!process.env.DATABASE_URL) {
    throw new Error('DATABASE_URL env var required (Supabase pooler URL)');
  }
  console.log(`[migrate] region=${REGION} dry_run=${DRY_RUN}`);

  const sm = new SecretsManagerClient({ region: REGION });
  const POOL_ID = await loadPoolId(sm);
  console.log(`[migrate] pool_id=${POOL_ID}`);

  const cog = new CognitoIdentityProviderClient({ region: REGION });
  const pg = new PgClient({
    connectionString: process.env.DATABASE_URL,
    ssl: { rejectUnauthorized: false },
  });
  await pg.connect();
  console.log('[migrate] connected to Supabase');

  const { rows } = await pg.query(`
    SELECT id::text AS id, email, raw_user_meta_data, created_at::text AS created_at
    FROM auth.users
    WHERE email_confirmed_at IS NOT NULL
    ORDER BY created_at
  `);
  console.log(`[migrate] ${rows.length} confirmed user(s) found in Supabase`);

  let created = 0;
  let skipped = 0;
  let dropped = 0;
  let errors = 0;

  for (const r of rows) {
    const emailLower = String(r.email).toLowerCase();

    // Drop deprecated alias (gmail + suffix test address)
    if (DEPRECATED_ALIAS_DROPS.has(emailLower)) {
      console.log('[drop-deprecated]', emailLower);
      dropped += 1;
      continue;
    }

    const role = ROLE_MAP[emailLower];
    if (!role) {
      console.log('[drop-unmapped]', emailLower, '— not in ROLE_MAP; add explicit role before migrating');
      dropped += 1;
      continue;
    }

    if (DRY_RUN) {
      console.log('[dry-run]', emailLower, '→ role=' + role + ' · supabase_sub=' + r.id);
      continue;
    }

    // Idempotency — skip if user exists
    try {
      await cog.send(new AdminGetUserCommand({ UserPoolId: POOL_ID, Username: emailLower }));
      console.log('[skip-exists]', emailLower);
      skipped += 1;
      continue;
    } catch (e: any) {
      if (e.name !== 'UserNotFoundException' && !(e instanceof UserNotFoundException)) {
        console.error('[error-get]', emailLower, e.name, e.message);
        errors += 1;
        continue;
      }
    }

    // Create
    try {
      await cog.send(new AdminCreateUserCommand({
        UserPoolId: POOL_ID,
        Username: emailLower,
        MessageAction: 'SUPPRESS', // Pitfall 5 — don't email the user during migration
        UserAttributes: [
          { Name: 'email',          Value: emailLower },
          { Name: 'email_verified', Value: 'true' },             // Pitfall 6 — string, not boolean
          { Name: 'custom:role',    Value: role },               // Pitfall 7 — declared in pool schema
          { Name: 'custom:supabase_sub', Value: r.id },          // forward audit link
          { Name: 'name',           Value: r.raw_user_meta_data?.name ?? '' },
        ],
      }));

      // Strong permanent password (users sign in via magic-link, not password)
      const pwd = crypto.randomBytes(32).toString('base64').replace(/\+/g, 'A').replace(/\//g, 'b').replace(/=/g, '') + 'A1!a';
      await cog.send(new AdminSetUserPasswordCommand({
        UserPoolId: POOL_ID,
        Username: emailLower,
        Password: pwd,
        Permanent: true,
      }));

      // Add to role group
      await cog.send(new AdminAddUserToGroupCommand({
        UserPoolId: POOL_ID,
        Username: emailLower,
        GroupName: role,
      }));

      // Rule 3 — verify the create succeeded by re-reading
      const got = await cog.send(new AdminGetUserCommand({ UserPoolId: POOL_ID, Username: emailLower }));
      const attrs = Object.fromEntries((got.UserAttributes ?? []).map(a => [a.Name, a.Value]));
      if (attrs['email'] !== emailLower) throw new Error(`email mismatch: ${attrs['email']}`);
      if (attrs['email_verified'] !== 'true') throw new Error(`email_verified not true: ${attrs['email_verified']}`);
      if (attrs['custom:role'] !== role) throw new Error(`custom:role mismatch: ${attrs['custom:role']}`);
      if (attrs['custom:supabase_sub'] !== r.id) throw new Error(`custom:supabase_sub mismatch`);

      console.log('[migrated]', emailLower, '→ role=' + role + ' · ' + got.UserStatus);
      created += 1;
    } catch (e: any) {
      if (e instanceof UsernameExistsException || e.name === 'UsernameExistsException') {
        console.log('[skip-race]', emailLower, '— created by concurrent run');
        skipped += 1;
        continue;
      }
      console.error('[error-create]', emailLower, e.name, e.message);
      errors += 1;
    }
  }

  await pg.end();

  console.log('');
  console.log('=== SUMMARY ===');
  console.log(`source rows (confirmed):  ${rows.length}`);
  console.log(`created:                  ${created}`);
  console.log(`skipped (already exists): ${skipped}`);
  console.log(`dropped (deprecated/unmapped): ${dropped}`);
  console.log(`errors:                   ${errors}`);

  if (errors > 0) process.exit(1);
}

main().catch(e => {
  console.error('[fatal]', e);
  process.exit(1);
});
```

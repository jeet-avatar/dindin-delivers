```typescript
// Add at the top (after existing imports):
import type { PoolClient } from 'pg';

// Append at end of file:
export type DrawingSourceUploaded = { source: 'uploaded'; cad_file_id: string; format: string; s3_key: string; revision: number };
export type DrawingSourceProcedural = { source: 'procedural'; template_name: string };
export type DrawingSource = DrawingSourceUploaded | DrawingSourceProcedural;

/**
 * Phase 60-02: DB-aware drawing source dispatcher.
 * Consults turion_satellite.part_cad_files FIRST (Pitfall 5 — uploaded files always win);
 * falls back to existing procedural regex `chooseTemplate()` only when no active upload exists.
 * MUST be called inside withTenantClient so RLS scopes the SELECT.
 */
export async function chooseDrawingSource(client: PoolClient, partId: string): Promise<DrawingSource> {
  const r = await client.query(
    `SELECT id, format, s3_key, revision
     FROM turion_satellite.part_cad_files
     WHERE part_id = $1 AND is_active = true
     ORDER BY revision DESC
     LIMIT 1`,
    [partId]
  );
  if (r.rows[0]) {
    return {
      source: 'uploaded',
      cad_file_id: r.rows[0].id,
      format: r.rows[0].format,
      s3_key: r.rows[0].s3_key,
      revision: r.rows[0].revision,
    };
  }
  const p = await client.query(
    `SELECT part_number FROM turion_satellite.part_definitions WHERE id = $1`,
    [partId]
  );
  const { name } = chooseTemplate({ part_number: p.rows[0]?.part_number ?? '' });
  return { source: 'procedural', template_name: name };
}
```

**Step A — Create `frontend/satellite/cad-viewer.js` (dispatcher — keeps occt off the critical path):**

```javascript
// Phase 60-02: CAD viewer dispatcher.
// Lazy-imports the format-specific mount so occt-import-js (~5 MB WASM) only loads
// when the user opens a STEP/IGES file (Pitfall 1 — never on the parts list).

export async function mountCadViewer(container, payload) {
  if (!container) return;
  container.innerHTML = '';  // clear any prior state

  if (payload.source === 'procedural' || payload.format === 'svg') {
    container.innerHTML = payload.drawing_svg || '<p class="hint">No drawing available.</p>';
    return { kind: 'svg' };
  }

  if (payload.format === 'stl') {
    const { mountStl } = await import('./cad-viewer-stl.js');
    await mountStl(container, payload.url);
    return { kind: 'stl' };
  }

  if (payload.format === 'step' || payload.format === 'stp' || payload.format === 'iges' || payload.format === 'igs') {
    const { mountStep } = await import('./cad-viewer-step.js');
    await mountStep(container, payload.url, payload.format);
    return { kind: 'step' };
  }

  // Fallback: download link
  container.innerHTML = `<p class="hint">Preview not available for <b>${payload.format}</b>. <a href="${payload.url}" target="_blank" rel="noopener">Download to view</a>.</p>`;
  return { kind: 'download' };
}
```

```javascript
import * as THREE from 'three';
import { STLLoader } from 'three/addons/loaders/STLLoader.js';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

export async function mountStl(container, url) {
  const spinner = document.createElement('div');
  spinner.className = 'hint';
  spinner.textContent = 'Loading STL…';
  container.appendChild(spinner);

  const w = container.clientWidth || 800;
  const h = container.clientHeight || 520;

  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0x101522);
  const camera = new THREE.PerspectiveCamera(45, w / h, 0.1, 10000);
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(w, h);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));  // Phase 30/31 size-blowup fix

  scene.add(new THREE.HemisphereLight(0xffffff, 0x404060, 0.8));
  const dir = new THREE.DirectionalLight(0xffffff, 0.6);
  dir.position.set(1, 1, 1);
  scene.add(dir);

  const loader = new STLLoader();
  let geometry;
  try {
    geometry = await loader.loadAsync(url);
  } catch (err) {
    spinner.textContent = `Failed to load STL: ${err.message || err}`;
    return;
  }
  geometry.computeVertexNormals();  // Pitfall 9 — fix degenerate normals

  const material = new THREE.MeshStandardMaterial({ color: 0x9090a0, flatShading: true });
  const mesh = new THREE.Mesh(geometry, material);
  scene.add(mesh);

  // Auto-fit
  const box = new THREE.Box3().setFromObject(mesh);
  const size = box.getSize(new THREE.Vector3()).length();
  const center = box.getCenter(new THREE.Vector3());
  function fit() {
    camera.position.copy(center).add(new THREE.Vector3(size, size, size));
    camera.lookAt(center);
  }
  fit();

  container.removeChild(spinner);
  container.appendChild(renderer.domElement);

  const controls = new OrbitControls(camera, renderer.domElement);
  controls.target.copy(center);

  // Reset-view button
  const btn = document.createElement('button');
  btn.textContent = '⟲ Reset view';
  btn.className = 'btn-small';
  btn.style.cssText = 'position:absolute;top:8px;right:8px;z-index:2';
  btn.onclick = () => { fit(); controls.target.copy(center); controls.update(); };
  container.style.position = 'relative';
  container.appendChild(btn);

  (function loop() {
    requestAnimationFrame(loop);
    controls.update();
    renderer.render(scene, camera);
  })();
}
```

```javascript
  spinner.innerHTML = '<span>Loading 3D viewer (~5 MB WASM, one-time)…</span>';
  container.appendChild(spinner);

  // Fetch + size warning
  const resp = await fetch(url);
  const contentLength = Number(resp.headers.get('content-length') || '0');
  if (contentLength > 20 * 1024 * 1024) {
    spinner.innerHTML = `<span>Large ${format.toUpperCase()} file (${(contentLength/1024/1024).toFixed(1)} MB) — this may take a minute…</span>`;
  }
  const buf = new Uint8Array(await resp.arrayBuffer());

  const occt = await occtInit({
    locateFile: f => `https://cdn.jsdelivr.net/npm/occt-import-js@0.0.23/dist/${f}`,
  });

  const fn = (format === 'iges' || format === 'igs') ? occt.ReadIgesFile : occt.ReadStepFile;
  const result = fn(buf, { linearUnit: 'millimeter', linearDeflection: 0.1, angularDeflection: 0.5 });
  if (!result.success) {
    spinner.innerHTML = `<span>Failed to parse ${format.toUpperCase()}: ${result.error || 'unknown error'}</span>`;
    return;
  }

  // Scene setup
  const w = container.clientWidth || 800;
  const h = container.clientHeight || 520;
  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0x101522);
  const camera = new THREE.PerspectiveCamera(45, w / h, 0.1, 100000);
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(w, h);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  scene.add(new THREE.HemisphereLight(0xffffff, 0x404060, 0.8));
  const dl = new THREE.DirectionalLight(0xffffff, 0.6); dl.position.set(1,1,1); scene.add(dl);

  const group = new THREE.Group();
  for (const m of result.meshes) {
    const geom = new THREE.BufferGeometry();
    geom.setAttribute('position', new THREE.Float32BufferAttribute(m.attributes.position.array, 3));
    if (m.attributes.normal) {
      geom.setAttribute('normal', new THREE.Float32BufferAttribute(m.attributes.normal.array, 3));
    } else {
      geom.computeVertexNormals();
    }
    if (m.index) geom.setIndex(new THREE.Uint32BufferAttribute(m.index.array, 1));
    const color = m.color ? new THREE.Color(m.color[0], m.color[1], m.color[2]) : new THREE.Color(0x9090a0);
    const mat = new THREE.MeshStandardMaterial({ color, flatShading: true });
    group.add(new THREE.Mesh(geom, mat));
  }
  scene.add(group);

  // Auto-fit
  const box = new THREE.Box3().setFromObject(group);
  const size = box.getSize(new THREE.Vector3()).length();
  const center = box.getCenter(new THREE.Vector3());
  function fit() {
    camera.position.copy(center).add(new THREE.Vector3(size, size, size));
    camera.lookAt(center);
  }
  fit();

  container.removeChild(spinner);
  container.appendChild(renderer.domElement);
  const controls = new OrbitControls(camera, renderer.domElement);
  controls.target.copy(center);

  const btn = document.createElement('button');
  btn.textContent = '⟲ Reset view';
  btn.className = 'btn-small';
  btn.style.cssText = 'position:absolute;top:8px;right:8px;z-index:2';
  btn.onclick = () => { fit(); controls.target.copy(center); controls.update(); };
  container.appendChild(btn);

  (function loop() {
    requestAnimationFrame(loop);
    controls.update();
    renderer.render(scene, camera);
  })();
}
```

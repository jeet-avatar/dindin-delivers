**Step B — Write `handler.mjs`** following RESEARCH §I lines 645-734 with Pitfall 5 dispatch (uploaded > procedural), Pitfall 10 fix (`browser.close()` in `finally`), and `SET LOCAL app.tenant_id` for RLS scoping:

```javascript
// zietra-cad-pdf-gen/handler.mjs
// Phase 60-04: SQS-triggered PDF generator. One job = one PDF.
// CRITICAL: honors Pitfall 5 template dispatch — uploaded CAD files take precedence
// over procedural SVG templates at the PDF render target too (not just on-screen viewer).
// Pitfall 10: browser.close() in finally + bound /tmp usage.
import chromium from '@sparticuz/chromium';
import puppeteer from 'puppeteer-core';
import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import { GetObjectCommand } from '@aws-sdk/client-s3';
import pg from 'pg';
import fs from 'fs/promises';
import path from 'path';

const s3 = new S3Client({ region: 'us-east-1' });
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL, max: 2 });
const BUCKET = process.env.CAD_BUCKET;

async function renderJob(job) {
  const { job_id, part_id, tenant_id } = job;
  let browser;
  const client = await pool.connect();
  try {
    await client.query(`SET LOCAL app.tenant_id = $1`, [tenant_id]);

    // Pull part header
    const partR = await client.query(
      `SELECT part_number, description, drawing_svg, drawing_rev FROM turion_satellite.part_definitions WHERE id = $1`,
      [part_id]
    );
    if (!partR.rows[0]) throw new Error('part not found');
    const part = partR.rows[0];

    // PITFALL 5: uploaded CAD > procedural template — query part_cad_files FIRST
    const cadR = await client.query(
      `SELECT id, filename, format, s3_key, revision, sha256
       FROM turion_satellite.part_cad_files
       WHERE part_id = $1 AND is_active = true
       ORDER BY revision DESC LIMIT 1`, [part_id]);
    const uploaded = cadR.rows[0] || null;

    // Optional markup
    let markupSvg = '';
    if (uploaded) {
      const mkR = await client.query(
        `SELECT markup_svg FROM turion_satellite.part_drawing_markups WHERE part_cad_file_id = $1 LIMIT 1`,
        [uploaded.id]);
      markupSvg = mkR.rows[0]?.markup_svg ?? '';
    }

    // BOM rows
    const bomR = await client.query(
      `SELECT c_pd.part_number, c_pd.description, bl.qty, bl.uom
       FROM turion_satellite.bom_lines bl
       JOIN turion_satellite.part_instances c_pi ON c_pi.id = bl.child_part_instance_id
       JOIN turion_satellite.part_definitions c_pd ON c_pd.id = c_pi.part_definition_id
       WHERE bl.parent_part_instance_id IN (
         SELECT id FROM turion_satellite.part_instances WHERE part_definition_id = $1 LIMIT 1
       )`, [part_id]);

    // Tenant name for title block
    const tnR = await client.query(`SELECT name FROM public.tenants WHERE id = $1`, [tenant_id]);
    const tenantName = tnR.rows[0]?.name ?? 'Zietra';

    // PDF source line — proves Pitfall 5 dispatch at render time
    const drawingSource = uploaded
      ? `Uploaded CAD: ${uploaded.filename} · rev ${uploaded.revision} · ${uploaded.format}`
      : `Procedural template · rev ${part.drawing_rev || 0}`;

    // Drawing slot:
    //  - uploaded: render a labeled placeholder + markup overlay (v1; Phase 61 = full 3D snapshot)
    //  - procedural: inline the existing drawing_svg
    let drawingHtml;
    if (uploaded) {
      const dlUrl = await getSignedUrl(s3, new GetObjectCommand({ Bucket: BUCKET, Key: uploaded.s3_key }), { expiresIn: 900 });
      drawingHtml = `<div class="uploaded-cad">
          <h2>Uploaded CAD file</h2>
          <p><b>File:</b> ${uploaded.filename}</p>
          <p><b>Format:</b> ${uploaded.format.toUpperCase()}</p>
          <p><b>Revision:</b> ${uploaded.revision}</p>
          <p><b>SHA-256:</b> ${uploaded.sha256.slice(0,16)}…</p>
          <p><b>Download (15-min link):</b> <a href="${dlUrl}">${uploaded.format} file</a></p>
          ${markupSvg ? `<div class="markup-overlay">${markupSvg}</div>` : ''}
        </div>`;
    } else {
      drawingHtml = part.drawing_svg || '<p>No drawing.</p>';
    }

    // Load template + substitute
    const tpl = await fs.readFile(path.join(import.meta.dirname, 'template.html'), 'utf8');
    const html = tpl
      .replaceAll('{{TENANT_NAME}}', tenantName)
      .replaceAll('{{PART_NUMBER}}', part.part_number)
      .replaceAll('{{DESCRIPTION}}', part.description || '')
      .replaceAll('{{REV}}', String(uploaded?.revision ?? part.drawing_rev ?? 0))
      .replaceAll('{{DRAWING_SOURCE}}', drawingSource)
      .replaceAll('{{DRAWING_HTML}}', drawingHtml)
      .replaceAll('{{BOM_ROWS}}',
        bomR.rows.map(b => `<tr><td>${b.part_number}</td><td>${b.description || ''}</td><td>${b.qty} ${b.uom || ''}</td></tr>`).join('')
        || '<tr><td colspan="3"><em>No BOM lines.</em></td></tr>'
      )
      .replaceAll('{{GENERATED_AT}}', new Date().toISOString());

    // Launch Chromium and render
    browser = await puppeteer.launch({
      args: chromium.args,
      defaultViewport: chromium.defaultViewport,
      executablePath: await chromium.executablePath(),
      headless: chromium.headless,
    });
    const page = await browser.newPage();
    await page.setContent(html, { waitUntil: 'networkidle0', timeout: 30000 });
    const pdfBuf = await page.pdf({
      format: 'A3', landscape: true, printBackground: true,
      margin: { top: '10mm', right: '10mm', bottom: '10mm', left: '10mm' },
    });

    // Upload PDF
    const pdfKey = `${tenant_id}/${part_id}/drawings/${job_id}.pdf`;
    await s3.send(new PutObjectCommand({
      Bucket: BUCKET, Key: pdfKey, Body: pdfBuf, ContentType: 'application/pdf',
    }));

    // Mark job ready + audit
    await client.query(
      `UPDATE turion_satellite.part_drawing_jobs SET status='ready', pdf_s3_key=$1, completed_at=now() WHERE id=$2`,
      [pdfKey, job_id]);
    await client.query(
      `INSERT INTO turion_satellite.audit_log (action, entity_type, entity_id, actor_cognito_sub, payload)
       VALUES ('cad_pdf_generate', 'part', $1, NULL, $2::jsonb)`,
      [part_id, JSON.stringify({ job_id, pdf_s3_key: pdfKey, source: uploaded ? 'uploaded' : 'procedural', uploaded_rev: uploaded?.revision ?? null })]);

    console.log('[pdf-gen] ok', { job_id, part_id, pdf_s3_key: pdfKey, source: uploaded ? 'uploaded' : 'procedural' });
  } catch (err) {
    console.error('[pdf-gen] FAIL', { job_id, err: String(err?.message ?? err) });
    try {
      await client.query(`SET LOCAL app.tenant_id = $1`, [tenant_id]);
      await client.query(`UPDATE turion_satellite.part_drawing_jobs SET status='failed', error=$1 WHERE id=$2`, [String(err?.message ?? err).slice(0,1000), job_id]);
    } catch (e2) {
      console.error('[pdf-gen] fail-update also failed', e2);
    }
  } finally {
    if (browser) {
      try { await browser.close(); } catch {}
    }
    client.release();
  }
}

export async function handler(event) {
  for (const record of event.Records || []) {
    const body = JSON.parse(record.body);
    await renderJob(body);
  }
}
```

```typescript
import { Router } from 'express';
import crypto from 'crypto';
import { SQSClient, SendMessageCommand } from '@aws-sdk/client-sqs';
import { requireAuth } from '../middleware/auth';
import { withTenantClient } from '../db';
import { presignGet } from '../lib/s3-presigner';
import { cadAudit } from '../lib/cad-audit';

const sqs = new SQSClient({ region: 'us-east-1' });
const QUEUE_URL = process.env.PDF_GEN_QUEUE_URL;
if (!QUEUE_URL) throw new Error('PDF_GEN_QUEUE_URL env var required');

export const cadPdfRouter = Router({ mergeParams: true });
cadPdfRouter.use(requireAuth);

function requireUploader(req: any, res: any, next: any) {
  if (!['admin','manager'].includes(req.user?.role)) return res.status(403).json({ error: 'forbidden' });
  next();
}

// POST /api/parts/:id/drawings/generate
cadPdfRouter.post('/:id/drawings/generate', requireUploader, async (req: any, res) => {
  const partId = req.params.id;
  const jobId = crypto.randomUUID();
  try {
    await withTenantClient(req, async (client) => {
      await client.query(
        `INSERT INTO turion_satellite.part_drawing_jobs (id, part_id, tenant_id, status, requested_by_cognito_sub)
         VALUES ($1, $2, current_setting('app.tenant_id')::uuid, 'queued', $3)`,
        [jobId, partId, req.user.sub]
      );
      await cadAudit(req, client, 'cad_pdf_generate', partId, { job_id: jobId, stage: 'enqueued' });
    });
    await sqs.send(new SendMessageCommand({
      QueueUrl: QUEUE_URL,
      MessageBody: JSON.stringify({ job_id: jobId, part_id: partId, tenant_id: req.tenant!.id }),
    }));
    res.status(202).json({ job_id: jobId, status: 'queued', poll_url: `/api/cad-pdf-jobs/${jobId}` });
  } catch (err: any) {
    console.error('[cad-pdf] generate failed:', err?.message ?? err);
    res.status(500).json({ error: 'failed to enqueue PDF job' });
  }
});

export const cadPdfJobsRouter = Router();
cadPdfJobsRouter.use(requireAuth);
cadPdfJobsRouter.get('/:jobId', async (req: any, res) => {
  try {
    const result = await withTenantClient(req, async (client) => {
      const r = await client.query(
        `SELECT id, part_id, status, pdf_s3_key, error, requested_at, completed_at
         FROM turion_satellite.part_drawing_jobs WHERE id = $1`,
        [req.params.jobId]);
      const row = r.rows[0];
      if (!row) return null;
      if (row.status === 'ready' && row.pdf_s3_key) {
        const url = await presignGet(row.pdf_s3_key, 900);
        return { ...row, pdf_url: url };
      }
      return row;
    });
    if (!result) return res.status(404).json({ error: 'job not found' });
    res.json(result);
  } catch (err: any) {
    console.error('[cad-pdf-jobs] failed:', err?.message ?? err);
    res.status(500).json({ error: 'failed to load job' });
  }
});
```

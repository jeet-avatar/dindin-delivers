```typescript
import DOMPurify from 'isomorphic-dompurify';

const MAX_SVG_BYTES = 524288;  // 512 KB — matches DB CHECK

export function sanitizeMarkupSvg(svg: string): string {
  if (typeof svg !== 'string' || svg.length === 0) {
    throw new Error('markup_svg must be a non-empty string');
  }
  if (svg.length > MAX_SVG_BYTES) {
    throw new Error(`markup_svg exceeds 512 KB limit (got ${svg.length} bytes)`);
  }
  // SVG-mode sanitization — strips <script>, on* handlers, javascript: URLs, foreignObject content
  const clean = DOMPurify.sanitize(svg, {
    USE_PROFILES: { svg: true, svgFilters: true },
    FORBID_TAGS: ['script', 'foreignObject'],
    FORBID_ATTR: ['onload', 'onclick', 'onerror', 'onmouseover'],
  });
  if (!clean || clean.length === 0) {
    throw new Error('sanitized markup_svg is empty — input may have contained only unsafe content');
  }
  return clean;
}
```

import { sanitizeMarkupSvg } from '../lib/markup-sanitizer';

// POST /:fileId/markup — UPSERT (last-write-wins)
cadFilesRouter.post('/:fileId/markup', requireUploader, async (req: any, res) => {
  const partId = req.params.id;
  const fileId = req.params.fileId;
  let clean: string;
  try {
    clean = sanitizeMarkupSvg(req.body?.markup_svg);
  } catch (err: any) {
    return res.status(400).json({ error: err.message });
  }
  try {
    const row = await withTenantClient(req, async (client) => {
      const existing = await client.query(
        `SELECT id FROM turion_satellite.part_drawing_markups WHERE part_cad_file_id=$1 LIMIT 1`, [fileId]);
      let ins;
      if (existing.rows[0]) {
        ins = await client.query(
          `UPDATE turion_satellite.part_drawing_markups
             SET markup_svg=$1, updated_at=now(), created_by_cognito_sub=$2
             WHERE part_cad_file_id=$3 RETURNING id, updated_at`,
          [clean, req.user.sub, fileId]);
      } else {
        ins = await client.query(
          `INSERT INTO turion_satellite.part_drawing_markups
             (part_cad_file_id, tenant_id, markup_svg, created_by_cognito_sub)
           VALUES ($1, current_setting('app.tenant_id')::uuid, $2, $3)
           RETURNING id, created_at AS updated_at`,
          [fileId, clean, req.user.sub]);
      }
      await cadAudit(req, client, 'cad_upload_commit', partId, {
        markup_size: clean.length, cad_file_id: fileId, markup_id: ins.rows[0].id,
      });
      return ins.rows[0];
    });
    res.status(201).json(row);
  } catch (err: any) {
    console.error('[cad-files] markup save failed:', err?.message ?? err);
    res.status(500).json({ error: 'failed to save markup' });
  }
});

// GET /:fileId/markup
cadFilesRouter.get('/:fileId/markup', async (req: any, res) => {
  try {
    const row = await withTenantClient(req, async (client) => {
      const r = await client.query(
        `SELECT id, markup_svg, created_by_cognito_sub, created_at, updated_at
         FROM turion_satellite.part_drawing_markups
         WHERE part_cad_file_id=$1 LIMIT 1`, [req.params.fileId]);
      return r.rows[0];
    });
    if (!row) return res.status(404).json({ error: 'no markup' });
    res.json(row);
  } catch (err: any) {
    console.error('[cad-files] markup get failed:', err?.message ?? err);
    res.status(500).json({ error: 'failed to load markup' });
  }
});

// DELETE /:fileId/markup
cadFilesRouter.delete('/:fileId/markup', requireUploader, async (req: any, res) => {
  const partId = req.params.id;
  try {
    await withTenantClient(req, async (client) => {

  } catch (err: any) {
    console.error('[cad-files] markup delete failed:', err?.message ?? err);
    res.status(500).json({ error: 'failed to delete markup' });
  }
});

partsRouter.get('/:id/revisions', requireAuth, async (req: any, res) => {
  try {
    const rows = await withTenantClient(req, async (client) => {
      // UNION procedural (part_revisions where cad_file_id IS NULL) + uploaded (part_cad_files)
      const r = await client.query(`
        SELECT
          pr.revision::text                                    AS revision,
          'procedural'                                         AS source,
          NULL::uuid                                           AS cad_file_id,
          NULL::text                                           AS filename,
          NULL::text                                           AS format,
          pr.modified_by_cognito_sub                           AS by_sub,
          pr.created_at                                        AS created_at,
          false                                                AS has_markup
        FROM turion_satellite.part_revisions pr
        WHERE pr.part_id = $1 AND pr.cad_file_id IS NULL
        UNION ALL
        SELECT
          cf.revision::text                                    AS revision,
          'uploaded'                                           AS source,
          cf.id                                                AS cad_file_id,
          cf.filename                                          AS filename,
          cf.format                                            AS format,
          cf.uploaded_by_cognito_sub                           AS by_sub,
          cf.uploaded_at                                       AS created_at,
          EXISTS(SELECT 1 FROM turion_satellite.part_drawing_markups m WHERE m.part_cad_file_id = cf.id) AS has_markup
        FROM turion_satellite.part_cad_files cf
        WHERE cf.part_id = $1
        ORDER BY created_at DESC
      `, [req.params.id]);
      return r.rows;
    });
    res.json(rows);
  } catch (err: any) {
    console.error('[parts] revisions failed:', err?.message ?? err);
    res.status(500).json({ error: 'failed to load revision history' });
  }
});

**Step B — Create `frontend/satellite/cad-markup.js`:**

Follow RESEARCH §I Fabric.js example (lines 604-639) with these critical adaptations:
- Pitfall 7: base image loads as a separate locked layer NOT included in `toSVG()`. Strategy: keep the base image rendered in a peer `<img>` sibling element (z-index below the canvas, `pointer-events:none`), NOT as a Fabric.js background. Only the user-drawn objects live in the Fabric canvas — `canvas.toSVG()` then naturally excludes the base.
- Use ESM Fabric.js v6 imports
- Save POSTs to `/api/parts/:id/cad-files/:fileId/markup`
- Load GETs from same endpoint; uses `fabric.loadSVGFromString()`
- Toolbar buttons: Text · Arrow · Rect · Freehand · Save · Clear
- Auto-save on blur (last-write-wins per Open Question 4)

```javascript
import * as fabric from 'fabric';
import { satelliteApi, currentRole } from './turion-config.js';

export async function mountMarkup(container, partId, cadFileId, baseImageUrl) {
  if (!['admin','manager'].includes(currentRole())) { container.style.display='none'; return; }

  // Layer 1: base image as DOM <img> behind the canvas (Pitfall 7 — NOT a Fabric background)
  container.style.position = 'relative';
  if (baseImageUrl) {
    const img = document.createElement('img');
    img.src = baseImageUrl;
    img.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;object-fit:contain;pointer-events:none;z-index:1';
    container.appendChild(img);
  }

  // Layer 2: Fabric.js canvas on top
  const canvasEl = document.createElement('canvas');
  canvasEl.width = container.clientWidth || 800;
  canvasEl.height = container.clientHeight || 520;
  canvasEl.style.cssText = 'position:relative;z-index:2';
  container.appendChild(canvasEl);
  const canvas = new fabric.Canvas(canvasEl, { backgroundColor: 'transparent' });

  // Load existing markup (if any)
  try {
    const existing = await satelliteApi(`/api/parts/${partId}/cad-files/${cadFileId}/markup`);
    if (existing?.markup_svg) {
      const { objects } = await fabric.loadSVGFromString(existing.markup_svg);
      objects.forEach(o => canvas.add(o));
      canvas.renderAll();
    }
  } catch (err) {
    // 404 is fine — no markup yet
  }

  async function save() {
    const svg = canvas.toSVG({ suppressPreamble: true });
    await satelliteApi(`/api/parts/${partId}/cad-files/${cadFileId}/markup`, {
      method: 'POST', body: JSON.stringify({ markup_svg: svg }),
    });
  }

  // Auto-save on every object modification (debounced 500ms)
  let saveTimer;
  function schedSave() { clearTimeout(saveTimer); saveTimer = setTimeout(() => save().catch(err => console.error('[markup] save failed', err)), 500); }
  canvas.on('object:added', schedSave);
  canvas.on('object:modified', schedSave);
  canvas.on('object:removed', schedSave);

  // Toolbar
  return {
    addText: () => canvas.add(new fabric.Textbox('Note', { left: 50, top: 50, fontSize: 16, fill: '#ff3366', fontFamily: 'Arial' })),
    addArrow: () => canvas.add(new fabric.Line([20, 20, 100, 80], { stroke: '#ff3366', strokeWidth: 2 })),
    addRect: () => canvas.add(new fabric.Rect({ left: 50, top: 50, width: 100, height: 60, fill: 'transparent', stroke: '#ff3366', strokeWidth: 2 })),
    enableFreehand: () => { canvas.isDrawingMode = true; canvas.freeDrawingBrush = new fabric.PencilBrush(canvas); canvas.freeDrawingBrush.color = '#ff3366'; canvas.freeDrawingBrush.width = 2; },
    disableFreehand: () => { canvas.isDrawingMode = false; },
    clear: () => { canvas.getObjects().forEach(o => canvas.remove(o)); schedSave(); },
    save,
    exportSvg: () => canvas.toSVG({ suppressPreamble: true }),
  };
}
```

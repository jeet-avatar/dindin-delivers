```typescript
  contentType?: string;
  contentLength: number;
  sha256: string;         // hex
  metadata?: Record<string, string>;
}

export async function presignPut(opts: PresignPutOpts): Promise<string> {
  // Convert hex sha256 → base64 for x-amz-checksum-sha256
  const sha256b64 = Buffer.from(opts.sha256, 'hex').toString('base64');
  const cmd = new PutObjectCommand({
    Bucket: BUCKET,
    Key: opts.key,
    ContentType: opts.contentType ?? 'application/octet-stream',
    ContentLength: opts.contentLength,
    ChecksumSHA256: sha256b64,
    Metadata: opts.metadata ?? {},
  });
  return getSignedUrl(s3, cmd, { expiresIn: 300 });  // 5-min PUT TTL (Anti-Patterns)
}

export async function presignGet(key: string, ttlSec = 900): Promise<string> {
  return getSignedUrl(s3, new GetObjectCommand({ Bucket: BUCKET, Key: key }), { expiresIn: ttlSec });
}

export async function headObject(key: string) {
  return s3.send(new HeadObjectCommand({ Bucket: BUCKET, Key: key }));
}
```

```typescript
const ALLOWED_FORMATS = new Set(['step','stp','stl','iges','igs','brep','dwg','sldprt','3dpdf','pdf']);
const MAX_SIZE = 100 * 1024 * 1024;  // 100 MB (matches DB CHECK)
const SHA256_RE = /^[a-f0-9]{64}$/;

export interface UploadShape {
  filename: string;
  file_size: number;
  sha256: string;
  format: string;
}

export function validateUploadShape(body: any): { ok: true; data: UploadShape } | { ok: false; error: string } {
  if (!body || typeof body !== 'object') return { ok: false, error: 'invalid body' };
  const { filename, file_size, sha256, format } = body;
  if (typeof filename !== 'string' || filename.length === 0 || filename.length > 500) return { ok: false, error: 'invalid filename' };
  if (typeof file_size !== 'number' || file_size <= 0) return { ok: false, error: 'invalid file_size' };
  if (file_size > MAX_SIZE) return { ok: false, error: `file too large (max ${MAX_SIZE} bytes / 100 MB)` };
  if (typeof sha256 !== 'string' || !SHA256_RE.test(sha256)) return { ok: false, error: 'invalid sha256 (expect 64 hex chars)' };
  if (typeof format !== 'string' || !ALLOWED_FORMATS.has(format.toLowerCase())) return { ok: false, error: `unsupported format (allowed: ${[...ALLOWED_FORMATS].join(',')})` };
  return { ok: true, data: { filename, file_size, sha256, format: format.toLowerCase() } };
}
```

```javascript
import { satelliteApi, currentRole } from './turion-config.js';

const ALLOWED_EXT = ['step','stp','stl','iges','igs','brep','dwg','sldprt','3dpdf','pdf'];
const MAX_BYTES = 100 * 1024 * 1024;

async function sha256Hex(file) {
  const buf = await file.arrayBuffer();
  const hash = await crypto.subtle.digest('SHA-256', buf);
  return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2,'0')).join('');
}

function hexToBase64(hex) {
  const bytes = new Uint8Array(hex.length / 2);
  for (let i = 0; i < bytes.length; i++) bytes[i] = parseInt(hex.substr(i*2, 2), 16);
  let bin = ''; bytes.forEach(b => bin += String.fromCharCode(b));
  return btoa(bin);
}

export function mountCadUpload(panelEl, partId) {
  if (!['admin','manager'].includes(currentRole())) { panelEl.style.display = 'none'; return; }
  const input = panelEl.querySelector('#cadFileInput');
  const progress = panelEl.querySelector('#cadUploadProgress');
  const list = panelEl.querySelector('#cadFileList');

  async function refreshList() {
    const r = await satelliteApi(`/api/parts/${partId}/cad-files`);
    list.innerHTML = r.map(f => `<li>rev ${f.revision} · ${f.filename} · ${(f.file_size/1024/1024).toFixed(2)} MB · ${f.is_active ? '✓ active' : '✗ inactive'} · ${new Date(f.uploaded_at).toLocaleString()}</li>`).join('') || '<li><em>No uploads yet.</em></li>';
  }

  input.addEventListener('change', async () => {
    const file = input.files?.[0]; if (!file) return;
    const ext = (file.name.split('.').pop() ?? '').toLowerCase();
    if (!ALLOWED_EXT.includes(ext)) { progress.textContent = `Unsupported extension: ${ext}`; return; }
    if (file.size > MAX_BYTES) { progress.textContent = `File too large: ${(file.size/1024/1024).toFixed(1)} MB (max 100 MB)`; return; }
    progress.textContent = `Hashing ${file.name}…`;
    const sha256 = await sha256Hex(file);
    progress.textContent = `Requesting upload URL…`;
    const { url, key } = await satelliteApi(`/api/parts/${partId}/cad-files/presign`, {
      method: 'POST', body: JSON.stringify({ filename: file.name, file_size: file.size, sha256, format: ext }),
    });
    progress.textContent = `Uploading 0%…`;
    await new Promise((resolve, reject) => {
      const xhr = new XMLHttpRequest();
      xhr.open('PUT', url);
      xhr.setRequestHeader('x-amz-checksum-sha256', hexToBase64(sha256));
      xhr.upload.onprogress = e => { if (e.lengthComputable) progress.textContent = `Uploading ${Math.round(e.loaded/e.total*100)}%…`; };
      xhr.onload = () => xhr.status >= 200 && xhr.status < 300 ? resolve() : reject(new Error(`PUT failed: ${xhr.status} ${xhr.responseText.slice(0,200)}`));
      xhr.onerror = () => reject(new Error('PUT network error'));
      xhr.send(file);
    });
    progress.textContent = `Committing…`;
    await satelliteApi(`/api/parts/${partId}/cad-files/commit`, {
      method: 'POST', body: JSON.stringify({ key, format: ext, file_size: file.size, sha256, filename: file.name }),
    });
    progress.textContent = `✓ Uploaded ${file.name}`;
    input.value = '';
    await refreshList();
  });

  refreshList();
}
```

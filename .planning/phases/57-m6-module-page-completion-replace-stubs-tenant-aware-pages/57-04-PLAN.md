**Step C — Add 2 NEW GET routes in agents.ts (at end of router, before `export default`):**

```typescript
const VALID_KINDS = new Set(['sales-cash','ncr-capa','evms','integration-sentinel']);

r.get('/runs', async (req, res) => {
  const kind = String(req.query.kind || '');
  if (!VALID_KINDS.has(kind)) return res.status(400).json({ error: 'kind required (one of: ' + Array.from(VALID_KINDS).join(',') + ')' });
  const limit = Math.min(parseInt(String(req.query.limit || '50'), 10) || 50, 200);
  const rows = await withTenantClient(req, async (client) => {
    const r = await client.query(
      `SELECT id, agent_kind, started_at, completed_at, status, triggered_by_sub, error_message,
              jsonb_build_object('summary', output->'summary', 'ids', output->'ids') AS output_summary
       FROM turion.agent_runs
       WHERE agent_kind = $1
       ORDER BY started_at DESC
       LIMIT $2`,
      [kind, limit]
    );
    return r.rows;
  });
  res.json(rows);
});

r.get('/runs/:id', async (req, res) => {
  const data = await withTenantClient(req, async (client) => {
    const r = await client.query(`SELECT * FROM turion.agent_runs WHERE id = $1`, [req.params.id]);
    return r.rows[0] ?? null;
  });
  if (!data) return res.status(404).json({ error: 'not found' });
  res.json(data);
});
```

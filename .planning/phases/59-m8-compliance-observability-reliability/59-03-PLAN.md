**Step C — Build `src/pages/ApiDocsPage.tsx` (~80 LOC):**

```tsx
import { useEffect } from 'react'
import { PageHelmet } from '../components/PageHelmet'
import { NavBar } from '../components/NavBar'
import { SiteFooter } from '../components/SiteFooter'

declare global {
  interface Window { SwaggerUIBundle?: any; SwaggerUIStandalonePreset?: any }
}

export default function ApiDocsPage() {
  useEffect(() => {
    // Inject CSS once
    if (!document.getElementById('swagger-css')) {
      const l = document.createElement('link')
      l.id = 'swagger-css'; l.rel = 'stylesheet'; l.href = '/swagger-ui/swagger-ui.css'
      document.head.appendChild(l)
    }
    // Inject bundle once + initialize after load
    if (!window.SwaggerUIBundle) {
      const s = document.createElement('script')
      s.src = '/swagger-ui/swagger-ui-bundle.js'
      s.onload = () => {
        if (window.SwaggerUIBundle) {
          window.SwaggerUIBundle({
            url: '/docs/openapi.yaml',
            dom_id: '#swagger-ui',
            deepLinking: true,
            tryItOutEnabled: true,
          })
        }
      }
      document.body.appendChild(s)
    } else {
      window.SwaggerUIBundle({
        url: '/docs/openapi.yaml',
        dom_id: '#swagger-ui',
        deepLinking: true,
        tryItOutEnabled: true,
      })
    }
  }, [])

  return (
    <>
      <PageHelmet
        title="API Documentation"
        description="Interactive REST API reference for the Zietra multi-tenant SaaS platform — top 30 endpoints with try-it-now."
        path="/docs/api"
      />
      <NavBar />
      <main className="api-docs-page" style={{ maxWidth: 1200, margin: '0 auto', padding: '24px' }}>
        <section className="hero">
          <h1>Zietra Platform API</h1>
          <p className="lede">Interactive reference for the top 30 endpoints. Full 169-route coverage ships with M9.</p>
        </section>
        <div id="swagger-ui" />
      </main>
      <SiteFooter />
    </>
  )
}
```

import satori from 'satori'

// Load Inter Bold font data (Satori needs explicit font bytes)
const interBold = await fs.readFile(path.join(__dirname, 'fonts/Inter-Bold.ttf'))

// SIMPLEST APPROACH: have separate static lists here that we sync once. OR run via tsx.

const MODULES = [
  { slug: 'crm',           title: 'Salesforce CRM',        tagline: 'Pipeline · contacts · deals — built into your ERP' },
  { slug: 'sales',         title: 'Sales Orders',          tagline: 'Multi-tier pricing · invoice in 1 click' },

  { slug: 'mes',           title: 'Manufacturing Execution', tagline: 'Work orders · stages · WIP dashboard' },
  { slug: 'quality',       title: 'Quality Management',    tagline: 'NCRs · CAPAs · internal audits' },

  { slug: 'qb-migration',  title: 'QuickBooks Migration',  tagline: 'COA wizard · 6-type guided cutover' },
]

]

function template({ kind, title, tagline, slug }) {
  return {
    type: 'div',
    props: {
      style: {
        width: '100%', height: '100%',
        display: 'flex', flexDirection: 'column', justifyContent: 'center',
        padding: '80px',
        background: 'linear-gradient(135deg, #6d28d9 0%, #4c1d95 100%)',
        color: 'white', fontFamily: 'Inter',
      },
      children: [
        { type: 'div', props: { style: { fontSize: 24, opacity: 0.75, marginBottom: 16 }, children: 'Zietra' } },
        { type: 'div', props: { style: { fontSize: 64, fontWeight: 700, marginBottom: 24, lineHeight: 1.1 }, children: title } },
        { type: 'div', props: { style: { fontSize: 28, opacity: 0.9, lineHeight: 1.4 }, children: tagline } },
        { type: 'div', props: { style: { fontSize: 20, marginTop: 'auto', opacity: 0.6 }, children: `zietra.com/${kind}/${slug}` } },
      ],
    },
  }
}

async function generate(items, kind, outDir) {
  await fs.mkdir(outDir, { recursive: true })
  for (const it of items) {

  }
}

```bash
#!/usr/bin/env bash
set -uo pipefail
BUCKET="zietra-status"
CF_DIST_ID_FILE=".cf-dist-id"
LAMBDA_NAME="zietra-status-api"
ROOT="$(cd "$(dirname "$0")" && pwd)"

# 1. Upload static assets to S3
aws s3 sync "$ROOT/" "s3://$BUCKET/" --exclude "lambda/*" --exclude "deploy.sh" --exclude ".*" \
  --delete --content-type "text/html; charset=utf-8" --include "*.html"
aws s3 sync "$ROOT/" "s3://$BUCKET/" --exclude "lambda/*" --exclude "deploy.sh" --exclude ".*" \
  --exclude "*.html" --delete

# 2. Package + update Lambda (only if lambda/ changed since last deploy — skip otherwise)
if [ -d "$ROOT/lambda" ]; then
  cd "$ROOT/lambda"
  npm install --silent --omit=dev
  rm -f /tmp/zietra-status-api.zip
  zip -r /tmp/zietra-status-api.zip handler.mjs package.json node_modules >/dev/null
  aws lambda update-function-code --function-name "$LAMBDA_NAME" --zip-file fileb:///tmp/zietra-status-api.zip --output text
  cd "$ROOT"
fi

# 3. Invalidate CF
if [ -f "$CF_DIST_ID_FILE" ]; then
  CF_DIST_ID=$(cat "$CF_DIST_ID_FILE")
  INV=$(aws cloudfront create-invalidation --distribution-id "$CF_DIST_ID" --paths "/*" --query 'Invalidation.Id' --output text)
  aws cloudfront wait invalidation-completed --distribution-id "$CF_DIST_ID" --id "$INV"
  echo "CF invalidation $INV Completed"
fi
echo "Status page deploy: done"
```

```typescript
import express from 'express'
import { withTenantClient } from '../db'
import { requireAdmin } from '../middleware/requireAdmin'  // from Phase 54.1 — adjust import path

const r = express.Router()

/**
 * GET /api/audit-log?cursor=<iso>&limit=50
 *
 * Admin-only paginated read of public.audit_log_v2 for the current tenant.
 * RLS enforces tenant scoping; this handler also explicitly filters via
 * withTenantClient + the policy.
 *
 * Response: { rows: [...], next_cursor: <iso|null> }
 */
r.get('/', requireAdmin, async (req, res) => {
  const limit = Math.min(parseInt(String(req.query.limit ?? '50'), 10) || 50, 200)
  const cursor = req.query.cursor ? String(req.query.cursor) : null

  try {
    const result = await withTenantClient(req, async (client) => {
      const params: unknown[] = []
      let where = ''
      if (cursor) { params.push(cursor); where = `WHERE created_at < $${params.length}` }
      params.push(limit + 1)  // fetch one extra to know if more pages exist
      const sql = `
        SELECT id, actor_cognito_sub, action, resource_type, resource_id,
               before_data, after_data, ip_address::text, user_agent, created_at
        FROM public.audit_log_v2
        ${where}
        ORDER BY created_at DESC
        LIMIT $${params.length}
      `
      const { rows } = await client.query(sql, params)
      const hasMore = rows.length > limit
      const slice = hasMore ? rows.slice(0, limit) : rows
      const next_cursor = hasMore ? slice[slice.length - 1].created_at.toISOString() : null
      return { rows: slice, next_cursor }
    })
    res.json(result)
  } catch (e) {
    // Hardened error shape per Phase 38 (no err.message leak to client)
    console.error('[audit-log] read failed:', e instanceof Error ? e.message : e)
    res.status(500).json({ error: 'audit log read failed' })
  }
})

export default r
```

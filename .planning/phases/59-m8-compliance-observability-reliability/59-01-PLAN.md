**Step B — Add `auditLog()` helper to `backend/src/db.ts` (RESEARCH Pattern 2 + Pitfall 4):**

Append (preserving existing `withTenantClient` + `audit` definitions):

```typescript
import type { Request } from 'express'
import type { PoolClient } from 'pg'

export interface AuditLogOpts {
  action: string
  resource_type: string
  resource_id: string | null
  before: unknown
  after: unknown
}

/**
 * Insert one row into public.audit_log_v2.
 *
 * MUST be called from inside a `withTenantClient(req, async (client) => { ... })`
 * callback so the SET LOCAL app.tenant_id is in scope. The 2nd arg is the
 * PoolClient yielded by withTenantClient — passing a fresh pool client will
 * fail RLS (Pitfall 4).
 *
 * The insert participates in the caller's transaction: if the caller's
 * mutation fails or rolls back, the audit row rolls back with it (no
 * orphaned audit rows).
 */
export async function auditLog(
  req: Request,
  client: PoolClient,
  opts: AuditLogOpts
): Promise<void> {
  // X-Forwarded-For format (CloudFront-aware): pick the penultimate IP per the
  // cache.ts:209 pattern — the last IP is CloudFront, the second-to-last is
  // the real client IP (attacker cannot inject past CF).
  const xff = req.headers['x-forwarded-for']?.toString().split(',').map(s => s.trim()) ?? []
  const ip = xff.length >= 2 ? xff[xff.length - 2] : (xff[0] ?? null)
  const ua = req.headers['user-agent']?.toString().slice(0, 500) ?? null

  await client.query(
    `INSERT INTO public.audit_log_v2
       (tenant_id, actor_cognito_sub, action, resource_type, resource_id,
        before_data, after_data, ip_address, user_agent)
     VALUES ($1, $2, $3, $4, $5, $6::jsonb, $7::jsonb, $8::inet, $9)`,
    [
      // @ts-ignore - tenant set by tenantContext middleware
      req.tenant!.id,
      // @ts-ignore - user set by requireAuth middleware
      req.user?.sub ?? null,
      opts.action,
      opts.resource_type,
      opts.resource_id,
      JSON.stringify(opts.before ?? null),
      JSON.stringify(opts.after ?? null),
      ip,
      ua,
    ]
  )
}
```

-- ============================================================================

DO $do$
DECLARE
  v_part_def_id UUID;
  v_part_inst_id UUID;
  v_pr_id UUID;
  v_vendor_id UUID;
  v_pn TEXT;
  v_subsystem TEXT;
  v_should_have_vo BOOLEAN;
  v_counter INT := 0;
  v_subsystem_pick_count INT;
BEGIN
  FOR v_part_def_id, v_part_inst_id, v_pn, v_subsystem IN
    SELECT pd.id, pi.id, pd.part_number, s.code
    FROM turion_satellite.part_definitions pd
    JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
    JOIN turion_satellite.part_instances pi
      ON pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
     AND pi.part_definition_id = pd.id
     AND pi.instance_index = 0
    WHERE pd.default_make_buy = 'buy'
      AND NOT EXISTS (
        SELECT 1 FROM turion_satellite.procurement_requests pr
        WHERE pr.part_instance_id = pi.id
      )
    ORDER BY s.code, pd.part_number
  LOOP
    INSERT INTO procurement_requests (
      part_instance_id, request_number, status, quantity, requested_by, created_at
    ) VALUES (
      v_part_inst_id,
      'PR-26-' || substring(v_part_inst_id::text, 1, 8),
      'approved',
      1,
      'procurement@turion-space.com',
      NOW() - INTERVAL '21 days'
    ) RETURNING id INTO v_pr_id;

    -- For ~30 representative parts: also create a vendor_order
    -- Sample 4-5 per subsystem (deterministic via counter modulo).
    v_counter := v_counter + 1;
    v_should_have_vo := (v_counter % 2 = 0);  -- ~half get a vendor_order ≈ 30 of ~60 buy-parts

    IF v_should_have_vo THEN
      -- Pick a vendor matching the subsystem category (or any if no match)
      SELECT id INTO v_vendor_id
      FROM turion_satellite.vendors
      WHERE category ILIKE CASE v_subsystem
             WHEN 'EPS'  THEN '%solar%'
             WHEN 'STR'  THEN '%struct%'
             WHEN 'ADCS' THEN '%sensor%'
             WHEN 'PROP' THEN '%prop%'
             WHEN 'PAY'  THEN '%optic%'
             WHEN 'COMM' THEN '%rf%'
             WHEN 'TCS'  THEN '%thermal%'
             WHEN 'CDH'  THEN '%avionic%'
             ELSE '%'
           END
      LIMIT 1;

      IF v_vendor_id IS NULL THEN
        SELECT id INTO v_vendor_id FROM turion_satellite.vendors ORDER BY name LIMIT 1;
      END IF;

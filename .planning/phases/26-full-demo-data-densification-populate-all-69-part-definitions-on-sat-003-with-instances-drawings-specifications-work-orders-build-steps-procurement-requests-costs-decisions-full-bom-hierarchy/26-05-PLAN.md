```bash
API_BASE="https://rjydekliee.execute-api.us-east-1.amazonaws.com"
SAT_ID="24587565-b15b-42ce-b590-87ecf9b6bb99"

while IFS='|' read -r subsystem part_number part_def_id part_inst_id; do
  echo "=== $subsystem: $part_number ==="

  curl -sS -H "Authorization: Bearer $TOKEN" "$API_BASE/api/parts/$part_def_id?sat=$SAT_ID" \

        has_drawing: (.drawing_svg != null and (.drawing_svg | length) > 100),

        spec_keys: (.specifications | keys),
        instance_count: (.instances | length // 0)
      }'

  # 2. GET make-buy decision
  curl -sS -H "Authorization: Bearer $TOKEN" "$API_BASE/api/make-buy-decisions?sat=$SAT_ID&part_definition_id=$part_def_id" \
    | jq '.[] | select(.superseded_by == null) | {decision, status, rationale_len: (.rationale | length)}'
```

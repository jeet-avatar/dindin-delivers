**Step 2 — DB-direct authoritative verification (PRIMARY proof — must pass before any curl).**

The 7 queries below verify the SAME truths the API endpoints would surface, but go directly to the database. The DB is the authoritative source — API endpoints are passthroughs. If any of these 7 queries fail their expected count, Phase 26 is NOT complete regardless of what curl returns.

```bash
psql "$DATABASE_URL" <<'SQL' | tee /tmp/phase26-db-verify.txt
\echo === DB-DIRECT VERIFICATION (Phase 26 acceptance gate) ===

-- Query 1: drawing_svg coverage = 80
\echo --- Query 1: drawing_svg coverage ---
SELECT
  COUNT(*) AS drawing_svg_populated,
  (SELECT COUNT(*) FROM turion_satellite.part_definitions) AS total_parts,
  CASE WHEN COUNT(*) = (SELECT COUNT(*) FROM turion_satellite.part_definitions)
       THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.part_definitions
WHERE drawing_svg IS NOT NULL;
-- Expected: drawing_svg_populated = total_parts (100% coverage, currently 80)

-- Query 2: specifications coverage = 80
\echo --- Query 2: specifications coverage (specifications != '{}'::jsonb) ---
SELECT
  COUNT(*) AS specs_populated,
  (SELECT COUNT(*) FROM turion_satellite.part_definitions) AS total_parts,
  CASE WHEN COUNT(*) = (SELECT COUNT(*) FROM turion_satellite.part_definitions)
       THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.part_definitions
WHERE specifications <> '{}'::jsonb;
-- Expected: specs_populated = total_parts (100% coverage)

-- Query 3: make_buy_decisions on SAT-003 = 80 (one per part_definition, status='approved')
\echo --- Query 3: make_buy_decisions on SAT-003 ---
SELECT
  COUNT(*) AS approved_decisions,
  (SELECT COUNT(*) FROM turion_satellite.part_definitions) AS total_parts,
  CASE WHEN COUNT(*) = (SELECT COUNT(*) FROM turion_satellite.part_definitions)
       THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.make_buy_decisions
WHERE satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
  AND status = 'approved'
  AND superseded_by IS NULL;
-- Expected: approved_decisions = total_parts (100% coverage, every part has approved decision)

-- Query 4: work_orders count per make-part >= 1
\echo --- Query 4: work_orders per make-part (every make-part has >=1 WO) ---
WITH make_parts AS (
  SELECT pd.id AS part_def_id, pi.id AS part_inst_id
  FROM turion_satellite.part_definitions pd
  JOIN turion_satellite.part_instances pi
    ON pi.part_definition_id = pd.id
   AND pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
   AND pi.instance_index = 0
  WHERE pd.default_make_buy = 'make'
),
wo_counts AS (
  SELECT mp.part_def_id, COUNT(wo.id) AS wo_count
  FROM make_parts mp
  LEFT JOIN turion_satellite.work_orders wo ON wo.part_instance_id = mp.part_inst_id
  GROUP BY mp.part_def_id
)
SELECT
  COUNT(*) FILTER (WHERE wo_count = 0) AS make_parts_missing_wo,
  COUNT(*) FILTER (WHERE wo_count >= 1) AS make_parts_with_wo,
  CASE WHEN COUNT(*) FILTER (WHERE wo_count = 0) = 0
       THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM wo_counts;
-- Expected: make_parts_missing_wo = 0 (every make-part has at least one work_order)

-- Query 5: procurement_requests count per buy-part >= 1
\echo --- Query 5: procurement_requests per buy-part (every buy-part has >=1 PR) ---
WITH buy_parts AS (
  SELECT pd.id AS part_def_id, pi.id AS part_inst_id
  FROM turion_satellite.part_definitions pd
  JOIN turion_satellite.part_instances pi
    ON pi.part_definition_id = pd.id
   AND pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
   AND pi.instance_index = 0
  WHERE pd.default_make_buy = 'buy'
),
pr_counts AS (
  SELECT bp.part_def_id, COUNT(pr.id) AS pr_count
  FROM buy_parts bp
  LEFT JOIN turion_satellite.procurement_requests pr ON pr.part_instance_id = bp.part_inst_id
  GROUP BY bp.part_def_id
)
SELECT
  COUNT(*) FILTER (WHERE pr_count = 0) AS buy_parts_missing_pr,
  COUNT(*) FILTER (WHERE pr_count >= 1) AS buy_parts_with_pr,
  CASE WHEN COUNT(*) FILTER (WHERE pr_count = 0) = 0
       THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM pr_counts;
-- Expected: buy_parts_missing_pr = 0 (every buy-part has at least one procurement_request)

-- Query 6: bom_lines count on SAT-003 >= 150
\echo --- Query 6: bom_lines on SAT-003 (target >= 150) ---
SELECT
  COUNT(*) AS bom_lines,
  CASE WHEN COUNT(*) >= 150 THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM turion_satellite.bom_lines bl
JOIN turion_satellite.part_instances pi ON pi.id = bl.parent_part_instance_id
WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99';
-- Expected: bom_lines >= 150 (CONTEXT.md target)

-- Query 7: cost rollup total on SAT-003 in $5M-$15M range
\echo --- Query 7: cost rollup total ---
WITH actuals AS (
  SELECT mc.material_cost + (mc.labor_hours * lr.hourly_rate) * (1 + mc.overhead_pct/100.0) AS unit_cost
  FROM turion_satellite.make_costs mc
  JOIN turion_satellite.part_instances pi ON pi.id = mc.part_instance_id
  LEFT JOIN turion_satellite.labor_rates lr ON lr.id = mc.labor_rate_id
  WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
    AND mc.superseded_by IS NULL
  UNION ALL
  SELECT bc.unit_price
  FROM turion_satellite.buy_costs bc
  JOIN turion_satellite.part_instances pi ON pi.id = bc.part_instance_id
  WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
    AND bc.superseded_by IS NULL
)
SELECT
  ROUND(SUM(unit_cost)::numeric, 2) AS total_cost_usd,
  CASE WHEN SUM(unit_cost) BETWEEN 5000000 AND 15000000
       THEN 'PASS' ELSE 'FAIL' END AS verdict
FROM actuals;
-- Expected: total_cost_usd between 5,000,000 and 15,000,000 (CONTEXT.md realism guard)

-- NOTE: if the make_costs schema doesn't carry labor_hours/labor_rate_id/material_cost/overhead_pct
-- (Phase 26-03 Task 0 schema introspection determines the actual column names), substitute the
-- computed unit_cost expression with whichever column represents the per-unit make_cost on this
-- schema. Same for buy_costs.unit_price — substitute the actual price column.
SQL
```

Required structure:

```javascript
#!/usr/bin/env node
// scripts/generate-densify-sql.js · Phase 26-01
// Introspects part_definitions and emits idempotent SQL filling drawing_svg + specifications.
// Usage: DATABASE_URL=... node scripts/generate-densify-sql.js > migrations/011_densify_drawings_and_specs.sql

import pg from 'pg';
import fs from 'node:fs';

// Mirror spec-keys.ts COMMON_SPEC_KEYS + SUBSYSTEM_SPEC_HINTS. Keep this in sync manually
// (SQL migration can't import TS). Validation: planner cross-checks before merge.
const SUBSYSTEM_SPEC_HINTS = {
  EPS:  ['efficiency_pct', 'output_voltage_v', 'output_current_ma'],
  STR:  ['thread_pitch', 'thread_size', 'head_type'],
  ADCS: ['momentum_capacity_mnms', 'max_torque_mnm', 'max_speed_rpm'],
  PROP: ['thrust_n', 'isp_s', 'propellant'],
  PAY:  ['focal_length_mm', 'aperture_mm', 'sensor_pixels'],
  COMM: ['frequency_band', 'gain_dbi', 'eirp_dbw'],
  TCS:  ['thermal_capacity_w', 'operating_pressure_bar'],
  CDH:  ['clock_mhz', 'ram_mb', 'storage_gb'],
};

// SVG generators per category. Reuse Phase 21 aesthetic from
// migrations/007_seed_eps_solar_array_drilldown.sql.
function genFastenerSvg(sizeLabel) { /* generic bolt template */ }
function genSubsystemSvg(subsystem, partType) { /* dispatch on subsystem code */ }

// Spec blob generator. Common keys always populated with plausible
// values derived from subsystem + part description; subsystem-specific
// keys layered on top via SUBSYSTEM_SPEC_HINTS dispatch.
function genSpecsBlob(part) { /* returns JSONB-safe JS object */ }

async function main() {
  const client = new pg.Client({ connectionString: process.env.DATABASE_URL });
  await client.connect();

  const rows = await client.query(`
    SELECT pd.id, pd.part_number, pd.description, pd.drawing_svg, pd.specifications, s.code AS subsystem_code
    FROM turion_satellite.part_definitions pd
    JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
    ORDER BY s.code, pd.part_number
  `);

  // emit file header (DO $$ database guard, comments, SET search_path)
  const out = [];
  out.push(`-- 011_densify_drawings_and_specs.sql · Phase 26-01\n`);
  out.push(`-- Generated by scripts/generate-densify-sql.js on ${new Date().toISOString()}\n`);
  out.push(`SET search_path TO turion_satellite, public;\n\n`);
  out.push(`DO $do$ BEGIN IF current_database() NOT IN ('postgres') THEN RAISE EXCEPTION 'Refusing to run on database %', current_database(); END IF; END $do$;\n\n`);

  // For each part missing drawing_svg, emit:
  //   UPDATE part_definitions SET drawing_svg = $svg$<svg>...</svg>$svg$
  //   WHERE part_number = '...' AND drawing_svg IS NULL;
  //
  // For each part with empty specifications, emit:
  //   UPDATE part_definitions SET specifications = '{...}'::jsonb
  //   WHERE part_number = '...' AND specifications = '{}'::jsonb;

  for (const part of rows.rows) {
    if (!part.drawing_svg) {
      const svg = genSubsystemSvg(part.subsystem_code, part.part_number);
      out.push(`-- ${part.part_number}: ${part.description}\n`);
      out.push(`UPDATE turion_satellite.part_definitions SET drawing_svg = $svg$${svg}$svg$ WHERE part_number = '${part.part_number}' AND drawing_svg IS NULL;\n\n`);
    }
    const hasSpecs = part.specifications && Object.keys(part.specifications).length > 0;
    if (!hasSpecs) {
      const specs = genSpecsBlob(part);
      const specsJson = JSON.stringify(specs).replace(/'/g, "''");
      out.push(`UPDATE turion_satellite.part_definitions SET specifications = '${specsJson}'::jsonb WHERE part_number = '${part.part_number}' AND specifications = '{}'::jsonb;\n\n`);
    }
  }

  process.stdout.write(out.join(''));
  await client.end();
}

main().catch(err => { console.error(err); process.exit(1); });
```

```sql
-- 014_seed_cross_system_fks.sql · Phase 26-04
-- Populate Phase 25 cross-schema FK columns with realistic sample linkages.
-- 15-20 part_instances linked + 5-8 vendor_orders linked.
--
-- ⚠️ APPLY ORDER: This migration depends on migration 015_extend_audit_log_actions.sql
-- being applied FIRST so the audit_log.chk_audit_log_action CHECK constraint
-- permits 'densify_seed'. Plan 26-05's apply list runs 015 then 014.
--
-- All target IDs resolved by JOIN against turion.* (no hardcoded turion IDs) so
-- this migration is portable across environments where legacy IDs may differ.
--
-- Idempotent: re-running is a no-op (UPDATE guards on `WHERE <fk_col> IS NULL`).

SET search_path TO turion_satellite, public;

DO $do$ BEGIN
  IF current_database() NOT IN ('postgres') THEN
    RAISE EXCEPTION 'Refusing to run on database %', current_database();
  END IF;
END $do$;

-- ============================================================================
-- Block 1: part_instances → turion.sales_orders (3-5 parts per SO × 6 SOs)
-- ============================================================================
-- Strategy: for each SO in turion.sales_orders, pick 3-5 part_instances from
-- SAT-003 whose subsystem matches the SO's customer/product context.
-- Since we don't know exact SO descriptions ahead of time, use a generic
-- subsystem-rotation approach: SO[1]→EPS parts, SO[2]→STR parts, SO[3]→ADCS, etc.
-- Each part_instance gets cross_links_updated_at set, and we add an audit_log row
-- with action='densify_seed' (new CHECK constraint value added in this migration).

-- Note: audit_log CHECK constraint expansion (adding 'densify_seed') is handled
-- by migration 015_extend_audit_log_actions.sql, which Plan 26-05 applies FIRST
-- in Phase 26-04's apply order. This migration assumes the constraint already
-- permits 'densify_seed' and writes audit_log rows freely.

-- Per-SO linkage. ROW_NUMBER assigns SO sequence; subsystem dispatch picks
-- corresponding parts; LIMIT bounds count per SO.
DO $do$
DECLARE
  v_so_id TEXT;
  v_so_seq INT := 0;
  v_subsystem TEXT;
  v_pi_id UUID;
BEGIN
  FOR v_so_id IN SELECT id FROM turion.sales_orders ORDER BY id LOOP
    v_so_seq := v_so_seq + 1;
    v_subsystem := CASE v_so_seq
      WHEN 1 THEN 'EPS'
      WHEN 2 THEN 'STR'
      WHEN 3 THEN 'ADCS'
      WHEN 4 THEN 'PROP'
      WHEN 5 THEN 'COMM'
      WHEN 6 THEN 'PAY'
      ELSE 'TCS'
    END;

    -- Pick 3-5 part_instances of this subsystem on SAT-003 that don't already
    -- have a sales_order_id, then UPDATE.
    FOR v_pi_id IN
      SELECT pi.id
      FROM turion_satellite.part_instances pi
      JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
      JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
      WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
        AND s.code = v_subsystem
        AND pi.sales_order_id IS NULL
      ORDER BY pi.id
      LIMIT 4
    LOOP
      UPDATE turion_satellite.part_instances
      SET sales_order_id = v_so_id,
          cross_links_updated_at = NOW()
      WHERE id = v_pi_id
        AND sales_order_id IS NULL;

      -- Audit
      INSERT INTO turion_satellite.audit_log (action, entity_type, entity_id, changes, actor, created_at)
      VALUES ('densify_seed', 'part_instance', v_pi_id::text,
              jsonb_build_object('sales_order_id', v_so_id),
              'phase26@turion-space.com', NOW())
      ON CONFLICT DO NOTHING;
    END LOOP;
  END LOOP;
END $do$;

-- ============================================================================
-- Block 2: vendor_orders → turion.invoices (5-8 high-value links)
-- ============================================================================
-- Pick 6 invoices from turion.invoices ordered by amount DESC, and link to 6
-- vendor_orders ordered by unit_price DESC (high-value pairings make narrative
-- sense — invoice "Cygnus comms hardware $250K" → vendor_order for PAY-IMAGER).

DO $do$
DECLARE
  v_inv_id TEXT;
  v_vo_id UUID;
  v_pairings RECORD;
BEGIN
  FOR v_pairings IN
    WITH invoice_rank AS (
      SELECT id, ROW_NUMBER() OVER (ORDER BY amount DESC) AS rnk
      FROM turion.invoices
    ),
    vo_rank AS (
      SELECT vo.id, ROW_NUMBER() OVER (ORDER BY vo.unit_price DESC) AS rnk
      FROM turion_satellite.vendor_orders vo
      JOIN turion_satellite.procurement_requests pr ON pr.id = vo.procurement_request_id
      JOIN turion_satellite.part_instances pi ON pi.id = pr.part_instance_id
      WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
        AND vo.ns_invoice_id IS NULL
    )
    SELECT ir.id AS inv_id, vr.id AS vo_id
    FROM invoice_rank ir
    JOIN vo_rank vr ON vr.rnk = ir.rnk
    WHERE ir.rnk <= 6
  LOOP
    UPDATE turion_satellite.vendor_orders
    SET ns_invoice_id = v_pairings.inv_id,
        cross_links_updated_at = NOW()
    WHERE id = v_pairings.vo_id
      AND ns_invoice_id IS NULL;

    INSERT INTO turion_satellite.audit_log (action, entity_type, entity_id, changes, actor, created_at)
    VALUES ('densify_seed', 'vendor_order', v_pairings.vo_id::text,
            jsonb_build_object('ns_invoice_id', v_pairings.inv_id),
            'phase26@turion-space.com', NOW())
    ON CONFLICT DO NOTHING;
  END LOOP;
END $do$;

-- ============================================================================
-- Block 3: part_instances → turion.arena_docs (3-5 thematic links)
-- ============================================================================
-- Pick 4 arena_docs and link to subsystem-matched part_instances.
-- Use ILIKE matching on arena_doc.title against subsystem-related keywords.

DO $do$
DECLARE
  v_arena_id TEXT;
  v_pi_id UUID;
  v_keyword TEXT;
BEGIN
  -- For each (subsystem-keyword × arena_doc match), link 1 part_instance
  FOR v_keyword, v_arena_id IN
    SELECT
      CASE
        WHEN ad.title ~* 'solar|power|eps' THEN 'EPS'
        WHEN ad.title ~* 'struct|frame|fastener' THEN 'STR'
        WHEN ad.title ~* 'adcs|wheel|sensor|gyro' THEN 'ADCS'
        WHEN ad.title ~* 'prop|thrust|tank' THEN 'PROP'
        WHEN ad.title ~* 'comm|antenna|radio' THEN 'COMM'
        WHEN ad.title ~* 'imager|optic|lens|payload' THEN 'PAY'
        WHEN ad.title ~* 'thermal|radiator|heat' THEN 'TCS'
        WHEN ad.title ~* 'avionic|obc|fpga|cdh' THEN 'CDH'
        ELSE 'STR'  -- fallback
      END AS subsystem,
      ad.id AS arena_id
    FROM turion.arena_docs ad
    ORDER BY ad.id
    LIMIT 5
  LOOP
    SELECT pi.id INTO v_pi_id
    FROM turion_satellite.part_instances pi
    JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
    JOIN turion_satellite.subsystems s ON s.id = pd.subsystem_id
    WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
      AND s.code = v_keyword
      AND pi.arena_doc_id IS NULL
    ORDER BY pi.id
    LIMIT 1;

    IF v_pi_id IS NOT NULL THEN
      UPDATE turion_satellite.part_instances
      SET arena_doc_id = v_arena_id,
          cross_links_updated_at = NOW()
      WHERE id = v_pi_id;

      INSERT INTO turion_satellite.audit_log (action, entity_type, entity_id, changes, actor, created_at)
      VALUES ('densify_seed', 'part_instance', v_pi_id::text,
              jsonb_build_object('arena_doc_id', v_arena_id),
              'phase26@turion-space.com', NOW())
      ON CONFLICT DO NOTHING;
    END IF;
  END LOOP;
END $do$;

-- ============================================================================
-- Block 4: part_instances → turion.work_orders (legacy MES WOs, 3-5 links)
-- ============================================================================
-- For 4 make-part instances on SAT-003, link to a legacy turion.work_orders row.
-- Pair by description keyword if possible, else round-robin.

DO $do$
DECLARE
  v_mes_wo_id TEXT;
  v_pi_id UUID;
BEGIN
  FOR v_mes_wo_id IN SELECT id FROM turion.work_orders ORDER BY id LIMIT 5 LOOP
    SELECT pi.id INTO v_pi_id
    FROM turion_satellite.part_instances pi
    JOIN turion_satellite.part_definitions pd ON pd.id = pi.part_definition_id
    WHERE pi.satellite_id = '24587565-b15b-42ce-b590-87ecf9b6bb99'
      AND pd.default_make_buy = 'make'
      AND pi.mes_work_order_id IS NULL
    ORDER BY pi.id
    LIMIT 1;

    IF v_pi_id IS NOT NULL THEN
      UPDATE turion_satellite.part_instances
      SET mes_work_order_id = v_mes_wo_id,
          cross_links_updated_at = NOW()
      WHERE id = v_pi_id;

      INSERT INTO turion_satellite.audit_log (action, entity_type, entity_id, changes, actor, created_at)
      VALUES ('densify_seed', 'part_instance', v_pi_id::text,
              jsonb_build_object('mes_work_order_id', v_mes_wo_id),
              'phase26@turion-space.com', NOW())
      ON CONFLICT DO NOTHING;
    END IF;
  END LOOP;
END $do$;
```

    ```typescript
    // Money helpers for cost-module arithmetic.
    // Source: RESEARCH.md Pattern + decimal.js docs (https://mikemcl.github.io/decimal.js/).
    import Decimal from 'decimal.js';

    // Initialised once in db.ts; restate here so importers that bypass db.ts still get correct settings.
    Decimal.set({ rounding: Decimal.ROUND_HALF_EVEN, precision: 30 });
    if (!(Decimal.prototype as any).toJSON) {
      (Decimal.prototype as any).toJSON = function () { return this.toString(); };
    }

    export type Money = Decimal;

    export function toMoney(v: string | number | Decimal | null | undefined): Money {
      if (v === null || v === undefined) return new Decimal(0);
      return v instanceof Decimal ? v : new Decimal(v);
    }

    export function sum(...vs: (string | number | Decimal | null | undefined)[]): Money {
      return vs.reduce<Decimal>((acc, v) => acc.plus(toMoney(v)), new Decimal(0));
    }

    export function diff(a: Money | string | number, b: Money | string | number): Money {
      return toMoney(a).minus(toMoney(b));
    }

    /** Percentage variance (a - b) / b * 100, returns Decimal. b===0 returns Decimal(0). */
    export function pct(a: Money | string | number, b: Money | string | number): Money {
      const bDec = toMoney(b);
      if (bDec.isZero()) return new Decimal(0);
      return toMoney(a).minus(bDec).div(bDec).times(100);
    }

    /** Format USD with thousands separators, 2 decimal places: "$1,500.50". */
    export function formatUSD(v: Money | string | number): string {
      const d = toMoney(v);
      const s = d.toFixed(2);
      // Insert thousands separators before the decimal.
      const [intPart, fracPart] = s.split('.');
      const sign = intPart.startsWith('-') ? '-' : '';
      const absInt = sign ? intPart.slice(1) : intPart;
      const withCommas = absInt.replace(/\B(?=(\d{3})+(?!\d))/g, ',');
      return `${sign}$${withCommas}.${fracPart}`;
    ```

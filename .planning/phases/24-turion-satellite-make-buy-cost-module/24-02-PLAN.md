    ASSUMPTION (RESEARCH.md Open Question #4, MEDIUM-confidence): "previous satellite" = the satellite with the most recent `program_start` strictly before the current satellite's `program_start`. Document this assumption in the route file's leading docstring AND surface it to the user in 24-05's human-verify checkpoint.    
    ```ts
    router.get('/:satId', requireAuth, async (req, res) => {
      const { satId } = req.params;
      try {
        // 1. by_subsystem rows for current sat
        const bySub = await query(
          `SELECT subsystem_code, subsystem_label,
                  make_total_usd, buy_invoiced_usd, buy_committed_usd, instance_count
           FROM turion_satellite.cost_rollup_v
           WHERE satellite_id=$1
           ORDER BY subsystem_code`,
          [satId]
        );
        // 2. totals: sum each numeric column via decimal.js (lib/money.ts sum())
        const totals = {
          make_total_usd:    sum(...bySub.map(r => r.make_total_usd)),
          buy_invoiced_usd:  sum(...bySub.map(r => r.buy_invoiced_usd)),
          buy_committed_usd: sum(...bySub.map(r => r.buy_committed_usd)),
        };
        // 3. previous-satellite by program_start (RESEARCH Open Question #4 recommendation)
        const prev = await queryOne(
          `SELECT id, name, program_start FROM turion_satellite.satellites
           WHERE id <> $1 AND program_start IS NOT NULL
             AND program_start < (SELECT program_start FROM turion_satellite.satellites WHERE id=$1)
           ORDER BY program_start DESC LIMIT 1`,
          [satId]
        );
        let prev_satellite_delta = null;
        if (prev?.id) {
          const prevSub = await query(
            `SELECT subsystem_code, make_total_usd, buy_invoiced_usd
             FROM turion_satellite.cost_rollup_v WHERE satellite_id=$1`,
            [prev.id]
          );
          // produce per-subsystem diff using diff() from lib/money.ts (current - prev)
          const prevMap = new Map(prevSub.map(r => [r.subsystem_code, r]));
          prev_satellite_delta = {
            previous_satellite: { id: prev.id, name: prev.name, program_start: prev.program_start },
            by_subsystem: bySub.map(r => ({
              subsystem_code: r.subsystem_code,
              make_delta_usd: diff(r.make_total_usd, prevMap.get(r.subsystem_code)?.make_total_usd ?? 0),
              buy_delta_usd:  diff(r.buy_invoiced_usd, prevMap.get(r.subsystem_code)?.buy_invoiced_usd ?? 0),
            })),
          };
        }
        res.json({ satellite_id: satId, by_subsystem: bySub, totals, prev_satellite_delta });
      } catch (err: any) {
        console.error('[cost-rollup] get failed:', err);
        res.status(500).json({ error: 'Failed to compute cost rollup' });
      }
    });
    ```

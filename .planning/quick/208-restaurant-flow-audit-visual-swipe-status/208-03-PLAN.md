private func calculateRemainingSeconds() -> Int {
    // Use sent_at (when the 3-minute window started) instead of placedAt (when customer placed order)
    // Backend sends sent_at as ISO 8601 string; fall back to placedAt if not available
    if let sentAt = order.sentAt,
       let sentDate = ISO8601DateFormatter().date(from: sentAt) {
        let elapsed = Date().timeIntervalSince(sentDate)
        return max(0, 180 - Int(elapsed))

}
